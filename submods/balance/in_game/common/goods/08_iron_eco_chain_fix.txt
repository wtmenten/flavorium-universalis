

# Deflates the metal chain to roughly a third of vanilla prices.
#
# Vanilla tunes the chain at iron = tools = 1.5 * coal (3 / 3 / 2) and the
# tools manufactories sit near break-even at that ratio, so iron and tools are
# cut together and their margins survive (their dominant input is iron, which
# got cheaper alongside the output).
#
# Coal is deliberately left at vanilla. Any building that spends coal to make
# iron is therefore underwater on price alone, so its coal input is scaled by
# the same factor in building_types/cc_bog_iron_recipe.txt rather than here.
# Do not "fix" that by cutting the coal price: coal feeds glass, porcelain,
# firearms and weapons, none of which are part of this retune.

REPLACE:iron = {
	method = mining
	category = raw_material
	color = goods_iron
	default_market_price = 0.88
	transport_cost = 2
	ai_rgo_size_importance = 10 # ai preference to avoid building a city on this rgo
	
	ai_rgo_expansion_priority = 0.025
	
	
}

REPLACE:tools = {		
	category = produced
	color = goods_tools
	default_market_price = 1.1
	
	transport_cost = 0.75
	
	demand_add = {
		laborers = 0.001
	}	
	
	wealth_impact_threshold = {
		all = 0.75
	}
	
}
