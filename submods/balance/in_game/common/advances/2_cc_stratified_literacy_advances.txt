# Flavorium Universalis — Class-Stratified Literacy Advances
#
# Converts flat global_max_literacy advances into class-stratified
# versions: upper classes (nobles / clergy / burghers) keep the full
# value, lower classes (laborers / soldiers / peasants) get less.

# Full fields preserved; only the literacy modifier changes.
############################################

# Vanilla: global_max_literacy = 5  (Novgorod "Birchbark Manuscripts")
# Change: upper 5 / lower 2
REPLACE_OR_CREATE:birch_bark_manuscript = {
	potential = { has_or_had_tag = NOV }

	age = age_1_traditions

	global_nobles_max_literacy = 5
	global_clergy_max_literacy = 5
	global_burghers_max_literacy = 5
	global_laborers_max_literacy = 2
	global_soldiers_max_literacy = 2
	global_peasants_max_literacy = 2

	requires = feudalism_advance
}
