############################################
# Flavorium Universalis — Library Building Override
#
# Replaces vanilla library's generic local_max_literacy = 5
# with estate-specific bonuses (nobles / clergy / burghers only).
# Peasants, soldiers, laborers, etc. do not benefit from
# a library the way literate upper classes do.
#
# All other fields are identical to vanilla.
############################################

REPLACE_OR_CREATE:library = {
	is_foreign = no
	pop_type = burghers
	max_levels = 1

	category = cultural_category
	employment_size = cultural_employment

	AI_ignore_available_worker_flag = yes

	town = yes
	city = yes
	megalopolis = yes

	forbidden_for_estates = yes

	graphical_tags = { medium_religious }

	build_time = cultural_building_time

	modifier = {
		local_cultural_influence = 0.1
		local_nobles_max_literacy = 3
		local_clergy_max_literacy = 3
		local_burghers_max_literacy = 2
		local_monthly_literacy = 0.03
		local_pop_promotion_speed = 0.001
		local_institution_growth_modifier = 0.25
	}

	possible_production_methods = {
		library_maintenance
	}

	construction_demand = library_construction
}

# Confucian School: vanilla local_max_literacy = 5 (all classes).
# Change: stratified — upper 5 / lower 2. Examination-school literacy
# benefits the literate gentry far more than the laboring classes.
REPLACE_OR_CREATE:confucian_school = {
	audio_tier = 1
	is_special = yes
	is_foreign = no
	max_levels = 1
	pop_type = nobles

	employment_size = noble_cultural_employment
	category = cultural_category

	AI_ignore_available_worker_flag = yes #since we may build this to increase pop promotion we don't want lack of workers to block it

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes

	build_time = cultural_building_time

	country_potential = {
		OR = {
			culture = { has_culture_group = culture_group:confucian_group }
			religion = religion:sanjiao
			any_international_organizations_member_of = {
				international_organization_type = international_organization_type:sect
				international_organization_has_policy = policy:confucianism_policy
			}
			custom_tooltip = {
				text = has_neo_confucianism_tt
				has_variable = has_neo_confucianism
			}
			is_member_of_international_organization = international_organization:middle_kingdom
		}
	}

	possible_production_methods = {
		confucian_school_maintenance
	}

	construction_demand = school_construction

	modifier = {
		local_nobles_max_literacy = 5
		local_clergy_max_literacy = 5
		local_burghers_max_literacy = 5
		local_laborers_max_literacy = 2
		local_soldiers_max_literacy = 2
		local_peasants_max_literacy = 2
		local_pop_promotion_speed = 0.001
		local_institution_growth_modifier = 0.25
	}
	capital_country_modifier = {
		monthly_towards_sinicized = societal_value_tiny_monthly_move
	}
}
