###############################################################################
# Flavorium Universalis - Factions: Factions situation
#
# A passive situation whose job is to host the party panel
# (in_game/gui/panels/situation/cc_pty_factions.gui), modelled on the main mod's Court
# Ledger (cc_cabinet_court.txt). Party state lives on each country, not on the situation,
# so the panel reads Player.MakeScope and every court sees only its own parties.
#
# Scopes (common/situations/readme.txt):
#   can_start / can_end                     root = situation
#   visible                                 root = country, scope:target = situation
#   map_color / tooltip                     root = location, scope:target = situation
###############################################################################

cc_pty_factions = {
	monthly_spawn_chance = monthly_spawn_chance_ultimate_high

	hint_tag = hint_cc_pty_factions

	can_start = {
		any_country = { has_variable = cc_pty_active }
	}

	# Permanent once spawned, like the Court Ledger. `visible` hides it from every court
	# without parties.
	can_end = {
		always = no
	}

	visible = {
		has_variable = cc_pty_active
	}

	# root = location; the viewer is not available here, so "the player" is is_ai = no.
	# From the age of absolutism, a player's cities shade by Liberal Thought (cc_pty_lt):
	# this is the location map the gauge in the panel averages. Elsewhere the realm is
	# drawn in a single colour.
	map_color = {
		if = {
			limit = {
				owner ?= {
					is_ai = no
					has_variable = cc_pty_active
				}
				trigger_if = {
					limit = { has_variable = cc_pty_lt }
					var:cc_pty_lt >= 60
				}
				trigger_else = { always = no }
			}
			value = rgb { 40 140 70 }
		}
		else_if = {
			limit = {
				owner ?= {
					is_ai = no
					has_variable = cc_pty_active
				}
				trigger_if = {
					limit = { has_variable = cc_pty_lt }
					var:cc_pty_lt >= 30
				}
				trigger_else = { always = no }
			}
			value = rgb { 110 175 95 }
		}
		else_if = {
			limit = {
				owner ?= {
					is_ai = no
					has_variable = cc_pty_active
				}
				has_variable = cc_pty_lt
			}
			value = rgb { 175 200 140 }
		}
		else_if = {
			limit = {
				owner ?= {
					is_ai = no
					has_variable = cc_pty_active
				}
			}
			value = rgb { 150 60 60 }
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}

	legend_key = {
		desc = "cc_pty_legend_realm"
		color = rgb { 150 60 60 }
	}
	legend_key = {
		desc = "cc_pty_legend_lt_low"
		color = rgb { 175 200 140 }
	}
	legend_key = {
		desc = "cc_pty_legend_lt_mid"
		color = rgb { 110 175 95 }
	}
	legend_key = {
		desc = "cc_pty_legend_lt_high"
		color = rgb { 40 140 70 }
	}

	tooltip = {
		if = {
			limit = {
				owner ?= { has_variable = cc_pty_active }
				has_variable = cc_pty_lt
			}
			custom_tooltip = cc_pty_map_tt_lt
		}
		else_if = {
			limit = { owner ?= { has_variable = cc_pty_active } }
			custom_tooltip = cc_pty_map_tt_realm
		}
	}
}
