###############################################################################
# Flavorium Universalis - Factions: disasters
#
# The last escalation step for two must-placate parties. Vanilla's coup_attempt and
# revolution_disaster cannot be started from script (no start_disaster effect, and their
# can_start reads conditions we do not control), so these are our own, armed by a
# variable that cc_pty_outcome_military / cc_pty_outcome_liberal set. Spawn chance 1
# means the disaster starts at the next monthly check once armed.
#
# Both end when the party is placated (satisfaction 0 or more), when it is gone, or
# after five years.
#
# Scopes (common/disasters/readme.txt): root = country, scope:disaster = the disaster type.
# Panels: in_game/gui/panels/disaster/<key>.gui. Art reuses vanilla's illustrations.
###############################################################################

cc_pty_praetorian_coup = {
	image = "gfx/interface/illustrations/disaster/coup_attempt.dds"

	monthly_spawn_chance = monthly_spawn_chance_unique

	can_start = {
		has_variable = cc_pty_coup_armed
		in_civil_war = no
	}

	can_end = {
		OR = {
			NOT = { cc_pty_has_type = { TYPE = 5 } }
			cc_pty_type_sat_ge = { TYPE = 5 VALUE = 0 }
			had_disaster_for_years = {
				disaster_type = scope:disaster
				years >= 5
			}
		}
	}

	modifier = {
		country_cabinet_efficiency = -0.15
		global_crown_estate_power = -0.10
		stability_cost_efficiency = -0.15
		monthly_rebel_growth = 0.005
	}

	on_start = {
		remove_variable = cc_pty_coup_armed
		trigger_event_non_silently = { id = cc_pty.40 }
	}

	on_end = {
		trigger_event_non_silently = { id = cc_pty.45 }
	}
}

# The Liberal Uprising runs on a momentum meter (cc_pty_lu_momentum, 0..100), not on the
# party's satisfaction: satisfaction is a one-off bump that a single event could push past
# zero. Momentum starts at 60 and moves every month by cc_pty_lu_drift (+1 while nothing is
# done; down only once the Liberals' grievance is actually addressed), and by the choices in
# the opening event cc_pty.42 and the crisis events cc_pty.70-75 (see cc_pty_lu_monthly).
#
#   momentum 0      defused                 -> cc_pty.46
#   momentum 100    boils over              -> cc_pty.47 (constitution or civil war)
#   five years      the clubs outlast it    -> cc_pty.48
#   party gone                              -> cc_pty.46
cc_pty_liberal_uprising = {
	image = "gfx/interface/illustrations/disaster/revolution_disaster.dds"

	monthly_spawn_chance = monthly_spawn_chance_unique

	can_start = {
		has_variable = cc_pty_uprising_armed
		in_civil_war = no
	}

	can_end = {
		OR = {
			NOT = { cc_pty_has_type = { TYPE = 11 } }
			custom_tooltip = {
				text = cc_pty_lu_end_defused_tt
				has_variable = cc_pty_lu_momentum
				NOT = { var:cc_pty_lu_momentum > 0 }
			}
			custom_tooltip = {
				text = cc_pty_lu_end_boiled_tt
				has_variable = cc_pty_lu_momentum
				var:cc_pty_lu_momentum >= 100
			}
			had_disaster_for_years = {
				disaster_type = scope:disaster
				years >= 5
			}
		}
	}

	modifier = {
		monthly_rebel_growth = 0.01
		country_cabinet_efficiency = -0.10
		monthly_towards_liberalism = societal_value_minor_monthly_move
	}

	on_start = {
		remove_variable = cc_pty_uprising_armed
		set_variable = { name = cc_pty_lu_momentum value = 60 }
		remove_variable = cc_pty_lu_repressed
		set_variable = { name = cc_pty_lu_event_cd value = yes months = 4 }
		trigger_event_non_silently = { id = cc_pty.42 }
	}

	on_monthly = {
		cc_pty_lu_monthly = yes
	}

	on_end = {
		if = {
			limit = {
				cc_pty_has_type = { TYPE = 11 }
				has_variable = cc_pty_lu_momentum
				var:cc_pty_lu_momentum >= 100
			}
			trigger_event_non_silently = { id = cc_pty.47 }
		}
		else_if = {
			limit = {
				cc_pty_has_type = { TYPE = 11 }
				has_variable = cc_pty_lu_momentum
				var:cc_pty_lu_momentum > 0
			}
			trigger_event_non_silently = { id = cc_pty.48 }
		}
		else = {
			trigger_event_non_silently = { id = cc_pty.46 }
		}
		remove_variable = cc_pty_lu_momentum
		remove_variable = cc_pty_lu_repressed
		remove_variable = cc_pty_lu_event_cd
	}
}
