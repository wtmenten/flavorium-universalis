namespace = cc_pty

###############################################################################
# Flavorium Universalis - Factions: court events (cc_pty.80-98)
#
# Rare personal chains between ministers, and between a minister and the ruler. Started by
# cc_pty_try_court_events (scripted_effects/cc_pty_court_effects.txt, which documents the
# pacing and the stored characters). Every event rebuilds its scopes with cc_pty_ce_load:
#   scope:cc_pty_ce_a, scope:cc_pty_ce_b   the ministers
#   scope:cc_pty_ce_ruler                  the ruler
#   Custom('cc_pty_ev_a_name') / _b_name   their parties' names
#
# Chains:
#   .80 Bad Blood -> .81 Whispers of a Plot -> .82 Murder at Court / .83 The Plot Unravels
#       (from .81 on, a is the plotter and b the target)
#   .85 An Affair of Honour -> .86 The Duel (a the winner, b the loser)
#   .88 The Power Behind the Throne -> .89 The Minister's Price
#   .91 Across the Aisle -> .92 An Unlikely Friendship
#   .94 A Friend at Court -> .95 Jealousy at Court
#   .97 Loyalty Tested
###############################################################################

# Two ministers of opposed parties fall out in front of the council.
cc_pty.80 = {
	type = country_event
	title = cc_pty.80.title
	desc = cc_pty.80.desc
	outcome = negative

	illustration_tags = {
		10 = interior
		10 = angry
	}

	trigger = {
		cc_pty_ce_seated = { WHO = a }
		cc_pty_ce_seated = { WHO = b }
	}

	immediate = { cc_pty_ce_load = yes }

	# Rebuke them both.
	option = {
		name = cc_pty.80.a
		add_political_influence = political_influence_weak_penalty
		custom_tooltip = cc_pty_ce_tt_both_m5
		hidden_effect = {
			cc_pty_char_add_sat = { CHAR = scope:cc_pty_ce_a VALUE = -5 }
			cc_pty_char_add_sat = { CHAR = scope:cc_pty_ce_b VALUE = -5 }
		}
		ai_chance = { base = 40 }
	}

	# Side with a. b is humiliated, and becomes the plotter.
	option = {
		name = cc_pty.80.b
		scope:cc_pty_ce_a = {
			add_character_modifier = { modifier = cc_minister_favoured years = 5 mode = replace }
		}
		scope:cc_pty_ce_b = {
			add_character_modifier = { modifier = cc_minister_slighted years = 10 mode = replace }
		}
		custom_tooltip = cc_pty_ce_tt_a_p10
		custom_tooltip = cc_pty_ce_tt_b_m15
		custom_tooltip = cc_pty_ce_tt_b_grudge
		hidden_effect = {
			cc_pty_char_add_sat = { CHAR = scope:cc_pty_ce_a VALUE = 10 }
			cc_pty_char_add_sat = { CHAR = scope:cc_pty_ce_b VALUE = -15 }
			random = {
				chance = 70
				cc_pty_run = { EFFECT = cc_pty_ce_swap }
				trigger_event_non_silently = { id = cc_pty.81 days = { 180 360 } }
			}
		}
		ai_chance = { base = 20 }
	}

	# Side with b. a is humiliated, and becomes the plotter.
	option = {
		name = cc_pty.80.c
		scope:cc_pty_ce_b = {
			add_character_modifier = { modifier = cc_minister_favoured years = 5 mode = replace }
		}
		scope:cc_pty_ce_a = {
			add_character_modifier = { modifier = cc_minister_slighted years = 10 mode = replace }
		}
		custom_tooltip = cc_pty_ce_tt_b_p10
		custom_tooltip = cc_pty_ce_tt_a_m15
		custom_tooltip = cc_pty_ce_tt_a_grudge
		hidden_effect = {
			cc_pty_char_add_sat = { CHAR = scope:cc_pty_ce_b VALUE = 10 }
			cc_pty_char_add_sat = { CHAR = scope:cc_pty_ce_a VALUE = -15 }
			random = {
				chance = 70
				trigger_event_non_silently = { id = cc_pty.81 days = { 180 360 } }
			}
		}
		ai_chance = { base = 20 }
	}

	# Let them settle it between themselves.
	option = {
		name = cc_pty.80.d
		custom_tooltip = cc_pty_ce_tt_unsettled
		hidden_effect = {
			random = {
				chance = 50
				trigger_event_non_silently = { id = cc_pty.81 days = { 180 360 } }
			}
		}
		ai_chance = { base = 20 }
	}
}

# The plot succeeds: b is murdered.
cc_pty.82 = {
	type = country_event
	title = cc_pty.82.title
	desc = cc_pty.82.desc
	outcome = negative

	illustration_tags = {
		10 = interior
		10 = angry
	}

	trigger = {
		cc_pty_ce_alive = { WHO = a }
		cc_pty_ce_alive = { WHO = b }
	}

	immediate = {
		cc_pty_ce_load = yes
		cc_pty_ce_save_victim_slot = yes
		cc_pty_ce_add_sat_victim = { VALUE = -30 }
		kill_character = {
			target = scope:cc_pty_ce_b
			reason = assassination
			killer = scope:cc_pty_ce_a
		}
	}

	# Hunt down the killers.
	option = {
		name = cc_pty.82.a
		add_gold = cc_pty_ce_inquiry_cost
		cc_pty_ce_remove = { CHAR = scope:cc_pty_ce_a YEARS = 20 }
		custom_tooltip = cc_pty_ce_tt_a_m20
		custom_tooltip = cc_pty_ce_tt_a_suppressed
		custom_tooltip = cc_pty_ce_tt_victim_p20
		hidden_effect = {
			cc_pty_char_add_sat = { CHAR = scope:cc_pty_ce_a VALUE = -20 }
			cc_pty_char_suppress = { CHAR = scope:cc_pty_ce_a }
			cc_pty_ce_add_sat_victim = { VALUE = 20 }
		}
		ai_chance = { base = 60 }
	}

	# Let the matter rest.
	option = {
		name = cc_pty.82.b
		add_stability = stability_mild_penalty
		add_country_modifier = { modifier = court_tension years = 3 mode = set_to_largest_and_extend }
		custom_tooltip = cc_pty_ce_tt_victim_m15
		hidden_effect = {
			cc_pty_ce_add_sat_victim = { VALUE = -15 }
		}
		ai_chance = { base = 40 }
	}
}

# Another minister resents the confidant's influence.
cc_pty.95 = {
	type = country_event
	title = cc_pty.95.title
	desc = cc_pty.95.desc
	outcome = negative

	illustration_tags = {
		10 = interior
		10 = angry
	}

	trigger = {
		exists = ruler
		cc_pty_ce_seated = { WHO = a }
		var:cc_pty_ce_a ?= { has_character_modifier = cc_pty_ce_confidant }
		any_cabinet_character = {
			has_variable = cc_pty_aff
			NOT = { has_character_modifier = cc_pty_ce_confidant }
		}
	}

	immediate = {
		cc_pty_ce_load = yes
		# The accuser: a minister of an opposed party if there is one, otherwise any other.
		if = {
			limit = {
				any_cabinet_character = {
					cc_pty_aff_active = yes
					cc_pty_aff_opposed = { OTHER = scope:cc_pty_ce_a }
				}
			}
			random_cabinet_character = {
				limit = {
					cc_pty_aff_active = yes
					cc_pty_aff_opposed = { OTHER = scope:cc_pty_ce_a }
				}
				save_scope_as = cc_pty_ce_pick_b
			}
		}
		else = {
			random_cabinet_character = {
				limit = {
					has_variable = cc_pty_aff
					NOT = { has_character_modifier = cc_pty_ce_confidant }
				}
				save_scope_as = cc_pty_ce_pick_b
			}
		}
		cc_pty_ce_store = { WHO = b CHAR = scope:cc_pty_ce_pick_b }
		cc_pty_ce_load = yes
	}

	# Stand by your friend.
	option = {
		name = cc_pty.95.a
		scope:cc_pty_ce_b = {
			add_character_modifier = { modifier = cc_minister_slighted years = 10 mode = replace }
		}
		custom_tooltip = cc_pty_ce_tt_a_p5
		custom_tooltip = cc_pty_ce_tt_b_m15
		hidden_effect = {
			cc_pty_char_add_sat = { CHAR = scope:cc_pty_ce_a VALUE = 5 }
			cc_pty_char_add_sat = { CHAR = scope:cc_pty_ce_b VALUE = -15 }
		}
		ai_chance = { base = 50 }
	}

	# Put some distance between you.
	option = {
		name = cc_pty.95.b
		scope:cc_pty_ce_a = {
			remove_character_modifier = cc_pty_ce_confidant
		}
		custom_tooltip = cc_pty_ce_tt_a_m10
		custom_tooltip = cc_pty_ce_tt_b_p10
		hidden_effect = {
			cc_pty_char_add_sat = { CHAR = scope:cc_pty_ce_a VALUE = -10 }
			cc_pty_char_add_sat = { CHAR = scope:cc_pty_ce_b VALUE = 10 }
		}
		ai_chance = { base = 50 }
	}
}
