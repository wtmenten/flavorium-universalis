# Country modifiers for the debt threads.
#
# Deliberately NOT duplicated from vanilla: BYZ_lacking_jewels (country.txt:7896,
# monthly_legitimacy -0.1) already covers the regalia sitting in Venice, and the regalia
# thread reads that modifier rather than adding a second one beside it.
#
# All modifier names here were checked with tools/vanilla_search.py modifier <name>.
# bank_interest is a rate added to the interest on new loans, so a positive value is a cost.

# A creditor holds a standing exemption on the customs. The revenue does not stop; it stops
# arriving at the treasury.
cc_byz_customs_conceded = {
	game_data = {
		category = country
	}

	tax_income_efficiency = -0.08
	global_trade_center_power = -0.1
	global_merchant_power = -0.05
}

# The republics have decided the emperor is a poor risk, and price him accordingly.
cc_byz_creditors_wary = {
	game_data = {
		category = country
	}

	bank_interest = 0.02
	total_loan_capacity_modifier = -0.2
}

# The opposite: a reputation for paying. Cheap credit is the reward for the expensive choice.
cc_byz_sound_credit = {
	game_data = {
		category = country
	}

	bank_interest = -0.015
	total_loan_capacity_modifier = 0.25
}

# Next year's revenue has already been sold. The money arrives once, and then the men who

	}
