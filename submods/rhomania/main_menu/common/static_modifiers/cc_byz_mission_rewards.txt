# Short-duration reward modifiers for the mission tree's stepping stones.
#
# SIX SHARED MODIFIERS, NOT TWENTY-SIX. Every small task draws from this pool rather than
# defining its own, because twenty-six near-identical entries would be twenty-six things to
# keep balanced and twenty-six more lines in a tooltip when several are live at once.
#
# These are deliberately small and deliberately temporary. The long-lived, structural
# modifiers belong to the nine cornerstone tasks (see cc_byz_reconquest_modifiers.txt); a
# stepping stone should feel like a good year, not a permanent upgrade. Applied with
# mode = add_and_extend so completing several tasks on one theme extends the window rather
# than silently doing nothing.
#
# All names checked with tools/vanilla_search.py modifier <name>.

# Walls repaired, arsenals stocked, the garrison paid on time.
cc_byz_reward_fortifications = {
	game_data = {
		category = country
	}

	fort_maintenance_efficiency = 0.1
	global_manpower_modifier = 0.05
}

# A season of shipbuilding.
cc_byz_reward_naval = {

	}

	global_sailors_modifier = 0.1

}

# Recruits from a province that has decided it is ours.
#
# LATE-GAME PAIR TO cc_byz_reward_levies. Standing manpower is what a fisc with a working
# assessment buys; levies are what an empire that has not got one calls out. The early Sword
# tasks draw levies, the later ones draw this.
cc_byz_reward_manpower = {
	game_data = {
		category = country
	}

	global_manpower_modifier = 0.08
	global_levy_size_modifier = 0.05
}

# The theme called out. Fourteenth-century Byzantine armies were levies and mercenaries long
# before they were anything a treasury could pay monthly, so the EARLY Sword tasks pay in levy
# recovery and levy size rather than standing manpower. Recovery in particular is what the
# opening decades are short of: an empire this size can raise a host once and then cannot raise
# it again for years.
cc_byz_reward_levies = {
	game_data = {
		category = country
	}

	levy_recovery_modifier = 0.15
	global_levy_size_modifier = 0.08
	peasants_estate_levy_size = 0.10
}

# Credit on better terms.
#
# bank_interest IS INVERTED: negative is cheaper. Checked against vanilla, where the positive
# values sit on penalty modifiers (country.txt:6235 = 0.03) and the negative on boons
# (country.txt:416 = -0.02). The early Fisc tasks land in exactly the window where a Byzantine
# player is debt-spending to hold Thrace, and a point of interest is worth more then than any
# amount of trade power.
cc_byz_reward_credit = {
	game_data = {
		category = country
	}

	bank_interest = -0.01
	total_loan_capacity_modifier = 0.10
}

# A public triumph, and a city that believes in the throne again for a while.
cc_byz_reward_confidence = {
	game_data = {
		category = country
	}

	monthly_legitimacy = 0.05
	stability_decay = -0.002
	global_estate_satisfaction_from_legitimacy = 0.05
}
