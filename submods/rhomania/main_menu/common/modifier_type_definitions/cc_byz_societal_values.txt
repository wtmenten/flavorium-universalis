# Drift modifier types for the cc_taxis_vs_cc_dynatoi societal value.
#
# NAMING IS NOT FREE. The engine splits the societal value's key on "_vs_" and derives
# three names per pole, all of which must exist or the value half-works silently:
#
#   value key   cc_taxis_vs_cc_dynatoi
#   poles       cc_taxis                    cc_dynatoi
#   this file   monthly_towards_cc_taxis    monthly_towards_cc_dynatoi
#   statics     cc_taxis_progress_cabinet_efficiency  (see ../static_modifiers/)
#   loc         cc_taxis_focus              cc_dynatoi_focus
#
# The pole name comes from the KEY, not from the left_modifier/right_modifier comments.
# An earlier key of cc_taxis_vs_dynatoi produced poles "cc_taxis" and "dynatoi", so the
# right side resolved and the left side threw:
#   "Modifier type definition monthly_towards_cc_taxis must exist in DB"
#   "Societal value is missing static modifier cc_taxis_progress_cabinet_efficiency"
# Both poles are prefixed now so the derived names stay symmetric. Cf. the Brandenburg
# Country Pack's pru_hre_vs_pru_empire, which prefixes both sides for the same reason.

monthly_towards_cc_taxis = {
	color = neutral
	game_data = {
		category = country
		min = 0
		is_societal_value_change = yes
	}
}

monthly_towards_cc_dynatoi = {
	color = neutral
	game_data = {
		category = country
		min = 0
		is_societal_value_change = yes
	}
}
