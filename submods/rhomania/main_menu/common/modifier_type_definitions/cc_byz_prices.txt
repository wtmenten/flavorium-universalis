# EVERY price entry needs a matching <price key>_cost_modifier registered here.
#
# The engine builds one cost modifier per price so players can be given discounts on it, and
# it logs at load if the modifier type is absent:
#     [price_database.cpp:117]: Missing modifier type for price. <name>_cost_modifier
#
# Vanilla registers 85 of these in main_menu/common/modifier_type_definitions/00_modifier_types.txt.
# Shape copied from mend_schism_price_cost_modifier there (line 13687): color=bad because a
# higher value means the action costs more, percent=yes because it scales the price rather
# than adding to it.
#
# The four below correspond to the four price entries in
# in_game/common/prices/cc_byz_union_prices.txt. Adding a price without adding one of these
# is a load-time error, so the two files must be kept in step.

cc_byz_union_contacts_price_cost_modifier = {
	color = bad
	percent = yes
	game_data = {
		category = country
	}
}

	}
}

}
