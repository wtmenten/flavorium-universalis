# The consistory of the Praetorian Prefectures: parliament participation modifier types.
#
# SIX ENTRIES, AND ALL SIX ARE LOAD-BEARING. Per CLAUDE.md, every special status that may vote
# in an IO parliament needs a registered <status>_can_participate_in_parliament type, and
# vanilla pairs every one of those with a matching <status>_agenda_impact (percent = yes).
# Vanilla's own emperor_can_participate_in_parliament in 00_modifier_types.txt is the shape.
#
# If any one of these is missing the consistory does not error. It votes, and the status whose
# type is absent simply contributes nothing, which reads in game as the exarchs having no
# opinion rather than as a bug. That is why the count is stated here: three statuses, two types
# each, six.
#
# category = internationalorganization, matching the main mod's cc_concert_parliament.txt.

cc_byz_augustus_can_participate_in_parliament = {
	boolean = yes
	game_data = {
		category = internationalorganization
	}
}

cc_byz_senior_exarch_can_participate_in_parliament = {
	boolean = yes
	game_data = {
		category = internationalorganization
	}
}

cc_byz_exarch_seat_can_participate_in_parliament = {
