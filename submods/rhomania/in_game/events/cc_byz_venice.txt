namespace = cc_byz_venice

# The Serenissima Question: Venice and the empire, over half a century.
#
# Venice was a Roman province before it was a republic. Its first doges were Byzantine
# officials, it held the title of dux from Constantinople into the eleventh century, and St
# Mark's is a Constantinopolitan church built on the lagoon. It was also the city that sacked
# Constantinople in 1204. This chain asks whether a restored empire can bring Venice back into
# its orbit, and what that would cost.
#
# ENTRY. Completing cc_byz_task_venetian_alliance calls cc_byz_ven_open_effect, which fires .1.
# Runs once per campaign. The Genoese task remains the exclusive alternative; inside the chain
# Genoa is the rival partner, and siding with it costs Concord.
#
#   .1   the chrysobull renewed                      privileges, Genoa, Italian standing
#   .2   the jewels and the loans                    vanilla crown jewels, creditor grip
#   .3   the Genoese war                             join Venice, stay out, or side with Genoa
#   .4   the sea league                              a joint fleet against the Turks
#   .5   Dalmatia                                    Venice pressed by Hungary
#   .6   the mainland question                       the Italian exarch, if there is one
#   .7   the Stato da Mar                            buy or swap a Morea or Aegean port
#   .20  (Venice) the port offer                     a human Venice answers .7
#   .22  the port answer                             back to us; the deal is applied here
#   .8   the Emperor's vicariate                     the HRE pull, against our old ducal title
#   .9   St Mark's and the purple                    union, Latin Empress and Academy read here
#   .10  the climax                                  Ducate, perpetual alliance, friendship, or rupture
#   .25  (Venice) the proposal                       a human Venice answers .10
#   .26  the proposal answer                         back to us; the ending is applied here
#   .80  hidden router
#   .90  the question lapses
#
# SCORES. Concord 0..10 (starts 3) decides the ending: 8+ with Leverage +2 makes the Ducate
# possible, 5+ the alliance, 3-4 a friendship, 2 or less a rupture. Leverage -5..5 is who needs
# whom. See scripted_effects/cc_byz_venice_effects.txt.
#
# PACING. 4-6 years between stages, so 40-55 years end to end. A war with Venice holds the
# router a year at a time; ten years of it closes the chain through .90. Wars elsewhere do not.
#
# CHARACTER STAT RANGES. adm, dip and mil are each 0..100. Roughly >= 65 above average.


#############################################
# .1  The chrysobull renewed
#############################################
cc_byz_venice.1 = {
	type = country_event
	title = cc_byz_venice.1.title
	desc = cc_byz_venice.1.desc
	outcome = neutral

	trigger = { has_variable = cc_byz_ven_active }

	illustration_tags = {
		10 = trading
		10 = economy
	}

	immediate = { cc_byz_ven_stage_setup_effect = yes }

	# Renew the 1082 terms in full.
	option = {
		name = cc_byz_venice.1.a
		ai_chance = { base = 35 }

		set_variable = { name = cc_byz_mem_ven_privileges_full  value = yes }
		add_country_modifier = { modifier = cc_byz_ven_old_privileges  years = 30  mode = add_and_extend }
		cc_byz_ven_venice_opinion_effect = { modifier = cc_byz_ven_courted }
		custom_tooltip = cc_byz_ven_concord_plus2_tt
		custom_tooltip = cc_byz_ven_leverage_minus1_tt
		hidden_effect = {
			cc_byz_ven_add_concord_effect = { value = 2 }
			cc_byz_ven_add_leverage_effect = { value = -1 }
			cc_byz_ven_add_italian_standing_effect = { value = 1 }
		}
		c:GEN ?= { add_opinion = { target = root  modifier = cc_byz_rival_privilege_granted } }
	}

	# Cut the exemption to two percent, as the 1265 treaty did for Genoa.
	option = {
		name = cc_byz_venice.1.b
		ai_chance = { base = 45 }

		add_country_modifier = { modifier = cc_byz_ven_reduced_privileges  years = 30  mode = add_and_extend }
		custom_tooltip = cc_byz_ven_leverage_plus1_tt
		hidden_effect = { cc_byz_ven_add_leverage_effect = { value = 1 } }
	}

	# Revoke it and charge the full kommerkion.
	option = {
		name = cc_byz_venice.1.c
		ai_chance = { base = 20 }

		set_variable = { name = cc_byz_mem_ven_privileges_revoked  value = yes }
		add_country_modifier = { modifier = cc_byz_ven_full_kommerkion  years = 30  mode = add_and_extend }
		cc_byz_ven_venice_opinion_effect = { modifier = cc_byz_ven_wronged }
		custom_tooltip = cc_byz_ven_concord_minus2_tt
		custom_tooltip = cc_byz_ven_leverage_plus1_tt
		hidden_effect = {
			cc_byz_ven_add_concord_effect = { value = -2 }
			cc_byz_ven_add_leverage_effect = { value = 1 }
			cc_byz_ven_add_italian_standing_effect = { value = -1 }
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate  value = estate_satisfaction_weak_bonus }
		c:GEN ?= { add_opinion = { target = root  modifier = cc_byz_ven_courted } }
	}

	# A merchant minister writes terms both sides can live with.
	option = {
		name = cc_byz_venice.1.d
		trigger = { cc_byz_ven_merchant_minister_trigger = yes }
		ai_chance = { base = 50 }

		add_country_modifier = { modifier = cc_byz_ven_reduced_privileges  years = 30  mode = add_and_extend }
		custom_tooltip = cc_byz_ven_concord_plus1_tt
		custom_tooltip = cc_byz_ven_leverage_plus1_tt
		hidden_effect = {
			cc_byz_ven_add_concord_effect = { value = 1 }
			cc_byz_ven_add_leverage_effect = { value = 1 }
		}
	}

	after = { hidden_effect = { cc_byz_ven_schedule_next_effect = { min = 4  max = 6 } } }
}

# .2  The jewels and the loans
#
# Reads vanilla's crown jewels (flavor_byz_ven.2): pawned to Venice, the jewels are leverage
# Venice holds over us. Otherwise Venice simply offers money.
#############################################
cc_byz_venice.2 = {
	type = country_event
	title = cc_byz_venice.2.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { cc_byz_ven_jewels_pawned_trigger = yes }
				desc = cc_byz_venice.2.desc_jewels
			}
			desc = cc_byz_venice.2.desc
		}
	}
	outcome = neutral

	trigger = { has_variable = cc_byz_ven_active }

	illustration_tags = {
		10 = bank
		10 = economy
	}

	immediate = { cc_byz_ven_stage_setup_effect = yes }

	# Jewels in pawn: pay the debt down and ask for them back in good faith.
	option = {
		name = cc_byz_venice.2.a
		trigger = {
			cc_byz_ven_jewels_pawned_trigger = yes
			gold >= 200
		}
		ai_chance = { base = 50 }

		add_gold = -200
		custom_tooltip = cc_byz_ven_concord_plus1_tt
		custom_tooltip = cc_byz_ven_leverage_plus1_tt
		hidden_effect = {
			cc_byz_ven_add_concord_effect = { value = 1 }
			cc_byz_ven_add_leverage_effect = { value = 1 }
			cc_byz_ven_add_creditor_grip_effect = { value = -1 }
		}
	}

	# Take the loan Venice offers.
	option = {
		name = cc_byz_venice.2.b
		ai_chance = { base = 40 }

		add_gold = 250
		custom_tooltip = cc_byz_ven_concord_plus1_tt
		custom_tooltip = cc_byz_ven_leverage_minus1_tt
		custom_tooltip = cc_byz_ven_creditor_grip_plus1_tt
		hidden_effect = {
			cc_byz_ven_add_concord_effect = { value = 1 }
			cc_byz_ven_add_leverage_effect = { value = -1 }
			cc_byz_ven_add_creditor_grip_effect = { value = 1 }
		}
	}

	# Owe Venice nothing.
	option = {
		name = cc_byz_venice.2.c
		ai_chance = { base = 30 }

		add_prestige = prestige_weak_bonus
		custom_tooltip = cc_byz_ven_leverage_plus1_tt
		hidden_effect = { cc_byz_ven_add_leverage_effect = { value = 1 } }
	}

	after = { hidden_effect = { cc_byz_ven_schedule_next_effect = { min = 4  max = 6 } } }
}

#############################################
# .6  The mainland question
#
# With an exarch in Italy the question is his: he governs the land next to the lagoon and wants
# more of it. Without one, it is Venice's own push onto the terraferma against the Empire.
#############################################
cc_byz_venice.6 = {
	type = country_event
	title = cc_byz_venice.6.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { cc_byz_ven_italian_exarch_trigger = yes }
				desc = cc_byz_venice.6.desc_exarch
			}
			desc = cc_byz_venice.6.desc
		}
	}
	outcome = neutral

	trigger = { has_variable = cc_byz_ven_active }

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		cc_byz_ven_stage_setup_effect = yes
		cc_byz_init_exarch_scores_effect = yes
		random_subject = {
			limit = {
				has_variable = cc_byz_is_exarchate
				capital.region = region:italy_region
			}
			save_scope_as = cc_ven_exarch
		}
	}

	# --- With an exarch ---

	# Restrain the exarch. Venice's land is not his to want.
	option = {
		name = cc_byz_venice.6.a
		trigger = { exists = scope:cc_ven_exarch }
		ai_chance = { base = 50 }

		custom_tooltip = cc_byz_ven_concord_plus2_tt
		custom_tooltip = cc_byz_ven_exarch_restrained_tt
		hidden_effect = {
			cc_byz_ven_add_concord_effect = { value = 2 }
			cc_byz_exarch_add_appetite_effect = { EXARCH = scope:cc_ven_exarch  VALUE = -1 }
			scope:cc_ven_exarch = {
				if = {
					limit = { has_variable = cc_byz_exarch_loyalty }
					change_variable = { name = cc_byz_exarch_loyalty  add = -1 }
					clamp_variable = { name = cc_byz_exarch_loyalty  min = -5  max = 5 }
				}
			}
		}
	}

	# Let the exarch press Venice. It reminds the lagoon whose neighbour it is.
	option = {
		name = cc_byz_venice.6.b
		trigger = { exists = scope:cc_ven_exarch }
		ai_chance = { base = 50 }

		custom_tooltip = cc_byz_ven_concord_minus2_tt
		custom_tooltip = cc_byz_ven_leverage_plus2_tt
		custom_tooltip = cc_byz_ven_exarch_unleashed_tt
		hidden_effect = {
			cc_byz_ven_add_concord_effect = { value = -2 }
			cc_byz_ven_add_leverage_effect = { value = 2 }
			cc_byz_exarch_add_appetite_effect = { EXARCH = scope:cc_ven_exarch  VALUE = 1 }
		}
	}

	# --- Without one ---

	# Support Venice on the terraferma against the Empire's lords.
	option = {
		name = cc_byz_venice.6.c
		trigger = { NOT = { exists = scope:cc_ven_exarch } }
		ai_chance = { base = 60 }

		custom_tooltip = cc_byz_ven_concord_plus1_tt
		hidden_effect = { cc_byz_ven_add_concord_effect = { value = 1 } }
	}

	# Venice's mainland is Venice's business.
	option = {
		name = cc_byz_venice.6.d
		trigger = { NOT = { exists = scope:cc_ven_exarch } }
		ai_chance = { base = 40 }

		custom_tooltip = cc_byz_ven_leverage_plus1_tt
		hidden_effect = { cc_byz_ven_add_leverage_effect = { value = 1 } }
	}

	after = { hidden_effect = { cc_byz_ven_schedule_next_effect = { min = 4  max = 6 } } }
}
