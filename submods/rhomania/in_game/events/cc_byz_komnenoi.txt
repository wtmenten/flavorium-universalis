namespace = cc_byz_komnenoi

# The Komnenian Question: Byzantium and Trebizond.
#
# Two Greek emperors have claimed the Roman title since 1204. This chain settles it, by
# union, by acclamation, or by war. It runs once per campaign.
#
# ENTRY. Either the decision (decisions/cc_byz_komnenoi_decisions.txt) or the yearly pulse
# (on_action/cc_byz_komnenoi.txt). Both call cc_byz_kom_open_effect, which records which of the
# five entrances held, stacks their starting scores, and fires .1.
#
#   .79                hidden opener for the passive route
#   .1                 the question opened
#   .2 .3 .4 .5        the stages: the Pontic court, the Turkmen frontier, Komnenian pride,
#                      rival suitors
#   .6                 the settlement (climax)
#   .20                Trebizond's answer, when Trebizond is human
#   .21                Trebizond accepted: acclamation or union
#   .30 .31 .32        outcomes: acclamation, union, turned away
#   .40                the pretext proclaimed
#   .50 .51            notifications to a human Trebizond
#   .80                hidden router
#   .90                the question lapses
#
# PACING. The passive entrance is behind the shared cc_byz_pulse_cd limiter. Once open, the
# chain schedules itself through the router: 1-2 years to the first stage, 2-3 years between
# the rest, so 8-13 years end to end, or about 5 fewer on the family shortcut in .4. Every
# stage stamps the cooldown so the other Rhomania threads give way while it runs, but the
# chain itself never waits on it.
#
# SCORES. cc_byz_kom_accord weights the climax roll; cc_byz_kom_pretext sets the CB's length
# and the peace-table discount. Both are 0..10. See scripted_effects/cc_byz_komnenoi_effects.txt
# for the state and memory variables.
#
# CHARACTER STAT RANGES. adm, dip and mil are each 0..100; total_abilities is their sum,
# 0..300. Roughly: >= 65 above average, >= 75 capable, >= 85 exceptional.


#############################################

cc_byz_komnenoi.6 = {
	type = country_event
	title = cc_byz_komnenoi.6.title
	desc = cc_byz_komnenoi.6.desc
	outcome = neutral

	trigger = {
		has_variable = cc_byz_kom_active
	}

	illustration_tags = {
		10 = interior
		10 = professional
	}

	immediate = {
		cc_byz_pulse_stamp_effect = yes
		save_scope_as = cc_kom_emperor
		c:TRE = { save_scope_as = cc_kom_tre }
	}

	# Propose the union. An AI Trebizond is settled by the roll here, with the odds shown;
	# a human Trebizond is asked in .20.
	option = {
		name = cc_byz_komnenoi.6.a
		ai_chance = {
			base = 50
			modifier = {
				factor = 2
				var:cc_byz_kom_accord >= 6
			}
			modifier = {
				factor = 0.3
				var:cc_byz_kom_accord < 3
			}
		}

		if = {
			limit = { c:TRE = { is_ai = yes } }
			random_list = {
				# Acclamation: Trebizond annexed.
				4 = {
					modifier = {
						factor = 2
						var:cc_byz_kom_accord >= 4
					}
					modifier = {
						factor = 2.5
						var:cc_byz_kom_accord >= 6
					}
					modifier = {
						factor = 2
						var:cc_byz_kom_accord >= 8
					}
					modifier = {
						factor = 1.5
						var:cc_byz_kom_accord >= 10
					}
					modifier = {
						factor = 0.5
						is_ai = yes
					}
					custom_tooltip = cc_byz_kom_roll_acclamation_tt
					hidden_effect = {
						trigger_event_non_silently = { id = cc_byz_komnenoi.30 }
					}
				}
				# Union: one emperor, two crowns.
				30 = {
					modifier = {
						factor = 1.5
						var:cc_byz_kom_accord >= 4
					}
					modifier = {
						factor = 1.2
						var:cc_byz_kom_accord >= 6
					}
					modifier = {
						factor = 0.8
						var:cc_byz_kom_accord >= 8
					}
					modifier = {
						factor = 0.7
						var:cc_byz_kom_accord >= 10
					}
					modifier = {
						factor = 0.5
						is_ai = yes
					}
					custom_tooltip = cc_byz_kom_roll_union_tt
					hidden_effect = {
						trigger_event_non_silently = { id = cc_byz_komnenoi.31 }
					}
				}
				# Refusal.
				66 = {
					modifier = {
						factor = 0.6
						var:cc_byz_kom_accord >= 4
					}
					modifier = {
						factor = 0.5
						var:cc_byz_kom_accord >= 6
					}
					modifier = {
						factor = 0.5
						var:cc_byz_kom_accord >= 8
					}
					modifier = {
						factor = 0.5
						var:cc_byz_kom_accord >= 10
					}
					custom_tooltip = cc_byz_kom_roll_refused_tt
					hidden_effect = {
						trigger_event_non_silently = { id = cc_byz_komnenoi.32 }
					}
				}
			}
		}
		else = {
			custom_tooltip = cc_byz_kom_human_tre_decides_tt
			hidden_effect = {
				c:TRE = { trigger_event_non_silently = { id = cc_byz_komnenoi.20 } }
			}
		}
	}

	# Proclaim Trebizond's crown a usurpation and claim it by right.
	option = {
		name = cc_byz_komnenoi.6.b
		ai_chance = {
			base = 30
			modifier = {
				factor = 2
				var:cc_byz_kom_pretext >= 6
			}
			modifier = {
				factor = 0.3
				var:cc_byz_kom_pretext < 3
			}
		}

		custom_tooltip = cc_byz_kom_pretext_outcome_tt
		hidden_effect = {
			trigger_event_non_silently = { id = cc_byz_komnenoi.40 }
		}
	}

	# Let the question rest. Nothing is gained and nothing is broken.
	option = {
		name = cc_byz_komnenoi.6.c
		ai_chance = { base = 10 }

		set_variable = { name = cc_byz_mem_kom_let_rest  value = yes }
		cc_byz_kom_close_effect = yes
	}
}
