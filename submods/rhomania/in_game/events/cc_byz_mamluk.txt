namespace = cc_byz_mamluk

# The Mamluk confrontation.
#
# Five threads, interleaved rather than sequential, so the situation reads as a long
# confrontation with many surfaces rather than a ladder:
#
#   .1  .10 .11        the holy places, custody, and the road to escalation
#   .2                 Cairo takes note
#   .3                 spice, grain and the Italians
#   .4                 the frontier
#   .5                 Cairo's own troubles (from the situation's on_monthly)
#   .14                the eastern sees once they are the emperor's
#   .12 .13            the pentarchy seats
#   .20                the ending
#
# WHO HOLDS WHAT, AND WHEN IT IS CHECKED. Every event below that has an opinion about who owns
# Jerusalem, Antioch or Alexandria now tests it, through the triggers in
# scripted_triggers/cc_byz_mamluk_triggers.txt. The thread previously assumed the 1337 map
# forever: .1 asked an emperor holding Jerusalem whether to keep paying the sultan for the keys to
# it, .10 told him the patriarchs he had annexed were the sultan's subjects, and .3 explained that
# Cairo controlled a spice trade running through his own port. The delays are what made it show,
# since .10 was scheduled twelve years ahead and .11 eight years after that.
#
# The consequence for the shape of the file: the chain is no longer strung through after blocks.
# A gated event that declines to fire also declines to schedule its successor, which would end the
# confrontation silently. .3, .10, .14 and .11 are all scheduled from the situation's on_start
# instead, and .12 from its on_monthly, so each event's guard is independent of every other's.
#
# Scores:
#   cc_byz_levant_pressure  -5..5  on Byzantium
#   cc_byz_levant_progress  0..100 on the SITUATION
#   cc_byz_mamluk_progress  0..100 on the SITUATION
#   cc_byz_mamluk_phase     1 or 2 on the SITUATION
#
# SCOPE NOTE. The progress counters live on the situation, not on the country. Every read and
# write below goes through situation:cc_byz_mamluk_confrontation explicitly. Writing them in
# root scope would create country variables that the situation's can_end never reads, and the
# confrontation would then run until the 1650 timeout no matter what the player did.
#
# Memory variables (on Byzantium):
#   cc_byz_mem_condominium       a negotiated sharing of the holy places was reached
#   cc_byz_mem_patriarchs_backed the eastern patriarchs answer to Constantinople. Written by .10
#                                when they were protected at a cost, and by every path through
#                                .14, where they are the emperor's subjects and the only question
#                                left is on what terms.
#   cc_byz_mem_patriarchs_sold   the eastern patriarchs were left to their own arrangements
#   cc_byz_mem_pentarchy_war     phase B was entered
#   cc_byz_mem_seat_jerusalem / _antioch / _alexandria   a pentarchy seat was taken
#
# CHARACTER STAT RANGES. adm, dip and mil are each 0..100; total_abilities is their sum,
# 0..300. Roughly: >= 65 above average, >= 75 capable, >= 85 exceptional.


#############################################
# .1  The holy places
#############################################
cc_byz_mamluk.1 = {
	type = country_event
	title = cc_byz_mamluk.1.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = cc_byz_union_stage
					var:cc_byz_union_stage >= 2
				}
				desc = cc_byz_mamluk.1.desc_unionist
			}
			desc = cc_byz_mamluk.1.desc
		}
	}
	outcome = neutral

	# JERUSALEM SPECIFICALLY, not the Levant generally. The whole event is the arrangement at the
	# Holy Sepulchre: dues paid annually to Cairo for rights in a city the sultan holds. An emperor
	# who already holds Jerusalem is not paying anyone for the keys to it, and one who is disputing
	# custody with a Latin kingdom or a Cypriot garrison is not disputing it with Cairo.
	trigger = {
		has_societal_value = societal_value_type:latinization_vs_hellenization
		cc_byz_jerusalem_under_cairo_trigger = yes
	}

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		cc_byz_init_levant_scores_effect = yes
	}

	# Pay the sultan for the rights, as every emperor has since the conquest. Cheap and it
	# concedes the principle that they are his to grant.
	option = {
		name = cc_byz_mamluk.1.a
		ai_chance = { base = 40 }

		add_gold = -150
		change_variable = { name = cc_byz_levant_pressure  add = -1 }
		clamp_variable = { name = cc_byz_levant_pressure  min = -5  max = 5 }
		situation:cc_byz_mamluk_confrontation ?= {
			change_variable = { name = cc_byz_mamluk_progress  add = 10 }
		}
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_penalty
		}
	}

	# Assert the custody as the emperor's by inheritance and stop paying.
	option = {
		name = cc_byz_mamluk.1.b
		ai_chance = { base = 35 }

		change_variable = { name = cc_byz_levant_pressure  add = 2 }
		clamp_variable = { name = cc_byz_levant_pressure  min = -5  max = 5 }
		situation:cc_byz_mamluk_confrontation ?= {
			change_variable = { name = cc_byz_levant_progress  add = 15 }
		}
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_bonus
		}
		add_prestige = prestige_mild_bonus
	}

	# Negotiate a standing arrangement: fixed dues, fixed rights, in writing. The route to
	# the condominium ending, and it takes a diplomat to open it.
	option = {
		name = cc_byz_mamluk.1.c
		ai_chance = { base = 25 }
		trigger = {
			ruler ?= { dip >= 75 }
		}

		add_gold = -100
		set_variable = { name = cc_byz_mem_condominium  value = yes }
		situation:cc_byz_mamluk_confrontation ?= {
			change_variable = { name = cc_byz_levant_progress  add = 20 }
		}
		c:MAM ?= {
			add_opinion = {
				target = root
				modifier = cc_byz_levant_accommodation
			}
		}
	}

	# NO after BLOCK. The chain used to be strung from here: .3 at five years, .10 at twelve. Now
	# that this event can decline to fire, an after block would take the whole thread down with it.
	# All four scheduled events moved to the situation's on_start, which runs unconditionally and
	# gives the same timings. See the note there.
}

#############################################
# .3  Spice, grain and the Italians
#
# Couples to the debt threads: an emperor deep in Italian debt finds his creditors have
# interests in Cairo, and those interests are not his.
#############################################
cc_byz_mamluk.3 = {
	type = country_event
	title = cc_byz_mamluk.3.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:cc_byz_creditor_grip >= 3 }
				desc = cc_byz_mamluk.3.desc_indebted
			}
			desc = cc_byz_mamluk.3.desc
		}
	}
	outcome = negative

	# ALEXANDRIA IS THE SUBJECT, so Alexandria is the test. The event's premise is that the spice
	# reaches Europe through Cairo's hands and the emperor's only lever is the straits. An emperor
	# who has taken Alexandria has the factories themselves, which is the eastern trade thread
	# (events/cc_byz_east.txt) and not this one.
	trigger = {
		cc_byz_mamluk_chain_open_trigger = yes
		cc_byz_alexandria_under_cairo_trigger = yes
	}

	illustration_tags = {
		10 = trading
		10 = economy
	}

	immediate = {
		cc_byz_init_levant_scores_effect = yes
		cc_byz_init_debt_scores_effect = yes
	}

	# Let the republics trade with Cairo. The customs are worth more than the principle.
	option = {
		name = cc_byz_mamluk.3.a
		ai_chance = { base = 45 }

		add_gold = 150
		change_variable = { name = cc_byz_levant_pressure  add = -1 }
		clamp_variable = { name = cc_byz_levant_pressure  min = -5  max = 5 }
		situation:cc_byz_mamluk_confrontation ?= {
			change_variable = { name = cc_byz_mamluk_progress  add = 10 }
		}
	}

	# Close the straits to the Alexandria trade. It costs the republics more than it costs
	# Cairo, which is the problem with it.
	option = {
		name = cc_byz_mamluk.3.b
		ai_chance = { base = 30 }

		change_variable = { name = cc_byz_levant_pressure  add = 2 }
		clamp_variable = { name = cc_byz_levant_pressure  min = -5  max = 5 }
		change_variable = { name = cc_byz_italian_standing  add = -2 }
		clamp_variable = { name = cc_byz_italian_standing  min = -5  max = 5 }
		situation:cc_byz_mamluk_confrontation ?= {
			change_variable = { name = cc_byz_levant_progress  add = 15 }
		}
		every_country = {
			limit = {
				OR = {
					has_or_had_tag = VEN
					has_or_had_tag = GEN
					has_or_had_tag = PIS
				}
			}
			add_opinion = {
				target = root
				modifier = cc_byz_rival_privilege_granted
			}
		}
	}

	# Take a cut of it instead. The trade continues and the empire is now a party to it,
	# which is a better position than either forbidding it or ignoring it.
	option = {
		name = cc_byz_mamluk.3.c
		ai_chance = { base = 25 }
		trigger = {
			ruler ?= { adm >= 75 }
		}

		add_gold = 250
		change_variable = { name = cc_byz_italian_standing  add = 1 }
		clamp_variable = { name = cc_byz_italian_standing  min = -5  max = 5 }
		situation:cc_byz_mamluk_confrontation ?= {
			change_variable = { name = cc_byz_levant_progress  add = 10 }
		}
	}

	after = {
		# ON CAIRO, NOT ON ROOT. This shipped as a bare trigger_event_silently, which fires on
		# the event's own scope, and .3 runs on Byzantium. So the Byzantine player was handed
		# .2, the one event in this file written from the SULTAN's side: an event whose option b
		# does `c:BYZ = { add_opinion = { target = root } }`, meaning the emperor improved his
		# own opinion of himself. Reported from playtesting as "as byz i am getting
		# cc_byz_mamluk.2, this seems like it should be a mam only event".
		#
		# .2 now also carries its own tag guard, so a future miswiring cannot repeat this.
		hidden_effect = {
			c:MAM ?= {
				trigger_event_silently = { id = cc_byz_mamluk.2  years = 4 }
			}
		}
	}
}

# .4  The frontier
#############################################
cc_byz_mamluk.4 = {
	type = country_event
	title = cc_byz_mamluk.4.title
	desc = {
		first_valid = {
			# Reads the SITUATION's phase rather than the country's cc_byz_mem_pentarchy_war.
			# The phase is a property of the confrontation, not of Byzantium, and this event
			# fires from the situation's on_monthly: a frontier during a declared war is a
			# different place from a frontier during a cold one, whoever is narrating it.
			triggered_desc = {
				trigger = {
					situation:cc_byz_mamluk_confrontation ?= { var:cc_byz_mamluk_phase = 2 }
				}
				desc = cc_byz_mamluk.4.desc_war
			}
			desc = cc_byz_mamluk.4.desc
		}
	}
	outcome = negative

	trigger = {
		cc_byz_mamluk_thread_open_trigger = yes
	}

	illustration_tags = {
		10 = military
		10 = armed
	}

	immediate = {
		cc_byz_init_levant_scores_effect = yes
		cc_byz_init_debt_scores_effect = yes

		# SIX-YEAR SHARED COOLDOWN, stamped by both events that fire from the situation's
		# monthly roll and read by cc_byz_mamluk_thread_open_trigger. Timed, so it expires on
		# its own with nothing to clear it. At the reworked 8% annual rate this rarely binds;
		# what it prevents is the frontier and Cairo both speaking in consecutive months,
		# which the monthly roll makes possible and which reads as noise.
		set_variable = { name = cc_byz_mamluk_cd  value = yes  years = 6 }
	}

	# Buy the border emirs. They change sides for money and will change back for more.
	option = {
		name = cc_byz_mamluk.4.a
		ai_chance = { base = 40 }

		add_gold = -400
		add_country_modifier = {
			modifier = cc_byz_emirs_retained
			years = 20
			mode = add_and_extend
		}
		add_manpower = 5
		situation:cc_byz_mamluk_confrontation ?= {
			change_variable = { name = cc_byz_levant_progress  add = 10 }
		}
	}

	# Garrison the border towns properly and let the raiding come to the walls.
	option = {
		name = cc_byz_mamluk.4.b
		ai_chance = { base = 35 }
		trigger = {
			ruler ?= { mil >= 70 }
		}

		add_gold = -300
		add_army_tradition = army_tradition_severe_bonus
		add_country_modifier = {
			modifier = cc_byz_syrian_garrisons
			years = 30
			mode = add_and_extend
		}
		change_variable = { name = cc_byz_levant_pressure  add = 1 }
		clamp_variable = { name = cc_byz_levant_pressure  min = -5  max = 5 }
		change_variable = { name = cc_byz_strain  add = 1 }
		clamp_variable = { name = cc_byz_strain  min = 0  max = 10 }
	}

	# Leave the frontier to the men who live on it and stop pretending it has a line.
	option = {
		name = cc_byz_mamluk.4.c
		ai_chance = { base = 25 }

		change_variable = { name = cc_byz_levant_pressure  add = -1 }
		clamp_variable = { name = cc_byz_levant_pressure  min = -5  max = 5 }
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_move_to_right
		}
		# The drift as well as the jump: handing a frontier to the men who hold it is a
		# process, and it is the clearest case in the thread of the magnates gaining ground
		# by the empire declining to govern.
		add_country_modifier = {
			modifier = cc_byz_dynatoi_creeping
			years = 25
			mode = add_and_extend
		}
		add_country_modifier = {
			modifier = cc_byz_frontier_left_alone
			years = 40
			mode = add_and_extend
		}
		situation:cc_byz_mamluk_confrontation ?= {
			change_variable = { name = cc_byz_mamluk_progress  add = 5 }
		}
	}
}

# .10  The eastern patriarchs
#

# modified or retargeted." Its first line is that the patriarchs of Antioch, Jerusalem and

# emperor who had spent those twelve years making it false.
#

# holds at least one of the three, and desc_partial narrates the case where some have been
# recovered and some have not, which is the ordinary mid-campaign state and the one the event
# read worst in. RETARGETED: when Cairo holds none of them, .14 fires instead, on the same

#############################################
# .14  The sees in the emperor's hand
#
# .10 RETARGETED. Same slot, same twelve-year schedule, complementary trigger: .10 requires Cairo
# to hold at least one eastern seat, this requires Cairo to hold none and the emperor to hold at
# least one, so exactly one of the two can land and neither fires when all three sit in some third
# power's hands.
#
# The question is genuinely a different one rather than the same one with the names changed. .10
# is about protection: whether an emperor spends anything on co-religionists he cannot reach, and
# whether saying so out loud gets them killed. Here he can reach them, and what he has acquired is
# the appointment of three patriarchs whose sees have run themselves under Muslim rule for eight
# centuries and have institutions of their own. The trap is the opposite one. .10 risks doing too
# little from too far away; this risks doing too much from too close.
#
# Writes cc_byz_mem_patriarchs_backed on every path, which is not a moral judgement but a fact:
# whatever the emperor decides here, nobody can say afterwards that the eastern sees were left to
# the sultan. That closes .10 permanently, as .10 closes this.
#############################################
cc_byz_mamluk.14 = {
	type = country_event
	title = cc_byz_mamluk.14.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = cc_byz_union_stage
					var:cc_byz_union_stage >= 2
				}
				desc = cc_byz_mamluk.14.desc_unionist
			}
			desc = cc_byz_mamluk.14.desc
		}
	}
	outcome = neutral

	# NOT cc_byz_mamluk_chain_open_trigger. That one refuses to fire during a war with Cairo, and
	# the likeliest way to be holding these sees at all is to have just taken them off him. The
	# exists test goes with it: the sees can perfectly well be the emperor's because the sultanate
	# no longer exists, and that is the case this event most needs to cover.
	trigger = {
		has_societal_value = societal_value_type:latinization_vs_hellenization
		cc_byz_any_eastern_see_held_trigger = yes
		NOT = { cc_byz_eastern_sees_under_cairo_trigger = yes }
		NOR = {
			has_variable = cc_byz_mem_patriarchs_backed
			has_variable = cc_byz_mem_patriarchs_sold
		}
	}

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		cc_byz_init_levant_scores_effect = yes
		set_variable = { name = cc_byz_mem_patriarchs_backed  value = yes }
	}

	# Appoint from Constantinople. The sees are imperial now and the patriarchs are imperial
	# officials, which is tidy, defensible, and resented by everyone who lives there.
	option = {
		name = cc_byz_mamluk.14.a
		ai_chance = { base = 35 }

		add_prestige = prestige_mild_bonus
		change_societal_value = {
			type = latinization_vs_hellenization
			value = societal_value_move_to_right
		}
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_minor_move_to_left
		}
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_penalty
		}
		situation:cc_byz_mamluk_confrontation ?= {
			change_variable = { name = cc_byz_levant_progress  add = 15 }
		}
	}

	# Confirm the sitting patriarchs and leave the sees their customs. Costs nothing the treasury
	# will miss and buys the one thing conquest cannot, which is the clergy on the ground.
	option = {
		name = cc_byz_mamluk.14.b
		ai_chance = { base = 40 }

		add_gold = -100
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_bonus
		}
		every_country = {
			limit = {
				religion = religion:orthodox
				NOT = { this = root }
			}
			add_opinion = {
				target = root
				modifier = cc_byz_restored_the_east
			}
		}
		situation:cc_byz_mamluk_confrontation ?= {
			change_variable = { name = cc_byz_levant_progress  add = 20 }
		}
	}

	# A synod, at which the sees keep their rites and the emperor keeps the appointments. Both of
	# the above at once, which is why it takes a ruler who can hold a room.
	option = {
		name = cc_byz_mamluk.14.c
		ai_chance = { base = 25 }
		trigger = {
			ruler ?= { dip >= 75 }
		}

		add_gold = -200
		add_prestige = prestige_severe_bonus
		change_societal_value = {
			type = latinization_vs_hellenization
			value = societal_value_move_to_right
		}
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_bonus
		}
		situation:cc_byz_mamluk_confrontation ?= {
			change_variable = { name = cc_byz_levant_progress  add = 25 }
		}
	}
}


#############################################
# .11  Escalation, or not
#
# The hinge between the phases. Phase B is entered ONLY here and only if the player chooses
# it, which is what makes the negotiated ending a real one rather than a timeout.
#############################################
cc_byz_mamluk.11 = {
	type = country_event
	title = cc_byz_mamluk.11.title
	desc = {
		first_valid = {
			# BEFORE THE CONDOMINIUM VARIANT. Both of the shipped descriptions close on the seats
			# of Jerusalem, Antioch and Alexandria being three of the five "and no emperor since
			# Heraclius has held them", which is a fine line to read to an emperor holding one of
			# them and an absurd one otherwise. Where the recovery has already started, the
			# decision point is a different decision.
			triggered_desc = {
				trigger = { cc_byz_any_eastern_see_held_trigger = yes }
				desc = cc_byz_mamluk.11.desc_holding
			}
			triggered_desc = {
				trigger = { has_variable = cc_byz_mem_condominium }
				desc = cc_byz_mamluk.11.desc_condominium
			}
			desc = cc_byz_mamluk.11.desc
		}
	}
	outcome = neutral

	trigger = {
		cc_byz_mamluk_chain_open_trigger = yes
		NOT = { has_variable = cc_byz_mem_pentarchy_war }
	}

	illustration_tags = {
		10 = military
		10 = interior
	}

	immediate = {
		cc_byz_init_levant_scores_effect = yes
		cc_byz_init_debt_scores_effect = yes
	}

	# Settle it. A condominium over the holy places, in writing, and the confrontation ends
	# without a war. Requires that the groundwork was actually done.
	option = {
		name = cc_byz_mamluk.11.a
		ai_chance = { base = 45 }
		trigger = {
			has_variable = cc_byz_mem_condominium
			var:cc_byz_levant_pressure <= 2
			# A condominium is a sharing of custody in a city somebody else governs. There is
			# nothing to share once the emperor holds Jerusalem, and signing a document conceding
			# the sultan rights in it would be a worse outcome than the war, not a better one.
			cc_byz_jerusalem_under_cairo_trigger = yes
		}

		situation:cc_byz_mamluk_confrontation ?= {
			set_variable = { name = cc_byz_levant_progress  value = 100 }
		}
		add_prestige = prestige_severe_bonus
		c:MAM ?= {
			add_opinion = {
				target = root
				modifier = cc_byz_levant_accommodation
			}
		}
	}

	# Keep the pressure on without committing to a war.
	option = {
		name = cc_byz_mamluk.11.b
		ai_chance = { base = 35 }

		change_variable = { name = cc_byz_levant_pressure  add = 1 }
		clamp_variable = { name = cc_byz_levant_pressure  min = -5  max = 5 }
		situation:cc_byz_mamluk_confrontation ?= {
			change_variable = { name = cc_byz_levant_progress  add = 10 }
		}
	}

	# The Pentarchy War. Jerusalem, Antioch and Alexandria, and the empire will hold all five
	# seats or lose the argument permanently.
	option = {
		name = cc_byz_mamluk.11.c
		ai_chance = { base = 20 }
		trigger = {
			var:cc_byz_levant_pressure >= 2
			# There has to be something left in Cairo's hands to go to war over. With all three
			# eastern seats already taken, this option was a war for objectives the emperor had
			# met, and .12 records them from the situation's monthly roll without it.
			cc_byz_eastern_sees_under_cairo_trigger = yes
		}

		set_variable = { name = cc_byz_mem_pentarchy_war  value = yes }
		situation:cc_byz_mamluk_confrontation ?= {
			set_variable = { name = cc_byz_mamluk_phase  value = 2 }
			change_variable = { name = cc_byz_levant_progress  add = 15 }
		}
		change_variable = { name = cc_byz_strain  add = 2 }
		clamp_variable = { name = cc_byz_strain  min = 0  max = 10 }
		if = {
			limit = {
				NOT = { has_unlocked_casus_belli_trigger = { type = cb_restore_roman_borders } }
			}
			unlock_casus_belli_effect = { type = cb_restore_roman_borders }
		}
		# No .12 schedule any more. Six years was a guess at how long a war takes; the situation's
		# on_monthly now fires it the month a seat is actually taken, in this war or any other.
	}
}


#############################################
# .12  The seats
#
# has_all_pentarchy (country_triggers.txt:1355) is PURE OWNERSHIP: owns constantinople, rome,
# jerusalem, antioch and alexandria, and nothing else. So the seats already count toward
# vanilla's mend_schism the moment they are conquered, without this event doing anything.
#
# What this event adds is add_pentarchy_modifier, the per-seat bonus, taken from
# autocephalous_patriarchates.txt:544-547. Vanilla pairs that with set_pentarch_variable, which
# is deliberately NOT called here: it needs a patriarch CHARACTER scope (scope:target_patriarch),
# supplied there by select_suitable_patriarch, and inventing one from a country-scope event
# would either error or install a patriarch the IO does not know about. The modifier is the
# half of the pair that is safe to apply on its own.
#
# Guarded on actually owning each seat, so an emperor who took Jerusalem and stalled at
# Antioch records only what he holds, and re-fires later to pick up the rest.
#############################################
cc_byz_mamluk.12 = {
	type = country_event
	title = cc_byz_mamluk.12.title
	desc = cc_byz_mamluk.12.desc
	outcome = positive

	# NOT WAR-GATED, unlike the rest of the thread. The other events narrate a cold war and
	# had to stop firing once it stopped being cold; this one records seats taken IN a war and
	# would never fire at all behind that guard. Same for .13 below.
	#
	# NOT PENTARCHY-WAR-GATED EITHER, any more. It carried has_variable = cc_byz_mem_pentarchy_war
	# and was scheduled only from .11's war option, which meant the seats were recorded when a
	# particular event chain took them and not when the player did. An emperor who took Antioch in
	# an ordinary border war, or off the Renovatio mission line, held it on the map with no
	# cc_byz_mem_seat_antioch, no add_pentarchy_modifier, and three mission tasks reading a
	# variable nothing would ever write.
	#
	# The unrecorded-seat test replaces both the war gate and the owns list: it is the same
	# ownership check the immediate below performs, hoisted, so the event cannot fire with nothing
	# to say. The situation's on_monthly evaluates it every month and fires this when it passes.
	trigger = {
		has_societal_value = societal_value_type:latinization_vs_hellenization
		cc_byz_mamluk_unrecorded_seat_trigger = yes
	}

	illustration_tags = {
		10 = happy
		10 = interior
	}

	immediate = {
		# Writes a score owned by another thread's initialiser, so that one is called too.
		cc_byz_init_debt_scores_effect = yes
		cc_byz_init_levant_scores_effect = yes

		if = {
			limit = {
				owns = location:jerusalem
				NOT = { has_variable = cc_byz_mem_seat_jerusalem }
			}
			set_variable = { name = cc_byz_mem_seat_jerusalem  value = yes }
			add_pentarchy_modifier = { location = location:jerusalem }
		}
		if = {
			limit = {
				owns = location:antioch
				NOT = { has_variable = cc_byz_mem_seat_antioch }
			}
			set_variable = { name = cc_byz_mem_seat_antioch  value = yes }
			add_pentarchy_modifier = { location = location:antioch }
		}
		if = {
			limit = {
				owns = location:alexandria
				NOT = { has_variable = cc_byz_mem_seat_alexandria }
			}
			set_variable = { name = cc_byz_mem_seat_alexandria  value = yes }
			add_pentarchy_modifier = { location = location:alexandria }
		}

		situation:cc_byz_mamluk_confrontation ?= {
			change_variable = { name = cc_byz_levant_progress  add = 25 }
		}
	}

	option = {
		name = cc_byz_mamluk.12.a
		ai_chance = { base = 60 }

		add_prestige = prestige_severe_bonus
		change_societal_value = {
			type = latinization_vs_hellenization
			value = societal_value_move_to_right
		}
	}

	# Press on for the rest. There are five seats and the argument only works with all of them.
	option = {
		name = cc_byz_mamluk.12.b
		ai_chance = { base = 40 }
		trigger = {
			NOT = { has_all_pentarchy = yes }
		}

		change_variable = { name = cc_byz_levant_pressure  add = 1 }
		clamp_variable = { name = cc_byz_levant_pressure  min = -5  max = 5 }
		change_variable = { name = cc_byz_strain  add = 1 }
		clamp_variable = { name = cc_byz_strain  min = 0  max = 10 }
		# The ten-year re-fire is gone with the six-year one in .11. Whichever seat is taken next,
		# the situation's monthly roll sees it that month.
	}

	after = {
		hidden_effect = {
			if = {
				limit = { has_all_pentarchy = yes }
				trigger_event_non_silently = { id = cc_byz_mamluk.13 }
			}
		}
	}
}
