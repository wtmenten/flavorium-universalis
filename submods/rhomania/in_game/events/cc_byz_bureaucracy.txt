namespace = cc_byz_bureaucracy

# The bureaucracy thread.
#
# Chained from cc_byz_bureaucracy.txt in common/on_action, which hangs off vanilla's empty
# on_bureaucracy_added / on_bureaucracy_removed hooks plus the yearly pulse.
#
# Running score: cc_byz_bureau_venality, clamped 0..5. It rises whenever an office is
# treated as property rather than a post, and it gates the later events in the thread.
# Nothing resets it cheaply, which is the point: venality is easy to permit and expensive
# to undo.
#
# Memory variables set here and read elsewhere in the thread:
#   cc_byz_mem_sold_offices     an office was sold outright
#   cc_byz_mem_purged_bureau    a bureau's staff were purged rather than reassigned
#   cc_byz_mem_eunuchs_backed   the palace service was backed against the army
#   cc_byz_mem_army_backed      the army was backed against the palace service
#
# CHARACTER STAT RANGES. adm, dip and mil are each 0..100 (MAX_ABILITY_VALUE = 100 in
# defines). total_abilities is their SUM and therefore runs 0..300, which is why vanilla
# compares it against 250 for an exceptional ruler and 150 for a poor one. Do not carry a
# total_abilities-sized number over to a single stat: a gate of mil >= 150 can never be
# satisfied and the option silently never appears.
#
# For calibration, the highest single-stat comparison anywhere in vanilla is 94. Roughly:
#   >= 65   above average
#   >= 75   genuinely capable
#   >= 85   exceptional


#############################################
# .1  A new bureau is founded. Who staffs it?
#############################################
cc_byz_bureaucracy.1 = {
	type = country_event
	title = cc_byz_bureaucracy.1.title
	desc = {
		first_valid = {
			# Ordered before the venality branch: a service that was purged inside living
			# memory is the more specific situation, and it is the reason there are posts
			# to fill at all.
			triggered_desc = {
				trigger = { has_variable = cc_byz_mem_purged_bureau }
				desc = cc_byz_bureaucracy.1.desc_purged
			}
			triggered_desc = {
				trigger = { var:cc_byz_bureau_venality >= 3 }
				desc = cc_byz_bureaucracy.1.desc_venal
			}
			desc = cc_byz_bureaucracy.1.desc
		}
	}
	outcome = neutral

	trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		exists = scope:bureaucracy_type
	}

	illustration_tags = {
		10 = interior
		10 = professional
	}

	immediate = {
		cc_byz_init_bureau_scores_effect = yes
	}

	# Career clerks. Slow, cheap, and the office outlives the man.
	option = {
		name = cc_byz_bureaucracy.1.a
		ai_chance = { base = 40 }

		add_country_modifier = {
			modifier = cc_byz_staffed_by_career_clerks
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_minor_move_to_left
		}
	}

	# Magnates' sons. Free, and every post is now owed to a family.
	option = {
		name = cc_byz_bureaucracy.1.b
		ai_chance = { base = 30 }

		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_bonus
		}
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_minor_move_to_right
		}
		change_variable = { name = cc_byz_bureau_venality  add = 1 }
		clamp_variable = { name = cc_byz_bureau_venality  min = 0  max = 5 }
	}

	# Sell the posts outright. Only worth considering once the practice is already normal,
	# which is what the venality gate represents.
	option = {
		name = cc_byz_bureaucracy.1.c
		ai_chance = { base = 20 }
		trigger = {
			var:cc_byz_bureau_venality >= 2
		}

		add_gold = 150
		set_variable = { name = cc_byz_mem_sold_offices  value = yes }
		change_variable = { name = cc_byz_bureau_venality  add = 2 }
		clamp_variable = { name = cc_byz_bureau_venality  min = 0  max = 5 }
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_move_to_right
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_penalty
		}
	}

	after = {
		# Whoever staffed it, the men already in post read the appointment as a precedent
		# and act on it before anyone asks them. Not offered as a choice.
		hidden_effect = {
			if = {
				limit = { var:cc_byz_bureau_venality >= 4 }
				add_country_modifier = {
					modifier = cc_byz_offices_are_property
					years = 25
					mode = add_and_extend
				}
			}
		}
	}
}

	}

			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_minor_move_to_left

	}

	option = {
		name = cc_byz_bureaucracy.10.b
		ai_chance = { base = 30 }

			type = estate_type:burghers_estate

		change_variable = { name = cc_byz_bureau_venality  add = 1 }
		clamp_variable = { name = cc_byz_bureau_venality  min = 0  max = 5 }

		name = cc_byz_bureaucracy.10.c
		ai_chance = { base = 10 }

			ruler ?= { mil >= 70 }
		}

			value = estate_satisfaction_mild_penalty

		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_move_to_left
		}
	}
}


#############################################
# .20  Underfunded bureaus and the quiet sale of offices
#############################################
cc_byz_bureaucracy.20 = {
	type = country_event
	title = cc_byz_bureaucracy.20.title
	desc = cc_byz_bureaucracy.20.desc
	outcome = negative

	trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		any_current_bureaucracy = {
			bureaucracy_maintenance < 0.5
			count > 0
		}
	}

	illustration_tags = {
		10 = interior
		10 = economy
	}

	immediate = {
		cc_byz_init_bureau_scores_effect = yes
	}

	# Fund them properly. THE ONLY THING IN THE THREAD THAT REDUCES VENALITY.
	#
	# Venality was a ratchet: eight ways up and none down, which made the score a countdown
	# rather than a decision. Paying the arrears in full is the one act that genuinely undoes
	# it, and it is priced accordingly.
	option = {
		name = cc_byz_bureaucracy.20.a
		ai_chance = { base = 40 }

		change_gold_effect = { scale = -5 }
		add_country_modifier = {
			modifier = cc_byz_bureaus_paid_in_full
			years = 25
			mode = add_and_extend
		}
		change_variable = { name = cc_byz_bureau_venality  add = -1 }
		clamp_variable = { name = cc_byz_bureau_venality  min = 0  max = 5 }
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_move_to_left
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_bonus
		}
	}

	# Let them find their own income. Everyone understands what this means.
	#
	# THE FOLLOW-UP IS THE POINT. In twelve years the fees will be the custom, and repealing a
	# custom in this empire is a different problem from repealing a policy. The player will
	# have forgotten choosing this.
	option = {
		name = cc_byz_bureaucracy.20.b
		ai_chance = { base = 60 }

		change_variable = { name = cc_byz_bureau_venality  add = 1 }
		clamp_variable = { name = cc_byz_bureau_venality  min = 0  max = 5 }
		add_country_modifier = {
			modifier = cc_byz_fees_became_custom
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_move_to_right
		}
		add_estate_satisfaction = {
			type = estate_type:peasants_estate
			value = estate_satisfaction_severe_penalty
		}
		trigger_event_silently = { id = cc_byz_bureaucracy.21  years = 12 }
	}

	# Available to a capable administrator: retrench without conceding the principle.
	option = {
		name = cc_byz_bureaucracy.20.c
		ai_chance = { base = 30 }
		trigger = {
			ruler ?= { adm >= 75 }
		}

		change_gold_effect = { scale = -2 }
		add_country_modifier = {
			modifier = cc_byz_retrenched_bureaus
			years = 30
			mode = add_and_extend
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_penalty
		}
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_minor_move_to_left
		}
	}
}

#############################################
# Funding x entrenchment
#
# Two independent numbers describe every bureau, and the interesting content is in how they
# combine rather than in either alone:
#
#                  low entrenchment          high entrenchment
#   funded         a new office               a pillar of the state   (.50)
#   unfunded       a paper office             a FIEFDOM               (.40)
#
# Vanilla names the dangerous cell itself, in china.txt's negative_modifier comment:
# "entrenched ministries without funding become fiefdoms of powerful officials".
#
# Scales, verified against vanilla usage:
#   bureaucracy_maintenance   0..1    (vanilla compares >= 0.25)
#   entrenchment              0..100  (hire_advisor.txt:74 steps 10..60)
#
# SIGN WARNING on change_entrenchment. The named values invert: entrenchment_severe_penalty
# is +5 and entrenchment_extreme_bonus is -10. High entrenchment is treated as the penalty,
# because it means a bureau that cannot easily be dislodged. So REDUCING entrenchment uses a
# _bonus name. Read those two option blocks carefully before editing them.
#############################################


#############################################
# .40  Entrenched and unfunded: the fiefdom
#############################################
cc_byz_bureaucracy.40 = {
	type = country_event
	title = cc_byz_bureaucracy.40.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:cc_byz_bureau_venality >= 3 }
				desc = cc_byz_bureaucracy.40.desc_venal
			}
			desc = cc_byz_bureaucracy.40.desc
		}
	}
	outcome = negative

	trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		any_current_bureaucracy = {
			entrenchment >= 60
			bureaucracy_maintenance <= 0.25
		}
	}

	illustration_tags = {
		10 = interior
		10 = angry
	}

	immediate = {
		cc_byz_init_bureau_scores_effect = yes

		random_current_bureaucracy = {
			limit = {
				entrenchment >= 60
				bureaucracy_maintenance <= 0.25
			}
			save_scope_as = cc_byz_fief_bureau
		}
	}

	# Pay it properly and take it back. Expensive, and the only option that restores the
	# principle rather than trading it away.
	option = {
		name = cc_byz_bureaucracy.40.a
		ai_chance = { base = 35 }

		change_gold_effect = { scale = -6 }
		scope:cc_byz_fief_bureau ?= {
			set_maintenance = 1
		}
		add_country_modifier = {
			modifier = cc_byz_bureaus_paid_in_full
			years = 30
			mode = add_and_extend
		}
		change_variable = { name = cc_byz_bureau_venality  add = -1 }
		clamp_variable = { name = cc_byz_bureau_venality  min = 0  max = 5 }
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_move_to_left
		}
	}

	# Break it up. Entrenchment falls, which is what the _bonus naming means here.
	# Cheap in gold and expensive in everything else.
	option = {
		name = cc_byz_bureaucracy.40.b
		ai_chance = { base = 25 }
		trigger = {
			ruler ?= { mil >= 70 }
		}

		scope:cc_byz_fief_bureau ?= {
			change_entrenchment = entrenchment_very_extreme_bonus
		}
		add_stability = stability_mild_penalty
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_penalty
		}
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_minor_move_to_left
		}
	}

	# Recognise what has already happened. The office becomes a family's property in law as
	# well as in fact, and the treasury is paid for the privilege.
	option = {
		name = cc_byz_bureaucracy.40.c
		ai_chance = { base = 40 }

		add_gold = 250
		set_variable = { name = cc_byz_mem_fiefdom_formalised  value = yes }
		change_variable = { name = cc_byz_bureau_venality  add = 2 }
		clamp_variable = { name = cc_byz_bureau_venality  min = 0  max = 5 }
		# PERMANENT. Selling the office outright is the one decision in this thread that
		# cannot be walked back: no later event unwinds a sale.
		add_country_modifier = {
			modifier = cc_byz_office_farmed_out
			years = -1
			mode = add
		}
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_bonus
		}
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_move_to_right
		}
	}

	after = {
		# Whatever was decided, the other entrenched offices were watching to see whether
		# the palace still enforces anything. Not offered as a choice.
		hidden_effect = {
			if = {
				limit = { has_variable = cc_byz_mem_fiefdom_formalised }
				every_current_bureaucracy = {
					limit = { bureaucracy_maintenance <= 0.25 }
					change_entrenchment = entrenchment_mild_penalty
				}
			}
		}
	}
}

	}

	# An administrator can see the risk: an office this secure answers to itself.
	option = {

#############################################
# .60  A funding cut lands
#
# Fired from on_maintenance_changed, which is declared per bureaucracy and gives
# scope:old_maintenance and scope:new_maintenance.
#
#
# FIVE YEAR COOLDOWN, AND IT LIVES HERE RATHER THAN IN THE HOOKS.
#
# on_maintenance_changed is declared on THIRTEEN bureaucracies: our three, plus the eleven
# vanilla Byzantine ones the submod REPLACE:s. Every one of them fires this event whenever its
# funding drops to a quarter or less. An emperor economising across the board in a bad year
# therefore triggers the same dialogue up to thirteen times in a row, and an emperor who
# oscillates funding triggers it forever.
#
# The guard is on the event's own trigger because a directly-fired event still evaluates it.
# Vanilla relies on exactly this: d008_flavor_byz.2 and .3 are fired by trigger_event and carry
# real trigger blocks that gate them. So one guard here covers all thirteen hooks, where
# thirteen copies in the bureaucracy files would be thirteen chances to miss one.
#
# cc_byz_bureau_cut_cd is a TIMED variable and expires on its own after five years. No pulse
# has to clear it and nothing can leak it, which is the same pattern the main mod uses for
# cc_bond_aor_cd (cc_bond_pulse.txt:70).
#############################################
cc_byz_bureaucracy.60 = {
	type = country_event
	title = cc_byz_bureaucracy.60.title
	desc = cc_byz_bureaucracy.60.desc
	outcome = negative

	trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		NOT = { has_variable = cc_byz_bureau_cut_cd }
	}

	illustration_tags = {
		10 = interior
		10 = economy
	}

	immediate = {
		cc_byz_init_bureau_scores_effect = yes
		set_variable = {
			name = cc_byz_bureau_cut_cd
			value = yes
			years = 5
		}
	}

	# Accept the consequences quietly. They arrive anyway, ten years later, as .61.
	option = {
		name = cc_byz_bureaucracy.60.a
		ai_chance = { base = 60 }

		change_variable = { name = cc_byz_bureau_venality  add = 1 }
		clamp_variable = { name = cc_byz_bureau_venality  min = 0  max = 5 }
		add_country_modifier = {
			modifier = cc_byz_fees_became_custom
			years = 20
			mode = add_and_extend
		}
		trigger_event_silently = { id = cc_byz_bureaucracy.61  years = 10 }
	}

	# Announce it as a reform so the cut is at least seen as a decision. A cut that is
	# explained is still a cut, but the men taking it keep their standing, which is most of
	# what stops an office quietly finding its own income.
	option = {
		name = cc_byz_bureaucracy.60.b
		ai_chance = { base = 40 }
		trigger = {
			ruler ?= { dip >= 65 }
		}

		add_prestige = prestige_mild_bonus
		add_country_modifier = {
			modifier = cc_byz_retrenched_bureaus
			years = 20
			mode = add_and_extend
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_penalty
		}
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_minor_move_to_left
		}
	}

	# Restore the funding and admit the cut was a mistake. Costs prestige and gold, and is the
	# only option here that leaves the service intact.
	option = {
		name = cc_byz_bureaucracy.60.c
		ai_chance = { base = 20 }
		trigger = {
			gold >= 300
		}

		change_gold_effect = { scale = -4 }
		add_prestige = prestige_weak_penalty
		add_country_modifier = {
			modifier = cc_byz_bureaus_paid_in_full
			years = 20
			mode = add_and_extend
		}
	}
}


#############################################
# .61  What the cut actually cost
#############################################
#
# Ten years after the funding was cut and nothing was said. The office still exists and still
# answers letters; what it no longer does is answer them the same way regardless of who is
# asking, and that change happened without a single decision being recorded anywhere.
cc_byz_bureaucracy.61 = {
	type = country_event
	title = cc_byz_bureaucracy.61.title
	desc = cc_byz_bureaucracy.61.desc
	outcome = negative

	trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
	}

	illustration_tags = {
		10 = interior
		10 = economy
	}

	immediate = {
		cc_byz_init_bureau_scores_effect = yes
	}

	# Rebuild it. Years of work, and the men who left are not coming back.
	option = {
		name = cc_byz_bureaucracy.61.a
		ai_chance = { base = 35 }

		change_gold_effect = { scale = -5 }
		add_country_modifier = {
			modifier = cc_byz_bureaus_paid_in_full
			years = 25
			mode = add_and_extend
		}
		change_variable = { name = cc_byz_bureau_venality  add = -1 }
		clamp_variable = { name = cc_byz_bureau_venality  min = 0  max = 5 }
	}

	# Let it stand. It is cheaper than it was, and it is not ours.
	option = {
		name = cc_byz_bureaucracy.61.b
		ai_chance = { base = 65 }

		change_variable = { name = cc_byz_bureau_venality  add = 1 }
		clamp_variable = { name = cc_byz_bureau_venality  min = 0  max = 5 }
		add_country_modifier = {
			modifier = cc_byz_office_farmed_out
			years = -1
			mode = add
		}
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_move_to_right
		}
	}
}
