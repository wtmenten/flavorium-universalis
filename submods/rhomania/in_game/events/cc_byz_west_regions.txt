namespace = cc_byz_prov

# The three new prefectures, each failing in its own way.
#
#   .1  .2  .3     AFRICA     the Moorish frontier
#   .10 .11 .12    SPANIA     the narrow province
#   .20 .21 .22    GAUL       the Rhone command
#
# Italy is not here. It already has two full threads (cc_byz_papal.txt and cc_byz_italy.txt)
# plus the shared friction and demand arcs, and adding a third would make the oldest prefecture
# the one with the most events for no reason other than that it was written first.
#
#
# WHAT THESE ARE FOR. The shared threads in cc_byz_exarch_friction.txt and
# cc_byz_exarch_demands.txt now run region-neutral, so an African exarch gets the same first
# independent decision, the same recall summons and the same unauthorised succession as an
# Italian one. That is correct: those are things governors do everywhere. What it cannot give
# is why Africa is a different problem from Italy, and these are that.
#
# Each is three events, gated on its own region through the cc_byz_exarch_in_*_trigger scripted
# triggers, and each ends somewhere the other two cannot reach.
#
#
# HISTORICAL POSITION.
#
# Only Ravenna and Carthage were ever exarchates. Spania was governed by a magister militum and
# never had one; Gaul was never recovered by the eastern empire at all. The PREFECTURES of
# Italy, Africa and Gaul, and the Spanish diocese under Gaul, are real late-Roman administrative
# units, which is why the four countries are named for those rather than for exarchates that
# mostly did not exist. The Gallic thread in particular is counterfactual and is written as one:
# it is about a Provencal aristocracy that never stopped being Roman being told it has been
# recovered.
#
# Memory variables, all set-and-read:
#   cc_byz_mem_africa_limes / _client / _coast
#   cc_byz_mem_spania_traded / _station / _latin
#   cc_byz_mem_gaul_march / _protectorate / _absorbed
#
# CHARACTER STAT RANGES. adm, dip and mil are each 0..100; total_abilities is their sum, 0..300.


#############################################
# AFRICA .1  The limes
#############################################
cc_byz_prov.1 = {
	type = country_event
	title = cc_byz_prov.1.title
	desc = cc_byz_prov.1.desc
	outcome = negative

	fire_only_once = yes

	trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		any_subject = { cc_byz_exarch_in_africa_trigger = yes }
	}

	illustration_tags = {
		10 = exterior
		10 = military
	}

	immediate = {
		cc_byz_init_debt_scores_effect = yes
		random_subject = {
			limit = { cc_byz_exarch_in_africa_trigger = yes }
			save_scope_as = cc_byz_africa_exarch
		}
	}

	# Pay the frontier. What Rome did, for as long as Rome could afford it.
	option = {
		name = cc_byz_prov.1.a
		ai_chance = { base = 40 }

		change_gold_effect = { scale = -4 }
		set_variable = { name = cc_byz_mem_africa_limes  value = yes }
		scope:cc_byz_africa_exarch ?= {
			change_variable = { name = cc_byz_exarch_loyalty  add = 1 }
			clamp_variable = { name = cc_byz_exarch_loyalty  min = -5  max = 5 }
		}
		trigger_event_silently = { id = cc_byz_prov.2  years = 6 }
	}

	# Buy a confederation instead of fighting it. Cheaper, and it makes somebody else's
	# authority the thing holding the frontier.
	option = {
		name = cc_byz_prov.1.b
		ai_chance = { base = 35 }
		trigger = {
			ruler ?= { dip >= 65 }
		}

		change_gold_effect = { scale = -2 }
		set_variable = { name = cc_byz_mem_africa_client  value = yes }
		save_scope_value_as = { name = cc_byz_west_delta  value = 5 }
		cc_byz_west_autonomy_up_effect = yes
		trigger_event_silently = { id = cc_byz_prov.2  years = 6 }
	}

	# Hold the coast and let the interior go. Honest, and it is what happened.
	option = {
		name = cc_byz_prov.1.c
		ai_chance = { base = 25 }

		set_variable = { name = cc_byz_mem_africa_coast  value = yes }
		change_variable = { name = cc_byz_strain  add = -1 }
		clamp_variable = { name = cc_byz_strain  min = 0  max = 10 }
		save_scope_value_as = { name = cc_byz_west_delta  value = 4 }
		cc_byz_west_grip_up_effect = yes
		trigger_event_silently = { id = cc_byz_prov.2  years = 6 }
	}
}

			}

		}
	}
	outcome = negative

	trigger = {
		any_subject = { cc_byz_exarch_in_africa_trigger = yes }
	}

	illustration_tags = {
		10 = armed
		10 = military
	}

	immediate = {
		cc_byz_init_debt_scores_effect = yes
		random_subject = {
			limit = { cc_byz_exarch_in_africa_trigger = yes }
			save_scope_as = cc_byz_africa_exarch
		}

		# THE HIDDEN OUTCOME. Whether the garrison waits is not the emperor's decision, and
		# an emperor who paid the limes in .1 has bought himself better odds here without
		# being told that was what he was buying.
		random_list = {
			50 = {
				modifier = {
					factor = 0.3
					has_variable = cc_byz_mem_africa_limes
				}
				modifier = {
					factor = 2
					is_during_bankruptcy = yes
				}
				set_variable = { name = cc_byz_mem_africa_mutiny  value = yes }
				change_variable = { name = cc_byz_strain  add = 1 }
				clamp_variable = { name = cc_byz_strain  min = 0  max = 10 }
			}
			50 = { }
		}
	}

	option = {
		name = cc_byz_prov.3.a
		trigger = { has_variable = cc_byz_mem_africa_mutiny }

		change_gold_effect = { scale = -5 }
		add_manpower = { value = monthly_manpower  multiply = -6 }
		scope:cc_byz_africa_exarch ?= {
			change_variable = { name = cc_byz_exarch_loyalty  add = -2 }
			clamp_variable = { name = cc_byz_exarch_loyalty  min = -5  max = 5 }
		}
		save_scope_value_as = { name = cc_byz_west_delta  value = 10 }
		cc_byz_west_autonomy_up_effect = yes
	}

	option = {
		name = cc_byz_prov.3.b
		trigger = { NOT = { has_variable = cc_byz_mem_africa_mutiny } }

		add_prestige = prestige_mild_bonus
		save_scope_value_as = { name = cc_byz_west_delta  value = 6 }
		cc_byz_west_grip_up_effect = yes
	}
}
