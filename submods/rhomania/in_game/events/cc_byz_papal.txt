namespace = cc_byz_papal

# An Orthodox emperor governing the see of Rome.
#
# The status question here is genuinely unresolvable and the thread does not resolve it. What
# it tracks is how the arrangement is managed, which is a different thing and is the only
# thing an emperor actually controls.
#
#   .1  .2  .3  .4      rite, the Pope's freedom, appointments, excommunication
#   .10                 THE PAP ANNEXATION GUARD, see below
#   .11                 substitute thread if PAP was erased anyway
#
# Score: cc_byz_papal_standing -5..5, shared with the exarch thread.
#
#
# THE GUARD AT .10, which is the single most important thing in this file.
#
# The Western Schism situation's can_start (common/situations/western_schism.txt:7) is:
#
#     current_date > 1360.1.1
#     current_date < 1402.1.1
#     country_exists = c:PAP
#     c:PAP = { has_regent = yes  regency_type = regency_type:papal_election_regency
#               num_cardinals > 0 }
#
# So annexing the Papal States before 1360 does not delay the Western Schism. It DELETES it,
# permanently and silently, from a campaign that otherwise has nothing to do with this submod.
# That is the largest way this content can damage a vanilla playthrough.
#
# Annexation DURING the schism is fine: western_schism_end_trigger includes
# not = { country_exists = c:PAP } and closes the situation cleanly.
#
# The guard does not forbid anything. It fires when PAP is nearly gone and offers subjugation
# as an explicit, mechanically better alternative, so a player who erases the papacy has
# chosen to rather than done it in passing. If they choose to anyway, .11 supplies a
# substitute thread so the 1360-1402 window is not simply empty, and the loc says plainly
# that the Western Schism will not occur in that timeline.
#
# Memory variables:
#   cc_byz_mem_latin_rite_kept     the Latin rite was left alone in the city's churches
#   cc_byz_mem_pope_is_guest       the Pope was treated as a guest with his own household
#   cc_byz_mem_pope_is_subject     the Pope was made a subject in fact
#   cc_byz_mem_pope_fled           the Pope left for a western protector
#   cc_byz_mem_excommunicated      the emperor or his exarch was excommunicated
#   cc_byz_mem_pap_warned          the annexation guard fired
#   cc_byz_mem_pap_erased          PAP was destroyed anyway
#
# CHARACTER STAT RANGES. adm, dip and mil are each 0..100; total_abilities is their sum,
# 0..300. Roughly: >= 65 above average, >= 75 capable, >= 85 exceptional.


#############################################
# .1  Which rite is celebrated
#############################################
cc_byz_papal.1 = {
	type = country_event
	title = cc_byz_papal.1.title
	desc = cc_byz_papal.1.desc
	outcome = neutral

	trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		any_subject = { has_variable = cc_byz_is_exarchate }
		NOT = { has_variable = cc_byz_mem_latin_rite_kept }
	}

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		cc_byz_init_exarch_scores_effect = yes
		random_subject = {
			limit = { has_variable = cc_byz_is_exarchate }
			save_scope_as = cc_byz_exarch
		}
	}

	# Leave the churches alone. The rite is not the emperor's business.
	option = {
		name = cc_byz_papal.1.a
		ai_chance = { base = 45 }

		set_variable = { name = cc_byz_mem_latin_rite_kept  value = yes }
		change_variable = { name = cc_byz_papal_standing  add = 2 }
		clamp_variable = { name = cc_byz_papal_standing  min = -5  max = 5 }
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_penalty
		}
	}

	# Establish Greek liturgy in the principal churches. The emperor's city, the emperor's rite.
	option = {
		name = cc_byz_papal.1.b
		ai_chance = { base = 30 }

		change_variable = { name = cc_byz_papal_standing  add = -2 }
		clamp_variable = { name = cc_byz_papal_standing  min = -5  max = 5 }
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_bonus
		}
		change_societal_value = {
			type = latinization_vs_hellenization
			value = societal_value_move_to_right
		}
		scope:cc_byz_exarch ?= {
			change_variable = { name = cc_byz_exarch_loyalty  add = -1 }
			clamp_variable = { name = cc_byz_exarch_loyalty  min = -5  max = 5 }
		}
	}

	# One Greek church, richly endowed, and nothing else touched. Requires a ruler who
	# understands that the gesture is the whole point.
	option = {
		name = cc_byz_papal.1.c
		ai_chance = { base = 25 }
		trigger = {
			ruler ?= { dip >= 70 }
		}

		set_variable = { name = cc_byz_mem_latin_rite_kept  value = yes }
		add_gold = -150
		change_variable = { name = cc_byz_papal_standing  add = 1 }
		clamp_variable = { name = cc_byz_papal_standing  min = -5  max = 5 }
		add_prestige = prestige_mild_bonus
	}

	after = {
		hidden_effect = {
			trigger_event_silently = { id = cc_byz_papal.2  years = 7 }
		}
	}
}


#############################################
# .2  Guest, subject or hostage
#############################################
cc_byz_papal.2 = {
	type = country_event
	title = cc_byz_papal.2.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:cc_byz_papal_standing <= -2 }
				desc = cc_byz_papal.2.desc_hostile
			}
			desc = cc_byz_papal.2.desc
		}
	}
	outcome = neutral

	trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		any_subject = { has_variable = cc_byz_is_exarchate }
		country_exists = c:PAP
		NOR = {
			has_variable = cc_byz_mem_pope_is_guest
			has_variable = cc_byz_mem_pope_is_subject
		}
	}

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		cc_byz_init_exarch_scores_effect = yes
	}

	# A guest, with his own household, his own guard and his own correspondence.
	option = {
		name = cc_byz_papal.2.a
		ai_chance = { base = 40 }

		set_variable = { name = cc_byz_mem_pope_is_guest  value = yes }
		change_variable = { name = cc_byz_papal_standing  add = 2 }
		clamp_variable = { name = cc_byz_papal_standing  min = -5  max = 5 }
		c:PAP ?= {
			add_opinion = {
				target = root
				modifier = cc_byz_papal_forbearance
			}
		}
	}

	# A subject, whose household is paid for and whose correspondence is read.
	option = {
		name = cc_byz_papal.2.b
		ai_chance = { base = 35 }

		set_variable = { name = cc_byz_mem_pope_is_subject  value = yes }
		change_variable = { name = cc_byz_papal_standing  add = -3 }
		clamp_variable = { name = cc_byz_papal_standing  min = -5  max = 5 }
		add_prestige = prestige_mild_bonus
		c:PAP ?= {
			add_opinion = {
				target = root
				modifier = cc_byz_papal_captivity
			}
		}
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_move_to_left
		}
	}

	# Leave the question formally open and let practice settle it. The historical Byzantine
	# answer, and it needs an administrator who can keep an ambiguity alive for a reign.
	option = {
		name = cc_byz_papal.2.c
		ai_chance = { base = 25 }
		trigger = {
			ruler ?= { adm >= 75 }
		}

		add_prestige = prestige_weak_bonus
		change_variable = { name = cc_byz_papal_standing  add = 1 }
		clamp_variable = { name = cc_byz_papal_standing  min = -5  max = 5 }
	}

	after = {
		hidden_effect = {
			trigger_event_silently = { id = cc_byz_papal.3  years = 8 }
		}
	}
}

# .3  Appointments made without imperial assent
#############################################
cc_byz_papal.3 = {
	type = country_event
	title = cc_byz_papal.3.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = cc_byz_mem_pope_is_subject }
				desc = cc_byz_papal.3.desc_subject
			}
			desc = cc_byz_papal.3.desc
		}
	}
	outcome = negative

	trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		any_subject = { has_variable = cc_byz_is_exarchate }
		country_exists = c:PAP
	}

	illustration_tags = {
		10 = interior
		10 = angry
	}

	immediate = {
		cc_byz_init_exarch_scores_effect = yes
		random_subject = {
			limit = { has_variable = cc_byz_is_exarchate }
			save_scope_as = cc_byz_exarch
		}
	}

	# Refuse the appointments and name your own men.
	option = {
		name = cc_byz_papal.3.a
		ai_chance = { base = 35 }

		change_variable = { name = cc_byz_papal_standing  add = -2 }
		clamp_variable = { name = cc_byz_papal_standing  min = -5  max = 5 }
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_move_to_left
		}
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_bonus
		}
	}

	# Accept them. A bishop is a bishop and the alternative is a quarrel in every diocese.
	option = {
		name = cc_byz_papal.3.b
		ai_chance = { base = 40 }

		change_variable = { name = cc_byz_papal_standing  add = 2 }
		clamp_variable = { name = cc_byz_papal_standing  min = -5  max = 5 }
		change_societal_value = {
			type = latinization_vs_hellenization
			value = societal_value_minor_move_to_left
		}
	}

	# Let the exarch settle it locally, which means letting him build a clientele in the
	# Italian church. Convenient now.
	option = {
		name = cc_byz_papal.3.c
		ai_chance = { base = 25 }

		scope:cc_byz_exarch ?= {
			change_variable = { name = cc_byz_exarch_loyalty  add = -2 }
			clamp_variable = { name = cc_byz_exarch_loyalty  min = -5  max = 5 }
		}
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_move_to_right
		}
	}

	after = {
		hidden_effect = {
			if = {
				limit = { var:cc_byz_papal_standing <= -3 }
				trigger_event_silently = { id = cc_byz_papal.4  years = 5 }
			}
		}
	}
}


#############################################
# .4  Excommunication
#############################################
cc_byz_papal.4 = {
	type = country_event
	title = cc_byz_papal.4.title
	desc = cc_byz_papal.4.desc
	outcome = negative

	trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		any_subject = { has_variable = cc_byz_is_exarchate }
		country_exists = c:PAP
		NOT = { has_variable = cc_byz_mem_excommunicated }
	}

	illustration_tags = {
		10 = interior
		10 = angry
	}

	immediate = {
		cc_byz_init_exarch_scores_effect = yes
		random_subject = {
			limit = { has_variable = cc_byz_is_exarchate }
			save_scope_as = cc_byz_exarch
		}
		set_variable = { name = cc_byz_mem_excommunicated  value = yes }
	}

	# Ignore it. An Orthodox emperor is not in the Pope's communion to begin with.
	option = {
		name = cc_byz_papal.4.a
		ai_chance = { base = 45 }

		change_societal_value = {
			type = latinization_vs_hellenization
			value = societal_value_move_to_right
		}
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_bonus
		}
		# The exarch, however, governs Catholics who take it seriously.
		scope:cc_byz_exarch ?= {
			change_variable = { name = cc_byz_exarch_loyalty  add = -1 }
			clamp_variable = { name = cc_byz_exarch_loyalty  min = -5  max = 5 }
		}
	}

	# Answer it: the exarch is removed and a more acceptable man appointed.
	option = {
		name = cc_byz_papal.4.b
		ai_chance = { base = 30 }

		change_variable = { name = cc_byz_papal_standing  add = 3 }
		clamp_variable = { name = cc_byz_papal_standing  min = -5  max = 5 }
		add_prestige = prestige_mild_penalty
		scope:cc_byz_exarch ?= {
			change_variable = { name = cc_byz_exarch_loyalty  add = -2 }
			clamp_variable = { name = cc_byz_exarch_loyalty  min = -5  max = 5 }
		}
	}

	# Have it lifted through the union negotiations rather than through submission. Only
	# available to an emperor who has a legation in Rome already.
	option = {
		name = cc_byz_papal.4.c
		ai_chance = { base = 25 }
		trigger = {
			has_variable = cc_byz_union_stage
			var:cc_byz_union_stage >= 1
			ruler ?= { dip >= 75 }
		}

		change_variable = { name = cc_byz_papal_standing  add = 2 }
		clamp_variable = { name = cc_byz_papal_standing  min = -5  max = 5 }
		remove_variable = cc_byz_mem_excommunicated
		change_societal_value = {
			type = latinization_vs_hellenization
			value = societal_value_minor_move_to_left
		}
	}

	after = {
		# Not offered. Whatever was decided, somebody in the Roman clergy has begun writing
		# to the north Italians about the state of the city.
		hidden_effect = {
			cc_byz_init_debt_scores_effect = yes
			change_variable = { name = cc_byz_italian_standing  add = -1 }
			clamp_variable = { name = cc_byz_italian_standing  min = -5  max = 5 }
		}
	}
}


#############################################
# .10  THE GUARD
#
# Fires when the Papal States are one step from being erased and the Western Schism has not
# started yet. See the file header for why this matters more than anything else here.
#
# Deliberately NOT a block on the player: it offers the alternative and makes it the better
# deal. The AI is given a strong preference for it too, so an AI Byzantium does not delete a
# major vanilla situation in a game the player is watching from elsewhere.
#############################################
cc_byz_papal.10 = {
	type = country_event
	title = cc_byz_papal.10.title
	desc = cc_byz_papal.10.desc
	outcome = neutral

	trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		country_exists = c:PAP
		current_date < 1360.1.1
		NOT = { has_variable = cc_byz_mem_pap_warned }
		# One step from gone: the papacy is down to a rump and we are the reason.
		c:PAP = {
			num_locations_owned_or_owned_by_subjects <= 2
			is_at_war_with = root
		}
	}

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		cc_byz_init_exarch_scores_effect = yes
		set_variable = { name = cc_byz_mem_pap_warned  value = yes }
	}

	# Leave him his city and take everything else. The papacy survives as a client.
	option = {
		name = cc_byz_papal.10.a
		ai_chance = { base = 70 }

		change_variable = { name = cc_byz_papal_standing  add = 2 }
		clamp_variable = { name = cc_byz_papal_standing  min = -5  max = 5 }
		add_prestige = prestige_mild_bonus
		add_country_modifier = {
			modifier = cc_byz_protector_of_the_see
			years = -1
			mode = add
		}
		c:PAP ?= {
			add_opinion = {
				target = root
				modifier = cc_byz_papal_forbearance
			}
		}
	}

	# Take the city and leave the office. He keeps his cardinals and his title and nothing else.
	option = {
		name = cc_byz_papal.10.b
		ai_chance = { base = 25 }

		change_variable = { name = cc_byz_papal_standing  add = -2 }
		clamp_variable = { name = cc_byz_papal_standing  min = -5  max = 5 }
		add_prestige = prestige_mild_bonus
		change_societal_value = {
			type = cc_taxis_vs_cc_dynatoi
			value = societal_value_move_to_left
		}
	}

	# Finish it. The emperor's decision, and the loc says what it costs.
	option = {
		name = cc_byz_papal.10.c
		ai_chance = { base = 5 }

		change_variable = { name = cc_byz_papal_standing  add = -4 }
		clamp_variable = { name = cc_byz_papal_standing  min = -5  max = 5 }
		add_prestige = prestige_severe_bonus
		change_societal_value = {
			type = latinization_vs_hellenization
			value = societal_value_move_to_right
		}
	}
}


#############################################
# .11  A papacy that no longer exists
#
# The substitute thread. If PAP was destroyed before 1360 the Western Schism will never fire,
# so the 1360-1402 window would otherwise be simply empty for a player who took Rome early.
# This fills it with the consequence that actually follows: a Catholic west with no pope in
# it, arguing about who gets to appoint one.
#############################################
cc_byz_papal.11 = {
	type = country_event
	title = cc_byz_papal.11.title
	desc = cc_byz_papal.11.desc
	outcome = neutral

	trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		NOT = { country_exists = c:PAP }
		current_date > 1360.1.1
		NOT = { has_variable = cc_byz_mem_pap_erased }
	}

	illustration_tags = {
		10 = interior
		10 = angry
	}

	immediate = {
		cc_byz_init_exarch_scores_effect = yes
		cc_byz_init_debt_scores_effect = yes
		set_variable = { name = cc_byz_mem_pap_erased  value = yes }
	}

	# Let the western powers argue it out among themselves.
	option = {
		name = cc_byz_papal.11.a
		ai_chance = { base = 45 }

		change_variable = { name = cc_byz_italian_standing  add = -1 }
		clamp_variable = { name = cc_byz_italian_standing  min = -5  max = 5 }
		change_societal_value = {
			type = latinization_vs_hellenization
			value = societal_value_move_to_right
		}
	}

	# Convene the western bishops ourselves. An emperor summoning a council to elect a pope
	# is not a novelty; it is simply one nobody has attempted for eight hundred years.
	option = {
		name = cc_byz_papal.11.b
		ai_chance = { base = 35 }

		add_gold = -250
		add_prestige = prestige_severe_bonus
		change_variable = { name = cc_byz_papal_standing  add = 2 }
		clamp_variable = { name = cc_byz_papal_standing  min = -5  max = 5 }
		change_societal_value = {
			type = latinization_vs_hellenization
			value = societal_value_move_to_left
		}
	}

	# Declare the Roman see vacant and its rights returned to the remaining patriarchs.
	# The maximal Orthodox position and it needs a ruler who can carry the argument.
	option = {
		name = cc_byz_papal.11.c
		ai_chance = { base = 20 }
		trigger = {
			ruler ?= { adm >= 80 }
		}

		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_bonus
		}
		change_societal_value = {
			type = latinization_vs_hellenization
			value = societal_value_large_move_to_right
		}
		every_country = {
			limit = {
				religion = religion:orthodox
				NOT = { this = root }
			}
			add_opinion = {
				target = root
				modifier = cc_byz_held_the_faith
			}
		}
	}
}
