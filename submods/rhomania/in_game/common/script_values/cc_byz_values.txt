# Counts consumed by cc_byz_taxis.txt auto_modifiers.
#
# auto_modifier scales_with takes a script value, not an inline iterator block.
# Accumulator pattern copied from vanilla byz_values.txt:5 (byz_romance_traders_count),
# which adds into root from inside the iterator.

# How many pronoia grants are outstanding. Each one is land and its soldiers
# passed out of the fisc and into a private hand.
cc_byz_pronoia_count = {
	every_subject = {
		limit = { is_subject_type = pronoia }
		root = { add = 1 }
	}
	min = 0
	max = 10
}

# How many bureaus are actually being funded, not merely established.
cc_byz_funded_bureau_count = {
	every_current_bureaucracy = {
		limit = { bureaucracy_maintenance > 0 }
		root = { add = 1 }
	}
	min = 0
	max = 8
}

# Tagmata cap. Deliberately the same shape as vanilla's legionaries_unit_limit
# (default_values.txt:595), but scaling off the Greek culture group the player already has
# rather than roman_culture, which only spreads via the Latin-gated roman_culture_movement.
cc_byz_tagmata_unit_limit = {
	add = {
		desc = "SUBUNIT_LIMIT_BASE"
		value = 4
	}
	add = {
		desc = "GREEK_CULTURE_PERCENTAGE"
		value = {
			add = "culture_group_percentage_in_country(culture_group:greek_group)"
			multiply = 20
		}
	}
	multiply = {
		desc = "[ShowModifierTypeName('cc_num_of_tagmata_modifier')]"
		value = {
			add = 1
			add = modifier:cc_num_of_tagmata_modifier
		}
	}
}

# Deeply entrenched bureaus, funded and unfunded. Consumed by the entrenchment tier in
# cc_byz_bureaucracy_overlay.txt. Entrenchment is 0..100 (hire_advisor.txt:74 steps 10..60),
# maintenance is 0..1 (vanilla compares >= 0.25).
#
# 60 is the threshold the .40 and .50 events use for "entrenched"; keep the three in sync.
cc_byz_entrenched_funded_count = {
	every_current_bureaucracy = {
		limit = {
			entrenchment >= 60
			bureaucracy_maintenance >= 0.5
		}
		root = { add = 1 }
	}
	min = 0
	max = 6
}

cc_byz_entrenched_unfunded_count = {
	every_current_bureaucracy = {
		limit = {
			entrenchment >= 60
			bureaucracy_maintenance < 0.5
		}
		root = { add = 1 }
	}
	min = 0
	max = 6
}


#############################################
# Consistory voting bias
#############################################
#
# WHAT THESE FIX. The four western laws shipped with flat constants in their wants_* blocks, so
# every prefect voted identically for autonomy by a fixed margin regardless of how he had been
# treated. cc_byz_exarch_loyalty did not enter the vote at all: it decided the recall roll and
# nothing else. A loyal prefect and a resentful one were the same voter.
#
# These two are added on top of each policy's own flat base, so the base still expresses the
# institutional interest (a prefect always prefers commanding his own troops) and these express
# how hard he is willing to push for it this session.
#
# SCOPE. Inside wants_this_policy_bias / wants_propose_policy / wants_keep_policy the root is
# the VOTING COUNTRY (laws/readme.txt:39-41), so root is the exarchate for a prefect's vote and
# Byzantium for the emperor's. Everything below is written from that scope, which is why the
# emperor branch tests is_leader_of_international_organization rather than a tag.
#
# EVERY VARIABLE READ IS GUARDED. These evaluate for any country the engine polls, including
# ones that have never had an exarchate, and an unguarded var: read in a script value is the
# error class in docs/scripting-gotchas.md.

# How hard this voter pushes for the WESTERN side of a question.
cc_byz_west_lean_value = {
	value = 0

	# The emperor is never on this side of it.
	if = {
		limit = {
			exists = international_organization:cc_byz_prefectures
			is_leader_of_international_organization = international_organization:cc_byz_prefectures
		}
		subtract = 40
	}

	# THE MAIN TERM. Loyalty runs -5..5, so this is a -30..+30 swing against a policy base of
	# 30 to 45: a well-treated prefect moderates almost to the point of voting with the
	# capital, and a mistreated one nearly doubles his push. This is what makes the donative,
	# the tribute and the census matter to the consistory rather than only to the recall.
	if = {
		limit = { has_variable = cc_byz_exarch_loyalty }
		subtract = {
			value = var:cc_byz_exarch_loyalty
			multiply = 6
		}
	}

	# A prefect whose son has already been confirmed thinks the post is his family's, and votes
	# accordingly whatever his personal feelings about the emperor.
	if = {
		limit = { overlord ?= { has_variable = cc_byz_mem_west_hereditary } }
		add = 10
	}

	# Somebody has already refused a summons and kept his province. Everyone west of the
	# Adriatic drew the same conclusion.
	if = {
		limit = { overlord ?= { has_variable = cc_byz_mem_west_defied } }
		add = 12
	}

	# A west that already runs itself argues from a stronger position.
	if = {
		limit = {
			is_situation_active = situation:cc_byz_the_west
			situation:cc_byz_the_west = {
				trigger_if = {
					limit = { has_variable = cc_byz_western_autonomy }
					var:cc_byz_western_autonomy >= 60
				}
				trigger_else = { always = no }
			}
		}
		add = 10
	}

	min = -50
	max = 50
}

# The mirror, for the IMPERIAL side of a question. Not simply the negation: the emperor's
# preference for his own prerogative is stronger and steadier than a prefect's opposition to
# it, which is what lets a determined emperor hold the line at two prefectures.
cc_byz_imperial_lean_value = {
	value = 0

	if = {
		limit = {
			exists = international_organization:cc_byz_prefectures
			is_leader_of_international_organization = international_organization:cc_byz_prefectures
		}
		add = 50
	}

	if = {
		limit = { has_variable = cc_byz_exarch_loyalty }
		add = {
			value = var:cc_byz_exarch_loyalty
			multiply = 6
		}
	}

	# Paid garrisons remember it in the chamber as well as in the field.
	if = {
		limit = { overlord ?= { has_country_modifier = cc_byz_west_donative_paid } }
		add = 8
	}

	# A summons that was obeyed is the clearest possible statement about where authority sits.
	if = {
		limit = { overlord ?= { has_country_modifier = cc_byz_west_recall_obeyed } }
		add = 10
	}

	if = {
		limit = {
			is_situation_active = situation:cc_byz_the_west
			situation:cc_byz_the_west = {
				trigger_if = {
					limit = { has_variable = cc_byz_imperial_grip }
					var:cc_byz_imperial_grip >= 60
				}
				trigger_else = { always = no }
			}
		}
		add = 10
	}

	min = -50
	max = 50
}


#############################################
# Bureau funding, sampled for the five-year average
#############################################
#
# WHAT THIS REPLACES. cc_byz_task_bureaus and cc_byz_task_logothete used to test
# any_current_bureaucracy with a bureaucracy_maintenance floor, which is an INSTANT test: fund
# everything for one month, let the task tick over, and defund immediately. The task recorded
# that the bureaus had been funded once, which is not what "the bureaus are funded" means.
#
# They now read a five-year running average maintained by cc_byz_bureau_sample_effect off the
# monthly pulse. The count below is half of the monthly sample; the other half, the total
# maintenance, is accumulated in the EFFECT rather than here.
#
# WHY THE SUM IS NOT A SCRIPT VALUE. It was, and it errored:
#
#     Event target link 'compare_value' returned an unset scope
#
# The accumulator idiom is root = { add = X }, and the moment you enter root you are in COUNTRY
# scope. bureaucracy_maintenance is a BUREAUCRACY-scope value, so reading it there asks the
# country for a property it does not have. The three accumulators in this file that work all
# add a constant 1, which is why none of them ever hit this.
#
# Vanilla's own summing accumulator (io_policy.txt:81) sidesteps it with a purpose-built
# prev-aware value. The effect does it by saving the number before changing scope, which needs
# no new script value at all. See scripted_effects/cc_byz_bureau_funding.txt.

# How many bureaucracies exist at all, funded or not. Distinct from
# cc_byz_funded_bureau_count above, which counts only those with maintenance above zero: an
# empire with four bureaus and one funded should sample 25 percent, not 100.
cc_byz_current_bureau_count = {
	every_current_bureaucracy = {
		root = { add = 1 }
	}
	min = 0
	max = 12
}


#############################################
# Redeeming a farmed-out office
#############################################
#
# WHAT SETS THE SCALE. country_tax_base is directly comparable to gold in this engine:
# vanilla's ask_for_money transfers exactly country_tax_base and its enabled block tests
# `gold >= country_tax_base`, so one unit of tax base is one large gift. Pricing the buy-back
# between 1.5 and 3.5 times that makes it a serious outlay at any realm size without ever
# becoming unpayable for a small one, which a flat number could not manage across a campaign
# that runs from a rump state to an empire.
#
# THE ENTRENCHMENT TERM IS THE POINT. An office sold last week and an office three generations
# of one family have been living off are the same modifier and should not be the same price.
# cc_byz_farmed_entrenchment is captured at SALE time by cc_byz_farm_out_office_effect, not
# read live, because the bureaucracy may not exist any more by the time the offer arrives.
#
#     entrenchment      0      25      50      75     100
#     multiplier      1.5     2.0     2.5     3.0     3.5
#
# cc_byz_bureaucracy.62 backfills the variable in its immediate before this is evaluated, so
# the read below is safe without a trigger_if.
cc_byz_office_buyback_cost = {
	value = country_tax_base
	multiply = {
		value = 1.5
		add = {
			value = var:cc_byz_farmed_entrenchment
			divide = 50
		}
	}
	add = 150

	# Floors and ceilings, because tax base is not bounded and this is a gold cost. The floor
	# stops a collapsed treasury buying an office back for pocket change; the ceiling stops a
	# late-game empire being quoted a number no realm has ever held.
	min = 250
	max = 8000
}


#############################################
# The imperial faith's share of a population
#############################################
#
# Read by cc_byz_faith_location_share_trigger and cc_byz_faith_country_share_trigger. See
# scripted_triggers/cc_byz_faith_triggers.txt for why the whole faith layer exists; briefly,
# the submod hard-coded religion:orthodox in 341 places and Fate of the Phoenix ships a fully
# supported route to hellenism_religion, which made six mission tasks impossible on conversion.
#
# BOTH FAITHS ADDED, NOT ALTERNATED. A city midway through a conversion is genuinely the
# emperor's in the way these objectives mean it. Requiring either faith to clear the bar on its
# own would turn the transition into a century-long hole in the mission tree, which is the
# opposite of the point.
#
# No clamp. The two are disjoint shares of the same population, so the sum cannot exceed 1.

# Location scope.
cc_byz_faith_share_location = {
	value = "religion_percentage(religion:orthodox)"
	add = "religion_percentage(religion:hellenism_religion)"
}

# Country scope.
cc_byz_faith_share_country = {
	value = "religion_percentage_in_country(religion:orthodox)"
	add = "religion_percentage_in_country(religion:hellenism_religion)"
}
