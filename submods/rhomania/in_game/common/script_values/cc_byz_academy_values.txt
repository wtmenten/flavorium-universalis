# The Academy chain (events/cc_byz_academy.txt). Country scope.

# Locations we own with a university. Counts locations, not building levels, because the task
# and the tactics reward speak of universities.
cc_byz_acad_university_count = {
	value = 0
	every_owned_location = {
		limit = { has_building = building_type:university }
		add = 1
	}
}

# Scale for the military tactics reward: 0.01 per university, capped at 20 (0.2 at full).
cc_byz_acad_tactics_scale = {
	value = cc_byz_acad_university_count
	max = 20
}
