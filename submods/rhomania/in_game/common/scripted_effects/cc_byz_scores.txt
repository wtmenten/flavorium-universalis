# Shared score variables, initialised in one place.
#
# Several pillars read the same three running scores, and the build order does not create
# them in dependency order: the debt threads (pillar 6b) are written before the exarchate
# event threads (pillar 2) and before strain (pillar 4d), yet all three touch these. Defining
# them here means a later pillar keys off a definition rather than retrofitting one, which is
# the failure mode that cost the main mod a whole review pass: variables written by one chain
# and never read by another, so the content looks deep and plays linear.
#
# Initialisation is LAZY. Every event that touches a score calls the matching init effect from
# its immediate block first. There is deliberately no on_game_start here: this is a submod, and
# both it and the main mod declaring on_game_start = { effect = {} } would silently leave only
# the last one loaded. The on_actions sub-list would merge safely, but lazy init needs no hook
# at all and cannot collide with anything.
#
#   cc_byz_italian_standing   -5..5   standing with the Italian maritime republics as a bloc.
#                                     Read by the debt threads, the Italian and papal threads
#                                     (pillar 2) and the spice/grain thread (pillar 5).
#
#   cc_byz_creditor_grip       0..5   how much of the fisc the creditors have a claim on.
#                                     Rises with concessions, not with the size of the debt:
#                                     an emperor can owe a great deal and concede nothing.
#
#   cc_byz_strain              0..10  overextension. Pillar 4d owns the pulse that drives this;
#                                     the debt threads only seed it, because borrowing to hold
#                                     ground is the mechanism by which reconquest becomes ruin.
#
# Ranges are enforced by clamp_variable at each call site rather than here, matching the
# existing cc_byz_bureau_venality pattern in the bureaucracy events.

cc_byz_init_debt_scores_effect = {
	if = {
		limit = { NOT = { has_variable = cc_byz_italian_standing } }
		set_variable = {
			name = cc_byz_italian_standing
			value = 0
		}
	}

	if = {
		limit = { NOT = { has_variable = cc_byz_creditor_grip } }
		set_variable = {
			name = cc_byz_creditor_grip
			value = 0
		}
	}

	if = {
		limit = { NOT = { has_variable = cc_byz_strain } }
		set_variable = {
			name = cc_byz_strain
			value = 0
		}
	}
}

# Union arc state.
#
#   cc_byz_union_stage      0..4   0 nothing, 1 contacts opened, 2 signed, 3 received at
#                                  home, 4 repudiated. Read by the four union actions'
#                                  potential blocks, so it must exist before any of them can
#                                  become visible: cc_byz_open_union_contacts therefore
#                                  accepts EITHER the variable being absent OR equal to 0,
#                                  and calls this effect before setting it.
#
#   cc_byz_anti_unionist    0..5   how hard the city is resisting. Rises with promulgation
#                                  and with Hellenic drift, falls when the union is
#                                  repudiated. Gates the ferment events and the AI's
#                                  willingness to renounce.
#
# Kept separate from the debt initialiser rather than merged into one effect: the union
# actions run this from a generic_action's effect block, where dragging in three unrelated
# debt variables would put them on every save that ever opened a legation.
# Bureaucracy thread state.
#
#   cc_byz_bureau_venality  0..5  how far offices have become property rather than posts.
#
# THIS EXISTS BECAUSE OF A REAL BUG. The score used to be initialised inline in the immediate
# of cc_byz_bureaucracy.1 only, on the assumption that .1 fires first because it hangs off
# on_bureaucracy_added. It does not always: .60 hangs off on_maintenance_changed, which fires
# whenever a bureau's funding moves, INCLUDING during game start before any bureau has been
# added. Its options then ran change_variable on an unset variable and errored.
#
# change_variable on an unset variable is a different failure from reading one in a trigger:
# it happens in an EFFECT, after the player has clicked, so there is no guard block to hide
# behind. The variable simply has to exist. Every event in the thread now calls this first.
cc_byz_init_bureau_scores_effect = {
	if = {
		limit = { NOT = { has_variable = cc_byz_bureau_venality } }
		set_variable = {
			name = cc_byz_bureau_venality
			value = 0
		}
	}
}

# Exarchate thread state, on the OVERLORD.
#
#   cc_byz_papal_standing  -5..5  the emperor's standing with the papacy as an institution,
#                                 distinct from the union arc's cc_byz_union_stage. A union
#                                 can be signed while this is deeply negative: the doctrinal
#                                 question and the question of who governs Rome are separate
#                                 quarrels and the threads treat them that way.
#
# NOT initialised here: cc_byz_exarch_loyalty and cc_byz_is_exarchate live on the SUBJECT,
# set by cc_byz_appoint_exarch and backfilled by the subject type's on_enable. The friction
# thread fires on the overlord and reaches them through a saved scope, so initialising them
# from the overlord's scope here would create a second, unrelated pair of variables on the
# wrong country.
#   cc_byz_exarch_appetite  0..5   how much more of Italy the exarch thinks should be his.
#                                  Owned by the demand thread and deliberately NOT the same
#                                  score as cc_byz_exarch_loyalty: an exarch can be perfectly
#                                  loyal and still want the Romagna, and a resentful one can
#                                  want nothing. Keeping them apart is also what stops a
#                                  recurring pulse thread from deciding the friction arc's
#                                  ending by sheer number of firings.
cc_byz_init_exarch_scores_effect = {
	if = {
		limit = { NOT = { has_variable = cc_byz_papal_standing } }
		set_variable = {
			name = cc_byz_papal_standing
			value = 0
		}
	}

	if = {
		limit = { NOT = { has_variable = cc_byz_exarch_appetite } }
		set_variable = {
			name = cc_byz_exarch_appetite
			value = 0
		}
	}
}

#############################################
# Who "the empire" is. Shared by both situations.
#############################################
#
# WHY THIS EXISTS. A Byzantine player who forms Rome (ROM_BYZ_f, or ROM_f once BYZ is gone)
# keeps the same country object and every variable, modifier and memory on it: has_or_had_tag
# = BYZ still passes, which is how the rest of this submod gates itself. What stops existing is
# the TAG. Both situations were written entirely in c:BYZ, and every one of those references
# resolved to nothing the moment the formable fired:
#
#   can_end had NOT = { country_exists = c:BYZ }, so forming Rome ENDED the situation on the
#   next evaluation, whatever state it was in
#   the ending hook ran through c:BYZ ?=, which then skipped, so it ended silently
#   can_start had the same test, so it could never start again either
#
# A scope variable does not have that problem. It is a handle on the country, and a country that
# renames itself is the same country. Vanilla stores a country on a situation exactly this way:
# rise_of_the_ottomans' strongest_beylik_variable, read back as
# country_exists = situation:rise_of_the_ottomans.var:strongest_beylik_variable
# (rise_of_the_ottomans.txt:22).
#
# Called from on_start and again from on_monthly. The second call is not redundant: it backfills
# a save whose situation began before this variable existed, and it repoints if the empire is
# annexed outright and the tag is later released to somebody else.
#
# ONE EFFECT FOR BOTH SITUATIONS rather than one each. It writes cc_byz_empire into whichever
# situation scope calls it, so the two copies would have been identical, and a resolver that
# exists twice is a resolver that gets fixed once.
#
# ROOT IS THE SITUATION.
cc_byz_resolve_empire_effect = {
	if = {
		# No has_variable guard, deliberately. exists is false both for "never set" and for
		# "set, but that country is dead", and those are the same case here.
		limit = { NOT = { exists = var:cc_byz_empire } }

		if = {
			limit = { exists = c:BYZ }
			c:BYZ = { save_scope_as = cc_byz_empire_found }
		}
		else = {
			# The City is the tiebreak. has_or_had_tag = BYZ on its own would also match a rump
			# Byzantium released by whoever conquered the empire, and both situations are about
			# the emperor rather than about the name.
			random_country = {
				limit = {
					has_or_had_tag = BYZ
					owns = location:constantinople
				}
				save_scope_as = cc_byz_empire_found
			}
		}

		# One write path for both branches, through a saved scope, which is the form vanilla uses
		# when it stores a country on a situation (golden_age_of_piracy.txt:103). If neither
		# branch found anybody the variable is simply left unset and every reader ?= past it;
		# can_end has its own separate test for the empire being gone.
		if = {
			limit = { exists = scope:cc_byz_empire_found }
			set_variable = { name = cc_byz_empire  value = scope:cc_byz_empire_found }
		}
	}
}


#############################################
# The three eastern seats, mirrored onto the situation for the panel
#############################################
#
# WHY THIS EXISTS. cc_byz_mamluk_confrontation.gui shows which of Jerusalem, Antioch and
# Alexandria the empire holds, because those three plus Constantinople and Rome are what
# has_all_pentarchy tests and therefore what the whole confrontation is for.
#
# The panel cannot read the map. The natural GUI form would be
# GetLocation('jerusalem').GetOwner.IsSameCountry(Player), and neither function exists: a grep
# across every vanilla .gui returns zero hits for GetLocation( and zero for IsSameCountry.
# GUI datafunctions are a different vocabulary from script triggers and there is no
# owns-this-location member among them.
#
# So script writes what the panel reads. Called from the situation's on_monthly, where root is
# the SITUATION, which is also where the variables need to live for the panel's
# GetActiveSituation.GetSituation.MakeScope idiom to find them.
#
# Flags rather than values, so the panel tests .IsSet and never reads an unset variable.
# Worst-case staleness is one month after a conquest, which is not worth more machinery.
#
# READS var:cc_byz_empire, NOT c:BYZ. See cc_byz_resolve_empire_effect above: with c:BYZ
# here, an emperor who formed Rome had all three seats blank out on the panel a month later,
# including seats he was standing in. The resolve effect is called immediately before this one in
# the situation's on_monthly, so the handle is current by the time this runs.
#
# TWO FLAGS PER SEAT, NOT ONE. The panel used to say "Held" or "Not held", and "not held" was
# doing too much work: it covered the sultan holding Jerusalem, a Latin kingdom holding it, and
# the emperor's own vassal holding it, which are three different positions in a confrontation
# whose entire subject is who holds these three places. A player looking at the panel could not
# tell whether the confrontation still had an object.
#
# The second flag is set only where Cairo holds the seat, so the panel reads ours / the sultan's /
# neither, and the three-way split matches the triggers the events gate on
# (cc_byz_mamluk_triggers.txt). Neither flag set means somebody else entirely.
cc_byz_mamluk_seat_sync_effect = {
	if = {
		limit = { var:cc_byz_empire ?= { owns = location:jerusalem } }
		set_variable = { name = cc_byz_seat_jerusalem  value = yes }
	}
	else = {
		remove_variable = cc_byz_seat_jerusalem
	}

	if = {
		limit = { var:cc_byz_empire ?= { owns = location:antioch } }
		set_variable = { name = cc_byz_seat_antioch  value = yes }
	}
	else = {
		remove_variable = cc_byz_seat_antioch
	}

	if = {
		limit = { var:cc_byz_empire ?= { owns = location:alexandria } }
		set_variable = { name = cc_byz_seat_alexandria  value = yes }
	}
	else = {
		remove_variable = cc_byz_seat_alexandria
	}

	if = {
		limit = { cc_byz_jerusalem_under_cairo_trigger = yes }
		set_variable = { name = cc_byz_seat_jerusalem_cairo  value = yes }
	}
	else = {
		remove_variable = cc_byz_seat_jerusalem_cairo
	}

	if = {
		limit = { cc_byz_antioch_under_cairo_trigger = yes }
		set_variable = { name = cc_byz_seat_antioch_cairo  value = yes }
	}
	else = {
		remove_variable = cc_byz_seat_antioch_cairo
	}

	if = {
		limit = { cc_byz_alexandria_under_cairo_trigger = yes }
		set_variable = { name = cc_byz_seat_alexandria_cairo  value = yes }
	}
	else = {
		remove_variable = cc_byz_seat_alexandria_cairo
	}
}


#############################################
# The Hellenic thread's state
#############################################
#
# See events/cc_byz_hellenic.txt. Initialised lazily like every other score in this file, so a
# save from before the thread existed backfills on first contact rather than erroring.
#
#   cc_byz_hellenic_stage      0..4  0 nothing, 1 edict, 2 temples, 3 reaction, 4 settled
#   cc_byz_christian_reaction  0..5  how hard the Church is pushing back
#
# BOTH ARE KEPT WHEN THE EMPIRE RETURNS TO ORTHODOXY rather than removed: cc_byz_hellenic.10
# resets them to 0 by hand on the way back, which is a different thing from deleting them. The
# same principle applies in the other direction to cc_byz_union_stage, which survives the
# conversion untouched so a returning empire resumes the union arc where it stopped.
cc_byz_init_hellenic_scores_effect = {
	if = {
		limit = { NOT = { has_variable = cc_byz_hellenic_stage } }
		set_variable = { name = cc_byz_hellenic_stage  value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = cc_byz_christian_reaction } }
		set_variable = { name = cc_byz_christian_reaction  value = 0 }
	}
}
