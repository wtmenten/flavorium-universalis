# Feeders for the cc_taxis_vs_cc_dynatoi societal value.
#
# Vanilla's byzantium.txt feeds latinization_vs_hellenization from language and culture
# state. This axis is fed from political state instead: who actually holds the land,
# the soldiers and the offices.
#
# Every entry gates on has_societal_value first. Countries without the value would
# otherwise error on the monthly_towards_* modifiers, exactly as vanilla's
# byzantium.txt guards each of its own entries.
#
# scales_with takes a script value (see cc_byz_values.txt), never an inline iterator.

#############################################
# Toward Dynatoi: the provincial magnates
#############################################

# Pronoia grants are how land and its soldiers pass out of the fisc into private hands.
cc_byz_pronoia_grants = {
	potential_trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		any_subject = {
			is_subject_type = pronoia
			count > 0
		}
	}

	scales_with = {
		value = cc_byz_pronoia_count
		divide = 4
	}

	monthly_towards_cc_dynatoi = societal_value_min_scaling_monthly_move
}

# A weak crown estate means the magnates are setting the terms.
#
# THRESHOLD LOWERED FROM 0.35 TO 0.25, and this was the single largest thing wrong with the
# axis. Every funded bureaucracy carries global_crown_estate_power = -0.02, so an emperor
# running four bureaus is down 0.08 crown power for the act of running them. At a 0.35 floor,
# building the administrative state was itself enough to trip this entry and pay the magnates
# 0.05 a month, while cc_byz_staffed_bureaus below paid only 0.013 for the same four bureaus.
# Four to one against the player for doing the thing the axis is about. Playtesting reported
# the axis "pushing dynatoi when it should be pushing taxis" and this is most of why.
#
# 0.25 is genuine crown weakness rather than ordinary administrative overhead.
cc_byz_weak_crown = {
	potential_trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		"estate_power(estate_type:crown_estate)" < 0.25
	}

	monthly_towards_cc_dynatoi = societal_value_minor_monthly_move
}

# A powerful and contented nobility entrenches itself.
cc_byz_ascendant_nobles = {
	potential_trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		"estate_power(estate_type:nobles_estate)" >= 0.4
		estate_satisfaction:nobles_estate >= 0.6
	}

	monthly_towards_cc_dynatoi = societal_value_minor_monthly_move
}

#############################################
# Toward Taxis: the imperial bureaus
#############################################

# Every bureau actually staffed and funded is an argument for government by office.
#
# RAISED FROM min_scaling TO minor. This is the axis's single positive feeder and it was paying
# 0.01 x (bureaus/3), so four funded bureaus earned 0.013 a month against 0.05 from either of
# the two Dynatoi feeders above. The player's whole administrative programme was worth a
# quarter of one magnate condition. At minor it pays 0.067 for the same four, which is one
# Dynatoi feeder's worth, and the two Dynatoi conditions are things the player can avoid.
#
# The bureau slot cap keeps this bounded on its own: four slots is the practical ceiling, so
# this cannot run away the way a per-location feeder would.
cc_byz_staffed_bureaus = {
	potential_trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		any_current_bureaucracy = {
			bureaucracy_maintenance > 0
			count > 0
		}
	}

	scales_with = {
		value = cc_byz_funded_bureau_count
		divide = 3
	}

	monthly_towards_cc_taxis = societal_value_minor_monthly_move
}

# A crown that can appoint and dismiss does not need to bargain.
cc_byz_strong_crown = {
	potential_trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		"estate_power(estate_type:crown_estate)" >= 0.5
	}

	monthly_towards_cc_taxis = societal_value_minor_monthly_move
}

# A state that runs on paper: literate clerks are the fisc's real infrastructure.
# average_location_literacy is location scope; average_country_literacy is the country one.
cc_byz_literate_administration = {
	potential_trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		average_country_literacy > 20
	}

	monthly_towards_cc_taxis = societal_value_tiny_monthly_move
}


#############################################
# The bureaus themselves
#############################################
#
# LIFTED OUT OF THE BUREAUCRACY DEFINITIONS. Until the Order/Magnates axis was generalised into
# the main mod, this drift sat inside each of the eleven overridden bureaucracies, one line per
# side. That works while exactly one axis exists and fails the moment two do: a definition
# carries a single positive_modifier block, so it can express drift for one axis only, and a
# Byzantine emperor holding a generic bureaucracy would have been pushed on an axis he does not
# have. monthly_towards_cc_order on a country without cc_order_vs_cc_magnates is precisely the
# error vanilla's byzantium.txt wraps every one of its own entries in has_societal_value to
# avoid.
#
# auto_modifiers take potential_trigger, so both axes can have a parallel set and neither ever
# applies to the wrong country. These mirror the main mod's cc_bureau_axis.txt entry for entry.
#
# THE POLE COUNTS ARE IN cc_byz_values.txt and follow vanilla's own estates_that_like /
# estates_that_dislike rather than our taste. That distinction is load-bearing: magister_militum
# was once "corrected" onto the Dynatoi pole here on the reasoning that the aristocracy supplies
# its officers, which is true of the officers and false of the institution, and vanilla's
# estates_that_dislike = nobles had said so all along.

# Funded offices of the fisc's own kind.
cc_byz_taxis_offices = {
	potential_trigger = {
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		any_current_bureaucracy = {
			bureaucracy_maintenance >= 0.5
			count > 0
		}
	}

	scales_with = {
		value = cc_byz_taxis_funded
		divide = 3
	}

	monthly_towards_cc_taxis = societal_value_minor_monthly_move
}
