# Opinion modifiers for the debt threads.
#
# Vanilla already ships most of what a debt chain needs and those are used directly rather
# than shadowed: opinion_took_on_debt, opinion_default_loans, opinion_defaulted_on_loan,
# opinion_increased_loan_interest, opinion_insulted, and stolen_jewels for the regalia.
# Only the leverage-specific ones are new, since vanilla has no concept of a creditor holding
# a standing claim on the customs.
#
# add_opinion has no years parameter; duration is value plus yearly_decay here.

# A concession of customs or quarter rights that the creditor got and expects to keep.
# Slow decay: a privilege granted once is cited for a century.
cc_byz_creditor_privilege = {
	value = 40
	yearly_decay = 0.5
}

# The other republic's view of the same concession. What one gets, the other resents.
cc_byz_rival_privilege_granted = {
	value = -35
	yearly_decay = 2
}

# The emperor bought his creditors out rather than conceding to them. Respected and resented
# in equal measure, which is why this decays quickly.
cc_byz_bought_out_creditors = {
	value = -25
	yearly_decay = 6
}

# An island or port pledged as security against a loan.
cc_byz_holds_pledged_territory = {
	value = 30
	yearly_decay = 0
}

# The pledge was redeemed and the creditor had to give it back.
cc_byz_pledge_redeemed = {
	value = -40
	yearly_decay = 3
}

# The emperor repudiated a debt to this creditor outright. Harsher than vanilla's
# opinion_defaulted_on_loan because repudiation is a decision, not a failure.
cc_byz_repudiated_debt = {
	value = -80
	min = -140
	yearly_decay = 2
}

# The creditors acted together. Held by the emperor toward each member of the syndicate.
cc_byz_creditor_syndicate = {
	value = -45
	yearly_decay = 3
}

# Revenue was farmed out to this power's merchants.
cc_byz_holds_tax_farm = {
	value = 25
	yearly_decay = 1
}


# ---- Union arc ----
#
# The union is worth something in the west only while it holds, so the Catholic biases decay
# faster than the Orthodox ones. A church that felt betrayed remembers longer than a court
# that felt encouraged, which is also why the repudiation biases are the slowest here.

# Catholic powers toward an emperor who has opened negotiations.
cc_byz_union_negotiating = {
	value = 20
	yearly_decay = 4
}

# Catholic powers toward an emperor whose union actually took hold at home.
cc_byz_union_upheld = {
	value = 60
	yearly_decay = 2
}

# Catholic powers toward an emperor who signed and then could not deliver his own capital.
cc_byz_union_undelivered = {
	value = -40
	yearly_decay = 3
}

# Catholic powers toward an emperor who renounced a signed union.
cc_byz_union_renounced = {
	value = -70
	min = -120
	yearly_decay = 1.5
}

# Orthodox powers toward an emperor who submitted to Rome. Rus, Serbia, Bulgaria, Georgia.
cc_byz_submitted_to_rome = {
	value = -60
	min = -110
	yearly_decay = 1
}

# Orthodox powers toward an emperor who renounced the union and returned to the fold.
cc_byz_held_the_faith = {
	value = 50
	yearly_decay = 1
}

# ---- Mamluk confrontation ----

# ---- Required by the IO system ----
#
# EVERY international_organization_type NEEDS ONE OF THESE OR IT FAILS TO LOAD:
#
#   [international_organization_type.cpp:1845]: International Organization Type needs an
#   opinion of other members, add a value for io_opinion_<type> in /biases/
#
# It is the member-to-member opinion bonus, and vanilla defines one for all thirty-odd of its
# IO types. The main mod hit the same error for all four of its own and answers it in
# in_game/common/biases/cc_biases.txt:237-253, which is the shape copied here.
#
# The value is deliberately LOW compared with the main mod's blocs (15 to 20 there). Those are
# alliances of sovereigns who chose each other. This is an administrative body whose members
# were appointed separately, at different times, to govern provinces a thousand miles apart,
# and whose only shared interest is the one the consistory gives them. Five is the warmth of
# colleagues who have met.
#
# yearly_decay is omitted on purpose, unlike the bloc biases. Prefectural collegiality does not
# fade while the prefectures exist; it ends when they do.
io_opinion_cc_byz_prefectures = {
	value = 5
	opinion_modifier = yes
}
