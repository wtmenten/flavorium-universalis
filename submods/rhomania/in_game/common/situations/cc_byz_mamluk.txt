# The Mamluk confrontation, in two phases.
#
# Phase A is a cold war over the holy places that is FULLY RESOLVABLE BY DIPLOMACY. A
# negotiated condominium is a real ending, not a consolation prize, and the situation is built
# so a player can reach it without ever declaring war.
#
# Phase B, the Pentarchy War, escalates only if the player presses for it. Taking a seat writes
# into vanilla's own pentarchy variable map through set_pentarch_variable and
# add_pentarchy_modifier (the idiom is autocephalous_patriarchates.txt:544-547), so seats count
# toward has_all_pentarchy and unlock vanilla's mend_schism, which is where pillar 3's union arc
# hands off.
#
#
# NO has_enabled_situation_trigger GATE, deliberately.
#
# The main mod gates its own two situations that way and calls enable_situation_effect from
# on_game_start (in_game/common/on_action/cc_game_start.txt:98,106). This submod cannot do the
# same: a second declaration of the on_game_start key is last-one-wins, exactly as CLAUDE.md
# documents, so declaring it here would risk silently clobbering the main mod's entire
# game-start block. That is far worse than not having a rule toggle.
#
# The gate is unnecessary anyway. Vanilla's own situations (western_schism, guelphs_and_
# ghibellines, italian_wars) carry no such trigger; only rise_of_timur does. can_start below is
# narrow enough on its own that this cannot fire in a game it has no business firing in.
#
#
# Situation scopes, per common/situations/readme.txt:
#   can_start / can_end / on_start / on_monthly / on_ending / on_ended   root = situation
#   visible                                                             root = country, scope:target = situation
#   map_color / secondary_map_color / tooltip                           root = location, scope:target = situation
#
# Situation variables:
#   cc_byz_levant_progress   0..100  the empire's position in the Levant
#   cc_byz_mamluk_progress   0..100  Cairo's
#   cc_byz_mamluk_phase      1 or 2  cold war, or the Pentarchy War
#   cc_byz_empire            scope   the country fighting this, as a handle rather than a tag
#
# Country variable (on Byzantium): cc_byz_levant_pressure -5..5
#
#
# NO c:BYZ ANYWHERE IN THIS FILE, deliberately.
#
# A player who forms Rome keeps the same country object, its variables and its memories, and
# has_or_had_tag = BYZ keeps passing, which is what the events and the four situation actions
# gate on. The tag BYZ is what stops existing. This file was written entirely in c:BYZ, so
# forming Rome ended the confrontation on the next can_end evaluation and on_ended's c:BYZ ?=
# then skipped, meaning the thread was deleted mid-run without even narrating an ending.
#
# Everything below goes through var:cc_byz_empire instead, set by
# cc_byz_resolve_empire_effect (cc_byz_scores.txt) and re-resolved every month so old
# saves backfill. The header on that effect is the full explanation.

cc_byz_mamluk_confrontation = {
	content_priority = 650

	monthly_spawn_chance = monthly_spawn_chance_unique

	# hint_tag is set on 22 of vanilla's 22 situations and was missing here. It is what makes
	# the "?" button in the situation tooltip exist at all: shared/situation_tooltips.gui:13
	# gates it on Situation.HasHint. Defined in common/scriptable_hints/cc_byz_hints.txt.
	hint_tag = hint_cc_byz_mamluk_confrontation

	can_start = {
		has_dlc = "d008_fate_of_the_phoenix"

		# Anatolia has to be settled first. This is the whole premise: the empire only looks
		# south once it is no longer fighting for its life in the east, which is what the
		# rise of the Ottomans represents.
		NOT = { is_situation_active = situation:rise_of_the_ottomans }

		country_exists = c:MAM

		# ANY_COUNTRY, NOT c:BYZ. A player who formed Rome before the Ottoman situation resolved
		# could otherwise never see this at all, since the tag test would fail forever. Holding
		# the City is the identifying condition, and ROM_BYZ_f requires it anyway.
		any_country = {
			has_or_had_tag = BYZ
			has_societal_value = societal_value_type:latinization_vs_hellenization
			owns = location:constantinople
			# A meaningful presence in Anatolia, not a toehold.
			any_owned_location = {
				count >= 15
				region = region:anatolia_region
			}
		}
	}

	can_end = {
		OR = {
			# Guarded even though on_start sets both and can_end only runs for an active
			# situation. The cost is four lines; the cost of being wrong about evaluation
			# order is an error every tick for the life of the confrontation.
			trigger_if = {
				limit = { has_variable = cc_byz_levant_progress }
				var:cc_byz_levant_progress >= 100
			}
			trigger_if = {
				limit = { has_variable = cc_byz_mamluk_progress }
				var:cc_byz_mamluk_progress >= 100
			}
			NOT = { country_exists = c:MAM }

			# THE EMPIRE IS GONE, tested through the handle rather than through the tag. Written
			# as an AND rather than a trigger_if because a trigger_if whose limit fails evaluates
			# to TRUE, and inside this OR that would end the confrontation on any save where the
			# variable has not been resolved yet.
			AND = {
				has_variable = cc_byz_empire
				NOT = { country_exists = situation:cc_byz_mamluk_confrontation.var:cc_byz_empire }
			}

			current_year > 1650
		}
	}

	visible = {
		OR = {
			has_or_had_tag = BYZ
			has_or_had_tag = MAM
			religion = religion:orthodox
		}
	}

	on_start = {
		set_variable = { name = cc_byz_levant_progress  value = 0 }
		set_variable = { name = cc_byz_mamluk_progress  value = 0 }
		set_variable = { name = cc_byz_mamluk_phase     value = 1 }

		cc_byz_resolve_empire_effect = yes

		var:cc_byz_empire ?= {
			cc_byz_init_levant_scores_effect = yes
			trigger_event_non_silently = cc_byz_mamluk.1
		}
		c:MAM = {
			trigger_event_non_silently = cc_byz_mamluk.2
		}
	}

	on_monthly = {
		# Re-resolve the empire before anything reads it. A no-op in the ordinary case; it is
		# what backfills a save from before this variable existed, and what picks the empire up
		# again on the tick after a formable renamed it.
		cc_byz_resolve_empire_effect = yes

		# Mirror the three eastern seats onto the situation so the panel can show them.
		# The panel cannot read location ownership itself; see the effect's header.
		cc_byz_mamluk_seat_sync_effect = yes

		# Cairo's own weather. Mostly hidden or forced outcomes, giving the player openings
		# they did not create and setbacks they could not have prevented, so the situation
		# has weather rather than a schedule.
		#
		# RATE CUT FROM 2+2 TO 0.7+0.7. Playtesting: "byz_mam events fire frequently but arent
		# impactful". At 4% a month these two alone fired 39% of years, forever, for the whole
		# life of a situation that can run three centuries. That is roughly fifty firings of a
		# two-event rotation, which is what turns weather into wallpaper.
		#
		# 1.4% a month is about 8% a year, so the frontier speaks up roughly twice a
		# generation. The events themselves are correspondingly heavier: see the payload
		# rework in cc_byz_mamluk.txt, which is the other half of this fix and the more
		# important one.
		#
		# The per-event cooldowns live on the events' own triggers, not here, for the same
		# reason cc_byz_bureaucracy.60's does: one guard where the event is defined beats a
		# guard at each call site.
		random_list = {
			986 = { }
			7 = {
				var:cc_byz_empire ?= { trigger_event_silently = cc_byz_mamluk.5 }
			}
			7 = {
				var:cc_byz_empire ?= { trigger_event_silently = cc_byz_mamluk.4 }
			}
		}
	}

	on_ended = {
		# Resolved once more here rather than trusting the last monthly tick, because the two
		# endings that matter most (levant_progress or mamluk_progress hitting 100) are written
		# by an event and can be followed by can_end before another month passes.
		cc_byz_resolve_empire_effect = yes

		var:cc_byz_empire ?= {
			trigger_event_non_silently = cc_byz_mamluk.20
		}
	}

	# Both sides are identified by the sticky tag test rather than by the current tag, so the map
	# does not go blank the month an emperor forms Rome or a sultan forms Egypt. The situation
	# variable would be marginally more exact, but it is reachable from here only as
	# scope:target.var:cc_byz_empire and would error on a save where it is not yet resolved,
	# which is a bad trade for a colour.
	map_color = {
		if = {
			limit = { owner ?= { has_or_had_tag = BYZ } }
			value = define:NMapColors|MAP_COLOR_HIGH
		}
		else_if = {
			limit = { owner ?= { has_or_had_tag = MAM } }
			value = define:NMapColors|MAP_COLOR_LOW
		}
		else = {
			value = define:NMapColors|MAP_COLOR_NULL
		}
	}

	# WHAT THE MAP SAYS WHEN YOU HOVER A COLOURED LOCATION. 21 of vanilla's 22 situations
	# define this and this one did not, so the map colouring above was unexplained: two shades
	# with nothing telling the player which was which.
	#
	# Root here is the LOCATION and scope:target is the situation, per situations/readme.txt.
	# Not executed, only rendered, which is why a bare custom_tooltip is the whole body.
	tooltip = {
		if = {
			limit = { owner ?= { has_or_had_tag = BYZ } }
			custom_tooltip = cc_byz_mam_map_tt_empire
		}
		else_if = {
			limit = { owner ?= { has_or_had_tag = MAM } }
			custom_tooltip = cc_byz_mam_map_tt_cairo
		}
		else = {
			custom_tooltip = cc_byz_mam_map_tt_other
		}
	}

	# The map legend. Vanilla defines 79 of these across its situations and both of ours had
	# none, so the situation's colours appeared in the legend box as nothing at all.
	legend_key = {
		desc = "CC_BYZ_MAM_LEGEND_EMPIRE"
		color = blue
		require_color_on_map = yes
	}
	legend_key = {
		desc = "CC_BYZ_MAM_LEGEND_CAIRO"
		color = red
		require_color_on_map = yes
	}
}
