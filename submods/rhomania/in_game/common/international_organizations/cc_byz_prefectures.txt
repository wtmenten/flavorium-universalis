# The Praetorian Prefectures of the West.
#
# Created by the cc_byz_the_west situation's on_start. The emperor leads it; every exarchate
# joins on creation. Its consistory is where the four western laws are voted, and the vote
# arithmetic is the whole point of the design:
#
#     the Augustus        3 votes
#     the Senior Exarch   2 votes
#     each other exarch   1 vote
#
# With one prefecture the emperor cannot lose, 3 to 1. With all four seated it is 3 against 5
# and the west carries the vote. The danger scales with the reconquest by itself, with no risk
# score to tune: the emperor built the body that outvotes him, one province at a time.
#
#
# AN OVERLORD AND ITS OWN SUBJECTS IN ONE IO IS A SHAPE VANILLA NEVER SHIPS.
#
# Three consequences, all handled below:
#
# 1. Both expel_members_who_* flags are OFF. They are on for the Balance of Power blocs because
#    those are alliances of sovereigns; here a routine war in which the emperor calls his
#    exarch in on one side would eject half the west from the body that governs it.
#
# 2. can_join_trigger REQUIRES being an exarchate rather than forbidding subjects. The vanilla
#    blocs write is_subject = no; that would exclude every intended member here.
#
# 3. has_limited_diplomacy = yes on the exarchate subject type is a SCRIPTED gate, not an
#    engine block. It is read by scripted_relations (guarantee.txt, food_access.txt) and by
#    independence_movement's join trigger, all of which are script we do not share. Nothing in
#    the definitions forbids a subject holding a special status or sitting in a parliament.
#    UNVERIFIED IN GAME: whether such a country can actually cast a vote. If it cannot, the
#    consistory degrades to a display and the four laws need another vehicle.
#
# disband_if_no_leader is NO. The acclamation ending is precisely the case where the west
# continues without the emperor, and disbanding the body at that moment would delete the
# ending's whole point.

cc_byz_prefectures = {
	has_target = no
	unique = yes
	show_on_diplomatic_map = yes

	# See note 1 above.
	expel_members_who_are_targets_of_other_members = no
	expel_members_who_are_attackers_at_war_with_other_members = no

	has_leader_country = yes
	leader_type = country
	disband_if_no_leader = no

	# The sacrum consistorium. Begins DORMANT: the prefectures exist as an administrative fact
	# long before they are a body that meets. cc_byz_io_convene_consistory swaps the parliament
	# form and is the emperor's decision alone.
	has_parliament = yes
	parliament_type = cc_byz_consistorium_dormant
	resolution_widget = parliament
	alert_view_tab = parliament
	special_statuses_implemented = { cc_byz_augustus cc_byz_senior_exarch cc_byz_exarch_seat }

	can_initiate_policy_votes = {
		is_leader_of_international_organization = scope:recipient
	}

	laws = {
		cc_byz_west_command_law    = cc_byz_command_imperial
		cc_byz_west_fisc_law       = cc_byz_fisc_remitted
		cc_byz_west_rite_law       = cc_byz_rite_greek
		cc_byz_west_succession_law = cc_byz_succession_appointed
	}

	# The emperor's armies move through the prefectures as a matter of course, and so do the
	# exarchs' through each other's. That is what a prefecture is.
	has_military_access = {
		scope:actor ?= {
			is_member_of_international_organization = international_organization:cc_byz_prefectures
		}
	}

	create_visible_trigger = { always = no }
	invite_visible_trigger = { always = no }

	can_lead_trigger = {
		has_or_had_tag = BYZ
		capital = location:constantinople
	}

	# See note 2 above: membership is defined positively.
	can_join_trigger = {
		OR = {
			has_variable = cc_byz_is_exarchate
			AND = {
				has_or_had_tag = BYZ
				capital = location:constantinople
			}
		}
		NOT = { is_member_of_international_organization = international_organization:cc_byz_prefectures }
	}

	# An exarch does not resign from the empire. Losing the status is something that happens to
	# him, through the friction thread or the situation's endings, not something he elects.
	can_leave_trigger = {
		always = no
	}

	# AUTOMATIC REMOVAL, AND THIS IS A CORRECTNESS FIX RATHER THAN POLISH.
	#
	# can_leave_trigger = always = no stops a prefect resigning. It does NOT remove a country
	# that has stopped being a prefecture some other way, and there are three such ways: the
	# dissolve action (cc_byz_west.30) strips cc_byz_is_exarchate and takes the territory, the
	# peace treaty strips it before annexing, and a subject type change strips it by hand.
	#
	# Without this the dissolved country keeps its seat. Worse, it keeps its VOTE: the special
	# statuses auto-rescind on the same variable, but a member with no status still counts
	# against the totals the consistory is measured on, so an emperor who dissolved a
	# prefecture to reduce the opposition would have found the opposition unchanged.
	#
	# 34 of vanilla's 35 IO types define this. Omitting it was an oversight.
	auto_leave_trigger = {
		NOT = { is_leader_of_international_organization = scope:recipient }
		NOT = { has_variable = cc_byz_is_exarchate }
	}

	# The body ends when there are no prefectures left to constitute it, and not before.
	#
	# NARROW ON PURPOSE. It deliberately does NOT disband when the emperor is gone or has
	# stopped leading, because the acclamation ending is precisely the case where the west
	# continues without him: the consistory goes on meeting, taking the view that it was
	# constituted by imperial authority and still is. That is also why disband_if_no_leader is
	# no, above.
	auto_disband_trigger = {
		OR = {
			total_members < 1
			NOT = {
				any_international_organization_member = {
					has_variable = cc_byz_is_exarchate
				}
			}
		}
	}

	# can_declare_war IS DELIBERATELY ABSENT, and it is the one common key this file omits on
	# purpose rather than by oversight.
	#
	# 33 of 35 vanilla IOs define it, and every one of them uses it to stop members making war
	# on each other. This body must permit exactly that: cc_byz_west.40 ends in the emperor
	# declaring on his own prefects while both remain members, and a restriction here would
	# make the civil war undeclarable no matter what the casus belli allowed.
}
