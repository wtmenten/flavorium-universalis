# FRONT-END 1: theatre actions, for players with mission packs OFF.
#
# This is the path MOST PLAYERS WILL SEE. mission_packs_enabled_rule declares
# default = mission_packs_disabled (main_menu/common/game_rules/00_game_rules.txt:155-167),
# and mission_packs_disabled additionally carries proficiency_expert, so it is also what an
# experienced player gets picked for them. A mission-only delivery of the reconquest would be
# invisible to the majority of the audience.
#
# The two front-ends are mutually exclusive on the game rule, so no player can claim a theatre
# twice and neither mode has content the other lacks. Both call the same
# cc_byz_theatre_*_effect, so rewards cannot drift apart.
#
# has_game_rule is an ordinary trigger usable in any trigger context; vanilla uses it for
# byz_name_greek in customizable_localization/countries.txt:27.
#
# No price block. The cost of a theatre is the war it took to hold the cities, which the
# trigger already required, plus the strain the effect applies afterwards. Charging gold on
# top would be taxing the player for having done the thing.

cc_byz_claim_theatre_illyricum = {
	type = owncountry
	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = monthly
	automation_tick_frequency = 12
	player_automated_category = religiousdoctrines

	potential = {
		has_dlc = "d008_fate_of_the_phoenix"
		NOT = { has_game_rule = mission_packs_enabled }
		scope:actor = {
			has_or_had_tag = BYZ
			has_societal_value = societal_value_type:latinization_vs_hellenization
			NOT = { has_variable = cc_byz_mem_theatre_illyricum }
		}
	}

	allow = {
		scope:actor = { cc_byz_theatre_illyricum_trigger = yes }
	}

	effect = {
		scope:actor = {
			cc_byz_theatre_illyricum_effect = yes
			trigger_event_non_silently = cc_byz_reconquest.1
		}
	}

	ai_will_do = { add = 10 }
}
