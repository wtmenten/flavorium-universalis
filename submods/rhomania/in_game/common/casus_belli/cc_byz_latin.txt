# The Latin crusade. Granted only by cc_byz_latin.5, to Byzantium and to its Catholic in-laws
# against the same Muslim neighbour. Never created by the player directly.
#
# Same structure as cc_byz_cb_reunify_pontus: create_visible and create_enabled are always = no,
# so the CB exists only when add_casus_belli grants it, and declare_enabled re-checks the
# target at declaration. take_border, because the war is to recover lost provinces, not to
# end a state.
cc_byz_cb_latin_crusade = {
	create_visible = {
		always = no
	}
	create_enabled = {
		always = no
	}
	declare_enabled = {
		scope:target = {
			religion.group = religion_group:muslim
		}
	}

	war_goal_type = take_border
}
