# VANILLA OVERRIDE of all eleven Byzantine bureaucracies from
# game/in_game/common/bureaucracies/byz.txt.
#
# THE REPLACE: PREFIX IS LOAD-BEARING. A bare `key = { ... }` for a key vanilla already
# defines does NOT win on load order. The engine refuses it and keeps vanilla:
#   [gamedatabase.h:408]: Duplicated key <name> will not be created from file: ...
# The file is then silently ignored while looking entirely correct. Directives available:
#   REPLACE:key  REPLACE_OR_CREATE:key  TRY_REPLACE:key  INJECT:key  TRY_INJECT:key
# INJECT: merges at the entry's TOP level and cannot reach inside an existing
# positive_modifier to alter its nested scale, which is why this file uses REPLACE:.
#
#
# ENTRENCHMENT IS THE ENGINE'S JOB. scope:entrenchment DOES NOT EXIST HERE.
#
# The engine ALREADY multiplies both the positive and the negative modifier of every
# bureaucracy by entrenchment by itself. See game_concept_bureaucracy_desc, "increasing the
# power of both positive and negative modifiers", and the ADDITIONAL_MODIFIER_ENTRENCHMENT
# line rendered under a bureaucracy tooltip as "Entrenchment +N%".
#
# TWO revisions of this file got entrenchment wrong, in opposite directions, and the history
# is kept here because the second mistake looked exactly like a fix for the first.
#
#   1. An early revision multiplied BOTH sides by scope:entrenchment, on the mistaken reading
#      that because no bureaucracy SCRIPT reads it, entrenchment must be an inert timer. It is
#      not; the base scaling is automatic and doing it again squares it.
#
#   2. The correction left ONE deliberate extra factor on the malus side, to soften a brand
#      new unfunded office. That looked reasonable and was still wrong, because
#      scope:entrenchment is not actually pushed into these blocks at all. The readme in
#      game/in_game/common/bureaucracies/ lists it alongside scope:maintenance, but NO vanilla
#      bureaucracy uses it anywhere, and the engine says so plainly:
#
#        Undefined event target 'entrenchment'
#        Value of wrong type in '...' Got value of type 'none'
#
#      The undefined read poisoned the whole enclosing `scale`, so every negative_modifier
#      carrying the factor evaluated to none. The maluses were not being softened. They were
#      not being applied.
#
# So the scale is vanilla's again: `value = 1  subtract = scope:maintenance`. scope:maintenance
# IS real and vanilla relies on it 22 times in byz.txt. If a brand new office still feels too
# punishing, the lever is the modifier VALUES in each negative_modifier body, not the scale.
#
# WHAT CHANGED, and nothing else. Vanilla text is otherwise untouched, so a patch can be
# re-synced by diffing against byz.txt and re-applying these per entry:
#
#   1. negative_modifier scale: UNCHANGED from vanilla. It once carried an entrenchment
#      factor; see the note above for why that had to come out.
#   2. negative_modifier body: a little noble power, universal to all eleven, since a starved
#      office of any kind ends up serving somebody local.
#   3. positive_modifier body: a small crown power cost. The positive SCALE is untouched.
#
#      THE AXIS DRIFT USED TO LIVE IN 2 AND 3 AND NO LONGER DOES. It was one line per
#      bureaucracy per side, twenty-one lines in all. A definition carries a single
#      positive_modifier block and can therefore express drift for exactly one axis, which
#      stopped being enough when the main mod lifted a generalised Order/Magnates version of
#      this quarrel for everybody else: a Byzantine emperor holding a generic bureaucracy would
#      have been pushed on an axis he does not have. Drift now lives in
#      auto_modifiers/cc_byz_taxis.txt, which takes potential_trigger and can gate on
#      has_societal_value so the two axes never both apply.
#   4. on_maintenance_changed firing cc_byz_bureaucracy.60 on a cut to <= 0.25. NOTE that
#      sixty_books_of_the_basilika_bureaucracy ALREADY has this hook in vanilla, where it
#      records maintenance into a variable; ours is merged into that block rather than
#      declared alongside it, since a duplicate key silently drops one of the two.
#
#      THE HOOK IS DELIBERATELY UNGUARDED HERE. Eleven bureaucracies in this file carry it and
#      two more are in cc_byz_bureaucracies.txt, so an emperor economising across the board
#      would fire the same dialogue thirteen times in one afternoon. The five-year cooldown
#      lives on cc_byz_bureaucracy.60's own trigger instead, where one guard covers all
#      thirteen call sites and cannot be missed on a twelfth copy.
#
# Three entries needed ARITHMETIC merges because vanilla already sets the same key in that
# block, and a repeated key inside one modifier block does not stack:
#   nomos_empsychos  positive global_crown_estate_power 0.15 - 0.02 = 0.13
#   magister_militum negative global_nobles_estate_power 0.5 + 0.03 = 0.53
#   themata          negative global_nobles_estate_power 0.5 + 0.03 = 0.53
#
# Pole assignment follows vanilla's own estates_that_like / estates_that_dislike: a bureau
# that dislikes nobles governs around the magnates (Taxis), one that likes them governs
# through them (Dynatoi). romanitas_bureaucracy is unassigned, being about imperial identity
# rather than about who governs.
#
# PATCH RISK. This shadows vanilla definitions. Any EU5 patch touching byz.txt diverges
# silently until re-synced. Re-check on every game update.

REPLACE:nomos_empsychos_bureaucracy = {
	on_maintenance_changed = {
		if = {
			limit = {
				has_dlc = "d008_fate_of_the_phoenix"
				has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
				scope:new_maintenance < scope:old_maintenance
				scope:new_maintenance <= 0.25
			}
			trigger_event_non_silently = cc_byz_bureaucracy.60
		}
	}

	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = cc_bureau_maintenance_base
		multiply = 0.003

		# STEWARD DISCOUNT, read from the variable the MAIN MOD's cabinet duty sets. The two must
		# name the same type or this silently does nothing while the tooltip still promises it.
		# See in_game/common/scripted_effects/cc_bureau_steward.txt in the main mod; these ten are
		# vanilla types, so it can name them whether or not this submod is loaded.
		if = {
			limit = { has_variable = cc_steward_nomos_empsychos_bureaucracy }
			multiply = 0.75
		}
	}

	potential = {
		has_dlc = "d008_fate_of_the_phoenix"
		OR = {
			has_or_had_tag = BYZ
			AND = {
				has_or_had_tag = TRE
				has_variable = tre_unlocked_byz_bureaucracies
			}
		}
	}

	allow = {
		uses_government_power = legitimacy
	}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	#neutral
	neutral_modifier = {
		global_estate_satisfaction_from_legitimacy = 0.05
	}

	#positive
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_crown_estate_power = 0.13
	}

	#negative
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_estate_satisfaction_decay = 0.025
		global_nobles_estate_power = 0.03
	}
}
