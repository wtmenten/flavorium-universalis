# The Academy chain: the yearly refresh of its location rewards.
#
# The vanilla hook (yearly_country_pulse) is declared in cc_byz_on_actions.txt. This file
# defines only the custom on_action it points at. Not behind cc_byz_pulse_cd: it is upkeep,
# not an event.
cc_byz_on_yearly_academy = {
	trigger = {
		has_or_had_tag = BYZ
		OR = {
			has_variable = cc_byz_acad_r_literacy
			has_variable = cc_byz_acad_r_institution
		}
	}
	effect = {
		cc_byz_acad_refresh_locations_effect = yes
	}
}
