# Exarchate thread chain points.
#
# The vanilla hook (yearly_country_pulse) is declared in cc_byz_on_actions.txt.
#
# All three threads share one pulse entry rather than three, because they share a
# precondition: an exarchate has to exist. The two papal events that do NOT require one
# (.10 the annexation guard and .11 the substitute thread) are in a separate on_action with
# its own trigger, since an emperor can erase the Papal States without ever appointing a
# governor and the guard has to fire for him too.
#
# Each thread's later events are reached by delayed continuation from its own first event, so
# the pulse only ever needs to open a thread, never to advance one.

cc_byz_on_yearly_exarchate = {
	trigger = {
		has_dlc = "d008_fate_of_the_phoenix"
		has_or_had_tag = BYZ
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		any_subject = { has_variable = cc_byz_is_exarchate }
		# Shared rate limiter across all six condition pulses. See
		# scripted_effects/cc_byz_pulse.txt for why this is one cooldown and not six
		# retuned weight tables.
		NOT = { has_variable = cc_byz_pulse_cd }
	}
	random_events = {
		100 = 0
		10 = cc_byz_exarch.1
		8 = cc_byz_papal.1
		8 = cc_byz_italy.1
		8 = cc_byz_demand.1
	}
}

# The prompt that offers a prefecture.
#
# THIS USED TO REQUIRE THAT NO EXARCHATE EXISTED ANYWHERE, which was correct while Italy was
# the only possibility and is wrong four ways over now. An emperor who seated Ravenna in 1400
# would never have been offered Carthage, Spania or the Gauls at all: the pulse would have
# stopped calling the event and the event's own fire_only_once would have finished the job.
#
# The guard is now per-region and lives in cc_byz_prefecture_open_any_trigger, and
# cc_byz_exarch.10 is no longer fire_only_once, so this fires again each time a NEW prefecture
# becomes seatable and never for one already seated.
cc_byz_on_yearly_exarch_prompt = {
	trigger = {
		has_dlc = "d008_fate_of_the_phoenix"
		has_or_had_tag = BYZ
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		cc_byz_prefecture_open_any_trigger = yes
	}
	random_events = {
		10 = 0
		10 = cc_byz_exarch.10
	}
}

# The three regional threads. Each is gated on its own prefecture existing, so an emperor who
# only ever holds Italy never sees any of them and an emperor who holds all four sees all three
# running at once alongside the shared friction and demand arcs.
#
# Italy is deliberately absent: it already has cc_byz_papal.1 and cc_byz_italy.1 in the list
# above, and giving the oldest prefecture a fourth thread purely because it was written first
# would be the wrong kind of depth.
cc_byz_on_yearly_west_regions = {
	trigger = {
		has_dlc = "d008_fate_of_the_phoenix"
		has_or_had_tag = BYZ
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
		any_subject = { has_variable = cc_byz_is_exarchate }
		# Shared rate limiter across all six condition pulses. See
		# scripted_effects/cc_byz_pulse.txt for why this is one cooldown and not six
		# retuned weight tables.
		NOT = { has_variable = cc_byz_pulse_cd }
	}
	random_events = {
		100 = 0
		10 = cc_byz_prov.1
		10 = cc_byz_prov.10
		10 = cc_byz_prov.20
	}
}

# The papal guard and its aftermath. Deliberately NOT gated on holding an exarchate: the
# Western Schism can be deleted by any Byzantine emperor who annexes the Papal States before
# 1360, whether or not he ever appointed a governor to Rome. See the header of
# in_game/events/cc_byz_papal.txt for what is at stake.
cc_byz_on_yearly_papal_guard = {
	trigger = {
		has_dlc = "d008_fate_of_the_phoenix"
		has_or_had_tag = BYZ
		has_societal_value = societal_value_type:cc_taxis_vs_cc_dynatoi
	}
	random_events = {
		# Weighted high because both events are one-shot and self-gating: .10 only fires
		# while PAP is one step from erasure and before 1360, .11 only once PAP is already
		# gone. A low weight would risk the guard never firing inside its own window.
		10 = 0
		10 = cc_byz_papal.10
		10 = cc_byz_papal.11
	}
}


# The deadline warning. Separate from the regional pulse because it must fire for an emperor
# whose prefectures are all in Italy, and because its own trigger already restricts it to one
# age window and one firing.
cc_byz_on_yearly_west_deadline = {
	trigger = {
		has_dlc = "d008_fate_of_the_phoenix"
		has_or_had_tag = BYZ
		is_situation_active = situation:cc_byz_the_west
	}
	random_events = {
		4 = 0
		10 = cc_byz_west.80
	}
}


# The western civil war flashpoint. Its own trigger carries the real gates (a convened
# consistory, a prefect at loyalty -3 or worse, autonomy at 55) and fire_only_once, so the
# weight here only decides how soon after qualifying the rising comes.
cc_byz_on_yearly_west_rising = {
	trigger = {
		has_dlc = "d008_fate_of_the_phoenix"
		has_or_had_tag = BYZ
		is_situation_active = situation:cc_byz_the_west
		NOT = { has_variable = cc_byz_mem_west_civil_war }
	}
	random_events = {
		6 = 0
		10 = cc_byz_west.40
	}
}
