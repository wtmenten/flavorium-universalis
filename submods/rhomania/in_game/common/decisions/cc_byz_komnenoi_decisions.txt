# The Komnenian Question, opened deliberately.
#
# THE MOD'S FIRST DECISION. Format follows vanilla 1.4 (in_game/common/decisions/
# flavor_decisions_byz.txt): decision_category, tags, potential/allow, inline options whose
# effects sit in an effect = { } block. There is no price block on decisions; costs are paid
# inside the effect and gated in allow, the way coal_prospecting.txt does it.
#
# The passive route (on_action/cc_byz_komnenoi.txt) opens the same chain without a decision;
# both call cc_byz_kom_open_effect.
#
# THE REQUIREMENTS ARE THE POINT. Political influence alone made this a button any emperor could
# press on day one. It now needs:
#   - Constantinople. A claim to the one Roman title made from anywhere else is not a claim.
#   - At least one of the five entrances. No cold start: the relationship with Trebizond has
#     to exist, friendly or hostile, before there is anything to negotiate.
#   - A settled court: stability above 0, legitimacy 50 or more, no regency.
# Thresholds follow vanilla decisions (stability > 0 in flavor_decisions_*, has_regent = no).

cc_byz_kom_raise_the_question_decision = {
	decision_category = country_specific_decisions
	only_once = yes

	tags = { BYZ ROM }

	potential = {
		cc_byz_kom_can_open_trigger = yes
	}

	allow = {
		political_influence >= 25
		at_war = no
		owns = location:constantinople
		stability > 0
		legitimacy >= 50
		has_regent = no
		custom_tooltip = {
			text = cc_byz_kom_needs_entry_tt
			cc_byz_kom_any_entry_trigger = yes
		}
	}

	ai_will_do = {
		value = 5
	}

	illustration_tags = {
		10 = interior
		10 = regular
	}

	illustration_scope = {
		ruler_or_regent ?= {
			save_scope_as = target_character
		}
		c:TRE ?= {
			save_scope_as = cc_kom_tre
		}
	}

	option = {
		name = cc_byz_kom_raise_the_question_decision.a
		ai_chance = { value = 100 }
		effect = {
			add_political_influence = -25
			cc_byz_kom_open_effect = yes
		}
	}
}
