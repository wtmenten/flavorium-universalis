# Prices for the union actions.
#
# Shape and currencies copied from vanilla's own religious action prices in
# in_game/common/prices/00_hardcoded.txt (mend_schism_price = { religious_influence = 50
# stability = 20 }). Valid currencies there are stability, gold, scaled_gold, prestige,
# religious_influence, government_power, honor, legitimacy, righteousness, war_exhaustion
# and karma.
#
# The cost curve is the argument the arc is making. Opening contacts is close to free,
# because it was; a legate and a letter cost an emperor nothing he could not deny later.
# The council is the expensive step and it is expensive in religious influence rather than
# gold, because what it spent was the patriarch's standing. Promulgation at home costs
# stability, which is the whole historical point: the signature was the easy part.

cc_byz_union_contacts_price = {
	religious_influence = 10
	gold = 40
}

	gold = 250
