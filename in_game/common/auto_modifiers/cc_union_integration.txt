###############################################################################
# Flavorium Universalis - union integration and the junior partner offer
#
# The junior_partner offer (subject_types/cc_subject_types.txt) loses 0.4 acceptance per point
# of the union member's tax base, which shuts out any sizeable member. An integrated union
# should not face that in full, so the senior partner gains Accept Subjugation Reasons, which
# the offer's diplo_chance_accept_subject weighs at 1, worth
#
#     0.4 x tax base of the largest other member x L / (1 + L)        L = union integration
#
# so the net tax-base penalty is 0.4 x tax base / (1 + L): all of it with no integration, half
# at level 1, a third at level 2.
#
# Why not scale the factor itself: diplo_chance factors take plain numbers only. A script value
# there is a load error ("Malformed token") and the factor is dropped. Tested 2026-10-25.
#
# Side effect, accepted: the modifier sits on the senior partner, so while the union is
# integrated it also eases that country's other diplomatic subjugation offers. It is sized
# against the LARGEST other member, so smaller members are offset by more than their own
# penalty.
###############################################################################

cc_union_integration_offer = {
	requires_real = yes

	potential_trigger = {
		union ?= {
			has_active_senior_partner = yes
			country_is_senior_partner = { country = root }
			modifier:union_integration_level > 0
		}
	}

	scales_with = cc_union_junior_offer_offset

	accept_subjugation_reasons = 1
}
