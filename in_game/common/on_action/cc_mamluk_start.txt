
# Mamluk Start — applied once at game start.
# Represents the Mamluks' tenuous hold over a largely foreign population.
# A permanent peasant levy penalty reflects the alienated native Egyptians,
# while decaying military and control penalties ease as the Sultanate consolidates
# its rule over the following two centuries.

on_game_start = {
	effect = {
		if = {
			limit = {
				has_game_rule = cc_mamluk_start_on
				country_exists = c:MAM
			}

			c:MAM = {
				# Permanent: foreign military caste ruling over a hostile peasantry.
				add_country_modifier = {
					modifier = cc_mam_foreign_rule
					days = -1
				}

				# Decaying: early military fragility and loose control over Egypt.
				# Fades over 200 years (~1644 in a 1444 start).
				add_country_modifier = {
					modifier = cc_mam_military_strain
					years = 200
				}
			}
		}
	}
}
