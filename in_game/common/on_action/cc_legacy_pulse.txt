
# Hooks into on_cabinet_death to fire legacy events.
# root = country, scope:target = dying minister character.
on_cabinet_death = {
	on_actions = {
		cc_on_cabinet_death
	}
}

cc_on_cabinet_death = {
	effect = {
		scope:target = { save_scope_as = cc_legacy_minister }

		# Track B: Legendary minister death → permanent modifier event (cc_legacy.2)
		if = {
			limit = {
				scope:cc_legacy_minister = {
					OR = {
						has_trait = grand_vizier_legendary
						has_trait = tribune_of_the_people
						has_trait = legendary_navigator
						has_trait = philosopher_king
					}
				}
				NOT = { has_variable = cc_legacy_legendary_fired }
			}
			set_variable = { name = cc_legacy_legendary_fired value = yes }
			trigger_event_silently = cc_legacy.2
		}

		# Track C: Retired legend dies → legacy modifier event (cc_legacy.3)
		else_if = {
			limit = {
				scope:cc_legacy_minister = { has_trait = retired_legend }
				NOT = { has_variable = cc_legacy_legendary_fired }
			}
			set_variable = { name = cc_legacy_legendary_fired value = yes }
			trigger_event_silently = cc_legacy.3
		}
	}
}
