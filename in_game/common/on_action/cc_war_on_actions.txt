
# Flavorium Universalis — War Council On-Actions (Pillar 04)
#
# Hooks into vanilla on_winning_war to set the military reform opportunity flag
# when the winning country has a reform-capable military minister in cabinet.

on_winning_war = {
	on_actions = { cc_wc_on_winning_war }
}

# Sets cc_military_reform_opportunity (60-month window) on the winning country
# when a standing_army_advocate, commissariat_officer, or siege_engineer is present.
# The reform proposal events (cc_wc.10-12) sweep yearly against this flag.
cc_wc_on_winning_war = {
	trigger = {
		scope:winner = {
			any_cabinet_character = {
				OR = {
					has_trait = standing_army_advocate
					has_trait = commissariat_officer
					has_trait = siege_engineer
				}
			}
		}
	}
	effect = {
		scope:winner = {
			set_variable = { name = cc_military_reform_opportunity value = yes months = 60 }
		}
	}
}
