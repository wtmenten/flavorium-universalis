
# Flavorium Universalis — Game Start On-Action Effects
# All one-time game start effects are combined here to avoid the
# "more than one effect defined" warning from having two separate files.
on_game_start = {
    on_actions = {
        cc_on_game_start
    }
}

cc_on_game_start = {
	effect = {
		# --- France Containment ---
		# Gives France's Lowlands-culture neighbors elevated antagonism, decaying over 200 years.
		if = {
			limit = {
				has_game_rule = cc_france_containment_on
				country_exists = c:FRA
			}

			c:FRA = {
				every_neighbor_country = {
					limit = {
						any_owned_location = {
							OR = {
								dominant_culture = culture:low_franconian
								dominant_culture = culture:picard
								dominant_culture = culture:walloon
							}
						}
					}
					add_antagonism = {
						target = c:FRA
						modifier = cc_fra_hegemony_threat
					}
				}
			}

			c:FRA = {
				add_country_modifier = {
					modifier = cc_fra_lowlands_ambition
					years = 200
				}
			}
		}

		# --- Hundred Years War Flavor Start ---
		# Applies starting levy/estate modifiers to FRA and ENG, decaying over 100 years.
		if = {
			limit = {
				has_game_rule = cc_hyw_flavor_on
				country_exists = c:FRA
			}

			c:FRA = {
				add_country_modifier = {
					modifier = cc_fra_hyw_feudal_strain
					years = 100
				}
			}

			if = {
				limit = { country_exists = c:ENG }
				c:ENG = {
					add_country_modifier = {
						modifier = cc_eng_hyw_martial_tradition
						years = 100
					}
				}
			}
		}

		# --- Mamluk Start ---
		# Represents the Mamluks' tenuous hold over a largely foreign population.
		if = {
			limit = {
				has_game_rule = cc_mamluk_start_on
				country_exists = c:MAM
			}

			c:MAM = {
				add_country_modifier = {
					modifier = cc_mam_foreign_rule
					days = -1
				}

				add_country_modifier = {
					modifier = cc_mam_military_strain
					years = 200
				}
			}
		}

		# --- Invasion of Mexico Situation ---
		# Enable the situation so it can fire when its can_start conditions are met.
		if = {
			limit = { has_game_rule = cc_invasion_mexico_on }
			enable_situation_effect = { type = cc_invasion_mexico }
		}

		# --- Ottoman Start ---
		# prevent the ottomans specifically from colonizing.
		if = {
			limit = {
				has_game_rule = cc_tur_start_on
				country_exists = c:TUR
			}

			c:TUR = {
				add_country_modifier = {
					modifier = cc_tur_no_colonization
					days = -1
				}

			}
		}

		# --- Hungary Start ---
		# Magyar nobility assert cultural dominance: -33% cultures cap, +50% noble power, small cav bonus.
		if = {
			limit = { country_exists = c:HUN }
			c:HUN = {
				unlock_estate_privilege_effect = { type = magyar_supremacy }
				grant_estate_privilege = estate_privilege:magyar_supremacy
			}
		}

		# --- Synergy Consolation Counter Initialization ---
		every_country = {
			set_variable = { name = cc_synergy_consolation_count value = 0 }
		}

		# --- AI Personality Monitor Variable Initialization ---
		# Pre-set all counter variables to 0 so var: comparisons in limits and
		# ai_chance blocks never fire before the monthly monitor has run once.
		if = {
			limit = {
				OR = {
					has_game_rule = cc_ai_personalities_dynamic
					has_game_rule = cc_ai_personalities_full
				}
			}
			every_country = {
				set_variable = { name = cc_p_at_war value = 0 }
				set_variable = { name = cc_p_high_exhaustion value = 0 }
				set_variable = { name = cc_p_treasury_low value = 0 }
				set_variable = { name = cc_p_treasury_rich value = 0 }
				set_variable = { name = cc_p_disloyal_subject value = 0 }
				set_variable = { name = cc_p_coalition_threat value = 0 }
				set_variable = { name = cc_p_low_prestige value = 0 }
				set_variable = { name = cc_p_great_power value = 0 }
			}
		}
	}
}
