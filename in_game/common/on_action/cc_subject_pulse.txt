# Dedicated holy protectorate periodic events — separated from the shared pool so they
# have their own chance_to_happen cadence and don't compete with other subject type events.
cc_holy_protectorate_pulse = {
	trigger = {
		any_subject = {
			is_subject_type = holy_protectorate
		}
	}
	random_events = {
		10 = cc.195        # The Faith Under Threat

		10 = cc.230        # A Shepherd for the Flock

		chance_to_happen = 25
	}
}
