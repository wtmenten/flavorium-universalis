# Colonial consolidation / border friction — mid-colonial period (Age 5+) only.
# At most 1 entry event fires per year from this pool; follow-up events fire via
# trigger_event_silently from their parents, not from here.
cc_colonial_consolidation_pulse = {
	trigger = {
		has_game_rule = cc_colonial_consolidation_on
		current_age_or_later = { age = age_5_absolutism }
		any_subject = { is_subject_type = colonial_nation }
	}
	random_events = {
		sample_count = 1
		10 = cc_colonial.20    # Border Friction — The Frontier Quarrel
		10 = cc_colonial.30    # Consolidation — A Petition for Unified Governance
		10 = cc_colonial.40    # Labor & Demographics — The Question of Labor
		chance_to_happen = 15
	}
}

# Scientific College periodic events — scholasticism & theology. Own cadence so they
# don't compete with the shared subject pool. cc.235 self-gates on is_protestant internally.
# Also re-offers the creation event (cc.104) while the college can still be founded — each
# event self-gates internally on whether a college exists.
cc_scientific_college_pulse = {
	trigger = {
		OR = {
			any_subject = { is_subject_type = scientific_college }
			has_estate_privilege = estate_privilege:ep_clergy_scholarly_college
		}
	}
	random_events = {
		10 = cc.104        # The Scholars Seek a Home (creation — gates on no college yet)

		10 = cc.232        # The Great Disputation
		10 = cc.233        # A Doctrine in Question
		10 = cc.234        # The Scholarly Synthesis
		10 = cc.235        # The Reformed Academy (Protestant/Reformed only)

		chance_to_happen = 18
	}
}

# Elite enclave / palatinate periodic events — nobility, prestige, autonomy & cavalry flavor.
# Own cadence so they don't compete in the shared pool. The palatinate sub-pool is deliberately
# larger (it is the bigger, more rambunctious, more prestigious subject). Events self-gate
# internally on subject type and per-subject cooldowns; cc.215 self-gates on bordering a
# non-overlord neighbor.
cc_noble_domain_pulse = {
	trigger = {
		OR = {
			any_subject = {
				OR = {
					is_subject_type = elite_enclave
					is_subject_type = palatinate
				}
			}
			has_estate_privilege = estate_privilege:ep_noble_elite_enclave
		}
	}
	random_events = {
		# Creation event — re-offered while the enclave can still be founded (gates internally)
		10 = cc.112        # A Seat for the Nobility (creation)

		# One-shot creation / elevation confirmations (fire_only_once, self-disable)
		10 = cc.164        # The Enclave Established
		10 = cc.165        # The Palatinate Elevated

		# Elite enclave pool
		10 = cc.160        # The Enclave Lord's Cultural Patronage
		10 = cc.162        # The Grand Tournament
		10 = cc.163        # A Noble Feud
		10 = cc.194        # Nobles Compete for Court Positions

		# Palatinate pool (larger — rambunctious & prestigious)
		10 = cc.161        # The Palatinate Demands Recognition
		10 = cc.210        # Dynastic Recognition
		10 = cc.211        # The Palatinate Tournament
		10 = cc.213        # The Palatinate Answers the Call
		10 = cc.214        # The Count Palatine Overreaches
		10 = cc.215        # The Palatinate Provokes a Neighbor

		chance_to_happen = 20
	}
}
