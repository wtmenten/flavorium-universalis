# COURT RIVALRY SUB-PULSES (Pillar 01)
###############################################################################

# Biyearly rivalry accumulation — builds or decays cc_court_rivalry variable;
# applies court_tension modifier when threshold is crossed.
cc_cabinet_biyearly_rivalry = {
	trigger = {
		any_cabinet_character = { in_cabinet = yes }
	}
	effect = {
		if = {
			limit = {
				OR = {
					AND = {
						any_cabinet_character = { has_trait = royal_absolutist }
						any_cabinet_character = { has_trait = liberal_reformer }
					}
					AND = {
						any_cabinet_character = { has_trait = hawk_minister }
						any_cabinet_character = { has_trait = peacemaker_counselor }
					}
					AND = {
						any_cabinet_character = { has_trait = efficiency_administrator }
						any_cabinet_character = { has_trait = regional_magnate }
					}
					AND = {
						any_cabinet_character = { has_trait = noble_champion }
						any_cabinet_character = { has_trait = merchant_prince }
					}
					AND = {
						any_cabinet_character = { has_trait = expansionist_advisor }
						any_cabinet_character = { has_trait = consolidation_minister }
					}
				}
			}
			# Opposing pair present — accumulate rivalry score
			if = {
				limit = { NOT = { has_variable = cc_court_rivalry } }
				set_variable = { name = cc_court_rivalry value = 1 }
			}
			else_if = {
				limit = { var:cc_court_rivalry < 5 }
				change_variable = { name = cc_court_rivalry add = 1 }
			}
			# Apply court_tension when score reaches 3+
			if = {
				limit = { var:cc_court_rivalry >= 3 }
				add_country_modifier = {
					modifier = court_tension
					years = 3
					mode = set_to_largest_and_extend
				}
			}
		}
		else = {
			# No opposing pair — rivalry decays
			if = {
				limit = { has_variable = cc_court_rivalry  var:cc_court_rivalry > 0 }
				change_variable = { name = cc_court_rivalry add = -1 }
			}
		}
	}
}
