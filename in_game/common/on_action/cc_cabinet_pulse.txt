# Monthly war flag — sets cc_recently_at_war (7-month duration) while country is at war.
# Wealth hoarding events check NOT = { has_variable = cc_recently_at_war } in their triggers.

	trigger = { at_war = yes }
	effect = {
		set_variable = { name = cc_recently_at_war  value = yes  months = 7 }
	}
}
