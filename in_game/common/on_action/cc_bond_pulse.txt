# YEARLY COUNTRY PULSE — Bond Event Triggers
# Chain events use random_events (sample_count = 1) so only 1 fires per subject
# type per year. Monitor, status reveals, and AoR payoffs stay in events = {}.
###############################################################################

yearly_country_pulse = {
	events = {
		cc_bonds.1		# Hidden monitor (10-year cycle)
	}

	on_actions = {
		cc_bonds_backfill_pulse
		cc_bonds_reveal_pulse
		cc_bonds_colonial_pulse

		cc_bonds_aor_pulse
	}
}

###############################################################################
# AGE OF REVOLUTIONS PAYOFF DISPATCH
#
# The payoffs used to sit in the events = {} list above with no mutual cooldown,
# so an empire with a dozen tracked subjects could resolve most of its subject
# relationships in the single year the age turned over. They now run through one
# dispatcher that resolves at most one payoff per window.
#
# The dispatcher also does the selection and the grading, rather than each event
# doing its own: a payoff's triggered_desc is evaluated BEFORE its immediate
# block runs, so scope:bond_subj and cc_bond_aor_tier have to already exist by
# the time the event is triggered.
#
# Hostile poles are attempted ahead of loyal ones so the empire's worst
# relationship is the one that surfaces first. Within a pole the candidate is
# picked at random rather than by severity, because ordered_key_in_variable_map's
# order_by is evaluated in the key's scope and the score maps live on the
# overlord.
#
# cc_bond_aor_cd doubles as the "something already fired this pass" guard, so
# each try self-guards and no separate latch is needed.
###############################################################################

cc_bonds_aor_pulse = {
	effect = {
		if = {
			limit = {
				has_game_rule = cc_subject_bonds_on
				current_age = age_6_revolutions
			}

			# One-time arming. Puts a jittered 5-15 year delay on the first payoff so
			# they do not all land on the age boundary in 1737.
			if = {
				limit = { NOT = { has_variable = cc_bond_aor_armed } }
				set_variable = { name = cc_bond_aor_armed  value = yes }
				random_list = {
					33 = { set_variable = { name = cc_bond_aor_cd  value = yes  years = 5 } }
					34 = { set_variable = { name = cc_bond_aor_cd  value = yes  years = 10 } }
					33 = { set_variable = { name = cc_bond_aor_cd  value = yes  years = 15 } }
				}
			}

			# --- Hostile poles ---
			cc_bonds_aor_try_hostile = { ID = 91   TYPE = colonial_nation         DIMA = eco  DIMB = pol  MAX = -2 }
			cc_bonds_aor_try_hostile = { ID = 93   TYPE = provincial_governorate  DIMA = pol  DIMB = eco  MAX = -2 }
			cc_bonds_aor_try_hostile = { ID = 95   TYPE = palatinate              DIMA = pol  DIMB = per  MAX = -2 }
			cc_bonds_aor_try_hostile = { ID = 97   TYPE = military_march          DIMA = mil  DIMB = per  MAX = -2 }
			cc_bonds_aor_try_hostile = { ID = 99   TYPE = crown_dependency        DIMA = per  DIMB = per  MAX = -3 }
			cc_bonds_aor_try_hostile = { ID = 151  TYPE = puppet_state            DIMA = pol  DIMB = per  MAX = -2 }
			cc_bonds_aor_try_hostile = { ID = 153  TYPE = imperial_council_member DIMA = pol  DIMB = per  MAX = -2 }
			cc_bonds_aor_try_hostile = { ID = 141  TYPE = equal_federal_member    DIMA = pol  DIMB = per  MAX = -3 }
			cc_bonds_aor_try_hostile = { ID = 143  TYPE = false_federal_member    DIMA = eco  DIMB = pol  MAX = -2 }
			cc_bonds_aor_try_cul_hostile = { ID = 155 }

			# --- Loyal poles ---
			cc_bonds_aor_try_loyal = { ID = 90   TYPE = colonial_nation         DIMA = eco  DIMB = pol  MIN = 2 }
			cc_bonds_aor_try_loyal = { ID = 92   TYPE = provincial_governorate  DIMA = eco  DIMB = pol  MIN = 2 }
			cc_bonds_aor_try_loyal = { ID = 94   TYPE = palatinate              DIMA = mil  DIMB = per  MIN = 2 }
			cc_bonds_aor_try_loyal = { ID = 96   TYPE = military_march          DIMA = mil  DIMB = per  MIN = 2 }
			cc_bonds_aor_try_loyal = { ID = 98   TYPE = crown_dependency        DIMA = per  DIMB = per  MIN = 2 }
			cc_bonds_aor_try_loyal = { ID = 150  TYPE = puppet_state            DIMA = pol  DIMB = per  MIN = 2 }
			cc_bonds_aor_try_loyal = { ID = 152  TYPE = imperial_council_member DIMA = pol  DIMB = per  MIN = 2 }
			cc_bonds_aor_try_loyal = { ID = 140  TYPE = equal_federal_member    DIMA = pol  DIMB = per  MIN = 3 }
			cc_bonds_aor_try_loyal = { ID = 142  TYPE = lesser_federal_member   DIMA = pol  DIMB = eco  MIN = 2 }
			cc_bonds_aor_try_loyal = { ID = 145  TYPE = client_state            DIMA = eco  DIMB = pol  MIN = 2 }
			cc_bonds_aor_try_loyal = { ID = 145  TYPE = shadow_state            DIMA = eco  DIMB = pol  MIN = 2 }
			cc_bonds_aor_try_loyal = { ID = 145  TYPE = chartered_company       DIMA = eco  DIMB = pol  MIN = 2 }
			cc_bonds_aor_try_cul_loyal = { ID = 154 }
			# Last, and at a higher bar than cc_bonds.90: the colony that has earned a
			# seat rather than merely stayed quiet.
			cc_bonds_aor_try_loyal = { ID = 144  TYPE = colonial_nation         DIMA = pol  DIMB = per  MIN = 3 }
		}
	}
}

# Status reveal events — pre-selects scope:bond_subj before firing so triggered_desc
# triggers can reference it (triggered_desc evaluates before immediate runs).
cc_bonds_reveal_pulse = {
	effect = {
		# cc_bonds.5 — generic subject reveal
		if = {
			limit = {
				has_game_rule = cc_subject_bonds_on
				NOT = { has_variable = cc_bond_5_cd }
				any_subject = { has_variable = cc_bond_initialized num_locations > 1 }
			}
			random_subject = {
				limit = { has_variable = cc_bond_initialized num_locations > 1 }
				save_scope_as = bond_subj
			}
			trigger_event_silently = { id = cc_bonds.5 }
		}
		# cc_bonds.6 — colonial nation reveal
		if = {
			limit = {
				has_game_rule = cc_subject_bonds_on
				NOT = { has_variable = cc_bond_6_cd }
				any_subject = { is_subject_type = colonial_nation  has_variable = cc_bond_initialized }
			}
			random_subject = {
				limit = { is_subject_type = colonial_nation  has_variable = cc_bond_initialized }
				save_scope_as = bond_subj
			}
			trigger_event_silently = { id = cc_bonds.6 }
		}
		# cc_bonds.7 — palatinate/march reveal
		if = {
			limit = {
				has_game_rule = cc_subject_bonds_on
				NOT = { has_variable = cc_bond_7_cd }
				any_subject = {
					OR = { is_subject_type = palatinate  is_subject_type = military_march }
					has_variable = cc_bond_initialized
				}
			}
			random_subject = {
				limit = {
					OR = { is_subject_type = palatinate  is_subject_type = military_march }
					has_variable = cc_bond_initialized
				}
				save_scope_as = bond_subj
			}
			trigger_event_silently = { id = cc_bonds.7 }
		}
		# cc_bonds.8 — crown dependency reveal
		if = {
			limit = {
				has_game_rule = cc_subject_bonds_on
				NOT = { has_variable = cc_bond_8_cd }
				any_subject = { is_subject_type = crown_dependency  has_variable = cc_bond_initialized }
			}
			random_subject = {
				limit = { is_subject_type = crown_dependency  has_variable = cc_bond_initialized }
				save_scope_as = bond_subj
			}
			trigger_event_silently = { id = cc_bonds.8 }
		}
		# cc_bonds.9 — provincial governorate reveal
		if = {
			limit = {
				has_game_rule = cc_subject_bonds_on
				NOT = { has_variable = cc_bond_9_cd }
				any_subject = { is_subject_type = provincial_governorate  has_variable = cc_bond_initialized }
			}
			random_subject = {
				limit = { is_subject_type = provincial_governorate  has_variable = cc_bond_initialized }
				save_scope_as = bond_subj
			}
			trigger_event_silently = { id = cc_bonds.9 }
		}
	}
}

###############################################################################
# SUBJECT ACQUISITION — Initialize bond tracking
# on_dependency_gained: root = new subject, scope:overlord = new overlord
###############################################################################

on_dependency_gained = {
	on_actions = {
		cc_bonds_on_dependency_gained
		cc_bonds_first_subject_on_action
	}
}

# Fires a major event the first time each broad subject category appears in the game.
# Uses global variables so the event fires at most once per game per subject group.
# root = new subject, scope:overlord = new overlord.
cc_bonds_first_subject_on_action = {
	effect = {
		# First colonial nation in this game
		if = {
			limit = {
				root = { is_subject_type = colonial_nation }
				NOT = { has_global_variable = cc_first_colonial_nation_seen }
			}
			set_global_variable = cc_first_colonial_nation_seen
			scope:overlord = {
				set_variable = { name = cc_first_subject_ref  value = root }
				trigger_event_silently = { id = cc_bonds.2 }
			}
		}
		# First military march in this game
		else_if = {
			limit = {
				root = { is_subject_type = military_march }
				NOT = { has_global_variable = cc_first_military_march_seen }
			}
			set_global_variable = cc_first_military_march_seen
			scope:overlord = {
				set_variable = { name = cc_first_subject_ref  value = root }
				trigger_event_silently = { id = cc_bonds.3 }
			}
		}
		# First palatinate, puppet, crown dependency, or provincial governorate in this game
		else_if = {
			limit = {
				root = {
					OR = {
						is_subject_type = palatinate
						is_subject_type = puppet_state
						is_subject_type = crown_dependency
						is_subject_type = provincial_governorate
					}
				}
				NOT = { has_global_variable = cc_first_formal_subject_seen }
			}
			set_global_variable = cc_first_formal_subject_seen
			scope:overlord = {
				set_variable = { name = cc_first_subject_ref  value = root }
				trigger_event_silently = { id = cc_bonds.4 }
			}
		}
	}
}

cc_bonds_on_dependency_gained = {
	effect = {
		# root = the newly-gained subject. Init logic lives in the shared
		# cc_bonds_initialize_subject effect (also used by the backfills below).
		if = {
			limit = { has_game_rule = cc_subject_bonds_on }
			cc_bonds_initialize_subject = yes
		}
	}
}

###############################################################################
# SAFETY BACKFILL — initialize any subject that predates bond tracking
# Catches subjects gained while the game rule was off (then toggled on); the
# registry-driven monitor only touches already-initialized subjects, so it
# cannot self-heal these. Runs from yearly_country_pulse (root = overlord).
###############################################################################
cc_bonds_backfill_pulse = {
	effect = {
		if = {
			limit = {
				has_game_rule = cc_subject_bonds_on
				any_subject = { NOT = { has_variable = cc_bond_initialized } }
			}
			every_subject = {
				limit = { NOT = { has_variable = cc_bond_initialized } }
				cc_bonds_initialize_subject = yes
			}
		}
	}
}

###############################################################################
# SUBJECT RELEASE — Freeze scores for old overlord's AoR use
# on_becoming_free: root = freed country, scope:overlord = former overlord
###############################################################################
on_becoming_free = {
	on_actions = {
		cc_bonds_on_becoming_free
	}
}
cc_bonds_on_becoming_free = {
	effect = {
		if = {
			limit = {
				has_game_rule = cc_subject_bonds_on
				has_variable = cc_bond_initialized
			}
			set_variable = { name = cc_bond_eco_frozen  value = var:cc_bond_eco }
			set_variable = { name = cc_bond_mil_frozen  value = var:cc_bond_mil }
			set_variable = { name = cc_bond_pol_frozen  value = var:cc_bond_pol }
			set_variable = { name = cc_bond_cul_frozen  value = var:cc_bond_cul }
			set_variable = { name = cc_bond_per_frozen  value = var:cc_bond_per }
			remove_variable = cc_bond_eco
			remove_variable = cc_bond_mil
			remove_variable = cc_bond_pol
			remove_variable = cc_bond_cul
			remove_variable = cc_bond_per
			remove_variable = cc_bond_initialized
		}
	}
}

###############################################################################
# WAR — Call to Arms
# scope:actor = caller, scope:recipient = callee
###############################################################################

on_accepted_call_to_arms = {
	on_actions = {
		cc_bonds_on_accepted_call_to_arms
	}
}
cc_bonds_on_accepted_call_to_arms = {
	effect = {
		if = {
			limit = { has_game_rule = cc_subject_bonds_on }

			# CASE A: Overlord calls subject to war
			if = {
				limit = {
					scope:recipient = {
						is_subject_of = scope:actor
						has_variable = cc_bond_initialized
					}
				}
				scope:recipient = {
					change_variable = { name = cc_bond_war_calls  add = 1 }
				}
				if = {
					limit = { scope:war = { is_a_defender = scope:actor } }
					scope:recipient = {
						change_variable = { name = cc_bond_mil  add = -1 }
						clamp_variable = { name = cc_bond_mil  max = 5  min = -5 }
						change_variable = { name = cc_bond_per  add = 1 }
						clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
					}
				}
				else = {
					scope:recipient = {
						change_variable = { name = cc_bond_mil  add = -1 }
						clamp_variable = { name = cc_bond_mil  max = 5  min = -5 }
					}
					if = {
						limit = { scope:war = { is_no_cb_war = yes } }
						scope:recipient = {
							change_variable = { name = cc_bond_unjust_wars  add = 1 }
						}
					}
				}
			}

			# CASE B: Subject calls overlord for help — overlord accepted
			if = {
				limit = {
					scope:actor = {
						is_subject_of = scope:recipient
						has_variable = cc_bond_initialized
					}
				}
				scope:actor = {
					change_variable = { name = cc_bond_mil  add = 1 }
					clamp_variable = { name = cc_bond_mil  max = 5  min = -5 }
					change_variable = { name = cc_bond_per  add = 1 }
					clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
				}
			}
		}
	}
}

###############################################################################
# GIFT SENT — Economic / personal dimension signal
# scope:actor = sender, scope:recipient = receiver, scope:target = gold amount
# A 4-year per-subject cooldown (cc_bond_gift_cd) stops gift-spam from pinning the
# score; the monitor's per-cycle decay caps the long-run resting value.
###############################################################################

on_gift_sent = {
	on_actions = {
		cc_bonds_on_gift_sent
	}
}
cc_bonds_on_gift_sent = {
	effect = {
		if = {
			limit = { has_game_rule = cc_subject_bonds_on }

			# Overlord gifts its subject → economic goodwill
			if = {
				limit = {
					scope:recipient = {
						is_subject_of = scope:actor
						has_variable = cc_bond_initialized
						NOT = { has_variable = cc_bond_gift_cd }
					}
				}
				scope:recipient = {
					set_variable = { name = cc_bond_gift_cd  value = yes  years = 4 }
					change_variable = { name = cc_bond_eco  add = 1 }
					clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
				}
			}

			# Subject gifts its overlord → personal goodwill
			if = {
				limit = {
					scope:actor = {
						is_subject_of = scope:recipient
						has_variable = cc_bond_initialized
						NOT = { has_variable = cc_bond_gift_cd }
					}
				}
				scope:actor = {
					set_variable = { name = cc_bond_gift_cd  value = yes  years = 4 }
					change_variable = { name = cc_bond_per  add = 1 }
					clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
				}
			}
		}
	}
}

# WAR — Subject Joins Against Overlord (betrayal, not via call to arms)
###############################################################################

on_join_war = {
	on_actions = {
		cc_bonds_on_join_war
	}
}
cc_bonds_on_join_war = {
	effect = {
		if = {
			limit = {
				has_game_rule = cc_subject_bonds_on
				has_variable = cc_bond_initialized
				is_subject = yes
				exists = overlord
				scope:war = {
					is_on_opposite_sides = {
						country = root
						target = root.overlord
					}
				}
			}
			change_variable = { name = cc_bond_mil  add = -1 }
			clamp_variable = { name = cc_bond_mil  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = -1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_betrayal_count  add = 1 }
		}
	}
}

# RELIGION CHANGE — Cultural dimension signal
###############################################################################

on_religion_changed = {
	on_actions = {
		cc_bonds_on_religion_changed
	}
}
cc_bonds_on_religion_changed = {
	effect = {
		if = {
			limit = {
				has_game_rule = cc_subject_bonds_on
				is_subject = yes
				has_variable = cc_bond_initialized
			}
			if = {
				limit = {
					religion = overlord.religion
					NOT = { scope:old_religion = overlord.religion }
				}
				if = {
					limit = { var:cc_bond_cul <= -2 }
					change_variable = { name = cc_bond_cul  add = -1 }
					clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
					set_variable = { name = cc_bond_mem_religion_forced  value = yes }
				}
				else = {
					change_variable = { name = cc_bond_cul  add = 1 }
					clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
				}
			}
			else_if = {
				limit = {
					NOT = { religion = overlord.religion }
					scope:old_religion = overlord.religion
				}
				if = {
					limit = { var:cc_bond_cul >= 2 }
					change_variable = { name = cc_bond_cul  add = 1 }
					clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
				}
				else = {
					change_variable = { name = cc_bond_cul  add = -1 }
					clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
					change_variable = { name = cc_bond_per  add = -1 }
					clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
				}
			}
			else = {
				change_variable = { name = cc_bond_cul  add = -1 }
				clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
				change_variable = { name = cc_bond_per  add = -1 }
				clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
			}
		}
	}
}

###############################################################################
# ROYAL MARRIAGE — Personal dimension signal
###############################################################################

on_royal_marriage = {
	on_actions = {
		cc_bonds_on_royal_marriage
	}
}
cc_bonds_on_royal_marriage = {
	effect = {
		if = {
			limit = { has_game_rule = cc_subject_bonds_on }

			if = {
				limit = {
					scope:actor = {
						is_subject = yes
						has_variable = cc_bond_initialized
						is_subject_of = scope:recipient
					}
				}
				scope:actor = {
					change_variable = { name = cc_bond_per  add = 1 }
					clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
				}
			}
			if = {
				limit = {
					scope:recipient = {
						is_subject = yes
						has_variable = cc_bond_initialized
						is_subject_of = scope:actor
					}
				}
				scope:recipient = {
					change_variable = { name = cc_bond_per  add = 1 }
					clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
				}
			}

			if = {
				limit = {
					scope:actor = {
						is_subject = yes
						has_variable = cc_bond_initialized
						NOT = { is_subject_of = scope:recipient }
					}
					scope:recipient = {
						NOT = { is_subject_of = scope:actor.overlord }
						NOT = { this = scope:actor.overlord }
					}
				}
				scope:actor = {
					change_variable = { name = cc_bond_pol  add = 1 }
					clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
					change_variable = { name = cc_bond_per  add = -1 }
					clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
				}
			}
			if = {
				limit = {
					scope:recipient = {
						is_subject = yes
						has_variable = cc_bond_initialized
						NOT = { is_subject_of = scope:actor }
					}
					scope:actor = {
						NOT = { is_subject_of = scope:recipient.overlord }
						NOT = { this = scope:recipient.overlord }
					}
				}
				scope:recipient = {
					change_variable = { name = cc_bond_pol  add = 1 }
					clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
					change_variable = { name = cc_bond_per  add = -1 }
					clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
				}
			}
		}
	}
}
