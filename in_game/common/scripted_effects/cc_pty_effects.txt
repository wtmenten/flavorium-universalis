###############################################################################
# Flavorium Universalis - Factions: scripted effects
#
# Country scope unless noted. Per-archetype dispatch (cc_pty_with_type and friends) is
# generated into cc_pty_catalogue_effects.txt by tools/generate_factions.py.
#
# PER-SLOT COUNTRY VARIABLES (N = 1..4). A slot with no cc_pty_N_type is empty, and
# nothing reads its other variables.
#   cc_pty_N_type         archetype id
#   cc_pty_N_eraw         raw elected-seat weight from the map (x0.6 while attacked)
#   cc_pty_N_app / _elected / _seats   appointed, elected and total seats   (panel)
#   cc_pty_N_cum          seats of slots 1..N together; cc_pty_N_start = slots 1..N-1
#   cc_pty_r<T>_lo / _hi  seat range held by archetype T, for the seat grid (panel)
#   cc_pty_N_power        share of the chamber's seats, 0..100           (panel)
#   cc_pty_N_leader       the party leader (character)                   (panel)
#   cc_pty_N_members      seated ministers of the party                  (panel)
#   cc_pty_N_c1 / _c2 / _c3   the party's three concerns this pass        (panel)
#   cc_pty_N_mood         their sum, roughly -45..45
#   cc_pty_N_target       what the party expects                         (panel)
#   cc_pty_N_t_cab / _t_head / _t_ruler / _t_comp   the target's other terms (panel)
#   cc_pty_N_sz_* / _dz_* / _fx<j>   aspect sizes and effect totals (see PANEL DISPLAY)
#   cc_pty_N_sat          satisfaction, -100..100                        (panel)
#   cc_pty_N_agenda       0 none, 1..3 which agenda                      (panel)
#   cc_pty_N_agenda_due   timed: the agenda's deadline
#   cc_pty_N_agenda_cd    timed: no new agenda until it lapses
#   cc_pty_N_neglect      passes spent angry and powerful (must-placate parties)
#   cc_pty_N_esc          escalation step 0..3                           (panel)
#   cc_pty_N_outcome_cd   timed: no second outcome for twenty years
#   cc_pty_N_suppressed   timed: attacked, strength x 0.6
#   cc_pty_N_bribe_cd / _attack_cd / _favour_cd   timed action cooldowns
#
# OTHER COUNTRY VARIABLES
#   cc_pty_active, cc_pty_sat_cd (timed), cc_pty_realign_cd (timed), cc_pty_popup_cd (timed)
#   cc_pty_seat_total, cc_pty_es_<estate> (raw seats; set by cc_pty_count_seats)
#   cc_pty_chamber (chamber size, 20..84), cc_pty_top_slot (agenda shown at the top)
#   cc_pty_election_cd (timed)
#   cc_pty_comp (0 mixed, 1 stacked, 2 balanced), cc_pty_cabinet_slot (the stacked party, 0 if none)
#   cc_pty_dominant (slot of the strongest party, 0 if none has 30%)
#   cc_pty_termlimits, cc_pty_liberal_thought, cc_pty_liberal_formed
#   cc_pty_coup_armed / cc_pty_uprising_armed (read by the disasters)
#   cc_pty_rivalry_feed (timed; stops the main mod's rivalry decay)
#   cc_pty_t<T>_*   aspect sizes by archetype, for the panel (see PANEL DISPLAY)
#   cc_pty_mem_won / _lost / _war / _law / _bankrupt   timed memories read by concerns
#                   (set in cc_pty_on_actions.txt)
#   cc_pty_event_slot / _type / _agenda, cc_pty_defect_from_type / _to_type (event loc)
#   scratch: cc_pty_tmp, cc_pty_total, cc_pty_seats, cc_pty_best, cc_pty_best_type,
#            cc_pty_best_slot, cc_pty_sc_T, cc_pty_picked_T
#
# CHARACTER VARIABLES
#   cc_pty_aff (archetype id) plus a cc_pty_member_<key> badge modifier
#   cc_pty_years (years seated), cc_pty_term_served
#
# change_variable/clamp_variable on an unset variable errors, so every slot variable is
# set by cc_pty_init_slot before anything changes it.
###############################################################################


###############################################################################
# LIFECYCLE
###############################################################################

cc_pty_init_country = {
	set_variable = { name = cc_pty_active value = yes }
	cc_pty_publish_rules = yes
	set_variable = { name = cc_pty_comp value = 0 }
	set_variable = { name = cc_pty_dominant value = 0 }
	set_variable = { name = cc_pty_cabinet_slot value = 0 }

	# The first parties come from the same choice a realignment makes, so a country starts
	# with parties that fit its government, age and estates.
	cc_pty_realign = yes
	remove_variable = cc_pty_realign_changed
	cc_pty_count_seats = yes

	save_scope_as = cc_pty_country
	every_cabinet_character = {
		limit = { NOT = { has_variable = cc_pty_aff } }
		cc_pty_assign_affiliation = yes
	}
	cc_pty_recompute_all = yes
	cc_pty_update_sat = { SLOT = 1 STEP = 1 }
	cc_pty_update_sat = { SLOT = 2 STEP = 1 }
	cc_pty_update_sat = { SLOT = 3 STEP = 1 }
	cc_pty_update_sat = { SLOT = 4 STEP = 1 }

	# Jittered first pass so tracked countries do not all recompute in the same year.
	random_list = {
		1 = { set_variable = { name = cc_pty_sat_cd value = yes months = 6 } }
		1 = { set_variable = { name = cc_pty_sat_cd value = yes months = 12 } }
		1 = { set_variable = { name = cc_pty_sat_cd value = yes months = 18 } }
	}

	# Leaders again a day later. At game start the player's country is not yet marked as a
	# player, so cc_pty_update_leaders above generated no leaders for it.
	trigger_event_silently = { id = cc_pty.6 days = 1 }
}

# The Court Ledger hides its cabinet experience roster and tabs when that rule is off, and
# GUI cannot read a game rule, so the factions side publishes it. Rules are fixed for a
# campaign; this runs at setup and yearly so a save that predates it catches up.
cc_pty_publish_rules = {
	if = {
		limit = { NOT = { has_game_rule = cc_cabinet_xp_on } }
		set_variable = { name = cc_xp_off value = yes }
	}
	else = {
		remove_variable = cc_xp_off
	}
}

	set_variable = { name = cc_pty_$SLOT$_type value = $TYPE$ }
	set_variable = { name = cc_pty_$SLOT$_eraw value = 1 }
	set_variable = { name = cc_pty_$SLOT$_app value = 0 }
	set_variable = { name = cc_pty_$SLOT$_elected value = 0 }
	set_variable = { name = cc_pty_$SLOT$_seats value = 0 }
	set_variable = { name = cc_pty_$SLOT$_power value = 25 }
	set_variable = { name = cc_pty_$SLOT$_members value = 0 }
	set_variable = { name = cc_pty_$SLOT$_mood value = 0 }
	set_variable = { name = cc_pty_$SLOT$_sat value = 0 }
	set_variable = { name = cc_pty_$SLOT$_target value = 0 }
	set_variable = { name = cc_pty_$SLOT$_agenda value = 0 }
	set_variable = { name = cc_pty_$SLOT$_neglect value = 0 }
	set_variable = { name = cc_pty_$SLOT$_esc value = 0 }
	set_variable = { name = cc_pty_$SLOT$_c1 value = 0 }
	set_variable = { name = cc_pty_$SLOT$_c2 value = 0 }
	set_variable = { name = cc_pty_$SLOT$_c3 value = 0 }
	set_variable = { name = cc_pty_$SLOT$_t_cab value = 0 }
	set_variable = { name = cc_pty_$SLOT$_t_head value = 0 }
	set_variable = { name = cc_pty_$SLOT$_t_ruler value = 0 }
	set_variable = { name = cc_pty_$SLOT$_t_comp value = 0 }
	cc_pty_init_display = { SLOT = $SLOT$ }

	random_list = {
		1 = { set_variable = { name = cc_pty_$SLOT$_agenda_cd value = yes years = 3 } }
		1 = { set_variable = { name = cc_pty_$SLOT$_agenda_cd value = yes years = 5 } }
		1 = { set_variable = { name = cc_pty_$SLOT$_agenda_cd value = yes years = 8 } }
	}
}

# Called through cc_pty_with_type_tk, so TYPE and KEY are the departing archetype.
cc_pty_clear_type_effects = {
	cc_pty_clear_country_mood = { KEY = $KEY$ }
	cc_pty_clear_standing = { KEY = $KEY$ }
	if = {
		limit = { has_country_modifier = cc_pty_push_$KEY$ }
		remove_country_modifier = cc_pty_push_$KEY$
	}
	# All three sets: per-key removal would name modifiers that only exist for the
	# must-placate archetypes. Any still due are reapplied at the next pass.
	cc_pty_clear_escalation_mods = yes
	cc_pty_clear_type_display = { TYPE = $TYPE$ }
	every_cabinet_character = {
		limit = { cc_pty_is_member = { TYPE = $TYPE$ } }
		cc_pty_clear_member_mood = yes
	}
}

###############################################################################
# REALIGNMENT
#
# Every fifteen to twenty-five years the court's four parties are chosen again: every
# archetype that is eligible here scores its base strength plus government and age
# bonuses (cc_pty_realign_score_type_T), incumbents get +20, and the best four hold the
# slots. Incumbents keep their slot and their satisfaction; a newcomer takes a vacated
# slot and starts at zero.
###############################################################################

cc_pty_realign = {
	save_scope_as = cc_pty_country
	cc_pty_score_all = yes
	cc_pty_pick_round = yes
	cc_pty_pick_round = yes
	cc_pty_pick_round = yes
	cc_pty_pick_round = yes

	cc_pty_vacate_if_unpicked = { SLOT = 1 }
	cc_pty_vacate_if_unpicked = { SLOT = 2 }
	cc_pty_vacate_if_unpicked = { SLOT = 3 }
	cc_pty_vacate_if_unpicked = { SLOT = 4 }
	cc_pty_place_new = yes

	cc_pty_clear_realign_scratch = yes
	remove_variable = cc_pty_best
	remove_variable = cc_pty_best_type

	random_list = {
		1 = { set_variable = { name = cc_pty_realign_cd value = yes years = 15 } }
		1 = { set_variable = { name = cc_pty_realign_cd value = yes years = 20 } }
		1 = { set_variable = { name = cc_pty_realign_cd value = yes years = 25 } }
	}
}

cc_pty_pick_round = {
	set_variable = { name = cc_pty_best value = -999 }
	set_variable = { name = cc_pty_best_type value = 0 }
	cc_pty_consider_all = yes
	cc_pty_mark_picked = yes
}

cc_pty_consider = {
	if = {
		limit = {
			NOT = { has_variable = cc_pty_picked_$TYPE$ }
			has_variable = cc_pty_sc_$TYPE$
		}
		if = {
			limit = { var:cc_pty_sc_$TYPE$ > var:cc_pty_best }
			set_variable = { name = cc_pty_best value = var:cc_pty_sc_$TYPE$ }
			set_variable = { name = cc_pty_best_type value = $TYPE$ }
		}
	}
}

cc_pty_vacate_if_unpicked = {
	if = {
		limit = {
			has_variable = cc_pty_$SLOT$_type
			NOT = { cc_pty_slot_type_picked = { SLOT = $SLOT$ } }
		}
		cc_pty_vacate_slot = { SLOT = $SLOT$ }
		set_variable = { name = cc_pty_realign_changed value = yes }
	}
}

cc_pty_place_type = {
	if = {
		limit = { NOT = { has_variable = cc_pty_1_type } }
		cc_pty_init_slot = { SLOT = 1 TYPE = $TYPE$ }
	}
	else_if = {
		limit = { NOT = { has_variable = cc_pty_2_type } }
		cc_pty_init_slot = { SLOT = 2 TYPE = $TYPE$ }
	}
	else_if = {
		limit = { NOT = { has_variable = cc_pty_3_type } }
		cc_pty_init_slot = { SLOT = 3 TYPE = $TYPE$ }
	}
	else_if = {
		limit = { NOT = { has_variable = cc_pty_4_type } }
		cc_pty_init_slot = { SLOT = 4 TYPE = $TYPE$ }
	}
	set_variable = { name = cc_pty_realign_changed value = yes }
}


###############################################################################
# AFFILIATION (character scope)
#
# cc_pty_assign_affiliation is generated (trait, then estate, then an even draw).
###############################################################################

# Country scope, player realms. Give every adult at court a party, not only ministers, so
# candidates show their party (badge modifier, in the character tooltip) before they are
# appointed. Also reassigns anyone whose party left at a realignment. Not run for AI realms:
# their ministers are affiliated when seated, and nobody inspects their courtiers.
cc_pty_affiliate_court = {
	if = {
		limit = { is_ai = no }
		save_scope_as = cc_pty_country
		every_character = {
			limit = {
				is_adult = yes
				OR = {
					NOT = { has_variable = cc_pty_aff }
					NOT = { cc_pty_aff_active = yes }
				}
			}
			cc_pty_assign_affiliation = yes
		}
	}
}

cc_pty_join = {
	cc_pty_clear_member_badge = yes
	cc_pty_clear_member_mood = yes
	set_variable = { name = cc_pty_aff value = $TYPE$ }
	add_character_modifier = { modifier = cc_pty_member_$KEY$ years = -1 mode = replace }
}


###############################################################################
# POWER: THE CHAMBER
#
# A party's power is its share of the chamber's seats.
#
# cc_pty_count_seats is the expensive part: one pass over owned locations storing the
# total raw seats and each estate's share of them. It runs at reviews, realignments,
# elections and setup, never on a cabinet change.
#
# cc_pty_recompute_all is cheap and runs on every cabinet change:
#   1. chamber size N from the raw total (cc_pty_chamber_size, 20..84)
#   2. appointed seats: 2 per seated minister, never more than half the chamber; if the
#      cabinet would take more, every party's appointed seats are scaled down (floored)
#   3. elected seats: the rest, shared by each party's raw seats (its estate weights,
#      x0.6 while attacked), floored, with the remainder going to the largest party
#   4. seats = appointed + elected, power = seats / N, and cumulative counts for the grid
###############################################################################

cc_pty_count_seats = {
	set_variable = { name = cc_pty_seat_total value = cc_pty_seat_total_value }
	if = {
		limit = { var:cc_pty_seat_total < 1 }
		set_variable = { name = cc_pty_seat_total value = 1 }
	}
	set_variable = { name = cc_pty_es_crown value = cc_pty_es_crown_value }
	set_variable = { name = cc_pty_es_nobles value = cc_pty_es_nobles_value }
	set_variable = { name = cc_pty_es_burghers value = cc_pty_es_burghers_value }
	set_variable = { name = cc_pty_es_clergy value = cc_pty_es_clergy_value }
	set_variable = { name = cc_pty_es_peasants value = cc_pty_es_peasants_value }
}

cc_pty_recompute_all = {
	if = {
		limit = { NOT = { has_variable = cc_pty_seat_total } }
		cc_pty_count_seats = yes
	}
	set_variable = { name = cc_pty_chamber value = cc_pty_chamber_size }

	cc_pty_with_type_st = { SLOT = 1 EFFECT = cc_pty_read_type }
	cc_pty_with_type_st = { SLOT = 2 EFFECT = cc_pty_read_type }
	cc_pty_with_type_st = { SLOT = 3 EFFECT = cc_pty_read_type }
	cc_pty_with_type_st = { SLOT = 4 EFFECT = cc_pty_read_type }

	# ---- Appointed seats ----
	set_variable = { name = cc_pty_app_total value = 0 }
	cc_pty_set_appointed = { SLOT = 1 }
	cc_pty_set_appointed = { SLOT = 2 }
	cc_pty_set_appointed = { SLOT = 3 }
	cc_pty_set_appointed = { SLOT = 4 }
	set_variable = { name = cc_pty_app_cap value = var:cc_pty_chamber }
	change_variable = { name = cc_pty_app_cap multiply = 0.5 }
	if = {
		limit = { var:cc_pty_app_total > var:cc_pty_app_cap }
		cc_pty_scale_appointed = { SLOT = 1 }
		cc_pty_scale_appointed = { SLOT = 2 }
		cc_pty_scale_appointed = { SLOT = 3 }
		cc_pty_scale_appointed = { SLOT = 4 }
		set_variable = { name = cc_pty_app_total value = 0 }
		cc_pty_sum_appointed = { SLOT = 1 }
		cc_pty_sum_appointed = { SLOT = 2 }
		cc_pty_sum_appointed = { SLOT = 3 }
		cc_pty_sum_appointed = { SLOT = 4 }
	}

	# ---- Elected seats ----
	set_variable = { name = cc_pty_elected_pool value = var:cc_pty_chamber }
	change_variable = { name = cc_pty_elected_pool subtract = var:cc_pty_app_total }
	set_variable = { name = cc_pty_eraw_total value = 0 }
	cc_pty_add_eraw = { SLOT = 1 }
	cc_pty_add_eraw = { SLOT = 2 }
	cc_pty_add_eraw = { SLOT = 3 }
	cc_pty_add_eraw = { SLOT = 4 }
	set_variable = { name = cc_pty_elected_sum value = 0 }
	if = {
		limit = { var:cc_pty_eraw_total > 0 }
		cc_pty_set_elected = { SLOT = 1 }
		cc_pty_set_elected = { SLOT = 2 }
		cc_pty_set_elected = { SLOT = 3 }
		cc_pty_set_elected = { SLOT = 4 }
	}
	# The seats lost to rounding go to the party with the most raw seats.
	set_variable = { name = cc_pty_tmp value = var:cc_pty_elected_pool }
	change_variable = { name = cc_pty_tmp subtract = var:cc_pty_elected_sum }
	set_variable = { name = cc_pty_best value = -1 }
	set_variable = { name = cc_pty_best_slot value = 0 }
	cc_pty_consider_largest = { SLOT = 1 }
	cc_pty_consider_largest = { SLOT = 2 }
	cc_pty_consider_largest = { SLOT = 3 }
	cc_pty_consider_largest = { SLOT = 4 }
	cc_pty_give_leftover = { SLOT = 1 }
	cc_pty_give_leftover = { SLOT = 2 }
	cc_pty_give_leftover = { SLOT = 3 }
	cc_pty_give_leftover = { SLOT = 4 }
	remove_variable = cc_pty_best
	remove_variable = cc_pty_best_slot

	# ---- Seats, power and the grid ----
	cc_pty_finish_slot = { SLOT = 1 }
	cc_pty_finish_slot = { SLOT = 2 }
	cc_pty_finish_slot = { SLOT = 3 }
	cc_pty_finish_slot = { SLOT = 4 }
	set_variable = { name = cc_pty_1_cum value = 0 }
	set_variable = { name = cc_pty_1_start value = 0 }
	cc_pty_add_cum = { SLOT = 1 }
	set_variable = { name = cc_pty_2_cum value = var:cc_pty_1_cum }
	set_variable = { name = cc_pty_2_start value = var:cc_pty_1_cum }
	cc_pty_add_cum = { SLOT = 2 }
	set_variable = { name = cc_pty_3_cum value = var:cc_pty_2_cum }
	set_variable = { name = cc_pty_3_start value = var:cc_pty_2_cum }
	cc_pty_add_cum = { SLOT = 3 }
	set_variable = { name = cc_pty_4_cum value = var:cc_pty_3_cum }
	set_variable = { name = cc_pty_4_start value = var:cc_pty_3_cum }
	cc_pty_add_cum = { SLOT = 4 }

	# The grid draws seats by party TYPE, not by slot: a GUI texture cannot be chosen by
	# script state (custom loc and Select_CString both failed to render), so every seat holds
	# one fixed-texture icon per archetype, shown when that archetype's range covers it.
	# cc_pty_r<T>_lo < seat <= cc_pty_r<T>_hi; both 0 for archetypes the court lacks.

	set_variable = { name = cc_pty_$SLOT$_c1 value = cc_pty_concern_$TYPE$_1 }
	set_variable = { name = cc_pty_$SLOT$_c2 value = cc_pty_concern_$TYPE$_2 }
	set_variable = { name = cc_pty_$SLOT$_c3 value = cc_pty_concern_$TYPE$_3 }
	set_variable = { name = cc_pty_$SLOT$_mood value = var:cc_pty_$SLOT$_c1 }
	change_variable = { name = cc_pty_$SLOT$_mood add = var:cc_pty_$SLOT$_c2 }
	change_variable = { name = cc_pty_$SLOT$_mood add = var:cc_pty_$SLOT$_c3 }
	set_variable = { name = cc_pty_$SLOT$_eraw value = cc_pty_seat_raw_type_$TYPE$ }
	if = {
		limit = { has_variable = cc_pty_$SLOT$_suppressed }
		change_variable = { name = cc_pty_$SLOT$_eraw multiply = 0.6 }
	}
	clamp_variable = { name = cc_pty_$SLOT$_eraw min = 0.01 max = 100000 }
}

# Called through cc_pty_with_type_sk. Every party with 10% of the seats or more grants its
# standing modifier, scaled by its share (size = share / 25%). Below 10% it is marginalised
# instead. cc_pty_N_sz_standing / _dz_standing carry the size to the panel.
cc_pty_apply_standing = {
	cc_pty_clear_standing = { KEY = $KEY$ }
	if = {
		limit = { var:cc_pty_$SLOT$_power >= 10 }
		set_variable = { name = cc_pty_tmp value = var:cc_pty_$SLOT$_power }
		change_variable = { name = cc_pty_tmp divide = 25 }
		set_variable = { name = cc_pty_$SLOT$_sz_standing value = var:cc_pty_tmp }
		set_variable = { name = cc_pty_$SLOT$_dz_standing value = var:cc_pty_tmp }
		set_variable = { name = cc_pty_$SLOT$_sz_marginal value = 0 }
		add_country_modifier = {
			modifier = cc_pty_standing_$KEY$
			years = -1
			mode = replace
			size = var:cc_pty_tmp
		}
	}
	else = {
		set_variable = { name = cc_pty_$SLOT$_sz_standing value = 0 }
		set_variable = { name = cc_pty_$SLOT$_dz_standing value = 1 }
		set_variable = { name = cc_pty_$SLOT$_sz_marginal value = 1 }
		add_country_modifier = { modifier = cc_pty_marginal_$KEY$ years = -1 mode = replace }
	}
}

cc_pty_clear_standing = {
	if = {
		limit = { has_country_modifier = cc_pty_standing_$KEY$ }
		remove_country_modifier = cc_pty_standing_$KEY$
	}
	if = {
		limit = { has_country_modifier = cc_pty_marginal_$KEY$ }
		remove_country_modifier = cc_pty_marginal_$KEY$
	}
}

###############################################################################
# POLITICAL INFLUENCE FROM THE CHAMBER
#
#   leaders in office  +0.05 monthly PI for each party whose leader sits in the cabinet
#   working majority   +15% PI gain while the head of the cabinet's party holds 40%+
#   hung chamber       -10% PI gain while no party holds 30% (cc_pty_dominant = 0)
# Published for the panel: cc_pty_leaders_in (0..4), cc_pty_majority (1, -1 or 0).
###############################################################################

cc_pty_update_influence = {
	save_scope_as = cc_pty_country

	set_variable = { name = cc_pty_leaders_in value = 0 }
	cc_pty_count_leader_in = { SLOT = 1 }
	cc_pty_count_leader_in = { SLOT = 2 }
	cc_pty_count_leader_in = { SLOT = 3 }
	cc_pty_count_leader_in = { SLOT = 4 }
	if = {
		limit = { has_country_modifier = cc_pty_leaders_in_office }
		remove_country_modifier = cc_pty_leaders_in_office
	}
	if = {
		limit = { var:cc_pty_leaders_in > 0 }
		add_country_modifier = {
			modifier = cc_pty_leaders_in_office
			years = -1
			mode = replace
			size = var:cc_pty_leaders_in
		}
	}

	set_variable = { name = cc_pty_majority value = 0 }
	cc_pty_check_majority = { SLOT = 1 }
	cc_pty_check_majority = { SLOT = 2 }
	cc_pty_check_majority = { SLOT = 3 }
	cc_pty_check_majority = { SLOT = 4 }
	if = {
		limit = {
			var:cc_pty_majority = 0
			var:cc_pty_dominant = 0
		}
		set_variable = { name = cc_pty_majority value = -1 }
	}
	if = {
		limit = { has_country_modifier = cc_pty_working_majority }
		remove_country_modifier = cc_pty_working_majority
	}
	if = {
		limit = { has_country_modifier = cc_pty_hung_chamber }
		remove_country_modifier = cc_pty_hung_chamber
	}
	if = {
		limit = { var:cc_pty_majority = 1 }
		add_country_modifier = { modifier = cc_pty_working_majority years = -1 mode = replace }
	}
	else_if = {
		limit = { var:cc_pty_majority = -1 }
		add_country_modifier = { modifier = cc_pty_hung_chamber years = -1 mode = replace }
	}
}

# Also publishes cc_pty_N_leader_in (1 or 0), so the panel can show which parties' leaders
# sit in the cabinet.
cc_pty_count_leader_in = {
	set_variable = { name = cc_pty_$SLOT$_leader_in value = 0 }
	if = {
		limit = {
			trigger_if = {
				limit = { has_variable = cc_pty_$SLOT$_leader }
				var:cc_pty_$SLOT$_leader = { in_cabinet = yes }
			}
			trigger_else = { always = no }
		}
		change_variable = { name = cc_pty_leaders_in add = 1 }
		set_variable = { name = cc_pty_$SLOT$_leader_in value = 1 }
	}
}

cc_pty_check_majority = {
	if = {
		limit = {
			cc_pty_slot_var_ge = { SLOT = $SLOT$ VAR = power VALUE = 40 }
			any_cabinet_character = {
				modifier:is_head_of_cabinet = yes
				cc_pty_member_of_slot = { SLOT = $SLOT$ }
			}
		}
		set_variable = { name = cc_pty_majority value = 1 }
	}
}

###############################################################################
# LEADERS
#
# Each party's leader is its most senior minister (cc_pty_leader_score). A party with no
# minister keeps its previous leader while they live and still belong to it. Otherwise,
# in a player realm, a leader is generated outside the cabinet by the hidden event
# cc_pty.5 (characters are created in an event's immediate block). AI realms get no
# generated leaders, so they never accumulate characters.
###############################################################################

cc_pty_update_leaders = {
	save_scope_as = cc_pty_country
	cc_pty_update_leader = { SLOT = 1 }
	cc_pty_update_leader = { SLOT = 2 }
	cc_pty_update_leader = { SLOT = 3 }
	cc_pty_update_leader = { SLOT = 4 }
	if = {
		limit = {
			is_ai = no
			OR = {
				has_variable = cc_pty_1_needs_leader
				has_variable = cc_pty_2_needs_leader
				has_variable = cc_pty_3_needs_leader
				has_variable = cc_pty_4_needs_leader
			}
		}
		trigger_event_silently = { id = cc_pty.5 }
	}
}

cc_pty_update_leader = {
	if = {
		limit = { has_variable = cc_pty_$SLOT$_type }
		if = {
			limit = { any_cabinet_character = { cc_pty_member_of_slot = { SLOT = $SLOT$ } } }
			ordered_cabinet_character = {
				limit = { cc_pty_member_of_slot = { SLOT = $SLOT$ } }
				order_by = cc_pty_leader_score
				max = 1
				check_range_bounds = no
				save_scope_as = cc_pty_ldr
			}
			set_variable = { name = cc_pty_$SLOT$_leader value = scope:cc_pty_ldr }
			remove_variable = cc_pty_$SLOT$_needs_leader
		}
		else_if = {
			limit = { cc_pty_slot_leader_valid = { SLOT = $SLOT$ } }
			remove_variable = cc_pty_$SLOT$_needs_leader
		}
		else = {
			remove_variable = cc_pty_$SLOT$_leader
			if = {
				limit = { is_ai = no }
				set_variable = { name = cc_pty_$SLOT$_needs_leader value = yes }
			}
		}
	}
}

# Called from cc_pty.5 through cc_pty_create_leader_dispatch (generated), which supplies
# the archetype and the estate its leaders come from.
cc_pty_create_leader = {
	if = {
		limit = { has_variable = cc_pty_$SLOT$_needs_leader }
		create_character = {
			create_in_limbo = yes
			estate = estate_type:$ESTATE$
			culture = root.culture
			religion = root.religion
			age = { 30 55 }
			save_scope_as = cc_pty_new_leader
		}
		scope:cc_pty_new_leader = {
			move_country = root
			cc_pty_join = { TYPE = $TYPE$ KEY = $KEY$ }
			set_variable = { name = cc_pty_generated_leader value = yes }
		}
		set_variable = { name = cc_pty_$SLOT$_leader value = scope:cc_pty_new_leader }
		remove_variable = cc_pty_$SLOT$_needs_leader
	}
}


###############################################################################
# CABINET COMPOSITION
#
#   stacked   one party holds more than half the seats (three seats or more)
#   balanced  no party holds more than 40% and at least three parties are seated
#   mixed     anything else
###############################################################################

cc_pty_compute_comp = {
	set_variable = { name = cc_pty_seats value = 0 }
	cc_pty_add_seats = { SLOT = 1 }
	cc_pty_add_seats = { SLOT = 2 }
	cc_pty_add_seats = { SLOT = 3 }
	cc_pty_add_seats = { SLOT = 4 }

	set_variable = { name = cc_pty_comp value = 0 }
	set_variable = { name = cc_pty_cabinet_slot value = 0 }

	if = {
		limit = { var:cc_pty_seats >= 3 }

		# Stacked?
		cc_pty_check_stacked = { SLOT = 1 }
		cc_pty_check_stacked = { SLOT = 2 }
		cc_pty_check_stacked = { SLOT = 3 }
		cc_pty_check_stacked = { SLOT = 4 }

		# Balanced? Count seated parties and look for one above 40%.
		if = {
			limit = { var:cc_pty_comp = 0 }
			set_variable = { name = cc_pty_best value = 0 }
			set_variable = { name = cc_pty_best_type value = 0 }
			cc_pty_check_balance = { SLOT = 1 }
			cc_pty_check_balance = { SLOT = 2 }
			cc_pty_check_balance = { SLOT = 3 }
			cc_pty_check_balance = { SLOT = 4 }
			if = {
				limit = {
					var:cc_pty_best >= 3
					var:cc_pty_best_type = 0
				}
				set_variable = { name = cc_pty_comp value = 2 }
			}
			remove_variable = cc_pty_best
			remove_variable = cc_pty_best_type
		}
	}

	cc_pty_clear_comp = yes
	if = {
		limit = { var:cc_pty_comp = 1 }
		add_country_modifier = { modifier = cc_pty_comp_stacked years = -1 mode = replace }
	}
	else_if = {
		limit = { var:cc_pty_comp = 2 }
		add_country_modifier = { modifier = cc_pty_comp_balanced years = -1 mode = replace }
	}
}

cc_pty_add_seats = {
	if = {
		limit = { has_variable = cc_pty_$SLOT$_type }
		change_variable = { name = cc_pty_seats add = var:cc_pty_$SLOT$_members }
	}
}

cc_pty_check_stacked = {
	if = {
		limit = { has_variable = cc_pty_$SLOT$_type }
		set_variable = { name = cc_pty_tmp value = var:cc_pty_$SLOT$_members }
		change_variable = { name = cc_pty_tmp multiply = 2 }
		if = {
			limit = { var:cc_pty_tmp > var:cc_pty_seats }
			set_variable = { name = cc_pty_comp value = 1 }
			set_variable = { name = cc_pty_cabinet_slot value = $SLOT$ }
		}
	}
}

# cc_pty_best counts seated parties; cc_pty_best_type is set to 1 if any holds over 40%.
cc_pty_check_balance = {
	if = {
		limit = { cc_pty_slot_var_ge = { SLOT = $SLOT$ VAR = members VALUE = 1 } }
		change_variable = { name = cc_pty_best add = 1 }
		set_variable = { name = cc_pty_tmp value = var:cc_pty_$SLOT$_members }
		change_variable = { name = cc_pty_tmp multiply = 5 }
		change_variable = { name = cc_pty_tmp divide = 2 }
		if = {
			limit = { var:cc_pty_tmp > var:cc_pty_seats }
			set_variable = { name = cc_pty_best_type value = 1 }
		}
	}
}

cc_pty_clear_comp = {
	if = {
		limit = { has_country_modifier = cc_pty_comp_stacked }
		remove_country_modifier = cc_pty_comp_stacked
	}
	if = {
		limit = { has_country_modifier = cc_pty_comp_balanced }
		remove_country_modifier = cc_pty_comp_balanced
	}
}


###############################################################################
# SATISFACTION
#
# Target (cc_pty_compute_target, shown in the panel) = the party's mood, the sum of its
# three concerns (cc_pty_concern_T_1..3, generated: what this particular party watches,
# such as court spending for the Court Party or army upkeep for the War Party)
#   + 5 for one seated minister, + 25 for two or more
#   - 20 with none, while holding 20% or more power
#   + 15 if the head of the cabinet belongs to it, + 10 if the ruler does
#   stacked cabinet: + 10 for the stacked party, - 10 for every other
#   balanced cabinet: never below - 20
# Satisfaction moves halfway to the target each pass.
###############################################################################

# Country scope, scope:cc_pty_country set. What the party in slot SLOT expects, stored as
# cc_pty_N_target. Computed whenever seats are recomputed, so the panel shows it at once.
cc_pty_compute_target = {
	if = {
		limit = { has_variable = cc_pty_$SLOT$_type }

		# Each term is also stored for the panel's expectations card (cc_pty_N_t_*), and the
		# rows there sum to the target.
		set_variable = { name = cc_pty_$SLOT$_t_cab value = 0 }
		set_variable = { name = cc_pty_$SLOT$_t_head value = 0 }
		set_variable = { name = cc_pty_$SLOT$_t_ruler value = 0 }

		if = {
			limit = { var:cc_pty_$SLOT$_members >= 2 }
			set_variable = { name = cc_pty_$SLOT$_t_cab value = 25 }
		}
		else_if = {
			limit = { var:cc_pty_$SLOT$_members >= 1 }
			set_variable = { name = cc_pty_$SLOT$_t_cab value = 5 }
		}
		else_if = {
			limit = { var:cc_pty_$SLOT$_power >= 20 }
			set_variable = { name = cc_pty_$SLOT$_t_cab value = -20 }
		}

		# The head of the cabinet is one of theirs.
		if = {
			limit = {
				any_cabinet_character = {
					modifier:is_head_of_cabinet = yes
					cc_pty_member_of_slot = { SLOT = $SLOT$ }
				}
			}
			set_variable = { name = cc_pty_$SLOT$_t_head value = 15 }
		}

		# The ruler is one of theirs.
		if = {
			limit = {
				ruler ?= { cc_pty_member_of_slot = { SLOT = $SLOT$ } }
			}
			set_variable = { name = cc_pty_$SLOT$_t_ruler value = 10 }
		}

		set_variable = { name = cc_pty_tmp value = var:cc_pty_$SLOT$_mood }
		change_variable = { name = cc_pty_tmp add = var:cc_pty_$SLOT$_t_cab }
		change_variable = { name = cc_pty_tmp add = var:cc_pty_$SLOT$_t_head }
		change_variable = { name = cc_pty_tmp add = var:cc_pty_$SLOT$_t_ruler }
		set_variable = { name = cc_pty_$SLOT$_t_comp value = var:cc_pty_tmp }

		if = {
			limit = { var:cc_pty_comp = 1 }
			if = {
				limit = { var:cc_pty_cabinet_slot = $SLOT$ }
				change_variable = { name = cc_pty_tmp add = 10 }
			}
			else = {
				change_variable = { name = cc_pty_tmp add = -10 }
			}
		}
		else_if = {
			limit = { var:cc_pty_comp = 2 }
			clamp_variable = { name = cc_pty_tmp min = -20 max = 100 }
		}

		clamp_variable = { name = cc_pty_tmp min = -100 max = 100 }
		set_variable = { name = cc_pty_$SLOT$_target value = var:cc_pty_tmp }

		# What the cabinet's shape (and the -100..100 clamp) changed: target minus the rest.
		change_variable = { name = cc_pty_$SLOT$_t_comp multiply = -1 }
		change_variable = { name = cc_pty_$SLOT$_t_comp add = var:cc_pty_tmp }
	}
}

# Move satisfaction toward the target by STEP (0.5 at a review, 1 at setup).
cc_pty_update_sat = {
	if = {
		limit = { has_variable = cc_pty_$SLOT$_type }
		set_variable = { name = cc_pty_tmp value = var:cc_pty_$SLOT$_target }
		change_variable = { name = cc_pty_tmp subtract = var:cc_pty_$SLOT$_sat }
		change_variable = { name = cc_pty_tmp multiply = $STEP$ }
		change_variable = { name = cc_pty_$SLOT$_sat add = var:cc_pty_tmp }
		clamp_variable = { name = cc_pty_$SLOT$_sat min = -100 max = 100 }
	}
}

###############################################################################
# AGENDAS
#
# A party with 25% power or more sets an agenda when its cooldown lapses. At most one new
# agenda per pass, and only while cc_pty_popup_cd is clear, which keeps announcements to
# about two a decade. An agenda has six years. Met: +30 satisfaction. Missed: -25.
# Either way the party waits five to twelve years before the next.
###############################################################################

cc_pty_resolve_agenda = {
	if = {
		limit = { cc_pty_slot_var_ge = { SLOT = $SLOT$ VAR = agenda VALUE = 1 } }
		if = {
			limit = { cc_pty_agenda_met = { SLOT = $SLOT$ } }
			cc_pty_add_sat = { SLOT = $SLOT$ VALUE = 30 }
			add_political_influence = cc_pty_agenda_dividend_slot_$SLOT$
			set_variable = { name = cc_pty_$SLOT$_last value = 1 }
			cc_pty_end_agenda = { SLOT = $SLOT$ }
		}
		else_if = {
			limit = { NOT = { has_variable = cc_pty_$SLOT$_agenda_due } }
			cc_pty_add_sat = { SLOT = $SLOT$ VALUE = -25 }
			set_variable = { name = cc_pty_$SLOT$_last value = 2 }
			cc_pty_end_agenda = { SLOT = $SLOT$ }
		}
	}
}

cc_pty_end_agenda = {
	set_variable = { name = cc_pty_$SLOT$_agenda value = 0 }
	remove_variable = cc_pty_$SLOT$_agenda_due
	random_list = {
		1 = { set_variable = { name = cc_pty_$SLOT$_agenda_cd value = yes years = 5 } }
		1 = { set_variable = { name = cc_pty_$SLOT$_agenda_cd value = yes years = 8 } }
		1 = { set_variable = { name = cc_pty_$SLOT$_agenda_cd value = yes years = 12 } }
	}
	cc_pty_publish_top_agenda = yes
}

# cc_pty.20 option b: the court refuses the agenda just announced (cc_pty_event_slot).
cc_pty_refuse_agenda = {
	if = {
		limit = { var:cc_pty_event_slot = 1 }
		cc_pty_add_sat = { SLOT = 1 VALUE = -15 }
		cc_pty_end_agenda = { SLOT = 1 }
	}
	else_if = {
		limit = { var:cc_pty_event_slot = 2 }
		cc_pty_add_sat = { SLOT = 2 VALUE = -15 }
		cc_pty_end_agenda = { SLOT = 2 }
	}
	else_if = {
		limit = { var:cc_pty_event_slot = 3 }
		cc_pty_add_sat = { SLOT = 3 VALUE = -15 }
		cc_pty_end_agenda = { SLOT = 3 }
	}
	else_if = {
		limit = { var:cc_pty_event_slot = 4 }
		cc_pty_add_sat = { SLOT = 4 VALUE = -15 }
		cc_pty_end_agenda = { SLOT = 4 }
	}
	save_scope_as = cc_pty_country
	cc_pty_apply_all = yes
}

# Pick the strongest party that wants an agenda, and give it one.
cc_pty_try_new_agenda = {
	if = {
		limit = { NOT = { has_variable = cc_pty_popup_cd } }
		set_variable = { name = cc_pty_best value = 0 }
		set_variable = { name = cc_pty_best_slot value = 0 }
		cc_pty_consider_agenda = { SLOT = 1 }
		cc_pty_consider_agenda = { SLOT = 2 }
		cc_pty_consider_agenda = { SLOT = 3 }
		cc_pty_consider_agenda = { SLOT = 4 }
		if = {
			limit = { var:cc_pty_best_slot = 1 }
			cc_pty_start_agenda = { SLOT = 1 }
		}
		else_if = {
			limit = { var:cc_pty_best_slot = 2 }
			cc_pty_start_agenda = { SLOT = 2 }
		}
		else_if = {
			limit = { var:cc_pty_best_slot = 3 }
			cc_pty_start_agenda = { SLOT = 3 }
		}
		else_if = {
			limit = { var:cc_pty_best_slot = 4 }
			cc_pty_start_agenda = { SLOT = 4 }
		}
		remove_variable = cc_pty_best
		remove_variable = cc_pty_best_slot
	}
}

cc_pty_consider_agenda = {
	if = {
		limit = { cc_pty_slot_wants_agenda = { SLOT = $SLOT$ } }
		if = {
			limit = { var:cc_pty_$SLOT$_power > var:cc_pty_best }
			set_variable = { name = cc_pty_best value = var:cc_pty_$SLOT$_power }
			set_variable = { name = cc_pty_best_slot value = $SLOT$ }
		}
	}
}

# An agenda the party already meets is never chosen. If all three are met, it is content
# and waits.
cc_pty_start_agenda = {
	if = {
		limit = {
			OR = {
				NOT = { cc_pty_slot_agenda_type_met = { SLOT = $SLOT$ K = 1 } }
				NOT = { cc_pty_slot_var_ge = { SLOT = $SLOT$ VAR = members VALUE = 2 } }
				NOT = { cc_pty_slot_agenda_type_met = { SLOT = $SLOT$ K = 3 } }
			}
		}
		random_list = {
			1 = {
				trigger = { NOT = { cc_pty_slot_agenda_type_met = { SLOT = $SLOT$ K = 1 } } }
				set_variable = { name = cc_pty_$SLOT$_agenda value = 1 }
			}
			1 = {
				trigger = { NOT = { cc_pty_slot_var_ge = { SLOT = $SLOT$ VAR = members VALUE = 2 } } }
				set_variable = { name = cc_pty_$SLOT$_agenda value = 2 }
			}
			1 = {
				trigger = { NOT = { cc_pty_slot_agenda_type_met = { SLOT = $SLOT$ K = 3 } } }
				set_variable = { name = cc_pty_$SLOT$_agenda value = 3 }
			}
		}
		set_variable = { name = cc_pty_$SLOT$_agenda_due value = yes years = 6 }
		remove_variable = cc_pty_$SLOT$_last

		set_variable = { name = cc_pty_event_slot value = $SLOT$ }
		set_variable = { name = cc_pty_event_type value = var:cc_pty_$SLOT$_type }
		set_variable = { name = cc_pty_event_agenda value = var:cc_pty_$SLOT$_agenda }
		set_variable = { name = cc_pty_popup_cd value = yes years = 5 }
		# The panel's top line is published on recompute, which ran before this; refresh it.
		cc_pty_publish_top_agenda = yes
		if = {
			limit = { is_ai = no }
			trigger_event_non_silently = { id = cc_pty.20 }
		}
	}
	else = {
		set_variable = { name = cc_pty_$SLOT$_agenda_cd value = yes years = 5 }
	}
}


###############################################################################
# CONSEQUENCES
#
#   above  30   pleased     goodwill modifier, members +5% efficiency
#   -20 .. 30   content     nothing
#   -50 ..-20   discontent  discontent modifier, members -10% efficiency
#   below -50   angry       discontent modifier, members -20% efficiency
#
# Country modifiers are scaled with `size` = |satisfaction| x power / 1250, so a
# quarter-share party at +/-50 applies its modifier at 1x, then by government
# (cc_pty_gov_weight). Goodwill is x1.25 for a stacked cabinet's party and x0.75 under a
# balanced cabinet: stacking concentrates the rewards, balance spreads them thin.
###############################################################################

cc_pty_apply_all = {
	cc_pty_with_type = { SLOT = 1 EFFECT = cc_pty_apply_mood }
	cc_pty_with_type = { SLOT = 2 EFFECT = cc_pty_apply_mood }
	cc_pty_with_type = { SLOT = 3 EFFECT = cc_pty_apply_mood }
	cc_pty_with_type = { SLOT = 4 EFFECT = cc_pty_apply_mood }
	cc_pty_apply_push = yes
	cc_pty_publish_display = yes
}

cc_pty_apply_slot = {
	cc_pty_with_type = { SLOT = $SLOT$ EFFECT = cc_pty_apply_mood }
	cc_pty_publish_display = yes
}

cc_pty_apply_mood = {
	cc_pty_clear_country_mood = { KEY = $KEY$ }
	set_variable = { name = cc_pty_$SLOT$_sz_goodwill value = 0 }
	set_variable = { name = cc_pty_$SLOT$_sz_discontent value = 0 }
	set_variable = { name = cc_pty_$SLOT$_dz_goodwill value = 1 }
	set_variable = { name = cc_pty_$SLOT$_dz_discontent value = 1 }

	# ---- Country ----
	if = {
		limit = { var:cc_pty_$SLOT$_sat < -20 }
		set_variable = { name = cc_pty_tmp value = var:cc_pty_$SLOT$_sat }
		change_variable = { name = cc_pty_tmp multiply = -1 }
		change_variable = { name = cc_pty_tmp multiply = var:cc_pty_$SLOT$_power }
		change_variable = { name = cc_pty_tmp divide = 1250 }
		change_variable = { name = cc_pty_tmp multiply = cc_pty_gov_weight }
		set_variable = { name = cc_pty_$SLOT$_sz_discontent value = var:cc_pty_tmp }
		set_variable = { name = cc_pty_$SLOT$_dz_discontent value = var:cc_pty_tmp }
		add_country_modifier = {
			modifier = cc_pty_discontent_$KEY$
			years = -1
			mode = replace
			size = var:cc_pty_tmp
		}
	}
	else_if = {
		limit = { var:cc_pty_$SLOT$_sat > 30 }
		set_variable = { name = cc_pty_tmp value = var:cc_pty_$SLOT$_sat }
		change_variable = { name = cc_pty_tmp multiply = var:cc_pty_$SLOT$_power }
		change_variable = { name = cc_pty_tmp divide = 1250 }
		change_variable = { name = cc_pty_tmp multiply = cc_pty_gov_weight }
		if = {
			limit = {
				var:cc_pty_comp = 1
				var:cc_pty_cabinet_slot = $SLOT$
			}
			change_variable = { name = cc_pty_tmp multiply = 1.25 }
		}
		else_if = {
			limit = { var:cc_pty_comp = 2 }
			change_variable = { name = cc_pty_tmp multiply = 0.75 }
		}
		set_variable = { name = cc_pty_$SLOT$_sz_goodwill value = var:cc_pty_tmp }
		set_variable = { name = cc_pty_$SLOT$_dz_goodwill value = var:cc_pty_tmp }
		add_country_modifier = {
			modifier = cc_pty_goodwill_$KEY$
			years = -1
			mode = replace
			size = var:cc_pty_tmp
		}
	}

	# ---- Members ----
	if = {
		limit = { var:cc_pty_$SLOT$_sat < -50 }
		every_cabinet_character = {
			limit = { cc_pty_is_member = { TYPE = $TYPE$ } }
			cc_pty_set_member_mood = { MOOD = angry }
		}
	}
	else_if = {
		limit = { var:cc_pty_$SLOT$_sat < -20 }
		every_cabinet_character = {
			limit = { cc_pty_is_member = { TYPE = $TYPE$ } }
			cc_pty_set_member_mood = { MOOD = discontent }
		}
	}
	else_if = {
		limit = { var:cc_pty_$SLOT$_sat > 30 }
		every_cabinet_character = {
			limit = { cc_pty_is_member = { TYPE = $TYPE$ } }
			cc_pty_set_member_mood = { MOOD = content }
		}
	}
	else = {
		every_cabinet_character = {
			limit = { cc_pty_is_member = { TYPE = $TYPE$ } }
			cc_pty_clear_member_mood = yes
		}
	}
}

cc_pty_clear_country_mood = {
	if = {
		limit = { has_country_modifier = cc_pty_discontent_$KEY$ }
		remove_country_modifier = cc_pty_discontent_$KEY$
	}
	if = {
		limit = { has_country_modifier = cc_pty_goodwill_$KEY$ }
		remove_country_modifier = cc_pty_goodwill_$KEY$
	}
}

###############################################################################
# PANEL DISPLAY
#
# What the selected party's aspect cards and the party cards' effects lists read. For each
# aspect (standing, marginal, goodwill, discontent, push, favour, esc1, esc2):
#   cc_pty_N_sz_<aspect>   the size its modifier is applied at now, 0 when not applied
#   cc_pty_N_dz_<aspect>   the size to display: sz while applied, 1 (full strength) while
#                          not; only for the scaled aspects (standing, goodwill, discontent,
#                          push). The GUI has no Select for fixed-point numbers, hence two.
# The apply effects set the scaled ones; this sets favour and escalation, then:
#   cc_pty_N_fx<j>         the j-th modifier the archetype can apply, summed over every
#                          aspect at its applied size (generated cc_pty_fx_sum_type_T)
#   cc_pty_t<T>_*          the sizes again, by archetype: the aspect grid is drawn once per
#                          archetype rather than once per slot and archetype
# Run after every apply, after every recompute, and yearly (favour lapses on its own).
###############################################################################

cc_pty_publish_display = {
	cc_pty_publish_slot_display = { SLOT = 1 }
	cc_pty_publish_slot_display = { SLOT = 2 }
	cc_pty_publish_slot_display = { SLOT = 3 }
	cc_pty_publish_slot_display = { SLOT = 4 }
	cc_pty_with_type_st = { SLOT = 1 EFFECT = cc_pty_publish_sums }
	cc_pty_with_type_st = { SLOT = 2 EFFECT = cc_pty_publish_sums }
	cc_pty_with_type_st = { SLOT = 3 EFFECT = cc_pty_publish_sums }
	cc_pty_with_type_st = { SLOT = 4 EFFECT = cc_pty_publish_sums }
}

cc_pty_publish_slot_display = {
	if = {
		limit = { has_variable = cc_pty_$SLOT$_type }
		# Older saves: slots created before these existed.
		if = {
			limit = { NOT = { has_variable = cc_pty_$SLOT$_sz_standing } }
			cc_pty_init_display = { SLOT = $SLOT$ }
		}
		# Whose leader sits in the cabinet (the chamber line's leader icons). Also set by
		# cc_pty_count_leader_in; refreshed here so a loaded save shows it without waiting
		# for a recompute.
		set_variable = { name = cc_pty_$SLOT$_leader_in value = 0 }
		if = {
			limit = {
				trigger_if = {
					limit = { has_variable = cc_pty_$SLOT$_leader }
					var:cc_pty_$SLOT$_leader = { in_cabinet = yes }
				}
				trigger_else = { always = no }
			}
			set_variable = { name = cc_pty_$SLOT$_leader_in value = 1 }
		}
		set_variable = { name = cc_pty_$SLOT$_sz_favour value = 0 }
		if = {
			limit = { has_variable = cc_pty_$SLOT$_favour_on }
			set_variable = { name = cc_pty_$SLOT$_sz_favour value = 1 }
		}
		set_variable = { name = cc_pty_$SLOT$_sz_esc1 value = 0 }
		set_variable = { name = cc_pty_$SLOT$_sz_esc2 value = 0 }
		if = {
			limit = { var:cc_pty_$SLOT$_esc >= 2 }
			set_variable = { name = cc_pty_$SLOT$_sz_esc2 value = 1 }
		}
		else_if = {
			limit = { var:cc_pty_$SLOT$_esc >= 1 }
			set_variable = { name = cc_pty_$SLOT$_sz_esc1 value = 1 }
		}
	}
}

# Called through cc_pty_with_type_st. Slot SLOT holds archetype TYPE.
cc_pty_publish_sums = {
	cc_pty_fx_sum_type_$TYPE$ = { SLOT = $SLOT$ }
	set_variable = { name = cc_pty_t$TYPE$_esc value = var:cc_pty_$SLOT$_esc }
	set_variable = { name = cc_pty_t$TYPE$_sz_standing value = var:cc_pty_$SLOT$_sz_standing }

			limit = { NOT = { has_variable = cc_court_rivalry } }

		}
		set_variable = { name = cc_pty_rivalry_feed value = yes months = 30 }
	}
}


###############################################################################
# ESCALATION (must-placate parties: War Party, Regional Party, Liberal Party)
#
# Each pass a must-placate party spends angry (below -50) with 20% power or more adds a
# step of neglect; any other pass takes one away.
#   1  warning modifier
#   2  unrest modifier (rebel growth, lost control)
#   3  the outcome: a coup, a regional rising or a breakaway of subjects, or a liberal
#      uprising. Then neglect resets, and that party cannot do it again for twenty years.
# Each step is visible in the panel before the next lands.
###############################################################################

cc_pty_escalate = {
	if = {
		limit = { cc_pty_slot_escalates = { SLOT = $SLOT$ } }
		if = {
			limit = {
				NOT = { cc_pty_slot_var_ge = { SLOT = $SLOT$ VAR = sat VALUE = -50 } }
				cc_pty_slot_var_ge = { SLOT = $SLOT$ VAR = power VALUE = 20 }
			}
			change_variable = { name = cc_pty_$SLOT$_neglect add = 1 }
		}
		else_if = {
			limit = { var:cc_pty_$SLOT$_neglect > 0 }
			change_variable = { name = cc_pty_$SLOT$_neglect add = -1 }
		}
		set_variable = { name = cc_pty_$SLOT$_esc value = var:cc_pty_$SLOT$_neglect }
		clamp_variable = { name = cc_pty_$SLOT$_esc min = 0 max = 3 }
		cc_pty_escalation_dispatch = { SLOT = $SLOT$ }
	}
}

cc_pty_clear_escalation_mods = {
	cc_pty_remove_esc_mods = { KEY = military }
	cc_pty_remove_esc_mods = { KEY = regionalist }
	cc_pty_remove_esc_mods = { KEY = liberal }
}

cc_pty_remove_esc_mods = {
	if = {
		limit = { has_country_modifier = cc_pty_esc_1_$KEY$ }
		remove_country_modifier = cc_pty_esc_1_$KEY$
	}
	if = {
		limit = { has_country_modifier = cc_pty_esc_2_$KEY$ }
		remove_country_modifier = cc_pty_esc_2_$KEY$
	}
}

# Called only for the must-placate archetypes (cc_pty_escalation_dispatch).
cc_pty_apply_escalation = {
	cc_pty_remove_esc_mods = { KEY = $KEY$ }
	if = {
		limit = { var:cc_pty_$SLOT$_esc >= 3 }
		if = {
			limit = { NOT = { has_variable = cc_pty_$SLOT$_outcome_cd } }
			cc_pty_outcome_$KEY$ = yes
			set_variable = { name = cc_pty_$SLOT$_outcome_cd value = yes years = 20 }
			set_variable = { name = cc_pty_$SLOT$_neglect value = 0 }
			set_variable = { name = cc_pty_$SLOT$_esc value = 0 }
		}
		else = {
			add_country_modifier = { modifier = cc_pty_esc_2_$KEY$ years = -1 mode = replace }
		}
	}
	else_if = {
		limit = { var:cc_pty_$SLOT$_esc >= 2 }
		add_country_modifier = { modifier = cc_pty_esc_2_$KEY$ years = -1 mode = replace }
	}
	else_if = {
		limit = { var:cc_pty_$SLOT$_esc >= 1 }
		add_country_modifier = { modifier = cc_pty_esc_1_$KEY$ years = -1 mode = replace }
	}
}

# Country scope, the disaster's on_monthly. Drift, then perhaps a crisis event: none for
# six months after one fires, then 30% a month, so about six over the five years.

		random = {
			chance = 30
			set_variable = { name = cc_pty_lu_event_cd value = yes months = 6 }
			random_list = {
				10 = { trigger_event_non_silently = { id = cc_pty.70 } }
				10 = { trigger_event_non_silently = { id = cc_pty.71 } }
				10 = { trigger_event_non_silently = { id = cc_pty.72 } }
				10 = {

				}

					}
					trigger_event_non_silently = { id = cc_pty.75 }

			}

	}

# Country scope. The uprising boiled over and the crown refused (cc_pty.47 option b): a
# burgher rebel faction takes the liberal cities' provinces, up to about a sixth of the
# realm and never the capital's, and starts a civil war. Same sequence as the regional
# rising below.
cc_pty_start_liberal_rising = {
	create_rebel = {
		category = estate
		estate = burghers_estate
		name = cc_pty_liberal_rebels
		save_scope_as = cc_pty_rebels
	}
	set_local_variable = { name = cc_pty_rebel_provinces value = num_provinces }
	change_local_variable = { name = cc_pty_rebel_provinces multiply = 0.15 }
	ordered_province = {
		limit = {
			NOT = { any_location_in_province = { is_capital = yes } }
			any_location_in_province = {
				owner = root
				has_variable = cc_pty_lt
				var:cc_pty_lt >= 50
			}
		}
		max = local_var:cc_pty_rebel_provinces
		every_location_in_province = {
			every_pop = {
				limit = { owner = root }
				change_pop_allegiance = scope:cc_pty_rebels
			}
		}
	}
	scope:cc_pty_rebels = { start_civil_war = {} }
}

###############################################################################
# TERM LIMITS
#
# In force while the dominant party is the Reform Party or the Liberal Party. A minister
# who has served ten years leaves office at the next pass and cannot return for ten.
###############################################################################

cc_pty_update_termlimits = {
	set_variable = { name = cc_pty_tmp value = 0 }
	if = {
		limit = { var:cc_pty_dominant = 1 cc_pty_slot_imposes_term_limits = { SLOT = 1 } }
		set_variable = { name = cc_pty_tmp value = 1 }
	}
	else_if = {
		limit = { var:cc_pty_dominant = 2 cc_pty_slot_imposes_term_limits = { SLOT = 2 } }
		set_variable = { name = cc_pty_tmp value = 1 }
	}
	else_if = {
		limit = { var:cc_pty_dominant = 3 cc_pty_slot_imposes_term_limits = { SLOT = 3 } }
		set_variable = { name = cc_pty_tmp value = 1 }
	}
	else_if = {
		limit = { var:cc_pty_dominant = 4 cc_pty_slot_imposes_term_limits = { SLOT = 4 } }
		set_variable = { name = cc_pty_tmp value = 1 }
	}

	if = {
		limit = {
			var:cc_pty_tmp = 1
			NOT = { has_variable = cc_pty_termlimits }
		}
		set_variable = { name = cc_pty_termlimits value = yes }
		if = {
			limit = { is_ai = no }
			trigger_event_non_silently = { id = cc_pty.60 }
		}
	}
	else_if = {
		limit = {
			var:cc_pty_tmp = 0
			has_variable = cc_pty_termlimits
		}
		remove_variable = cc_pty_termlimits
		if = {
			limit = { is_ai = no }
			trigger_event_non_silently = { id = cc_pty.61 }
		}
	}

	if = {
		limit = { has_variable = cc_pty_termlimits }
		every_cabinet_character = {
			limit = {
				trigger_if = {
					limit = { has_variable = cc_pty_years }
					var:cc_pty_years >= 10
				}
				trigger_else = { always = no }
			}
			save_scope_as = cc_pty_term_ender
			set_variable = { name = cc_pty_term_served value = yes }
			add_character_modifier = { modifier = banned_from_cabinet years = 10 mode = replace }
			scope:cc_pty_country = { remove_from_cabinet = scope:cc_pty_term_ender }
		}
	}
}

###############################################################################
# LIBERAL THOUGHT
#
# From the age of absolutism, every pass grows cc_pty_lt in the capital and every city
# (cc_pty_lt_growth), and the country gauge is their average. Past 50 the Liberal Party
# forms: the next yearly update realigns at once and the Liberal Party is a strong
# candidate (cc_pty.50).
###############################################################################

cc_pty_update_liberal_thought = {
	if = {
		limit = { current_age_or_later = { age = age_5_absolutism } }

		every_owned_location = {
			limit = {
				OR = {
					is_capital = yes
					is_city = yes
				}
			}
			if = {
				limit = { NOT = { has_variable = cc_pty_lt } }
				set_variable = { name = cc_pty_lt value = 0 }
			}
			change_variable = { name = cc_pty_lt add = cc_pty_lt_growth }
			clamp_variable = { name = cc_pty_lt min = 0 max = 100 }
		}

		set_variable = { name = cc_pty_liberal_thought value = cc_pty_lt_sum }
		if = {
			limit = { cc_pty_lt_count > 0 }
			change_variable = { name = cc_pty_liberal_thought divide = cc_pty_lt_count }
		}
		clamp_variable = { name = cc_pty_liberal_thought min = 0 max = 100 }

		if = {
			limit = {
				var:cc_pty_liberal_thought >= 50
				NOT = { has_variable = cc_pty_liberal_formed }
			}
			set_variable = { name = cc_pty_liberal_formed value = yes }
			remove_variable = cc_pty_realign_cd
			if = {
				limit = { is_ai = no }
				set_variable = { name = cc_pty_popup_cd value = yes years = 5 }
				trigger_event_non_silently = { id = cc_pty.50 }
			}
		}
	}
}


###############################################################################
# DEFECTION
#
# A minister whose party is angry may cross the floor to the strongest other party. One
# roll per pass, 25%. For a player this is the personal event cc_pty.30, and it shares
# the main mod's rivalry cooldown (cc_rival_event_cooldown) so the court's personal
# events never stack. The AI simply defects.
###############################################################################

cc_pty_try_defection = {
	if = {
		limit = {
			NOT = { has_variable = cc_rival_event_cooldown }
			any_cabinet_character = { cc_pty_member_party_angry = yes }
		}
		random = {
			chance = 25
			random_cabinet_character = {
				limit = { cc_pty_member_party_angry = yes }
				save_scope_as = cc_pty_defector
			}
			cc_pty_pick_defection_target = yes
			if = {
				limit = { var:cc_pty_best_slot > 0 }
				if = {
					limit = { is_ai = no }
					set_variable = { name = cc_rival_event_cooldown value = yes months = 30 }
					trigger_event_non_silently = { id = cc_pty.30 }
				}
				else = {
					scope:cc_pty_defector = { cc_pty_join_from_country_var = { VAR = cc_pty_defect_to_type } }
				}
			}
			remove_variable = cc_pty_best
			remove_variable = cc_pty_best_slot
		}
	}
}

# The strongest party the defector does not already belong to.
cc_pty_pick_defection_target = {
	set_variable = { name = cc_pty_best value = -1 }
	set_variable = { name = cc_pty_best_slot value = 0 }
	cc_pty_consider_defection = { SLOT = 1 }
	cc_pty_consider_defection = { SLOT = 2 }
	cc_pty_consider_defection = { SLOT = 3 }
	cc_pty_consider_defection = { SLOT = 4 }
	scope:cc_pty_defector = {
		if = {
			limit = { has_variable = cc_pty_aff }
			scope:cc_pty_country = {
				set_variable = { name = cc_pty_defect_from_type value = scope:cc_pty_defector.var:cc_pty_aff }
			}
		}
	}
}

cc_pty_consider_defection = {
	if = {
		limit = {
			has_variable = cc_pty_$SLOT$_type
			NOT = { scope:cc_pty_defector = { cc_pty_member_of_slot = { SLOT = $SLOT$ } } }
		}
		if = {
			limit = { var:cc_pty_$SLOT$_power > var:cc_pty_best }
			set_variable = { name = cc_pty_best value = var:cc_pty_$SLOT$_power }
			set_variable = { name = cc_pty_best_slot value = $SLOT$ }
			set_variable = { name = cc_pty_defect_to_type value = var:cc_pty_$SLOT$_type }
		}
	}
}

# Buy a party's goodwill. Applied at once so the player sees what they paid for.
cc_pty_bribe = {
	cc_pty_add_sat = { SLOT = $SLOT$ VALUE = 25 }
	set_variable = { name = cc_pty_$SLOT$_bribe_cd value = yes years = 2 }
	cc_pty_apply_slot = { SLOT = $SLOT$ }
}

# Marginalise a party: elected seats x 0.6 for five years, satisfaction -30, and every party
# opposed to it +10. Marginalising a must-placate party below 20% power also stops its
# escalation, which is the other way out besides placating it.
cc_pty_attack = {
	set_variable = { name = cc_pty_$SLOT$_suppressed value = yes years = 5 }
	set_variable = { name = cc_pty_$SLOT$_attack_cd value = yes years = 5 }
	cc_pty_add_sat = { SLOT = $SLOT$ VALUE = -30 }
	cc_pty_opposed_ripple = { SLOT = $SLOT$ OTHER = 1 VALUE = 10 }
	cc_pty_opposed_ripple = { SLOT = $SLOT$ OTHER = 2 VALUE = 10 }
	cc_pty_opposed_ripple = { SLOT = $SLOT$ OTHER = 3 VALUE = 10 }
	cc_pty_opposed_ripple = { SLOT = $SLOT$ OTHER = 4 VALUE = 10 }
	save_scope_as = cc_pty_country
	cc_pty_recompute_all = yes
	cc_pty_apply_all = yes
}

# Every party opposed to slot SLOT gains VALUE satisfaction (negative to lose it).
cc_pty_opposed_ripple = {
	if = {
		limit = { cc_pty_slots_opposed = { A = $SLOT$ B = $OTHER$ } }
		cc_pty_add_sat = { SLOT = $OTHER$ VALUE = $VALUE$ }
	}
}

# Call in a favour: the party's favour modifier for five years, at 15 satisfaction.
cc_pty_favour = {
	cc_pty_with_type_k = { SLOT = $SLOT$ EFFECT = cc_pty_grant_favour }
	cc_pty_add_sat = { SLOT = $SLOT$ VALUE = -15 }
	set_variable = { name = cc_pty_$SLOT$_favour_cd value = yes years = 5 }
	set_variable = { name = cc_pty_$SLOT$_favour_on value = yes years = 5 }
	cc_pty_apply_slot = { SLOT = $SLOT$ }
}

cc_pty_grant_favour = {
	add_country_modifier = { modifier = cc_pty_favour_$KEY$ years = 5 mode = replace }
}

# Seat the party's leader in the cabinet: an open seat if there is one, otherwise the
# seat of the least senior minister from another party.
cc_pty_appoint = {
	save_scope_as = cc_pty_country
	var:cc_pty_$SLOT$_leader = { save_scope_as = cc_pty_appointee }
	if = {
		limit = { NOT = { any_cabinet = { NOT = { exists = cabinet_member } } } }
		if = {
			limit = { any_cabinet_character = { NOT = { cc_pty_member_of_slot = { SLOT = $SLOT$ } } } }
			ordered_cabinet_character = {
				limit = { NOT = { cc_pty_member_of_slot = { SLOT = $SLOT$ } } }
				order_by = cc_pty_leader_score_inverse
				max = 1
				check_range_bounds = no
				save_scope_as = cc_pty_displaced
			}
			remove_from_cabinet = scope:cc_pty_displaced
		}
	}
	ordered_cabinet = {
		limit = { NOT = { exists = cabinet_member } }
		max = 1
		check_range_bounds = no
		add_to_cabinet = scope:cc_pty_appointee
	}
	cc_pty_recompute_all = yes
	cc_pty_apply_all = yes
}

# Meet the party's agenda by decree: it counts as met now, and the parties opposed to it
# resent the favour.
cc_pty_concede = {
	cc_pty_add_sat = { SLOT = $SLOT$ VALUE = 30 }
	add_political_influence = cc_pty_agenda_dividend_slot_$SLOT$
	set_variable = { name = cc_pty_$SLOT$_last value = 1 }
	cc_pty_end_agenda = { SLOT = $SLOT$ }
	cc_pty_opposed_ripple = { SLOT = $SLOT$ OTHER = 1 VALUE = -10 }
	cc_pty_opposed_ripple = { SLOT = $SLOT$ OTHER = 2 VALUE = -10 }
	cc_pty_opposed_ripple = { SLOT = $SLOT$ OTHER = 3 VALUE = -10 }
	cc_pty_opposed_ripple = { SLOT = $SLOT$ OTHER = 4 VALUE = -10 }
	save_scope_as = cc_pty_country
	cc_pty_recompute_all = yes
	cc_pty_apply_all = yes
}

# Call an election: recount the seats and run a realignment now. Ten-year cooldown.
cc_pty_call_election = {
	set_variable = { name = cc_pty_election_cd value = yes years = 10 }

	# Which party leads the cabinet going in, and its seats.
	save_scope_as = cc_pty_country
	set_variable = { name = cc_pty_el_type value = 0 }
	set_variable = { name = cc_pty_el_before value = 0 }
	cc_pty_record_head_party = { SLOT = 1 }
	cc_pty_record_head_party = { SLOT = 2 }
	cc_pty_record_head_party = { SLOT = 3 }
	cc_pty_record_head_party = { SLOT = 4 }

	cc_pty_realign_pass = yes

	# The same party's seats coming out (0 if a realignment dissolved it).
	if = {
		limit = { var:cc_pty_el_type > 0 }
		set_variable = { name = cc_pty_el_after value = 0 }
		cc_pty_measure_head_party = { SLOT = 1 }
		cc_pty_measure_head_party = { SLOT = 2 }
		cc_pty_measure_head_party = { SLOT = 3 }
		cc_pty_measure_head_party = { SLOT = 4 }
		add_political_influence = cc_pty_election_mandate
		set_variable = { name = cc_pty_last_mandate value = cc_pty_election_mandate }
	}
	remove_variable = cc_pty_el_type
	remove_variable = cc_pty_el_before
	remove_variable = cc_pty_el_after
}

cc_pty_record_head_party = {
	if = {
		limit = {
			has_variable = cc_pty_$SLOT$_type
			any_cabinet_character = {
				modifier:is_head_of_cabinet = yes
				cc_pty_member_of_slot = { SLOT = $SLOT$ }
			}
		}
		set_variable = { name = cc_pty_el_type value = var:cc_pty_$SLOT$_type }
		set_variable = { name = cc_pty_el_before value = var:cc_pty_$SLOT$_seats }
	}
}

cc_pty_measure_head_party = {
	if = {
		limit = { has_variable = cc_pty_$SLOT$_type }
		if = {
			limit = { var:cc_pty_$SLOT$_type = var:cc_pty_el_type }
			set_variable = { name = cc_pty_el_after value = var:cc_pty_$SLOT$_seats }
		}
	}
}


###############################################################################
# TOOLTIP GUARD
#
# A button's or an option's effect is evaluated to build its tooltip, and hidden_effect
# does NOT stop that: it only hides the text. In that pass variable writes do not commit,
# so the party effects' scratch comparisons (var:cc_pty_best and friends) read unset
# variables and log "Invalid right side during comparison".
#
# These wrappers set a marker and run the body only if the marker exists. On a real click
# the write commits and the body runs; in the tooltip pass it does not, and the body is
# skipped. Callers still wrap them in hidden_effect so the marker lines are not shown.
###############################################################################

cc_pty_run_slot = {
	set_variable = { name = cc_pty_exec value = yes }
	if = {
		limit = { has_variable = cc_pty_exec }
		$EFFECT$ = { SLOT = $SLOT$ }
	}
	remove_variable = cc_pty_exec
}

cc_pty_run = {
	set_variable = { name = cc_pty_exec value = yes }
	if = {
		limit = { has_variable = cc_pty_exec }
		$EFFECT$ = yes
	}
	remove_variable = cc_pty_exec
}

# cc_pty.30 option a. scope:cc_pty_defector and cc_pty_defect_to_type are set by the pass.

	}
	save_scope_as = cc_pty_country
	cc_pty_recompute_all = yes
}
