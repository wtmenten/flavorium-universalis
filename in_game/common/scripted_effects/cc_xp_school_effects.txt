# Flavorium Universalis — Cabinet Experience: school state
#
# Everything that keeps a founded school's record true after founding day: where it sits,
# what kind of institution its seat makes it, what it teaches, and how many belong to it.
# All effects take N, the slot (1-3), and run in COUNTRY scope.
#
# STATE PER SLOT (country variables, cc_xp_school_N_*):
#   founder      character ref, set at founding (cc_xp_schools.txt)
#   track        1 adm / 2 dip / 3 mil, set at founding
#   loc          the seat. Kept while in exile, so the panel can say where it was
#   prev_loc     the seat it last left, for the notification events only
#   exile        flag. Set while the school has no university to sit in
#   kind         1-8, the kind of institution the seat's religion makes it (below)
#   religion     the seat's dominant religion when seated
#   culture      the seat's dominant culture when seated
#   curriculum   by discipline (track). Administrative: 1 law, 2 theology, 3 natural
#                philosophy, 6 medicine, 8 public finance. Diplomatic: 4 letters, 9 rhetoric,
#                10 languages, 11 commerce, 12 history. Military: 5 military science,
#                7 navigation, 13 fortification, 14 gunnery, 15 logistics.
#   members      living members, republished monthly for the panel
#
# THE SEAT FLAG holds the owning country, not `yes`: cc_xp_school_seat on a location is
# only a seat while `owner = var:cc_xp_school_seat` (cc_xp_location_is_school_seat). A bare
# flag outlived its school whenever the seat changed hands, kept the location gold on the
# map, and kept its new owner from seating a school there. Older saves stored `yes`;
# cc_xp_school_maintain rewrites the flag on every live seat each month, which migrates
# them on the first pass.
#
# KIND, from the seat's dominant religion:
#   1 studium    Western Christian
#   2 academy    Orthodox and the Eastern churches
#   3 madrasa    Muslim
#   4 shuyuan    East Asian: sanjiao, mahayana, shinto
#   5 vihara     theravada, tibetan_buddhism, sammitiya
#   6 gurukula   Dharmic
#   7 yeshiva    Israelite
#   8 school     anything else, or no settled population
###############################################################################


# Seat slot N at scope:cc_xp_seat (a location) and record what that seat makes the school.
# Used at founding, on relocation and on return from exile.
cc_xp_school_seat_at = {
	save_scope_as = cc_xp_school_court
	scope:cc_xp_seat = {
		set_variable = { name = cc_xp_school_seat  value = scope:cc_xp_school_court }
	}
	set_variable = { name = cc_xp_school_$N$_loc  value = scope:cc_xp_seat }
	remove_variable = cc_xp_school_$N$_exile

	if = {
		limit = { scope:cc_xp_seat = { exists = dominant_religion } }
		set_variable = { name = cc_xp_school_$N$_religion  value = scope:cc_xp_seat.dominant_religion }
	}
	if = {
		limit = { scope:cc_xp_seat = { exists = dominant_culture } }
		set_variable = { name = cc_xp_school_$N$_culture  value = scope:cc_xp_seat.dominant_culture }
	}

	if = {
		limit = {
			scope:cc_xp_seat = {
				exists = dominant_religion
				OR = {
					dominant_religion = religion:orthodox
					dominant_religion = religion:miaphysite
					dominant_religion = religion:nestorianism
					dominant_religion = religion:bogomilism
					dominant_religion = religion:paulicianism
					dominant_religion = religion:strigolniki
				}
			}
		}
		set_variable = { name = cc_xp_school_$N$_kind  value = 2 }
	}
	else_if = {
		limit = { scope:cc_xp_seat = { exists = dominant_religion  dominant_religion.group = religion_group:christian } }
		set_variable = { name = cc_xp_school_$N$_kind  value = 1 }
	}
	else_if = {
		limit = { scope:cc_xp_seat = { exists = dominant_religion  dominant_religion.group = religion_group:muslim } }
		set_variable = { name = cc_xp_school_$N$_kind  value = 3 }
	}
	else_if = {
		limit = {
			scope:cc_xp_seat = {
				exists = dominant_religion
				OR = {
					dominant_religion = religion:sanjiao
					dominant_religion = religion:mahayana
					dominant_religion = religion:shinto
				}
			}
		}
		set_variable = { name = cc_xp_school_$N$_kind  value = 4 }
	}
	else_if = {
		limit = { scope:cc_xp_seat = { exists = dominant_religion  dominant_religion.group = religion_group:buddhist } }
		set_variable = { name = cc_xp_school_$N$_kind  value = 5 }
	}
	else_if = {
		limit = { scope:cc_xp_seat = { exists = dominant_religion  dominant_religion.group = religion_group:dharmic } }
		set_variable = { name = cc_xp_school_$N$_kind  value = 6 }
	}
	else_if = {
		limit = { scope:cc_xp_seat = { exists = dominant_religion  dominant_religion.group = religion_group:israelite_group } }
		set_variable = { name = cc_xp_school_$N$_kind  value = 7 }
	}
	else = {
		set_variable = { name = cc_xp_school_$N$_kind  value = 8 }
	}
}


# The curriculum a new school starts with, from its discipline. A school always has one, so
# the panel and the event pools never meet an empty state. Changing it is a school action.
#
# TRACK is passed in rather than read back from cc_xp_school_N_track: at founding that
# variable was written a line earlier, and writes do not commit while the interaction's
# tooltip is built, so reading it there logs "Failed to fetch variable".
cc_xp_school_default_curriculum = {
	if = {
		limit = { $TRACK$ = 2 }
		set_variable = { name = cc_xp_school_$N$_curriculum  value = 4 }
	}
	else_if = {
		limit = { $TRACK$ = 3 }
		set_variable = { name = cc_xp_school_$N$_curriculum  value = 5 }
	}
	else = {
		set_variable = { name = cc_xp_school_$N$_curriculum  value = 1 }
	}
}


# Find slot N a new seat, or send it into exile. Capital first, then the most developed free
# university. NOTICE is the per-slot flag suffix the notification event looks for, and
# EVENT its id in the cc_xp_school namespace: `moved` / 1 when leaving a lost seat,
# `returned` / 3 when coming back from exile. Exile itself is cc_xp_school.2.
cc_xp_school_find_seat = {
	# Saved scopes live for the whole on_action run. Without this, a slot that finds no
	# university would see the seat an earlier slot found this month and claim it too.
	clear_saved_scope = cc_xp_seat

	if = {
		limit = {
			capital ?= {
				has_building = building_type:university
				NOT = { cc_xp_location_is_school_seat = yes }
			}
		}
		capital = { save_scope_as = cc_xp_seat }
	}
	else = {
		ordered_owned_location = {
			limit = {
				has_building = building_type:university
				NOT = { cc_xp_location_is_school_seat = yes }
			}
			order_by = development
			position = 0
			check_range_bounds = no
			save_scope_as = cc_xp_seat
		}
	}

	if = {
		limit = { exists = scope:cc_xp_seat }
		cc_xp_school_seat_at = { N = $N$ }
		if = {
			limit = { is_ai = no }
			set_variable = { name = cc_xp_school_$N$_$NOTICE$  value = yes  months = 2 }
			trigger_event_non_silently = cc_xp_school.$EVENT$
		}
	}
	else_if = {
		limit = { NOT = { has_variable = cc_xp_school_$N$_exile } }
		set_variable = cc_xp_school_$N$_exile
		if = {
			limit = { is_ai = no }
			set_variable = { name = cc_xp_school_$N$_exiled  value = yes  months = 2 }
			trigger_event_non_silently = cc_xp_school.2
		}
	}
}


# Monthly upkeep for slot N. A seat is lost when the realm no longer owns it or its
# university is gone; an exiled school is re-seated as soon as a university is free.
cc_xp_school_maintain = {
	if = {
		limit = { has_variable = cc_xp_school_$N$_founder }
		save_scope_as = cc_xp_school_court
		clear_saved_scope = cc_xp_lost_seat

		# Seated, and the seat still holds.
		if = {
			limit = {
				NOT = { has_variable = cc_xp_school_$N$_exile }
				var:cc_xp_school_$N$_loc ?= {
					owner ?= scope:cc_xp_school_court
					has_building = building_type:university
				}
			}
			# Rewritten every month: migrates the old `yes` flag and costs nothing.
			var:cc_xp_school_$N$_loc = {
				set_variable = { name = cc_xp_school_seat  value = scope:cc_xp_school_court }
			}
			# Schools seated before kinds existed have none recorded. Re-seat them in place
			# once to fill in kind, religion and culture.
			if = {
				limit = { NOT = { has_variable = cc_xp_school_$N$_kind } }
				var:cc_xp_school_$N$_loc = { save_scope_as = cc_xp_seat }
				cc_xp_school_seat_at = { N = $N$ }
			}
			if = {
				limit = { has_variable = cc_xp_school_$N$_curriculum }
				cc_xp_school_apply_modifiers = { N = $N$ }
			}
		}
		# Seated, but the seat is lost.
		else_if = {
			limit = { NOT = { has_variable = cc_xp_school_$N$_exile } }
			var:cc_xp_school_$N$_loc ?= {
				save_scope_as = cc_xp_lost_seat
				cc_xp_school_strip_seat = yes
				# Clear it unless the location's new owner has seated a school of their own
				# there since. Still ours (the university is what was lost) always clears.
				# This also clears a pre-migration `yes` flag.
				if = {
					limit = {
						OR = {
							owner ?= scope:cc_xp_school_court
							NOT = { cc_xp_location_is_school_seat = yes }
						}
					}
					remove_variable = cc_xp_school_seat
				}
			}
			if = {
				limit = { exists = scope:cc_xp_lost_seat }
				set_variable = { name = cc_xp_school_$N$_prev_loc  value = scope:cc_xp_lost_seat }
			}
			cc_xp_school_find_seat = { N = $N$  NOTICE = moved  EVENT = 1 }
		}
		# In exile, and a university has come free.
		else_if = {
			limit = { cc_xp_has_free_university = yes }
			cc_xp_school_find_seat = { N = $N$  NOTICE = returned  EVENT = 3 }
		}

		# Schools founded before renown and quality existed.
		if = {
			limit = { NOT = { has_variable = cc_xp_school_$N$_renown } }
			cc_xp_school_init_standing = { N = $N$ }
		}
		# Schools founded before admission existed start at Low.
		cc_xp_school_init_admission = { N = $N$ }

		# A Rector who has left the cabinet without the duty ending cleanly.
		if = {
			limit = {
				has_variable = cc_xp_school_$N$_rector
				NOT = { var:cc_xp_school_$N$_rector ?= { in_cabinet = yes } }
			}
			remove_variable = cc_xp_school_$N$_rector
		}

		# Schools founded before curricula existed get their discipline's default.
		if = {
			limit = {
				NOT = { has_variable = cc_xp_school_$N$_curriculum }
				has_variable = cc_xp_school_$N$_track
			}
			cc_xp_school_default_curriculum = { N = $N$  TRACK = var:cc_xp_school_$N$_track }
		}
		# Curricula are per discipline. A school teaching one outside its own (possible before
		# that rule, or after a track change) goes back to its discipline's default, with no
		# cooldown spent.
		if = {
			limit = {
				has_variable = cc_xp_school_$N$_curriculum
				has_variable = cc_xp_school_$N$_track
				NOT = { cc_xp_school_curriculum_fits = { N = $N$ } }
			}
			cc_xp_school_default_curriculum = { N = $N$  TRACK = var:cc_xp_school_$N$_track }
			var:cc_xp_school_$N$_loc ?= { cc_xp_school_strip_seat = yes }
			cc_xp_school_refresh_realm = yes
		}
	}
}


# Republish slot N's living membership for the panel, which cannot count characters by
# variable. Human courts only; nothing else reads it.
cc_xp_school_count_members = {
	set_variable = { name = cc_xp_school_$N$_members  value = 0 }
	if = {
		limit = { has_variable = cc_xp_school_$N$_founder }
		save_scope_as = cc_xp_school_court
		every_character = {
			limit = {
				is_alive = yes
				has_variable = cc_xp_school
				var:cc_xp_school = $N$
			}
			scope:cc_xp_school_court = {
				change_variable = { name = cc_xp_school_$N$_members  add = 1 }
			}
		}
	}
}

#

###############################################################################
# RENOWN AND QUALITY
#
# Two numbers per school, both 0-100, both recomputed once a year by drifting toward a
# target, and both pushed directly by actions and events (which then fade back toward
# the target over the following years). Neither grows without limit: each settles where
# its target is, and the target is set by the school's circumstances, not its age.
#
# RENOWN is how widely the school is known. It competes with the world.
#   target = established x (  established = 20% at founding, +0.8% a year, full at 100 years
#              rank         cc_xp_renown_rank        realm rank, great power      at most 25
#            + prestige     cc_xp_renown_prestige    realm prestige               at most 10
#            + wealth       cc_xp_renown_wealth      seat development             at most 15
#            + learning     cc_xp_renown_learning    seat noble/burgher literacy  at most 20
#            + teaching )   0.3 x quality                                         at most 30
#          halved in exile, 0 while dormant. Rises 8% of the gap a year, falls 15%.
#   The terms are defined in script_values/cc_xp_values.txt with the reasoning behind
#   the weights. Foreign universities are scored on the same terms (cc_xp_university_score)
#   so the two stay comparable.
#   standing = 1 + the number of foreign universities whose cc_xp_university_score beats
#   this school's renown (cc_xp_school_scan_rivals). Falling behind the leading foreign
#   university brings rival events; standing first brings its own.
#   Feeds: the realm renown modifier, scholar and artist arrivals, event options,
#   capacity and self-enrolment.
#
# QUALITY is how good the teaching is. It is bought and staffed, not won.
#   target = funding        20 + 50 x the Culture budget slider (cultural_maintenance),
#                           or a flat 35 before the realm can invest in culture
#          + Rector         5, plus their seniority level, plus a tenth of their skill in
#                           the school's discipline: 5 to 25
#          + stability      5 while the curriculum has not changed in 10 years
#          + members        a fifth of the members' AVERAGE aptitude in the discipline, at
#                           most 20. An average, so a small school of able members beats a
#                           large school of indifferent ones.
#          10 flat in exile, 0 while dormant. Drifts 20% of the gap a year.
#
# DORMANT (cc_xp_school_N_dormant): no living members at the yearly update. The school does
# not age, costs nothing, rolls no events, and both targets are 0, so renown and quality fade.
#   Feeds: qscale = 0.5 + quality / 100, which sizes every seat and curriculum modifier
#   and the members' yearly experience; disputation options; curriculum unlocks.
#
# Per-slot state (country): cc_xp_school_N_ renown, quality, qscale, age, founded,
# renown_target, quality_target, standing, cost, capacity, apt (admission), apt_min,
# apt_avg, enrol_chance, and the target breakdown maturity, r_rank, r_prestige, r_wealth,
# r_learning, r_quality, q_funding, q_rector, q_stable, q_members, all for the panel.
# Per-realm: cc_xp_school_rival_loc, cc_xp_school_rival_score, cc_xp_school_best_renown,
# cc_xp_school_culture_pos and cc_xp_school_no_culture (the funding source, for the panel).
###############################################################################

# Country scope. Starting values for a new school, and for one founded before this existed.
cc_xp_school_init_standing = {
	# Unknown at first: renown takes about a century to build (see RENOWN AND QUALITY).
	set_variable = { name = cc_xp_school_$N$_renown   value = 0 }
	set_variable = { name = cc_xp_school_$N$_quality  value = 40 }
	set_variable = { name = cc_xp_school_$N$_age      value = 0 }
	# The year it was founded, for the panel. A school founded before this existed is
	# dated to the month it was first given renown and quality.
	set_variable = { name = cc_xp_school_$N$_founded  value = current_year }
	set_variable = { name = cc_xp_school_$N$_standing value = 0 }
	# Written directly rather than through cc_xp_school_set_qscale: at founding this runs
	# while the interaction tooltip is built, when the quality written above cannot yet be
	# read back. 0.9 is 0.5 + 40 / 100.
	set_variable = { name = cc_xp_school_$N$_qscale   value = 0.9 }
	set_variable = { name = cc_xp_school_$N$_capacity value = 4 }
	cc_xp_school_init_admission = { N = $N$ }
}

cc_xp_school_set_qscale = {
	set_variable = {
		name = cc_xp_school_$N$_qscale
		value = {
			value = var:cc_xp_school_$N$_quality
			divide = 100
			add = 0.5
		}
	}
}

# Country scope. Push slot N's renown and quality by RENOWN and QUALITY (either may be 0
# or negative), clamped to 0-100.
cc_xp_school_change = {
	if = {
		limit = { has_variable = cc_xp_school_$N$_renown }
		change_variable = { name = cc_xp_school_$N$_renown  add = $RENOWN$ }
		clamp_variable = { name = cc_xp_school_$N$_renown  min = 0  max = 100 }
		change_variable = { name = cc_xp_school_$N$_quality  add = $QUALITY$ }
		clamp_variable = { name = cc_xp_school_$N$_quality  min = 0  max = 100 }
		cc_xp_school_set_qscale = { N = $N$ }
	}
}


# Country scope, yearly. Age, recompute both targets with their breakdowns, drift.
cc_xp_school_yearly_standing = {
	if = {
		limit = { has_variable = cc_xp_school_$N$_founder }
		save_scope_as = cc_xp_school_court

		if = {
			limit = { NOT = { has_variable = cc_xp_school_$N$_renown } }
			cc_xp_school_init_standing = { N = $N$ }
		}

		# A school with no living members is dormant: nobody teaches there and nobody
		# carries its name, so it does not age, is not paid for, and both targets are 0
		# (renown and quality fade). It wakes as soon as anyone is enrolled again.
		cc_xp_school_count_members = { N = $N$ }
		if = {
			limit = { var:cc_xp_school_$N$_members <= 0 }
			set_variable = { name = cc_xp_school_$N$_dormant  value = yes }
		}
		else = {
			remove_variable = cc_xp_school_$N$_dormant
			change_variable = { name = cc_xp_school_$N$_age  add = 1 }
		}

		# ---- Renown target ----
		# Established (20-100%, full at 100 years) scales every term, so a new school is
		# known only locally however fine its seat. The published terms are the scaled
		# ones, so the panel's rows add up to the target.
		set_variable = {
			name = cc_xp_school_$N$_maturity
			value = { value = var:cc_xp_school_$N$_age  multiply = 0.8  add = 20  max = 100 }
		}
		set_variable = {
			name = cc_xp_school_$N$_r_rank
			value = { value = cc_xp_renown_rank  multiply = var:cc_xp_school_$N$_maturity  divide = 100 }
		}
		set_variable = {
			name = cc_xp_school_$N$_r_prestige
			value = { value = cc_xp_renown_prestige  multiply = var:cc_xp_school_$N$_maturity  divide = 100 }
		}
		set_variable = { name = cc_xp_school_$N$_r_wealth  value = 0 }
		set_variable = { name = cc_xp_school_$N$_r_learning  value = 0 }
		clear_saved_scope = cc_xp_school_seat_scope
		var:cc_xp_school_$N$_loc ?= { save_scope_as = cc_xp_school_seat_scope }
		if = {
			limit = {
				exists = scope:cc_xp_school_seat_scope
				NOT = { has_variable = cc_xp_school_$N$_exile }
			}
			set_variable = {
				name = cc_xp_school_$N$_r_wealth
				value = {
					value = scope:cc_xp_school_seat_scope.cc_xp_renown_wealth
					multiply = var:cc_xp_school_$N$_maturity
					divide = 100
				}
			}
			set_variable = {
				name = cc_xp_school_$N$_r_learning
				value = {
					value = scope:cc_xp_school_seat_scope.cc_xp_renown_learning
					multiply = var:cc_xp_school_$N$_maturity
					divide = 100
				}
			}
		}
		set_variable = {
			name = cc_xp_school_$N$_r_quality
			value = { value = var:cc_xp_school_$N$_quality  multiply = 0.3  multiply = var:cc_xp_school_$N$_maturity  divide = 100 }
		}
		set_variable = {
			name = cc_xp_school_$N$_renown_target
			value = {
				value = var:cc_xp_school_$N$_r_rank
				add = var:cc_xp_school_$N$_r_prestige
				add = var:cc_xp_school_$N$_r_wealth
				add = var:cc_xp_school_$N$_r_learning
				add = var:cc_xp_school_$N$_r_quality
				if = {
					limit = { has_variable = cc_xp_school_$N$_exile }
					multiply = 0.5
				}
				if = {
					limit = { has_variable = cc_xp_school_$N$_dormant }
					multiply = 0
				}
				max = 100
			}
		}
		# The terms the previous formula published. Nothing reads them any more.
		remove_variable = cc_xp_school_$N$_r_age
		remove_variable = cc_xp_school_$N$_r_seat
		remove_variable = cc_xp_school_$N$_r_realm

		# ---- Quality target ----
		# Funding rides the vanilla Culture budget slider. Before the realm can invest in
		# culture at all, schools live on a customary allowance worth 35.
		if = {
			limit = { modifier:has_cultural_maintenance = yes }
			set_variable = {
				name = cc_xp_school_$N$_q_funding
				value = { value = cultural_maintenance  multiply = 50  add = 20 }
			}
		}
		else = {
			set_variable = { name = cc_xp_school_$N$_q_funding  value = 35 }
		}

		# Rector: 5, plus seniority, plus a tenth of their skill in this school's discipline.
		set_variable = { name = cc_xp_school_$N$_q_rector  value = 0 }
		clear_saved_scope = cc_xp_school_rector_scope
		var:cc_xp_school_$N$_rector ?= { save_scope_as = cc_xp_school_rector_scope }
		if = {
			limit = { exists = scope:cc_xp_school_rector_scope }
			set_variable = { name = cc_xp_school_$N$_q_rector  value = 5 }
			if = {
				limit = { scope:cc_xp_school_rector_scope = { has_variable = cc_level } }
				change_variable = { name = cc_xp_school_$N$_q_rector  add = scope:cc_xp_school_rector_scope.var:cc_level }
			}
			if = {
				limit = { var:cc_xp_school_$N$_track ?= 1 }
				change_variable = { name = cc_xp_school_$N$_q_rector  add = { value = scope:cc_xp_school_rector_scope.adm  divide = 10  max = 10 } }
			}
			else_if = {
				limit = { var:cc_xp_school_$N$_track ?= 2 }
				change_variable = { name = cc_xp_school_$N$_q_rector  add = { value = scope:cc_xp_school_rector_scope.dip  divide = 10  max = 10 } }
			}
			else_if = {
				limit = { var:cc_xp_school_$N$_track ?= 3 }
				change_variable = { name = cc_xp_school_$N$_q_rector  add = { value = scope:cc_xp_school_rector_scope.mil  divide = 10  max = 10 } }
			}
		}

		set_variable = { name = cc_xp_school_$N$_q_stable  value = 0 }
		if = {
			limit = { NOT = { has_variable = cc_xp_school_$N$_curr_cd } }
			set_variable = { name = cc_xp_school_$N$_q_stable  value = 5 }
		}

		# Members: one pass that also counts each member's years enrolled (for graduation)
		# and sums their aptitude, so quality reads the average rather than a head count.
		set_variable = { name = cc_xp_school_$N$_apt_sum  value = 0 }
		set_variable = { name = cc_xp_school_$N$_apt_n  value = 0 }
		every_character = {
			limit = {
				is_alive = yes
				var:cc_xp_school ?= $N$
			}
			if = {
				limit = { NOT = { has_variable = cc_xp_school_years } }
				set_variable = { name = cc_xp_school_years  value = 0 }
			}
			change_variable = { name = cc_xp_school_years  add = 1 }
			save_scope_as = cc_xp_school_member
			scope:cc_xp_school_court = {
				change_variable = { name = cc_xp_school_$N$_apt_sum  add = scope:cc_xp_school_member.cc_xp_aptitude_school }
				change_variable = { name = cc_xp_school_$N$_apt_n  add = 1 }
			}
		}
		set_variable = { name = cc_xp_school_$N$_apt_avg  value = 0 }
		if = {
			limit = { var:cc_xp_school_$N$_apt_n > 0 }
			set_variable = {
				name = cc_xp_school_$N$_apt_avg
				value = { value = var:cc_xp_school_$N$_apt_sum  divide = var:cc_xp_school_$N$_apt_n }
			}
		}
		set_variable = {
			name = cc_xp_school_$N$_q_members
			value = { value = var:cc_xp_school_$N$_apt_avg  multiply = 0.2  max = 20  min = 0 }
		}
		remove_variable = cc_xp_school_$N$_apt_sum
		remove_variable = cc_xp_school_$N$_apt_n

		if = {
			limit = { has_variable = cc_xp_school_$N$_dormant }
			set_variable = { name = cc_xp_school_$N$_quality_target  value = 0 }
		}
		else_if = {
			limit = { has_variable = cc_xp_school_$N$_exile }
			set_variable = { name = cc_xp_school_$N$_quality_target  value = 10 }
		}
		else = {
			set_variable = {
				name = cc_xp_school_$N$_quality_target
				value = {
					value = var:cc_xp_school_$N$_q_funding
					add = var:cc_xp_school_$N$_q_rector
					add = var:cc_xp_school_$N$_q_stable
					add = var:cc_xp_school_$N$_q_members
					max = 100
				}
			}
		}

		# ---- Drift ----
		# Renown builds slowly (8% of the gap a year) and is lost faster (15%).
		if = {
			limit = { var:cc_xp_school_$N$_renown_target > var:cc_xp_school_$N$_renown }
			change_variable = {
				name = cc_xp_school_$N$_renown
				add = {
					value = var:cc_xp_school_$N$_renown_target
					subtract = var:cc_xp_school_$N$_renown
					multiply = 0.08
				}
			}
		}
		else = {
			change_variable = {
				name = cc_xp_school_$N$_renown
				add = {
					value = var:cc_xp_school_$N$_renown_target
					subtract = var:cc_xp_school_$N$_renown
					multiply = 0.15
				}
			}
		}
		change_variable = {
			name = cc_xp_school_$N$_quality
			add = {
				value = var:cc_xp_school_$N$_quality_target
				subtract = var:cc_xp_school_$N$_quality
				multiply = 0.2
			}
		}
		clamp_variable = { name = cc_xp_school_$N$_renown  min = 0  max = 100 }
		clamp_variable = { name = cc_xp_school_$N$_quality  min = 0  max = 100 }
		cc_xp_school_set_qscale = { N = $N$ }
		cc_xp_school_set_capacity = { N = $N$ }

		# How far the leading foreign university is ahead, for the rival events and panel.
		# Uses last year's scan; cc_xp_school_scan_rivals runs after this.
		set_variable = { name = cc_xp_school_$N$_gap  value = 0 }
		if = {
			limit = { has_variable = cc_xp_school_rival_score }
			set_variable = {
				name = cc_xp_school_$N$_gap
				value = { value = var:cc_xp_school_rival_score  subtract = var:cc_xp_school_$N$_renown }
			}
		}
	}
}


###############################################################################
# CAPACITY, ADMISSION, SELF-ENROLMENT AND GRADUATION
#
# A school fills itself. Once a year each school may draw members from the realm's
# characters of their own accord, and members who are not in the cabinet graduate after
# a fixed term, which frees their places. The player can always enrol anyone by hand
# (cc_xp_enroll_in_school); hand enrolment ignores both capacity and admission.
#
# CAPACITY  3 + renown / 20 + quality / 25, rounded down: 3 to 12. Self-enrolment stops
#           at capacity; hand enrolment does not.
# ADMISSION Low / Medium / High (cc_xp_school_N_apt 1-3), the least aptitude in the
#           school's discipline (cc_xp_aptitude_school) a character needs to join of their
#           own accord: 0 / 40 / 60. Changed from the school card, then 5 years before it
#           can be changed again. Raising it never expels anyone. A Low school grows large
#           and broad (see BREADTH below); a High school stays small and, because quality
#           reads the members' average aptitude, good.
# ENROLMENT Two rolls a year. Each succeeds with chance
#               (15 + 0.3 x renown + 0.2 x quality)% x free places / capacity
#           and enrols one eligible character, weighted toward aptitude, a primary or
#           accepted culture, the state religion, the seat's culture, and proteges.
#           Eligible: adult, alive, not the ruler, not away, not already a member or a
#           graduate, aptitude at or above the admission bar. Cabinet ministers, proteges,
#           the ruler's family and every other character of the realm all qualify.
# GRADUATE  A member not in the cabinet graduates after GRADUATION_YEARS (8) years. They
#           leave the school with cc_xp_alumnus = N and a permanent cc_xp_school_alumnus
#           character modifier sized by the school's quality on the day (1x to 4x).
#           Ministers stay enrolled while they hold office, and graduate at the first yearly
#           update after leaving it if their term is served.
#
# Silent by design: the school card shows the membership against capacity.
###############################################################################

# Country scope. New and pre-existing schools start at Low admission.
cc_xp_school_init_admission = {
	if = {
		limit = { NOT = { has_variable = cc_xp_school_$N$_apt } }
		set_variable = { name = cc_xp_school_$N$_apt  value = 1 }
		set_variable = { name = cc_xp_school_$N$_apt_min  value = 0 }
	}
}

# Country scope. Yearly, after the drift.
cc_xp_school_set_capacity = {
	set_variable = {
		name = cc_xp_school_$N$_capacity
		value = {
			value = 3
			add = { value = var:cc_xp_school_$N$_renown  divide = 20 }
			add = { value = var:cc_xp_school_$N$_quality  divide = 25 }
			floor = yes
		}
	}
	# The chance each self-enrolment roll succeeds, published for the panel.
	set_variable = {
		name = cc_xp_school_$N$_enrol_chance
		value = {
			value = 15
			add = { value = var:cc_xp_school_$N$_renown  multiply = 0.3 }
			add = { value = var:cc_xp_school_$N$_quality  multiply = 0.2 }
			multiply = {
				value = var:cc_xp_school_$N$_capacity
				subtract = var:cc_xp_school_$N$_members
				min = 0
			}
			divide = var:cc_xp_school_$N$_capacity
			min = 0
		}
	}
}

# Country scope, monthly (human courts). Keeps the card's capacity and chance current.
cc_xp_school_refresh_capacity = {
	if = {
		limit = {
			has_variable = cc_xp_school_$N$_founder
			has_variable = cc_xp_school_$N$_renown
			has_variable = cc_xp_school_$N$_members
		}
		cc_xp_school_set_capacity = { N = $N$ }
	}
}

# The Admission button on a school card: Low -> Medium -> High -> Low.
cc_xp_school_do_admission = {
	if = {
		limit = { var:cc_xp_school_$N$_apt ?= 1 }
		custom_tooltip = { text = cc_xp_school_tt_admission_to_2 }
		hidden_effect = {
			set_variable = { name = cc_xp_school_$N$_apt  value = 2 }
			set_variable = { name = cc_xp_school_$N$_apt_min  value = 40 }
		}
	}
	else_if = {
		limit = { var:cc_xp_school_$N$_apt ?= 2 }
		custom_tooltip = { text = cc_xp_school_tt_admission_to_3 }
		hidden_effect = {
			set_variable = { name = cc_xp_school_$N$_apt  value = 3 }
			set_variable = { name = cc_xp_school_$N$_apt_min  value = 60 }
		}
	}
	else = {
		custom_tooltip = { text = cc_xp_school_tt_admission_to_1 }
		hidden_effect = {
			set_variable = { name = cc_xp_school_$N$_apt  value = 1 }
			set_variable = { name = cc_xp_school_$N$_apt_min  value = 0 }
		}
	}
	custom_tooltip = { text = cc_xp_school_tt_admission_keeps }
	hidden_effect = {
		set_variable = { name = cc_xp_school_$N$_adm_cd  value = yes  years = 5 }
	}
}

# Country scope, yearly. Members not in the cabinet whose term is served graduate.
cc_xp_school_graduate = {
	if = {
		limit = { has_variable = cc_xp_school_$N$_founder }
		save_scope_as = cc_xp_school_court
		every_character = {
			limit = {
				is_alive = yes
				var:cc_xp_school ?= $N$
				NOT = { in_cabinet = yes }
				has_variable = cc_xp_school_years
				var:cc_xp_school_years >= 8
			}
			cc_xp_school_graduate_member = { N = $N$ }
		}
	}
}

# Character scope; scope:cc_xp_school_court is their country.
cc_xp_school_graduate_member = {
	add_character_modifier = {
		modifier = cc_xp_school_alumnus
		years = -1
		mode = replace
		size = {
			value = scope:cc_xp_school_court.var:cc_xp_school_$N$_quality
			divide = 33
			add = 1
		}
	}
	set_variable = { name = cc_xp_alumnus  value = $N$ }
	remove_variable = cc_xp_school
	remove_variable = cc_xp_school_track
	remove_variable = cc_xp_school_years
}

# Character scope; scope:cc_xp_school_court is the realm. Enrols them in slot N.
cc_xp_school_self_enrol_member = {
	cc_xp_init = yes
	set_variable = { name = cc_xp_school  value = $N$ }
	set_variable = { name = cc_xp_school_track  value = scope:cc_xp_school_court.var:cc_xp_school_$N$_track }
	set_variable = { name = cc_xp_school_years  value = 0 }
	scope:cc_xp_school_court = {
		remove_variable = cc_xp_school_$N$_dormant
		change_variable = { name = cc_xp_school_$N$_members  add = 1 }
	}
}

# Country scope, yearly, after graduation. Two rolls for slot N.
cc_xp_school_self_enrol = {
	if = {
		limit = {
			has_variable = cc_xp_school_$N$_founder
			has_variable = cc_xp_school_$N$_capacity
		}
		save_scope_as = cc_xp_school_court
		cc_xp_school_init_admission = { N = $N$ }
		# Graduation has just freed places; recount before rolling.
		cc_xp_school_count_members = { N = $N$ }
		cc_xp_school_set_capacity = { N = $N$ }
		cc_xp_school_self_enrol_roll = { N = $N$ }
		cc_xp_school_set_capacity = { N = $N$ }
		cc_xp_school_self_enrol_roll = { N = $N$ }
		cc_xp_school_set_capacity = { N = $N$ }
	}
}

cc_xp_school_self_enrol_roll = {
	if = {
		limit = {
			var:cc_xp_school_$N$_members < var:cc_xp_school_$N$_capacity
			any_character = { cc_xp_school_would_enrol = { N = $N$ } }
		}
		random = {
			chance = { value = var:cc_xp_school_$N$_enrol_chance }
			random_character = {
				limit = { cc_xp_school_would_enrol = { N = $N$ } }
				weight = {
					factor = 1
					# Aptitude: 0.4x at 0, 1x at 30, 2.4x at 100.
					modifier = {
						factor = { value = cc_xp_aptitude_adm  add = 20  divide = 50 }
						trigger = { scope:cc_xp_school_court = { var:cc_xp_school_$N$_track ?= 1 } }
					}
					modifier = {
						factor = { value = cc_xp_aptitude_dip  add = 20  divide = 50 }
						trigger = { scope:cc_xp_school_court = { var:cc_xp_school_$N$_track ?= 2 } }
					}
					modifier = {
						factor = { value = cc_xp_aptitude_mil  add = 20  divide = 50 }
						trigger = { scope:cc_xp_school_court = { var:cc_xp_school_$N$_track ?= 3 } }
					}
					# Culture: at home in the realm, tolerated, or neither.
					modifier = {
						factor = 1.5
						trigger = {
							exists = culture
							scope:cc_xp_school_court = { has_primary_or_accepted_culture = prev.culture }
						}
					}
					modifier = {
						factor = 0.4
						trigger = {
							exists = culture
							NOT = { scope:cc_xp_school_court = { has_primary_or_accepted_culture = prev.culture } }
							NOT = { culture = { is_tolerated_in = scope:cc_xp_school_court } }
						}
					}
					# The seat's own culture: the school teaches in their tongue.
					modifier = {
						factor = 1.5
						trigger = {
							scope:cc_xp_school_court = { has_variable = cc_xp_school_$N$_culture }
							culture ?= scope:cc_xp_school_court.var:cc_xp_school_$N$_culture
						}
					}
					# Religion: the state faith, or not.
					modifier = {
						factor = 1.5
						trigger = { religion ?= scope:cc_xp_school_court.religion }
					}
					modifier = {
						factor = 0.5
						trigger = { NOT = { religion ?= scope:cc_xp_school_court.religion } }
					}
					# Proteges are being prepared for office; ministers have less time.
					modifier = {
						factor = 2
						trigger = { has_variable = cc_protege }
					}
					modifier = {
						factor = 0.75
						trigger = { in_cabinet = yes }
					}
				}
				cc_xp_school_self_enrol_member = { N = $N$ }
			}
		}
	}
}


###############################################################################
# BREADTH
#
# Every member, of whatever calibre, carries the school's discipline into the realm. One
# country modifier per discipline, sized by the living members of all schools teaching it,
# with diminishing returns: the first 4 count fully, the next 4 half, the next 8 a quarter
# (at most 8, at 16 members). This is what a large Low-admission school is for.
# Monthly, human courts (cc_xp_monthly_schools), after the members are counted.
###############################################################################

cc_xp_school_apply_breadth = {
	cc_xp_school_breadth_track = { TRACK = 1  KEY = adm }
	cc_xp_school_breadth_track = { TRACK = 2  KEY = dip }
	cc_xp_school_breadth_track = { TRACK = 3  KEY = mil }
}

cc_xp_school_breadth_track = {
	set_variable = { name = cc_xp_school_breadth_n  value = 0 }
	if = {
		limit = { var:cc_xp_school_1_track ?= $TRACK$  has_variable = cc_xp_school_1_members }
		change_variable = { name = cc_xp_school_breadth_n  add = var:cc_xp_school_1_members }
	}
	if = {
		limit = { var:cc_xp_school_2_track ?= $TRACK$  has_variable = cc_xp_school_2_members }
		change_variable = { name = cc_xp_school_breadth_n  add = var:cc_xp_school_2_members }
	}
	if = {
		limit = { var:cc_xp_school_3_track ?= $TRACK$  has_variable = cc_xp_school_3_members }
		change_variable = { name = cc_xp_school_breadth_n  add = var:cc_xp_school_3_members }
	}
	if = {
		limit = { var:cc_xp_school_breadth_n > 0 }
		add_country_modifier = {
			modifier = cc_xp_school_breadth_$KEY$
			months = 2
			mode = replace
			desc = cc_xp_school_while_members
			size = {
				value = { value = var:cc_xp_school_breadth_n  max = 4 }
				add = { value = var:cc_xp_school_breadth_n  subtract = 4  min = 0  max = 4  multiply = 0.5 }
				add = { value = var:cc_xp_school_breadth_n  subtract = 8  min = 0  max = 8  multiply = 0.25 }
			}
		}
	}
	remove_variable = cc_xp_school_breadth_n
}


# Country scope, yearly. Score every foreign university and rank our schools against them.
# A university is any foreign location with the vanilla university building. Runs for
# human courts only: nobody else sees the ranking, and it walks every realm's locations.
cc_xp_school_scan_rivals = {
	save_scope_as = cc_xp_school_court
	set_variable = { name = cc_xp_school_rival_score  value = 0 }
	remove_variable = cc_xp_school_rival_loc
	set_variable = { name = cc_xp_school_1_standing  value = 1 }
	set_variable = { name = cc_xp_school_2_standing  value = 1 }
	set_variable = { name = cc_xp_school_3_standing  value = 1 }

	every_country = {
		limit = { NOT = { this = scope:cc_xp_school_court } }
		every_owned_location = {
			limit = { has_building = building_type:university }
			save_scope_as = cc_xp_school_rival_candidate

			scope:cc_xp_school_court = {
				# THE RIVAL IS CONTINENTAL. The rival behind the poaching and student events
				# (cc_xp_school.40/.41) and the renown gap that rolls them is the best university
				# on our capital's continent: a Byzantine master is courted by Paris or Bologna,
				# not by a school in Vietnam. Standing below stays a world ranking, as the panel
				# and cc_xp_school.42 describe it.
				if = {
					limit = {
						scope:cc_xp_school_rival_candidate = { continent = scope:cc_xp_school_court.capital.continent }
						scope:cc_xp_school_rival_candidate.cc_xp_university_score > var:cc_xp_school_rival_score
					}
					set_variable = { name = cc_xp_school_rival_score  value = scope:cc_xp_school_rival_candidate.cc_xp_university_score }
					set_variable = { name = cc_xp_school_rival_loc  value = scope:cc_xp_school_rival_candidate }
				}
				if = {
					limit = {
						has_variable = cc_xp_school_1_renown
						scope:cc_xp_school_rival_candidate.cc_xp_university_score > var:cc_xp_school_1_renown
					}
					change_variable = { name = cc_xp_school_1_standing  add = 1 }
				}
				if = {
					limit = {
						has_variable = cc_xp_school_2_renown
						scope:cc_xp_school_rival_candidate.cc_xp_university_score > var:cc_xp_school_2_renown
					}
					change_variable = { name = cc_xp_school_2_standing  add = 1 }
				}
				if = {
					limit = {
						has_variable = cc_xp_school_3_renown
						scope:cc_xp_school_rival_candidate.cc_xp_university_score > var:cc_xp_school_3_renown
					}
					change_variable = { name = cc_xp_school_3_standing  add = 1 }
				}
			}
		}
	}

	# The realm's renown modifier follows its most renowned seated school.
	set_variable = { name = cc_xp_school_best_renown  value = 0 }
	if = {
		limit = { cc_xp_school_is_seated = { N = 1 }  var:cc_xp_school_1_renown > var:cc_xp_school_best_renown }
		set_variable = { name = cc_xp_school_best_renown  value = var:cc_xp_school_1_renown }
	}
	if = {
		limit = { cc_xp_school_is_seated = { N = 2 }  var:cc_xp_school_2_renown > var:cc_xp_school_best_renown }
		set_variable = { name = cc_xp_school_best_renown  value = var:cc_xp_school_2_renown }
	}
	if = {
		limit = { cc_xp_school_is_seated = { N = 3 }  var:cc_xp_school_3_renown > var:cc_xp_school_best_renown }
		set_variable = { name = cc_xp_school_best_renown  value = var:cc_xp_school_3_renown }
	}
}


# Country scope, monthly. The realm renown modifier, sized by the best renown.
cc_xp_school_apply_renown = {
	if = {
		limit = {
			has_variable = cc_xp_school_best_renown
			var:cc_xp_school_best_renown > 0
		}
		add_country_modifier = {
			modifier = cc_xp_school_renown
			months = 2
			mode = replace
			desc = cc_xp_school_while_renowned
			size = { value = var:cc_xp_school_best_renown  divide = 100 }
		}
	}
}


# Country scope, monthly. What slot N costs, published for the panel and paid for a seated
# school: 5% of the realm's monthly trade and tax income at a full Culture budget slider,
# scaled down with the slider. Nothing before the realm can invest in culture, when schools
# live on their customary allowance.
#
# The engine's budget sliders are a fixed list (EconomyView.GetMaintenanceSettings), so a
# school cannot have a slider of its own; it follows the Culture slider instead. The cost
# is a plain gold deduction rather than a budget line, because vanilla has no modifier
# that raises the culture slider's own expense.
cc_xp_school_fund = {
	set_variable = { name = cc_xp_school_$N$_cost  value = 0 }
	if = {
		limit = {
			modifier:has_cultural_maintenance = yes
			cc_xp_school_is_seated = { N = $N$ }
			NOT = { has_variable = cc_xp_school_$N$_dormant }
		}
		set_variable = {
			name = cc_xp_school_$N$_cost
			value = {
				value = monthly_income_trade_and_tax
				multiply = 0.05
				multiply = cultural_maintenance
				min = 0
			}
		}
		if = {
			limit = { var:cc_xp_school_$N$_cost > 0 }
			add_gold = { value = var:cc_xp_school_$N$_cost  multiply = -1 }
		}
	}
}

# Country scope, monthly. Publish the funding source for the panel, which cannot read
# cultural_maintenance or a modifier flag itself.
cc_xp_school_publish_funding = {
	if = {
		limit = { modifier:has_cultural_maintenance = yes }
		remove_variable = cc_xp_school_no_culture
		set_variable = { name = cc_xp_school_culture_pos  value = cultural_maintenance }
	}
	else = {
		set_variable = cc_xp_school_no_culture
		set_variable = { name = cc_xp_school_culture_pos  value = 0 }
	}
}


# Country scope, yearly. Roll slot N's renown events. Each sets the slot's flag and fires
# an event that finds the school by it, like every other school event. Checked rarest
# first; the first to fire sets a one-year cooldown, so a school raises at most one a year.
#   first    standing first in the world, renown 60+, 10%      cc_xp_school.42
#   rival    the leading foreign university is 10+ ahead, 20%   cc_xp_school.40 / .41
#   artist   renown 50+, chance renown / 12 percent             cc_xp_school.32
#   scholar  renown 40+, chance renown / 20 percent, and at most one
#            a decade for the whole realm (cc_xp_school_scholar_cd)  cc_xp_school.30
#   student  renown 30+, 6%                                     cc_xp_school.31
cc_xp_school_roll_events = {
	if = {
		limit = {
			is_ai = no
			cc_xp_school_is_seated = { N = $N$ }
			has_variable = cc_xp_school_$N$_renown
			NOT = { has_variable = cc_xp_school_$N$_dormant }
		}

		if = {
			limit = {
				NOT = { has_variable = cc_xp_school_$N$_event_cd }
				var:cc_xp_school_$N$_standing ?= 1
				var:cc_xp_school_$N$_renown >= 60
			}
			random = {
				chance = 10
				cc_xp_school_fire = { N = $N$  FLAG = first  EVENT = 42 }
			}
		}
		if = {
			limit = {
				NOT = { has_variable = cc_xp_school_$N$_event_cd }
				has_variable = cc_xp_school_rival_loc
				has_variable = cc_xp_school_$N$_gap
				var:cc_xp_school_$N$_gap > 10
			}
			random = {
				chance = 10
				cc_xp_school_fire = { N = $N$  FLAG = poached  EVENT = 40 }
			}
		}
		if = {
			limit = {
				NOT = { has_variable = cc_xp_school_$N$_event_cd }
				has_variable = cc_xp_school_rival_loc
				has_variable = cc_xp_school_$N$_gap
				var:cc_xp_school_$N$_gap > 10
			}
			random = {
				chance = 10
				cc_xp_school_fire = { N = $N$  FLAG = leaving  EVENT = 41 }
			}
		}
		if = {
			limit = {
				NOT = { has_variable = cc_xp_school_$N$_event_cd }
				var:cc_xp_school_$N$_renown >= 50
			}
			random = {
				chance = { value = var:cc_xp_school_$N$_renown  divide = 12 }
				cc_xp_school_fire = { N = $N$  FLAG = artist  EVENT = 32 }
			}
		}
		if = {
			limit = {
				NOT = { has_variable = cc_xp_school_$N$_event_cd }
				NOT = { has_variable = cc_xp_school_scholar_cd }
				var:cc_xp_school_$N$_renown >= 40
			}
			random = {
				chance = { value = var:cc_xp_school_$N$_renown  divide = 20 }
				set_variable = { name = cc_xp_school_scholar_cd  value = yes  years = 10 }
				cc_xp_school_fire = { N = $N$  FLAG = scholar  EVENT = 30 }
			}
		}
		if = {
			limit = {
				NOT = { has_variable = cc_xp_school_$N$_event_cd }
				var:cc_xp_school_$N$_renown >= 30
			}
			random = {
				chance = 6
				cc_xp_school_fire = { N = $N$  FLAG = student  EVENT = 31 }
			}
		}
	}
}

cc_xp_school_fire = {
	set_variable = { name = cc_xp_school_$N$_event_cd  value = yes  years = 1 }
	set_variable = { name = cc_xp_school_$N$_$FLAG$  value = yes  months = 2 }
	trigger_event_non_silently = cc_xp_school.$EVENT$
}
