
ep_noble_elite_enclave = {
	estate = nobles_estate

	allow = {
		has_advance = noble_patronage_advance
		"estate_power(estate_type:nobles_estate)" >= 0.35
	}

	can_revoke = {
		NOT = {
			any_subject = {
				OR = {
					is_subject_type = elite_enclave
					is_subject_type = palatinate
				}
			}
		}
	}

	country_modifier = {
		global_nobles_estate_power = 0.5
		nobles_estate_target_satisfaction = small_privilege_target_satisfaction
		monthly_prestige = 0.05
		cultural_influence = 50
	}

	on_activate = {
		trigger_event_non_silently = { id = cc.112 }
	}
}
