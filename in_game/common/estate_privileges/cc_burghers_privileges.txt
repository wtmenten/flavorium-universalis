
ep_burgher_naval_charter = {
	estate = burghers_estate

	allow = {
		has_advance = naval_charter_advance
		"estate_power(estate_type:burghers_estate)" >= 0.30
	}

	can_revoke = {
		NOT = {
			any_subject = {
				is_subject_type = naval_administration
			}
		}
	}

	country_modifier = {
		global_burghers_estate_power = 0.25
		burghers_estate_target_satisfaction = small_privilege_target_satisfaction
		global_sailors_modifier = 0.05
		naval_morale = 0.05
		num_naval_governors = 1
	}
}
