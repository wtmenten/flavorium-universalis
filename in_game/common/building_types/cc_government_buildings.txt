###############################################################################
# Frontier Governance — Lesser Local Governor
# The land mirror of the vanilla naval_governor: a governor for territory that is
# land-connected to the capital (incl. through subjects) but lacks a road link and
# is too undeveloped for a true local_governor. Buildable in RURAL locations, gives
# LESS proximity than a real governor, and draws from its own count pool
# (cc_num_frontier_governors) granted by cc_frontier_administration_advance and the
# ep_frontier_governance privilege.
###############################################################################

cc_lesser_local_governor = {
	audio_tier = 2
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	important_for_UI = yes
	important_for_AI = yes
	
	employment_size = large_noble_employment

	category = government_category

	rural_settlement = yes      # key relaxation vs local_governor (which is city/megalopolis only)
	town = yes
	city = yes
	megalopolis = yes

	allow = {
		custom_tooltip = {
			text = cc_lesser_local_governor_tt
			scope:actor = {
				total_building_levels_including_construction:cc_lesser_local_governor < modifier:cc_num_frontier_governors
			}
		}
		owner = scope:actor      # own territory, not subjects'
		scope:actor = { has_advance = cc_frontier_administration_advance }
	}

	location_potential = {
		is_connected_to_through_realm = owner.capital   # land-linked, incl. through vassals
		has_road_to_capital = no                        # niche: only where a true governor cannot go
		owner.capital != this
	}

	remove_if = {
		OR = {
			location = { has_building = building_type:local_governor }   # superseded by a real local governor in this location
			location = { NOT = { is_connected_to_through_realm = owner.capital } }   # lost land link / transferred away
			owner = { NOT = { has_advance = cc_frontier_administration_advance } }
			owner = {
				total_building_levels_including_construction:cc_lesser_local_governor > modifier:cc_num_frontier_governors
			}
		}
	}

	expensive = yes

	automation_build_allowed = no
	ai_forbid_shutdown = yes
	forbidden_for_estates = yes
	important_for_AI = yes

	build_time = government_build_time

	unique_production_methods = {
		cc_lesser_local_governor_maintenance = {
			tools = 1
			masonry = 1
			paper = 2
			books = 1
			goods_gold = 1
			furniture = 1

			category = building_maintenance
		}
	}

	construction_demand = local_governor_construction

	modifier = {
		local_proximity_source = 50   # slightly more than half of local_governor's 80 -> less proximity than a real governor
	}
}
