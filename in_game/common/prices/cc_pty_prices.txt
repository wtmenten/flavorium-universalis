###############################################################################
# Flavorium Universalis - Factions: prices
#
# Party actions are paid mainly in political influence, the 1.4 currency vanilla already
# uses for cabinet changes. Each price needs a matching <price>_cost_modifier type in
# main_menu/common/modifier_type_definitions/cc_pty_prices.txt. A balanced cabinet takes
# 25% off both (cc_pty_comp_balanced). Asking a favour has no price: it is paid in the
# party's satisfaction.
###############################################################################

# Buy a party's goodwill: places, pensions and promises.
cc_pty_bribe = {
	political_influence = 10
	gold = 25
	scaled_gold = 0.5
}

# Move against a party. Political capital only: the cost is in who you have to lean on.
cc_pty_attack = {
	political_influence = 20
}

# Appointing a party's leader has no price: it EARNS political influence (+10, in the
# action's effect), because the party owes the crown for the seat.

# Meet a party's agenda by decree: the government's authority, some political capital,
# and money to make it happen.
cc_pty_concede = {
	government_power = 10
	political_influence = 10
	gold = 50
	scaled_gold = 1
}

# Call an election (the header button). Also costs stability, in the effect.
cc_pty_election = {
	political_influence = 10
	gold = 25
	scaled_gold = 0.5
}
