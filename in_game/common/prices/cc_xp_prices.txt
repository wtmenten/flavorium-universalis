###############################################################################
# Flavorium Universalis — Cabinet Experience: prices
#
# Costs for the training interactions. Using the engine's price system rather than
# add_gold in the effect means the cost is shown in the interaction UI, checked for
# affordability before the button is enabled, and paid by the engine.
#
# Every entry pairs a flat component with a scaled one (a multiple of monthly income),
# so training stays a real decision for a small realm without becoming free for a large
# one. min_scale sets the floor a tiny country actually pays.
#
# Available currencies are listed in the vanilla readme at
# game/in_game/common/prices/readme.txt: gold, scaled_gold, prestige, stability,
# government_power, manpower, sailors, war_exhaustion, inflation and the religious ones,
# plus political_influence since 1.4 (vanilla prices cabinet and advisor changes in it).
# Legitimacy is NOT among them. Political cost goes in political_influence here, and any
# estate cost goes in the interaction's own effect.
###############################################################################

# Send a minister to be taught. The workhorse cost, paid repeatedly over a career.
cc_xp_training = {
	gold = 50
	scaled_gold = 1.5
	min_scale = 40
}

# Take an outsider into the crown's care as a protege. Gold plus the political capital
# spent persuading the court that this person is worth the attention.
#
# The political_influence component is deliberately token (vanilla hires an advisor for 10). Sponsoring is the entry point to
# the whole grooming path, and a court is meant to keep two or three protegees running at
# once, so a real political price here would gate the system rather than season it. The
# ongoing cost is the training and mentorship each protege then needs.
#
# THE GOLD COMPONENTS ARE MIRRORED in script_values/cc_xp_values.txt as
# cc_xp_protege_refund_full, because a burghers or peasants candidate has their own
# estate pay some of it back and the engine gives no way to read a price back out of a
# price definition. If the gold below changes, change the refund with it.
cc_xp_sponsor_protege = {
	gold = 25
	scaled_gold = 0.5
	political_influence = 5
}

# Put a senior minister's time at a junior's disposal. Cheap in coin: the real cost is
# the mentor's attention, which the effect charges by occupying them.
cc_xp_mentorship = {
	gold = 15
	scaled_gold = 0.25
}

# Hold a formal examination of the whole court. A set piece, priced like one.
cc_xp_examination = {
	gold = 100
	scaled_gold = 2.5
	min_scale = 80
}

# Put a minister's name and money behind an artist for a generation.
cc_xp_patronage = {
	gold = 60
	scaled_gold = 1.0
	min_scale = 50
}

# Found a prestigious institution. The largest single outlay in the system, and one a
# realm can make at most three times.
cc_xp_found_institution = {
	gold = 400
	scaled_gold = 6.0
	min_scale = 300
	prestige = 10
}

# Enroll someone in an existing school. Fees, board, and a letter of introduction.
cc_xp_enrollment = {
	gold = 40
	scaled_gold = 0.8
	min_scale = 30
}

# Hire the hall. The upfront cost only: what the evening actually costs is decided on
# the night, in cc_xp.50.
cc_xp_soiree = {
	gold = 80
	scaled_gold = 1.2
	min_scale = 60
}

# Establish a resident at a subject's court: household, staff, and years of expenses.
cc_xp_secondment = {
	gold = 120
	scaled_gold = 2.0
	min_scale = 90
}

# Raise a minister to a post in the household. Priced between mentorship and training:
# the office pays for itself through its own modifier, so the cost is the elevation
# rather than an ongoing salary.
#
# The political component used to be 15 government power, on the reasoning that it was
# the only thing stopping a court filling all six slots the moment it could afford the
# gold. It is not any more: cc_office_household_upkeep charges -2% court spending
# efficiency for every post filled, so a full household runs at -12% and the brake is now
# ONGOING rather than a toll at the door. Paying a heavy political price on top charged
# twice for the same restraint. Since 1.4 the political part is political_influence,
# matching vanilla's cabinet and advisor prices, and the gold was halved to compensate.
cc_office_appointment = {
	gold = 35
	scaled_gold = 0.5
	min_scale = 25
	political_influence = 10
}

# Dismiss a minister to end court tension. Same as vanilla's replace_cabinet_member.
cc_dismiss_rival_minister_price = {
	political_influence = 20
}

# France forces a new ruler onto a disloyal appanage.
cc_hyw_replace_appanage_ruler_price = {
	political_influence = 20
	government_power = 10
}
