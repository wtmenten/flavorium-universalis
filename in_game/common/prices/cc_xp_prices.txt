###############################################################################
# Flavorium Universalis — Cabinet Experience: prices
#
# Costs for the training interactions. Using the engine's price system rather than
# add_gold in the effect means the cost is shown in the interaction UI, checked for
# affordability before the button is enabled, and paid by the engine.
#
# Every entry pairs a flat component with a scaled one (a multiple of monthly income),
# so training stays a real decision for a small realm without becoming free for a large
# one. min_scale sets the floor a tiny country actually pays.
#
# Available currencies are listed in the vanilla readme at
# game/in_game/common/prices/readme.txt: gold, scaled_gold, prestige, stability,
# government_power, manpower, sailors, war_exhaustion, inflation and the religious ones.
# Note that legitimacy is NOT among them, which is why the political cost of elevating
# an outsider is charged as government_power here and as estate satisfaction in the
# interaction's own effect.
###############################################################################

# Send a minister to be taught. The workhorse cost, paid repeatedly over a career.
cc_xp_training = {
	gold = 50
	scaled_gold = 1.5
	min_scale = 40
}

# Take an outsider into the crown's care as a protege. Gold plus the political capital
# spent persuading the court that this person is worth the attention.
cc_xp_sponsor_protege = {
	gold = 25
	scaled_gold = 0.5
	government_power = 10
}

# Put a senior minister's time at a junior's disposal. Cheap in coin: the real cost is
# the mentor's attention, which the effect charges by occupying them.
cc_xp_mentorship = {
	gold = 15
	scaled_gold = 0.25
}

# Hold a formal examination of the whole court. A set piece, priced like one.
cc_xp_examination = {
	gold = 100
	scaled_gold = 2.5
	min_scale = 80
}
