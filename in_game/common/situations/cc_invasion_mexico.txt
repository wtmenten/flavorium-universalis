
# Cabinets and Choices — Invasion of Mexico
# Fires when a colonial or non-American power establishes a foothold in Mesoamerica (or the
# adjacent Caribbean/Central America by 1540). Lasts until all native Mesoamerican powers are
# subjugated or the year 1600 is reached.

cc_invasion_mexico = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	hint_tag = hint_cc_invasion_mexico

	can_start = {
		has_enabled_situation_trigger = { type = cc_invasion_mexico }
		OR = {
			# A colonial nation or foreign power (non-Americas capital) owns locations in mesoamerica
			any_country = {
				OR = {
					is_subject_type = colonial_nation
					NOT = {
						capital ?= {
							OR = {
								sub_continent = sub_continent:north_america
								sub_continent = sub_continent:south_america
							}
						}
					}
				}
				has_presence_in = region:mesoamerica_region
			}
			# OR: same powers reach Caribbean/Central America after 1540
			AND = {
				current_year >= 1540
				any_country = {
					OR = {
						is_subject_type = colonial_nation
						NOT = {
							capital ?= {
								OR = {
									sub_continent = sub_continent:north_america
									sub_continent = sub_continent:south_america
								}
							}
						}
					}
					OR = {
						has_presence_in = region:central_america_region
						has_presence_in = region:caribbean_region
					}
				}
			}
		}
	}

	can_end = {
		OR = {
			current_year > 1600
			# All independent native powers in mesoamerica are subjugated or gone
			NOT = {
				region:mesoamerica_region = {
					any_country_with_capital_in_geography = {
						is_capital_mesoamerica = yes
						is_subject = no
					}
				}
			}
		}
	}

	visible = {
		NOT = { country_type = pop }
		NOT = { country_type = building }
		OR = {
			is_capital_mesoamerica = yes
			has_presence_in = region:mesoamerica_region
			has_presence_in = region:central_america_region
			has_presence_in = region:caribbean_region
		}
	}

	on_start = {
		# Save top 3 native powers by population for targeted start events
		region:mesoamerica_region = {
			ordered_country_with_capital_in_geography = {
				limit = { is_capital_mesoamerica = yes is_subject = no }
				order_by = total_population
				max = 1
				check_range_bounds = no
				situation:cc_invasion_mexico = {
					set_variable = { name = ccim_native_1 value = prev }
				}
			}
		}
		region:mesoamerica_region = {
			ordered_country_with_capital_in_geography = {
				limit = {
					is_capital_mesoamerica = yes
					is_subject = no
					NOT = { this = situation:cc_invasion_mexico.var:ccim_native_1 }
				}
				order_by = total_population
				max = 1
				check_range_bounds = no
				situation:cc_invasion_mexico = {
					set_variable = { name = ccim_native_2 value = prev }
				}
			}
		}
		region:mesoamerica_region = {
			ordered_country_with_capital_in_geography = {
				limit = {
					is_capital_mesoamerica = yes
					is_subject = no
					NOT = { this = situation:cc_invasion_mexico.var:ccim_native_1 }
					NOT = { this = situation:cc_invasion_mexico.var:ccim_native_2 }
				}
				order_by = total_population
				max = 1
				check_range_bounds = no
				situation:cc_invasion_mexico = {
					set_variable = { name = ccim_native_3 value = prev }
				}
			}
		}

		# Announcement event to all participants
		every_country = {
			limit = {
				NOT = { country_type = pop }
				NOT = { country_type = building }
				OR = {
					is_capital_mesoamerica = yes
					has_presence_in = region:mesoamerica_region
					has_presence_in = region:central_america_region
					has_presence_in = region:caribbean_region
				}
			}
			trigger_event_non_silently = cc_invasion_mexico.1
		}

		# Federation Request to the three largest native powers (staggered)
		situation:cc_invasion_mexico.var:ccim_native_1 ?= {
			trigger_event_non_silently = { id = cc_invasion_mexico.2 days = 5 }
		}
		situation:cc_invasion_mexico.var:ccim_native_2 ?= {
			trigger_event_non_silently = { id = cc_invasion_mexico.3 days = 10 }
		}
		situation:cc_invasion_mexico.var:ccim_native_3 ?= {
			trigger_event_non_silently = { id = cc_invasion_mexico.4 days = 15 }
		}

		# Defensive Preparation to native countries bordering colonial locations
		region:mesoamerica_region = {
			every_country_with_capital_in_geography = {
				limit = {
					is_capital_mesoamerica = yes
					any_neighbor_country = {
						NOT = { is_capital_mesoamerica = yes }
						OR = {
							has_presence_in = region:mesoamerica_region
							has_presence_in = region:central_america_region
							has_presence_in = region:caribbean_region
						}
					}
				}
				trigger_event_non_silently = { id = cc_invasion_mexico.5 days = 20 }
			}
		}

		# Colonial Neighbor Interaction to non-native countries adjacent to native provinces
		every_country = {
			limit = {
				NOT = { is_capital_mesoamerica = yes }
				NOT = { country_type = pop }
				NOT = { country_type = building }
				OR = {
					has_presence_in = region:mesoamerica_region
					has_presence_in = region:central_america_region
					has_presence_in = region:caribbean_region
				}
				any_neighbor_country = {
					is_capital_mesoamerica = yes
				}
			}
			trigger_event_non_silently = { id = cc_invasion_mexico.20 days = 25 }
		}
	}

	on_monthly = {
		# Monthly confederation solidarity events for members
		every_country = {
			limit = {
				is_member_of_international_organization = international_organization:cc_mesoamerican_confederation
				can_see_situation = root
			}
			random_list = {
				3 = {
					trigger_event_non_silently = cc_invasion_mexico.10
				}
				97 = { }
			}
		}

		# Monthly opportunity events for colonizers with regional presence
		every_country = {
			limit = {
				NOT = { is_capital_mesoamerica = yes }
				NOT = { country_type = pop }
				NOT = { country_type = building }
				OR = {
					has_presence_in = region:mesoamerica_region
					has_presence_in = region:central_america_region
					has_presence_in = region:caribbean_region
				}
				any_neighbor_country = {
					is_capital_mesoamerica = yes
				}
				can_see_situation = root
			}
			random_list = {
				3 = {
					trigger_event_non_silently = cc_invasion_mexico.25
				}
				97 = { }
			}
		}
	}

	on_ended = {
		# End flavor events
		region:mesoamerica_region = {
			every_country_with_capital_in_geography = {
				limit = { is_capital_mesoamerica = yes }
				trigger_event_non_silently = cc_invasion_mexico.35
			}
		}
		every_country = {
			limit = {
				NOT = { is_capital_mesoamerica = yes }
				NOT = { country_type = pop }
				NOT = { country_type = building }
				has_presence_in = region:mesoamerica_region
			}
			trigger_event_non_silently = cc_invasion_mexico.36
		}
		# Remove situation variables to keep savefile clean
		situation:cc_invasion_mexico = {
			remove_variable = ccim_native_1
			remove_variable = ccim_native_2
			remove_variable = ccim_native_3
		}
	}

	# Map tooltip per location
	tooltip = {
		if = {
			limit = {
				owner ?= {
					is_member_of_international_organization = international_organization:cc_mesoamerican_confederation
				}
			}
			custom_tooltip = ccim_confederation_member_tt
		}
		else_if = {
			limit = { owner ?= { is_capital_mesoamerica = yes } }
			custom_tooltip = ccim_native_unaffiliated_tt
		}
		else_if = {
			limit = {
				owner ?= { NOT = { is_capital_mesoamerica = yes } }
			}
			custom_tooltip = ccim_colonial_power_tt
		}
	}

	# Map colors: confederation=green, unaffiliated native=amber, colonizers=own color
	map_color = {
		if = {
			limit = {
				owner ?= {
					is_member_of_international_organization = international_organization:cc_mesoamerican_confederation
				}
			}
			value = rgb { 0 160 80 }
		}
		else_if = {
			limit = { owner ?= { is_capital_mesoamerica = yes } }
			value = rgb { 200 140 40 }
		}
		else_if = {
			limit = {
				owner ?= { NOT = { is_capital_mesoamerica = yes } }
			}
			value = owner.country_color
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}

	legend_key = {
		color = rgb { 0 160 80 }
		require_color_on_map = yes
		desc = CCIM_CONFEDERATION_LEGEND
	}
	legend_key = {
		color = rgb { 200 140 40 }
		require_color_on_map = yes
		desc = CCIM_NATIVE_UNAFFILIATED_LEGEND
	}
}
