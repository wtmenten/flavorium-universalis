
# Flavorium Universalis — Subject Relations Ledger
# A quiet, passive situation that exists purely to host the bond-status panel
# (in_game/gui/panels/situation/cc_subject_bonds.gui). It carries no map colour,
# no legend, no announcement event and no situation variables — the panel reads
# each subject's own cc_bond_* variables directly via Player.GetDiplomacy.GetSubjects,
# so each player only ever sees their own subjects. Visible only to overlords that
# have at least one bond-tracked subject while the bond game rule is on.

cc_subject_bonds = {
	# High spawn chance so the ledger appears promptly once an overlord has a tracked bond.
	monthly_spawn_chance = monthly_spawn_chance_ultimate_high

	can_start = {
		has_game_rule = cc_subject_bonds_on
		any_country = {
			is_overlord = yes
			any_subject = { has_variable = cc_bond_initialized }
		}
	}

	# Permanent once spawned. Deliberately never ends: this is an informational panel, not a
	# crisis, and tying it to "does any overlord still track a bond" would make it spawn and
	# despawn as dependencies come and go. The visible block below already hides it from every
	# court that has nothing to show, so a resident situation with no viewers costs nothing.
	can_end = {
		always = no
	}

	# Only overlords with a tracked subject see the header icon / panel. The panel body
	# is player-scoped (Player.GetDiplomacy.GetSubjects), so nothing leaks between courts.
	visible = {
		is_overlord = yes
		any_subject = { has_variable = cc_bond_initialized }
	}

	# ---------------------------------------------------------------------------------
	# MAP COLOUR
	#
	# root = location, scope:target = situation. Those are the ONLY scopes map_color gets:
	# the situations readme says so and no vanilla map_color in the game references a
	# player or an observer. The colour is computed once per location, globally, not once
	# per viewer.
	#
	# So "the player" has to be written as `is_ai = no`. In single player that is exactly
	# right. In multiplayer it colours every human realm and every human's subjects, which
	# is a knowing trade for having the ledger legible on the map at all.
	#
	# Subjects are shaded by the same thing the panel ranks them by, so the map and the
	# ledger cannot tell different stories. cc_bond_overall_factor is in
	# script_values/cc_bond_values.txt.
	# ---------------------------------------------------------------------------------
	map_color = {
		if = {
			limit = { owner ?= { is_ai = no } }
			value = rgb { 45 120 65 }
		}
		else_if = {
			limit = {
				owner ?= {
					has_variable = cc_bond_initialized
					overlord ?= { is_ai = no }
				}
			}
			lerp = {
				min_color = rgb { 175 205 230 }
				max_color = rgb { 20 55 115 }
				factor = "cc_bond_overall_factor"
			}
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}

	legend_key = {
		desc = "cc_bond_legend_overlord"
		color = rgb { 45 120 65 }
	}
	legend_key = {
		desc = "cc_bond_legend_weak"
		color = rgb { 175 205 230 }
	}
	legend_key = {
		desc = "cc_bond_legend_strong"
		color = rgb { 20 55 115 }
	}

	# Deliberately omitted: on_start announcement and tooltip. This is an informational
	# ledger, not a crisis.
}
