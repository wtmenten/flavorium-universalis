
# Flavorium Universalis — The European Balance of Power
# A two-pole great-power situation spanning Europe and the Near-East. An aspiring
# preponderant bloc (cc_bop_preponderance), led by the strongest power, is checked by
# a balancing coalition (cc_bop_coalition) led by the chief balancer. The situation
# tracks a 0-100 preponderance share (50 = equilibrium) and ends when the balance is
# broken — by hegemony, an eastern deluge, or revolution — or institutionalised as a
# peaceful Concert of Europe. Begins in the Age of Absolutism once a great power adopts
# the balance-of-power doctrine, or automatically once the revolutionary era opens.

cc_balance_of_power = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	hint_tag = hint_cc_balance_of_power

	can_start = {
		has_enabled_situation_trigger = { type = cc_balance_of_power }
		current_age_or_later = { age = age_5_absolutism }
		cc_bop_doctrine_or_era = yes
		# Hold until the earlier crises of the age have run their course — no stable
		# balance can form while Italy is still contested or the wars of religion rage.
		# (situation_is_active = no also covers games where these never fired at all,
		# so the gate can never deadlock.)
		# situation:italian_wars = { situation_is_active = no }
		situation:war_of_religions = { situation_is_active = no }
		# A decisive break (hegemony / deluge / revolution) settles the question for good.
		NOT = { has_global_variable = cc_bop_concluded }
		# A prior Concert of Europe may give way to a new balance, but only after its
		# ~25-year cooldown has lapsed.
		OR = {
			NOT = { has_global_variable = cc_bop_concert_restart_year }
			cc_bop_concert_cooldown_elapsed = yes
		}
		# At least two distinct eligible great powers — otherwise there is no balance to keep.
		any_country = {
			cc_bop_eligible_country = yes
			is_subject = no
			is_great_power = yes
			any_country = {
				cc_bop_eligible_country = yes
				is_subject = no
				is_great_power = yes
				NOT = { this = prev }
			}
		}
	}

	can_end = {
		situation:cc_balance_of_power = {
			OR = {
				has_variable = cc_bop_flag_hegemony
				has_variable = cc_bop_flag_coalition
				has_variable = cc_bop_flag_stagnation
				has_variable = cc_bop_flag_deluge
				has_variable = cc_bop_flag_revolution
				has_variable = cc_bop_flag_concert
			}
		}
	}

	visible = {
		cc_bop_eligible_country = yes
	}

	on_start = {
		cc_bop_designate_poles = yes

		# Mark the engaged participants. Great powers are always in the game; secondary
		# powers join the contest only sometimes; most minors remain bystanders.
		every_country = {
			limit = {
				cc_bop_eligible_country = yes
				is_subject = no
				NOT = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
				NOT = { is_member_of_international_organization = international_organization:cc_bop_coalition }
				NOT = { has_variable = cc_bop_engaged }
			}
			if = {
				limit = { is_great_power = yes }
				set_variable = cc_bop_engaged
			}
			else = {
				random_list = {
					30 = { set_variable = cc_bop_engaged }
					70 = { }
				}
			}
		}

		situation:cc_balance_of_power = {
			# Three end-state tickers (0-60). See on_monthly for the band logic.
			set_variable = { name = cc_bop_hegemony_months value = 0 }
			set_variable = { name = cc_bop_coalition_months value = 0 }
			set_variable = { name = cc_bop_stagnation_months value = 0 }
			# Escalating-cost counter for the preponderant's Call to Arms (resets per war).
			set_variable = { name = cc_bop_callins_this_war value = 0 }
			# Event-driven momentum layered on top of the structural share by
			# cc_bop_recompute_share; starts neutral and decays toward 0 monthly.
			set_variable = { name = cc_bop_share_offset value = 0 }
			# How long the contest has run (months). Drives the rising appetite for a durable
			# settlement — see the cc_bop_enact_concert policy's wants_propose_policy.
			set_variable = { name = cc_bop_situation_months value = 0 }
		}
		cc_bop_recompute_share = yes

		# Grand announcement to every eligible power.
		every_country = {
			limit = { cc_bop_eligible_country = yes is_subject = no }
			trigger_event_non_silently = cc_balance_of_power.1
		}

		# Role intros for the two leaders.
		international_organization:cc_bop_preponderance.leader_country ?= {
			trigger_event_non_silently = { id = cc_balance_of_power.2 days = 4 }
		}
		international_organization:cc_bop_coalition.leader_country ?= {
			trigger_event_non_silently = { id = cc_balance_of_power.3 days = 6 }
		}

		# Alignment choice to engaged powers that are not yet aligned.
		every_country = {
			limit = {
				cc_bop_eligible_country = yes
				is_subject = no
				has_variable = cc_bop_engaged
				NOT = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
				NOT = { is_member_of_international_organization = international_organization:cc_bop_coalition }
			}
			trigger_event_non_silently = { id = cc_balance_of_power.10 days = 12 }
		}

		# --- A balance re-forming from the ashes of a lapsed Concert of Europe ---
		# When the cooldown has elapsed and a fresh balance arises, the old concert is
		# formally broken: the courts resent the power whose renewed ambition shattered
		# the peace, and a heralding event plays. Clear the cooldown stamp so a future
		# concert can set its own.
		if = {
			limit = { has_global_variable = cc_bop_concert_restart_year }
			international_organization:cc_bop_preponderance.leader_country ?= {
				save_scope_as = bop_new_aspirant
			}
			every_country = {
				limit = {
					cc_bop_eligible_country = yes
					is_subject = no
					exists = scope:bop_new_aspirant
					NOT = { this = scope:bop_new_aspirant }
				}
				add_opinion = { target = scope:bop_new_aspirant  modifier = cc_bop_concert_betrayal }
				trigger_event_non_silently = { id = cc_balance_of_power.75 days = 8 }
			}
			remove_global_variable = cc_bop_concert_restart_year
		}
	}

	on_monthly = {
		cc_bop_recompute_share = yes

		# Tally the months the contest has run — fuels the rising appetite for a durable
		# settlement (the cc_bop_enact_concert policy weights itself by this).
		situation:cc_balance_of_power = {
			if = {
				limit = { has_variable = cc_bop_situation_months }
				change_variable = { name = cc_bop_situation_months add = 1 }
			}
			else = {
				set_variable = { name = cc_bop_situation_months value = 1 }
			}
		}

		# Reset the Call-to-Arms escalation once the hegemon-coalition war is over, so the
		# next war's first call-in starts cheap again.
		if = {
			limit = {
				situation:cc_balance_of_power = { var:cc_bop_callins_this_war > 0 }
				NOT = {
					any_country = {
						is_member_of_international_organization = international_organization:cc_bop_coalition
						exists = international_organization:cc_bop_preponderance
						is_at_war_with = international_organization:cc_bop_preponderance.leader_country
					}
				}
			}
			situation:cc_balance_of_power = { set_variable = { name = cc_bop_callins_this_war value = 0 } }
		}

		# War footing for the unaligned: engaged-but-unaligned participants forage freely and
		# march without attrition, just as the bloc IO modifiers grant their members. Refreshed
		# monthly (2-month window); once such a court joins a bloc the IO modifier takes over and
		# this one harmlessly lapses.
		# every_country = {
		# 	limit = {
		# 		cc_bop_eligible_country = yes
		# 		is_subject = no
		# 		has_variable = cc_bop_engaged
		# 		NOT = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
		# 		NOT = { is_member_of_international_organization = international_organization:cc_bop_coalition }
		# 	}
		# 	add_country_modifier = { modifier = cc_bop_war_footing  months = 2  mode = add_and_extend }
		# }

		# Event-driven momentum decays back toward equilibrium each month.
		situation:cc_balance_of_power = {
			if = {
				limit = { var:cc_bop_share_offset > 0.5 }
				change_variable = { name = cc_bop_share_offset subtract = 0.5 }
			}
			else_if = {
				limit = { var:cc_bop_share_offset < -0.5 }
				change_variable = { name = cc_bop_share_offset add = 0.5 }
			}
			else = {
				set_variable = { name = cc_bop_share_offset value = 0 }
			}
			clamp_variable = { name = cc_bop_share_offset min = -25 max = 25 }
		}

		# ============================================================================
		# END-STATE TICKERS — three roads can break the balance, each tracked by its own
		# 0-60 months counter. INVARIANT: the share bands are disjoint (>=65 hegemony,
		# <=35 coalition, the 35<share<65 middle ground for stagnation), so at most one
		# counter climbs in any month; the others bleed back toward 0. A *primed* reckoning
		# (a full hegemony/coalition counter) FREEZES all three counters until its final
		# war resolves, and no break flag may be set while another flag is set or a counter
		# is primed — guaranteeing the two victory races can never advance or fire together.
		# ============================================================================

		# --- Counters: fill in-band, bleed out-of-band; frozen while a reckoning is primed.
		situation:cc_balance_of_power = {
			if = {
				limit = {
					NOT = { has_variable = cc_bop_hegemony_primed }
					NOT = { has_variable = cc_bop_coalition_primed }
				}
				# Hegemony — the aspirant holds preponderance.
				if = {
					limit = { var:cc_bop_preponderance_share >= 65 }
					change_variable = { name = cc_bop_hegemony_months add = 1 }
				}
				else = {
					change_variable = { name = cc_bop_hegemony_months subtract = 1 }
				}
				clamp_variable = { name = cc_bop_hegemony_months min = 0 max = 60 }

				# Coalition — the balancing bloc is ascendant.
				if = {
					limit = { var:cc_bop_preponderance_share <= 35 }
					change_variable = { name = cc_bop_coalition_months add = 1 }
				}
				else = {
					change_variable = { name = cc_bop_coalition_months subtract = 1 }
				}
				clamp_variable = { name = cc_bop_coalition_months min = 0 max = 60 }
			}
		}

		# Stagnation — the aspirant individually eclipsed in the middle band (country-scope check).
		if = {
			limit = {
				situation:cc_balance_of_power = {
					NOT = { has_variable = cc_bop_hegemony_primed }
					NOT = { has_variable = cc_bop_coalition_primed }
					var:cc_bop_preponderance_share > 35
					var:cc_bop_preponderance_share < 65
				}
				cc_bop_hegemon_eclipsed = yes
			}
			situation:cc_balance_of_power = { change_variable = { name = cc_bop_stagnation_months add = 1 } }
		}
		else = {
			situation:cc_balance_of_power = { change_variable = { name = cc_bop_stagnation_months subtract = 1 } }
		}
		situation:cc_balance_of_power = { clamp_variable = { name = cc_bop_stagnation_months min = 0 max = 60 } }

		# --- Priming: a full counter does not end the game outright. Hegemony and coalition
		# prime a decisive war; stagnation simply ends the contest, quietly.
		situation:cc_balance_of_power = {
			if = {
				limit = {
					var:cc_bop_hegemony_months >= 60
					years_since_situation_start > 5
					NOT = { has_variable = cc_bop_hegemony_primed }
					NOT = { has_variable = cc_bop_coalition_primed }
					NOT = { has_variable = cc_bop_flag_coalition }
					NOT = { has_variable = cc_bop_flag_stagnation }
				}
				set_variable = cc_bop_hegemony_primed
				set_variable = { name = cc_bop_final_war_grace value = 0 }
				remove_variable = cc_bop_final_war_called
			}
			else_if = {
				limit = {
					var:cc_bop_coalition_months >= 60
					years_since_situation_start > 5
					NOT = { has_variable = cc_bop_coalition_primed }
					NOT = { has_variable = cc_bop_hegemony_primed }
					NOT = { has_variable = cc_bop_flag_hegemony }
					NOT = { has_variable = cc_bop_flag_stagnation }
				}
				set_variable = cc_bop_coalition_primed
				set_variable = { name = cc_bop_final_war_grace value = 0 }
				remove_variable = cc_bop_final_war_called
			}
			else_if = {
				limit = {
					var:cc_bop_stagnation_months >= 60
					years_since_situation_start > 5
					NOT = { has_variable = cc_bop_hegemony_primed }
					NOT = { has_variable = cc_bop_coalition_primed }
					NOT = { has_variable = cc_bop_flag_hegemony }
					NOT = { has_variable = cc_bop_flag_coalition }
				}
				set_variable = cc_bop_flag_stagnation
			}
		}

		# Announce the reckoning the month it is primed (one-shot, all engaged courts).
		if = {
			limit = {
				situation:cc_balance_of_power = {
					OR = {
						has_variable = cc_bop_hegemony_primed
						has_variable = cc_bop_coalition_primed
					}
					var:cc_bop_final_war_grace = 0
				}
			}
			every_country = {
				limit = { cc_bop_eligible_country = yes is_subject = no can_see_situation = root }
				trigger_event_non_silently = cc_balance_of_power.112
			}
		}

		# --- HEGEMONY reckoning: force the coalition's last stand, then resolve by share.
		if = {
			limit = { situation:cc_balance_of_power = { has_variable = cc_bop_hegemony_primed } }
			situation:cc_balance_of_power = { change_variable = { name = cc_bop_final_war_grace add = 1 } }
			# Light the fire while no coalition member yet fights the hegemon.
			if = {
				limit = {
					situation:cc_balance_of_power = { NOT = { has_variable = cc_bop_final_war_called } }
					exists = international_organization:cc_bop_preponderance
					exists = international_organization:cc_bop_coalition
					NOT = {
						any_country = {
							is_member_of_international_organization = international_organization:cc_bop_coalition
							exists = international_organization:cc_bop_preponderance
							is_at_war_with = international_organization:cc_bop_preponderance.leader_country
						}
					}
				}
				cc_bop_ignite_final_war = yes
			}
			# Resolve once the fought war has ended, on the real treaty (cc_bop_final_war_outcome,
			# set by on_winning_war / on_losing_war). Outcome 1 = the aspirant prevailed; anything
			# else — a coalition victory (2) or a white peace (unset) — denies the bid.
			if = {
				limit = {
					situation:cc_balance_of_power = { has_variable = cc_bop_final_war_called }
					NOT = {
						any_country = {
							is_member_of_international_organization = international_organization:cc_bop_coalition
							exists = international_organization:cc_bop_preponderance
							is_at_war_with = international_organization:cc_bop_preponderance.leader_country
						}
					}
				}
				if = {
					limit = { situation:cc_balance_of_power = { var:cc_bop_final_war_outcome = 1 } }
					situation:cc_balance_of_power = { set_variable = cc_bop_flag_hegemony }
				}
				else = {
					# The coalition won or fought the colossus to a standstill — deny and fight on.
					situation:cc_balance_of_power = {
						remove_variable = cc_bop_hegemony_primed
						remove_variable = cc_bop_final_war_called
						remove_variable = cc_bop_final_war_outcome
						set_variable = { name = cc_bop_hegemony_months value = 30 }
						change_variable = { name = cc_bop_share_offset subtract = 8 }
						clamp_variable = { name = cc_bop_share_offset min = -25 max = 25 }
					}
				}
			}
			# Safety valve: if no war could be forced for two years, the colossus wins by default.
			if = {
				limit = {
					situation:cc_balance_of_power = {
						NOT = { has_variable = cc_bop_final_war_called }
						var:cc_bop_final_war_grace >= 24
					}
				}
				situation:cc_balance_of_power = { set_variable = cc_bop_flag_hegemony }
			}
		}

		# --- COALITION reckoning: the cut-down-to-size coup. Requires the coalition to have
		# enacted its coup-de-grâce policy; until then the primed state holds and the bloc is
		# pressed to pass it (see the congress pulse). Then force the war and resolve by share.
		if = {
			limit = { situation:cc_balance_of_power = { has_variable = cc_bop_coalition_primed } }
			situation:cc_balance_of_power = { change_variable = { name = cc_bop_final_war_grace add = 1 } }
			if = {
				limit = {
					situation:cc_balance_of_power = { NOT = { has_variable = cc_bop_final_war_called } }
					exists = international_organization:cc_bop_preponderance
					exists = international_organization:cc_bop_coalition
					international_organization:cc_bop_coalition = {
						international_organization_has_policy = policy:cc_bop_coup_de_grace
					}
					NOT = {
						any_country = {
							is_member_of_international_organization = international_organization:cc_bop_coalition
							exists = international_organization:cc_bop_preponderance
							is_at_war_with = international_organization:cc_bop_preponderance.leader_country
						}
					}
				}
				cc_bop_ignite_final_war = yes
			}
			if = {
				limit = {
					situation:cc_balance_of_power = { has_variable = cc_bop_final_war_called }
					NOT = {
						any_country = {
							is_member_of_international_organization = international_organization:cc_bop_coalition
							exists = international_organization:cc_bop_preponderance
							is_at_war_with = international_organization:cc_bop_preponderance.leader_country
						}
					}
				}
				if = {
					# The aspirant was beaten on the field (outcome 2) → cut down to size.
					limit = { situation:cc_balance_of_power = { var:cc_bop_final_war_outcome = 2 } }
					situation:cc_balance_of_power = { set_variable = cc_bop_flag_coalition }
				}
				else = {
					# The aspirant prevailed or held to a draw — deny the coup and fight on.
					situation:cc_balance_of_power = {
						remove_variable = cc_bop_coalition_primed
						remove_variable = cc_bop_final_war_called
						remove_variable = cc_bop_final_war_outcome
						set_variable = { name = cc_bop_coalition_months value = 30 }
						change_variable = { name = cc_bop_share_offset add = 8 }
						clamp_variable = { name = cc_bop_share_offset min = -25 max = 25 }
					}
				}
			}
		}

		# Eastern deluge — an eastern great power leads the preponderant bloc and dominates.
		if = {
			limit = {
				situation:cc_balance_of_power = {
					years_since_situation_start > 5
					var:cc_bop_preponderance_share >= 62
					# Yield to a reckoning already under way.
					NOT = { has_variable = cc_bop_hegemony_primed }
					NOT = { has_variable = cc_bop_coalition_primed }
				}
				any_country = {
					cc_bop_eligible_country = yes
					cc_bop_eastern_power = yes
					is_great_power = yes
					is_leader_of_international_organization = international_organization:cc_bop_preponderance
				}
			}
			situation:cc_balance_of_power = { set_variable = cc_bop_flag_deluge }
		}

		# Revolutionary collapse.
		if = {
			limit = { is_situation_active = situation:the_revolution }
			situation:cc_balance_of_power = { set_variable = cc_bop_flag_revolution }
		}

		# Concert of Europe — sustained equilibrium into the revolutionary era.
		if = {
			limit = {
				current_age_or_later = { age = age_6_revolutions }
				situation:cc_balance_of_power = {
					years_since_situation_start > 40
					var:cc_bop_preponderance_share >= 40
					var:cc_bop_preponderance_share <= 60
					# Yield to a reckoning already under way.
					NOT = { has_variable = cc_bop_hegemony_primed }
					NOT = { has_variable = cc_bop_coalition_primed }
				}
			}
			situation:cc_balance_of_power = { set_variable = cc_bop_flag_concert }
		}

		# Overstretch settles on a runaway preponderant leader.
		if = {
			limit = {
				situation:cc_balance_of_power = { var:cc_bop_preponderance_share >= 60 }
				exists = international_organization:cc_bop_preponderance
			}
			international_organization:cc_bop_preponderance.leader_country ?= {
				add_country_modifier = {
					modifier = cc_bop_hegemonic_overstretch
					months = 2
					mode = add_and_extend
				}
			}
		}

		# --- Pulse events ---

		# Ongoing alignment temptation for engaged neutrals.
		every_country = {
			limit = {
				cc_bop_eligible_country = yes
				is_subject = no
				has_variable = cc_bop_engaged
				can_see_situation = root
				NOT = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
				NOT = { is_member_of_international_organization = international_organization:cc_bop_coalition }
			}
			# A minister on Free Hands embodies the unaligned court keeping its options
			# open — such powers are courted (and tempted) more often.
			random_list = {
				3 = {
					modifier = {
						add = 3
						any_cabinet_character = {
							has_cabinet_action = yes
							cabinet_action = cabinet_action:cc_duty_free_hands
						}
					}
					trigger_event_non_silently = cc_balance_of_power.10
				}
				97 = { }
			}
		}

		# The aspiring hegemon leans on a weaker neighbour (threats / bribery).
		every_country = {
			limit = {
				is_leader_of_international_organization = international_organization:cc_bop_preponderance
				can_see_situation = root
				situation:cc_balance_of_power = { var:cc_bop_preponderance_share >= 45 }
			}
			random_list = {
				4 = {
					# A confident bloc presses the unaligned harder.
					modifier = {
						add = 4
						international_organization:cc_bop_preponderance.var:cc_bop_ambition >= 50
					}
					trigger_event_non_silently = cc_balance_of_power.20
				}
				96 = { }
			}
		}

		# Coalition members stoke fear of the hegemon.
		every_country = {
			limit = {
				is_member_of_international_organization = international_organization:cc_bop_coalition
				can_see_situation = root
			}
			random_list = {
				3 = {
					# A resolute coalition raises its voice more readily.
					modifier = {
						add = 3
						international_organization:cc_bop_coalition.var:cc_bop_resolve >= 50
					}
					trigger_event_non_silently = cc_balance_of_power.30
				}
				97 = { }
			}
		}

				exists = international_organization:cc_bop_preponderance
				NOT = { has_variable = cc_bop_recently_engaged_hegemon }
				international_organization:cc_bop_preponderance.leader_country ?= {
					NOT = { this = prev }
					NOT = { is_at_war_with = prev }
