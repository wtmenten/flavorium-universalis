
# Balance of Power — shared triggers

# A country eligible to take part in the European balance of power: a real country
# (not a pop/rebel/colony abstraction) whose capital lies in Europe or the Near-East.
cc_bop_eligible_country = {
	NOT = { country_type = pop }
	NOT = { country_type = building }
	exists = capital
	OR = {
		capital.sub_continent = sub_continent:western_europe
		capital.sub_continent = sub_continent:eastern_europe
		capital.sub_continent = sub_continent:north_africa
		capital.region = region:crescent_region
		capital.region = region:anatolia_region
	}
}

# An "eastern" power whose surge westward would constitute a deluge breaking the balance.
#
# Every call site pairs this with cc_bop_eligible_country, so the effective set is the
# intersection of the two:
#
#   eastern_europe   fully eligible above, so fully eastern
#   middle_east      only its anatolia_region and crescent_region are eligible above, so the
#                    Ottomans and the Levantine powers qualify; Arabia and Persia do not
#
# north_asia (west/east Siberia) and central_asia (Khorasan) used to be listed here too, but
# neither contains a region cc_bop_eligible_country accepts, so those clauses could never
# contribute and were removed. Widening the deluge to Persia or the steppe khanates is a
# design change: add their regions to cc_bop_eligible_country first, not here.
cc_bop_eastern_power = {
	exists = capital
	OR = {
		capital.sub_continent = sub_continent:eastern_europe
		capital.sub_continent = sub_continent:middle_east
	}
}

# True once a prior Concert of Europe has lapsed: the cooldown year stamped at the
# end of that concert (its end year + ~25) has been reached, so a new balance may form.
cc_bop_concert_cooldown_elapsed = {
	has_global_variable = cc_bop_concert_restart_year
	current_year >= global_var:cc_bop_concert_restart_year
}

# The aspiring hegemon has been individually eclipsed: some other eligible power has grown
# to outweigh it in raw great-power score by a clear margin (cc_bop_eclipse_target, recomputed
# monthly = aspirant score x 1.10). Drives the neutral "Stagnation" ending — the would-be
# hegemon fades into irrelevance, overtaken by a power risen in its shadow.
cc_bop_hegemon_eclipsed = {
	exists = international_organization:cc_bop_preponderance
	exists = international_organization:cc_bop_preponderance.leader_country
	situation:cc_balance_of_power = { has_variable = cc_bop_eclipse_target }
	any_country = {
		cc_bop_eligible_country = yes
		is_subject = no
		NOT = { is_leader_of_international_organization = international_organization:cc_bop_preponderance }
		great_power_score >= situation:cc_balance_of_power.var:cc_bop_eclipse_target
	}
}

# True once a great power has the doctrine, or the revolutionary era has opened.
cc_bop_doctrine_or_era = {
	OR = {
		current_age_or_later = { age = age_6_revolutions }
		any_country = {
			cc_bop_eligible_country = yes
			is_great_power = yes
			has_advance = cc_balance_of_power_advance
		}
	}
}
