###############################################################################
# Flavorium Universalis — Cabinet Experience: tuning values
#
# The whole balance surface of the experience system lives here. Everything else
# (cc_xp_effects.txt, cc_xp_pulse.txt, the generated duty lookup) reads these and
# holds no magic numbers of its own.
#
# NOTE: EU5 caches script values. After editing this file, run `reload script_values`
# in the console or restart, or the old numbers keep being used.
#
# Sections:
#   1. Level thresholds       — total XP required for each of the ten levels
#   2. Track tier thresholds  — per-track XP required for each of the four tiers
#   3. Award magnitudes       — what each XP source is worth
#   4. Derived helpers        — values computed from the above
###############################################################################


###############################################################################
# 1. LEVEL THRESHOLDS
#
# Level is seniority: derived from TOTAL experience across all three tracks, and
# it drives the scaling service modifier plus any gate that cares about how long
# a minister has been consequential rather than what they are good at.
#
# Steps rise by 50 each level (100, 150, 200, ... 500), so early levels arrive fast
# enough to be felt and level 10 stays a deliberate long-campaign investment.
# A minister on continuous duty for 25 years earns roughly 600 from service and
# 125 from tenure, so reaching the top of the ladder requires milestones and paid
# training as well, not merely surviving.
###############################################################################

cc_xp_level_2_at  = { value = 100 }
cc_xp_level_3_at  = { value = 250 }
cc_xp_level_4_at  = { value = 450 }
cc_xp_level_5_at  = { value = 700 }
cc_xp_level_6_at  = { value = 1000 }
cc_xp_level_7_at  = { value = 1350 }
cc_xp_level_8_at  = { value = 1750 }
cc_xp_level_9_at  = { value = 2200 }
cc_xp_level_10_at = { value = 2700 }


###############################################################################
# 2. TRACK TIER THRESHOLDS
#
# Tier is specialization: derived per track from that track's own XP, and it is what
# gates trait advancement. A minister with 1200 administrative XP may reach tier 3
# on the administrative ladder no matter how little diplomacy they have ever done.
#
# Tier 0 is the entry band (the malus and kiss-curse traits), 1 competent,
# 2 specialist, 3 master.
###############################################################################

cc_xp_tier_1_at = { value = 150 }
cc_xp_tier_2_at = { value = 500 }
cc_xp_tier_3_at = { value = 1200 }


###############################################################################
# 3. AWARD MAGNITUDES
###############################################################################

# Monthly, to a minister holding a cabinet duty, credited to that duty's track
# (see the generated cc_xp_duty_tracks.txt). 24 per year of continuous service.
cc_xp_duty_tick = { value = 2 }

# Yearly, to every seated minister regardless of what they are doing. Deliberately
# slow: tenure alone reaches level 4 in a very long career and never clears tier 2,
# so parking a minister is never a substitute for employing one.
cc_xp_tenure_tick = { value = 5 }

# Country milestones, credited to the ministers whose track the milestone belongs to.
cc_xp_milestone_war     = { value = 40 }
cc_xp_milestone_law     = { value = 25 }
cc_xp_milestone_treaty  = { value = 20 }


###############################################################################
# 3b. ADVANCEMENT PACING
###############################################################################

# Experience bought outright by a training interaction. Worth about five years of
# continuous duty service, so paying for it is a genuine shortcut without making the
# duty roster pointless. Costs are in common/prices/cc_xp_prices.txt.
cc_xp_training_gain = { value = 120 }

# POSTING LENGTHS ARE NOT HERE, AND CANNOT BE.
#
# The engine parses event delays and variable durations as raw tokens, not as values: a
# script value in `trigger_event_silently = { months = X }` is a parse error, and in
# `set_variable = { months = X }` it is worse, because it fails silently. Vanilla never
# uses anything but literals in either place.
#
# So the durations live as literals in the interactions themselves, written as random
# ranges the way vanilla does it (ibn_battuta_events.txt:174):
#
#   chancery      months = { 5 7 }    common/character_interactions/cc_xp_training.txt
#   embassy       months = { 10 14 }  (longest: it is the one that involves travelling)
#   staff college months = { 8 10 }
#   examination   months = { 5 7 }    common/character_interactions/cc_xp_examination.txt
#   away flag     months = 36         backstop, so a minister who dies abroad is not
#                                     marked absent forever

# A posting can go better or worse than expected. The ordinary result is
# cc_xp_training_gain above; these are the tails. Weighted 20/60/20, with an already
# accomplished minister more likely to excel and less likely to flounder, so training
# compounds rather than leveling everyone out.
cc_xp_training_gain_good = { value = 210 }
cc_xp_training_gain_poor = { value = 45 }

# Yearly experience a sponsored protege earns while being groomed out of office. Below
# the seated tenure trickle: watching is not the same as doing.
cc_xp_protege_tick = { value = 3 }

# What a dying minister's accumulated knowledge is worth to whoever inherits it. A
# fraction of a long career, not the whole of it: institutions remember imperfectly.
cc_xp_legacy_transfer = { value = 150 }

# Yearly experience a mentored character receives, credited to their mentor's track,
# for as long as the mentorship lasts.
cc_xp_mentor_tick = { value = 25 }

# What a court examination awards the best performer, and everyone else.
cc_xp_exam_winner = { value = 60 }
cc_xp_exam_participant = { value = 15 }


###############################################################################
# 3c. FIELD AWARDS
#
# What a random court event gives a HUMAN court's minister in place of the ladder
# trait it used to hand out.
#
# Fifty-six options across twenty-four events could grant a ladder rung outright,
# which meant a player could be given supreme_commander by a monthly roll without
# ever having earned mil tier 3. Those options now award experience instead, and the
# trait is left to the ladder. AI courts keep the trait, because nothing offers an AI
# a choice and the random path is the only path it has.
#
# The amounts are the TIER BAND the rung represents, so the experience path is worth
# roughly what the trait was:
#
#   tier 0 -> 1   150   (cc_xp_tier_1_at)
#   tier 1 -> 2   350   (cc_xp_tier_2_at minus cc_xp_tier_1_at)
#   tier 2 -> 3   700   (cc_xp_tier_3_at minus cc_xp_tier_2_at)
#
# Entry is half a band: joining a career is not itself a tier. These are the first
# numbers to tune if court events feel too generous, and they are the only place the
# conversion's balance lives.
###############################################################################

cc_xp_field_award_entry  = { value = 75 }
cc_xp_field_award_low    = { value = 150 }
cc_xp_field_award_mid    = { value = 350 }
cc_xp_field_award_high   = { value = 700 }


###############################################################################
# 3d. PAYOFF SCALING
#
# Seventy-four court events read a minister's traits and pay out without ever asking
# how senior that minister is: a level 1 minister and a level 10 one bought the same
# synergy. This multiplies those payouts by the level of the minister the event is
# actually about.
#
# WHY IT READS A COUNTRY VARIABLE. add_country_modifier evaluates `size` in COUNTRY
# scope, and the events do not agree on what the minister is called: 34 save 'minister',
# 8 'rehabilitated_minister', 8 a 'minister_a'/'minister_b' pair, and a long tail saves
# things like 'military_minister' or 'muslim_partner'. cc_xp_publish_payoff_level takes
# the scope name as a parameter and writes the level here, so one shared value serves
# every event regardless of what its minister is called.
#
# CAPPED AT 2.5x DELIBERATELY. Level runs 1-10 and an uncapped linear scale would make
# a mature court's synergies worth ten times a young one's, which turns flavour events
# into the dominant income source. 2.5x is enough to feel like seniority matters without
# that. Level 1 yields exactly 1.0, so nothing is ever weaker than it is today.
###############################################################################

cc_xp_payoff_scale = {
	value = 1
	if = {
		limit = { has_variable = cc_xp_payoff_level }
		add = {
			value = var:cc_xp_payoff_level
			subtract = 1
			multiply = 0.1667
		}
	}
	min = 1
	max = 2.5
}


###############################################################################
# 4. DERIVED HELPERS
###############################################################################

# Multiplier passed to `size` on the cc_xp_service_record character modifier, so one
# static modifier scales across all ten levels instead of ten near-identical ones.
# Level 1 yields zero, so a freshly seated minister gets no unearned bonus.
cc_xp_level_bonus_size = {
	value = var:cc_level
	add = -1
	min = 0
}


###############################################################################
# 5. PATRONAGE OF THE ARTS
###############################################################################

# What a discreet, unnegotiated sale fetches. Deliberately below what vanilla's
# diplomatic request_work_of_art_purchase would pay: the discount IS the mechanic.
# Nobody refuses, nobody haggles, and nobody pays full price either.
#
# scope:cc_xp_sale_work is set by cc_xp_sell_work from its $WORK$ parameter. Guarded on
# `exists` because an event option's effect is evaluated to build the option TOOLTIP
# before anything is chosen, exactly as vanilla guards the same read in
# country_interactions/sell_work_of_art.txt:22 (`if = { limit = { exists = scope:target } }`).
# Without the guard this logged "Undefined event target" twice per option per open.
#
# art_price already carries the piece's quality, so nothing here re-applies it.
cc_xp_discreet_sale_price = {
	if = {
		limit = { exists = scope:cc_xp_sale_work }
		value = scope:cc_xp_sale_work.art_price
	}
	else = { value = 0 }

	multiply = 0.7
	multiply = cc_xp_art_culture_factor
	multiply = cc_xp_art_wealth_factor

	min = 25
}

# How much the origin culture's standing is worth on top of the piece itself.
#
# Measured as a RATIO against our own culture rather than against an absolute number,
# because cultural_influence is an accumulating stat with no documented ceiling and any
# fixed reference point would drift as the game goes on. Our own work is the 1.0 case,
# which is what "or your own by default" means here: a piece by one of our own artists is
# priced at face value and everything else is measured against it.
#
# Bands rather than a division, because a script value cannot divide by a scope read
# safely when that read can be zero. The comparison form is vanilla's own, from
# hegemons/4_cultural_hegemon.txt:16.
cc_xp_art_culture_factor = {
	value = 1

	if = {
		# Half again our own standing or better: a name that carries abroad.
		limit = {
			exists = scope:cc_xp_sale_culture
			scope:cc_xp_sale_culture = {
				cultural_influence > {
					value = root.culture.cultural_influence
					multiply = 1.5
				}
			}
		}
		add = 0.35
	}
	else_if = {
		limit = {
			exists = scope:cc_xp_sale_culture
			scope:cc_xp_sale_culture = {
				cultural_influence > root.culture.cultural_influence
			}
		}
		add = 0.15
	}
	else_if = {
		# Less than half our standing: provincial work, and priced like it.
		limit = {
			exists = scope:cc_xp_sale_culture
			scope:cc_xp_sale_culture = {
				cultural_influence < {
					value = root.culture.cultural_influence
					multiply = 0.5
				}
			}
		}
		add = -0.25
	}
	else_if = {
		limit = {
			exists = scope:cc_xp_sale_culture
			scope:cc_xp_sale_culture = {
				cultural_influence < root.culture.cultural_influence
			}
		}
		add = -0.1
	}
	# else: our own culture, or a work whose creator is gone. Face value.
}

# The market gets richer as the world does. Without this a sale worth a month's income in
# 1337 is worth an afternoon by 1750, and the whole action quietly stops mattering.
#
# country_tax_base is the growth measure rather than monthly_income_total, which the rest
# of this file uses for COSTS: income swings with war, loans and trade, and the price of a
# painting should not. Tax base only ratchets.
cc_xp_art_wealth_factor = {
	value = 1
	add = {
		value = country_tax_base
		multiply = 0.0015
	}
	# A large realm pays more for everything, but not without limit: past this the piece
	# is simply the most anyone in the room will pay for it.
	max = 3.0
}

# Diplomatic experience for the minister who arranged the sale. Brokerage is
# diplomacy conducted about objects rather than borders.
cc_xp_brokerage_gain = { value = 45 }

# Yearly, to a minister currently sponsoring an artist. Patronage is reputation work
# done at dinner tables, so it credits the diplomatic track.
cc_xp_patron_tick = { value = 12 }


###############################################################################
# 6. SCHOOLS OF THOUGHT
###############################################################################

# How many prestigious institutions this realm may hold. Each source is unlocked
# separately and a century or so apart, so the third is a late-game achievement rather
# than something a wealthy court simply buys three of on day one.
#
# Script values do support if/limit blocks (vanilla does this in
# script_values/albanian_migrations.txt), so the capacity is computed rather than
# maintained as another variable.
cc_xp_school_capacity = {
	value = 0
	if = {
		limit = { capital ?= { has_building = building_type:university } }
		add = 1
	}
	if = {
		limit = { has_advance = scholarly_orders_advance }
		add = 1
	}
	if = {
		limit = { has_advance = enlightened_administration }
		add = 1
	}
}

# Yearly experience a school member earns in the school's discipline, whether or not
# they hold office. Between the protege trickle and a mentorship: a school is less
# attentive than a mentor and more consistent than watching from the sidelines.
cc_xp_school_tick = { value = 14 }

###############################################################################
# 9. COURT HEADCOUNTS
#
# WHY THESE EXIST RATHER THAN `count = N` INSIDE AN any_ TRIGGER. In EU5 `count` on an
# any_ iterator means EXACTLY that many, not at least: vanilla writes
# `any_character = { is_ruler = no  is_adult = yes  count = 0 }` in
# generic_actions/invite_foreign_cleric.txt to mean "there are none of them", and the
# form documented in the trigger reference is
# `any_x = { filter = { <triggers> } count = num/all <triggers> }`, where the triggers
# BEFORE count are the filter and count says how many of them must meet the ones after.
#
# So `any_character = { count = 2  cc_xp_is_examinable = yes }` did not mean "at least
# two people can sit the examination". It meant "exactly two can", and a court with
# five ministers failed it. There is no at-least form of `count`, so anything needing a
# threshold counts here and compares the result.
###############################################################################

# Everyone eligible to sit a court examination: the seated cabinet and the proteges,
# provided the system tracks them.
cc_xp_num_examinable = {
	value = 0
	every_character = {
		limit = { cc_xp_is_examinable = yes }
		add = 1
	}
}

# Proteges currently sponsored. Counted live rather than kept in a running total, for the
# reason the old cc_xp_has_protege_room comment gave: a protege who dies or is appointed
# to the cabinet would otherwise burn a slot forever.
cc_xp_num_proteges = {
	value = 0
	every_character = {
		limit = { has_variable = cc_protege }
		add = 1
	}
}

# People at court who could be taken on as a protege, and ministers senior enough to be
# given a pupil. Both exist so an interaction's `allow` can state a threshold in the same
# terms its selector uses, instead of asserting a bare existence check that passes while
# the selector still finds nobody to offer.
cc_xp_num_sponsorable = {
	value = 0
	every_character = {
		limit = { cc_xp_can_be_sponsored = yes }
		add = 1
	}
}

# Everyone the experience system tracks, for events that need two people who could
# plausibly have met rather than one.
cc_xp_num_tracked = {
	value = 0
	every_character = {
		limit = {
			is_alive = yes
			cc_xp_is_tracked = yes
		}
		add = 1
	}
}

# Seated ministers of seniority 5 or more. Read by the rivalry event, which needs two
# accomplished people to have a rivalry between.
cc_xp_num_senior_ministers = {
	value = 0
	every_cabinet_character = {
		limit = {
			in_cabinet = yes
			cc_xp_level_at_least = { LEVEL = 5 }
		}
		add = 1
	}
}

###############################################################################
# 11. ORDERING VALUES
#
# `ordered_x = { order_by = <script value> position = N }` is how a follow-up event picks
# its candidate list, so each of those lists needs a value to sort on. Guarded with
# `exists = var:` the way vanilla guards the same read in
# script_values/economy_satisfaction_target.txt: an unguarded var: read logs
# "Failed to fetch variable ... due to not being set" once per candidate per evaluation.
###############################################################################

# Character scope. Total experience, or zero for someone the system has never touched.
cc_xp_order_seniority = {
	if = {
		limit = { exists = var:cc_xp_total }
		value = var:cc_xp_total
	}
	else = { value = 0 }
}
