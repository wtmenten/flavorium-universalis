# Country scope, the senior partner. Size of cc_union_integration_offer: the share of the
# junior_partner offer's tax-base penalty that union integration cancels, measured against the
# largest other member. See auto_modifiers/cc_union_integration.txt.
cc_union_junior_offer_offset = {
	value = 0
	union ?= {
		if = {
			limit = { modifier:union_integration_level > 0 }
			ordered_international_organization_member = {
				limit = { NOT = { this = root } }
				order_by = country_tax_base
				add = country_tax_base
			}
			multiply = 0.4
			multiply = {
				value = modifier:union_integration_level
				divide = {
					value = 1
					add = modifier:union_integration_level
				}
			}
		}
	}
}
