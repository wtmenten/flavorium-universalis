# Subject Relations Ledger — scaled gold magnitudes
# Evaluated in SUBJECT scope (reference the subject's `overlord` link for the
# overlord-side income), so they invoke identically from the scripted_gui
# is_valid (subject = scope:subject) and the engage events
# (subject = scope:cc_engage_subject), e.g.:
#   gold >= scope:subject.cc_bond_gift_cost
#   add_gold = { value = scope:cc_engage_subject.cc_bond_gift_cost  multiply = -1 }
#
# Operators evaluate linearly top-to-bottom; min = floor, max = ceiling.
###############################################################################

# Gift cost a gracious overlord pays a subject (underwrite / review / festival / envoy).
# Scales off the overlord's monthly income, clamped to a band of the subject's own
# monthly income so the gesture is never trivial nor ruinous relative to the subject.
cc_bond_gift_cost = {
	value = overlord.monthly_income_total
	multiply = 3
	min = {
		value = monthly_income_total
		multiply = 2
		add = 100
	}
	max = {
		value = monthly_income_total
		multiply = 12
		add = 100
	}
}

# Tribute the overlord squeezes from a subject (demand tribute).
# Scales off the subject's current treasury, clamped by the subject's monthly income
# so a cash-poor subject still yields a little and a hoard cannot be drained at once.
cc_bond_tribute_gain = {
	value = gold
	multiply = 0.20
	min = {
		value = monthly_income_total
		multiply = 1
		add = 50
	}
	max = {
		value = monthly_income_total
		multiply = 8
		add = 50
	}
}


###############################################################################
# HEADCOUNTS
#
# `count = N` on an any_ iterator is an EXACT match in EU5, not a threshold, so a
# trigger meaning "two or more" has to count and compare. See the note at the head of
# section 9 in cc_xp_values.txt.
###############################################################################

# Subjects held as shadow states.
cc_num_shadow_states = {
	value = 0
	every_subject = {
		limit = { is_subject_type = shadow_state }
		add = 1
	}
}

# MAP COLOURING
###############################################################################

# LOCATION scope. The subject's overall bond, normalised to 0-1 for a map_color lerp.
#
# Evaluated on the LOCATION rather than on the country because that is the only scope
# map_color offers: the situations readme is explicit that map_color runs with
# root = location and scope:target = situation, and there is no country or player scope
# anywhere in it. So the owner hop happens through `owner.var:` reads rather than by
# entering owner scope, which would move `add` onto the wrong accumulator.
#
# The five dimensions each run -5 to +5, so the sum runs -25 to +25; +25 then /50 puts it
# in 0-1. Each read is guarded, because a subject the ledger has not initialised carries
# none of these and an unguarded var: read logs a fetch failure per location per frame.
cc_bond_overall_factor = {
	value = 25
	if = {
		limit = { owner ?= { exists = var:cc_bond_pol } }
		add = owner.var:cc_bond_pol
	}
	if = {
		limit = { owner ?= { exists = var:cc_bond_eco } }
		add = owner.var:cc_bond_eco
	}
	if = {
		limit = { owner ?= { exists = var:cc_bond_mil } }
		add = owner.var:cc_bond_mil
	}
	if = {
		limit = { owner ?= { exists = var:cc_bond_cul } }
		add = owner.var:cc_bond_cul
	}
	if = {
		limit = { owner ?= { exists = var:cc_bond_per } }
		add = owner.var:cc_bond_per
	}
	divide = 50
	min = 0
	max = 1
}
