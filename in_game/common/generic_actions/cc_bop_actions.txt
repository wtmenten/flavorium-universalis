
# Flavorium Universalis — Balance of Power Actions
# Player-driven actions for the European Balance of Power situation, rendered by the
# engine directly into the situation panel (type = situation) and the IO view
# (type = internationalorganization). Modelled on cc_invasion_mexico.txt.
#
# Engaged unaligned powers can petition a bloc (the leader decides) or take flavorful
# neutral stances gated on a cabinet minister's duty. Bloc leaders spend their
# accumulated IO currency — cc_bop_ambition (preponderance) / cc_bop_resolve
# (coalition) — on coercive or balancing measures. Non-leader members may leave.

################################################################################
# SITUATION ACTION 1 — Petition the Preponderant Bloc
# An engaged, unaligned power asks the aspiring hegemon to admit it. The leader
# decides (event .80).
################################################################################
cc_bop_petition_preponderance = {
	type = situation

	ai_tick = monthly
	ai_tick_frequency = 36

	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			cc_bop_eligible_country = yes
			is_subject = no
			NOT = { country_type = pop }
			NOT = { country_type = building }
			NOT = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
			NOT = { is_member_of_international_organization = international_organization:cc_bop_coalition }
			exists = international_organization:cc_bop_preponderance
		}
	}

	allow = {
		scope:actor = {
			NOT = { has_variable = cc_bop_recently_petitioned }
			can_join_international_organization = international_organization:cc_bop_preponderance
		}
		situation:cc_balance_of_power = { situation_is_active = yes }
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:cc_balance_of_power = this
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			set_variable = { name = cc_bop_recently_petitioned value = yes years = 4 }
			save_scope_as = cc_bop_petitioner
		}
		international_organization:cc_bop_preponderance.leader_country ?= {
			save_scope_as = cc_bop_petition_target_bloc
			trigger_event_non_silently = { id = cc_balance_of_power.80 days = 3 }
		}
	}

	ai_will_do = {
		# Petitioning is the exception, not the reflex — courts mostly keep to themselves.
		add = {
			value = -40
			desc = BASE_VALUE
		}
		# Great powers play the game readily; lesser courts hang back unless pressed.
		if = {
			limit = { scope:actor = { is_great_power = yes } }
			add = { desc = "CC_BOP_GREAT_POWER" value = 20 }
		}
		# Proximity to the aspiring hegemon — a friendly neighbour seeks its protection.
		if = {
			limit = {
				exists = international_organization:cc_bop_preponderance
				international_organization:cc_bop_preponderance.leader_country ?= { any_neighbor_country = { this = scope:actor } }
			}
			add = { desc = "CC_BOP_NEAR_HEGEMON" value = 30 }
		}
		if = {
			limit = {
				exists = international_organization:cc_bop_preponderance
				international_organization:cc_bop_preponderance.leader_country ?= { this_opinion_of_prev >= 0 }
			}
			add = { desc = "CC_BOP_FRIENDLY_LEADER" value = 25 }
		}
		if = {
			limit = { situation:cc_balance_of_power = { var:cc_bop_preponderance_share >= 55 } }
			add = { desc = "CC_BOP_BANDWAGON" value = 20 }
		}
	}
}

################################################################################
# IO LEADER ACTION (COALITION) — Convene the Congress
# A structural act: the chief balancer calls the dormant congress into session, swapping the
# coalition's parliament form to the standing assembly. Only then can the bloc vote on laws
# and table parliament issues. Requires a quorum of members. Unilateral — the leader convenes,
# the bloc deliberates thereafter.
################################################################################
cc_bop_io_convene_the_congress = {
	type = internationalorganization

	ai_tick = monthly
	ai_tick_frequency = 6

	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			is_leader_of_international_organization = international_organization:cc_bop_coalition
			NOT = { country_type = pop }
			NOT = { country_type = building }
		}
		international_organization:cc_bop_coalition = {
			parliament_type = parliament_type:cc_bop_coalition_dormant
		}
	}

	allow = {
		# A congress needs a quorum of powers to be worth convening.
		international_organization:cc_bop_coalition = {
			total_members >= 3
		}
		international_organization:cc_bop_coalition.var:cc_bop_resolve >= 20
		situation:cc_balance_of_power = { situation_is_active = yes }
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		column = { data = name }
		visible = {
			international_organization_type = international_organization_type:cc_bop_coalition
		}
	}

	effect = {
		international_organization:cc_bop_coalition = {
			set_parliament_type = parliament_type:cc_bop_coalition_assembly
			# Calling the powers together rallies the cause.
			change_variable = { name = cc_bop_resolve add = 5 }
		}
		scope:actor = {
			trigger_event_non_silently = { id = cc_balance_of_power.93 days = 2 }
		}
	}

	ai_will_do = {
		add = {
			value = 25
			desc = BASE_VALUE
		}
		# The more the coalition is threatened, the keener the leader is to organise it.
		if = {
			limit = { situation:cc_balance_of_power = { var:cc_bop_preponderance_share >= 52 } }
			add = { desc = "CC_BOP_HEGEMON_RISING" value = 25 }
		}
	}
}

################################################################################
# IO MEMBER ACTION — Leave the Bloc
# A non-leader member withdraws from its bloc. The engine then runs the bloc's
# on_left (defector penalty, currency loss).
################################################################################
cc_bop_io_leave_bloc = {
	type = internationalorganization

	ai_tick = monthly
	ai_tick_frequency = 12

	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			NOT = { country_type = pop }
			NOT = { country_type = building }
			OR = {
				AND = {
					is_member_of_international_organization = international_organization:cc_bop_preponderance
					NOT = { is_leader_of_international_organization = international_organization:cc_bop_preponderance }
				}
				AND = {
					is_member_of_international_organization = international_organization:cc_bop_coalition
					NOT = { is_leader_of_international_organization = international_organization:cc_bop_coalition }
				}
			}
		}
	}

	allow = {
		scope:actor = {
			NOT = { has_variable = cc_bop_recently_left_bloc }
			# A bloc closes ranks in wartime — no member may desert while under arms.
			custom_tooltip = {
				text = cc_bop_cannot_leave_at_war_tt
				at_war = no
			}
		}
		situation:cc_balance_of_power = { situation_is_active = yes }
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		column = { data = name }
		visible = {
			scope:actor = { is_member_of_international_organization = root }
			NOT = { scope:actor = { is_leader_of_international_organization = root } }
			OR = {
				international_organization_type = international_organization_type:cc_bop_preponderance
				international_organization_type = international_organization_type:cc_bop_coalition
			}
		}
	}

	effect = {
		scope:actor = {
			set_variable = { name = cc_bop_recently_left_bloc value = yes years = 3 }
		}
		scope:recipient = {
			remove_country_from_international_organization = scope:actor
		}
	}

	ai_will_do = {
		# Desertion is near-unthinkable — a member stands by its bloc save for an
		# internal pull toward neutrality, or a cause it judges already lost.
		add = {
			value = -90
			desc = BASE_VALUE
		}
		# Foreign gold tempting us to the sidelines.
		if = {
			limit = { scope:actor = { has_country_modifier = cc_bop_bought_ally } }
			add = { desc = "CC_BOP_NEUTRALIST_PULL" value = 35 }
		}
		# A minister keeping our hands free.
		if = {
			limit = {
				scope:actor = {
					any_cabinet_character = {
						has_cabinet_action = yes
						cabinet_action = cabinet_action:cc_duty_free_hands
					}
				}
			}
			add = { desc = "CC_BOP_NEUTRALIST_COUNSEL" value = 30 }
		}
		# Our bloc is visibly failing — the cause looks lost.
		if = {
			limit = {
				OR = {
					AND = {
						scope:actor = { is_member_of_international_organization = international_organization:cc_bop_coalition }
						international_organization:cc_bop_coalition.var:cc_bop_resolve < 25
					}
					AND = {
						scope:actor = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
						international_organization:cc_bop_preponderance.var:cc_bop_ambition < 25
					}
				}
			}
			add = { desc = "CC_BOP_BLOC_FALTERING" value = 30 }
		}
	}
}
