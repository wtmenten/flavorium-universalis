
# Flavorium Universalis — Balance of Power Actions
# Player-driven actions for the European Balance of Power situation, rendered by the
# engine directly into the situation panel (type = situation) and the IO view
# (type = internationalorganization). Modelled on cc_invasion_mexico.txt.
#
# Engaged unaligned powers can petition a bloc (the leader decides) or take flavorful
# neutral stances gated on a cabinet minister's duty. Bloc leaders spend their
# accumulated IO currency — cc_bop_ambition (preponderance) / cc_bop_resolve
# (coalition) — on coercive or balancing measures. Non-leader members may leave.

################################################################################
# SITUATION ACTION 1 — Petition the Preponderant Bloc
# An engaged, unaligned power asks the aspiring hegemon to admit it. The leader
# decides (event .80).
################################################################################
cc_bop_petition_preponderance = {
	type = situation

	ai_tick = monthly
	ai_tick_frequency = 12

	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			cc_bop_eligible_country = yes
			is_subject = no
			NOT = { country_type = pop }
			NOT = { country_type = building }
			NOT = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
			NOT = { is_member_of_international_organization = international_organization:cc_bop_coalition }
			exists = international_organization:cc_bop_preponderance
		}
	}

	allow = {
		scope:actor = {
			NOT = { has_variable = cc_bop_recently_petitioned }
			can_join_international_organization = international_organization:cc_bop_preponderance
		}
		situation:cc_balance_of_power = { situation_is_active = yes }
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:cc_balance_of_power = this
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			set_variable = { name = cc_bop_recently_petitioned value = yes years = 4 }
			save_scope_as = cc_bop_petitioner
		}
		international_organization:cc_bop_preponderance.leader_country ?= {
			save_scope_as = cc_bop_petition_target_bloc
			trigger_event_non_silently = { id = cc_balance_of_power.80 days = 3 }
		}
	}

	ai_will_do = {
		add = {
			value = -20
			desc = BASE_VALUE
		}
		if = {
			limit = {
				exists = international_organization:cc_bop_preponderance
				international_organization:cc_bop_preponderance.leader_country ?= { this_opinion_of_prev >= 0 }
			}
			add = { desc = "CC_BOP_FRIENDLY_LEADER" value = 30 }
		}
		if = {
			limit = { situation:cc_balance_of_power = { var:cc_bop_preponderance_share >= 55 } }
			add = { desc = "CC_BOP_BANDWAGON" value = 25 }
		}
	}
}

################################################################################
# IO MEMBER ACTION — Leave the Bloc
# A non-leader member withdraws from its bloc. The engine then runs the bloc's
# on_left (defector penalty, currency loss).
################################################################################
cc_bop_io_leave_bloc = {
	type = internationalorganization

	ai_tick = monthly
	ai_tick_frequency = 12

	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			NOT = { country_type = pop }
			NOT = { country_type = building }
			OR = {
				AND = {
					is_member_of_international_organization = international_organization:cc_bop_preponderance
					NOT = { is_leader_of_international_organization = international_organization:cc_bop_preponderance }
				}
				AND = {
					is_member_of_international_organization = international_organization:cc_bop_coalition
					NOT = { is_leader_of_international_organization = international_organization:cc_bop_coalition }
				}
			}
		}
	}

	allow = {
		scope:actor = {
			NOT = { has_variable = cc_bop_recently_left_bloc }
		}
		situation:cc_balance_of_power = { situation_is_active = yes }
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		column = { data = name }
		visible = {
			scope:actor = { is_member_of_international_organization = root }
			NOT = { scope:actor = { is_leader_of_international_organization = root } }
			OR = {
				international_organization_type = international_organization_type:cc_bop_preponderance
				international_organization_type = international_organization_type:cc_bop_coalition
			}
		}
	}

	effect = {
		scope:actor = {
			set_variable = { name = cc_bop_recently_left_bloc value = yes years = 3 }
		}
		scope:recipient = {
			remove_country_from_international_organization = scope:actor
		}
	}

	ai_will_do = {
		add = {
			value = -50
			desc = BASE_VALUE
		}
	}
}
