###############################################################################
# Flavorium Universalis — Cabinet Experience: the two actions with no target
#
# WHY THESE ARE generic_actions AND NOT character_interactions. A character_interaction
# must have a target. Every one of the 60-odd in vanilla declares at least one
# select_trigger, without exception, and an interaction that declares none still has the
# engine ask for its `recipient` flag: the panel evaluates the button every frame and
# error.log fills with
#
#   [interaction_target.cpp:877]: Asking for a flag that's not in the interaction target
#   chooser specified
#
# at tens of thousands of lines per session, and the button does not work.
#
# A generic_action with `type = owncountry` has no such requirement. Ten of vanilla's own
# declare no select_trigger, and two of those (train_general in unit_overview.gui,
# hire_advisor in government_lateralview.gui) are driven from exactly the kind of panel
# button used here: an `actor` and no `parameter` at all.
#
# These two actions need that because their targets are on no row of the court panel.
# Sponsorship picks among courtiers, who appear in no roster anywhere; a discreet sale
# needs both a broker and a work. Both offer their candidates as options in
# events/cc_xp_choice_events.txt instead.
#
# `type = owncountry` also replaces the character-interaction pair `message = yes` and
# `on_own_nation = yes`, which do not belong on a generic_action.
###############################################################################


###############################################################################
# TAKE ON A PROTEGE
#
# Elevating an outsider costs coin and political capital (charged as government_power by
# the price). What it costs beyond that depends on WHICH outsider, and is settled in
# cc_xp_take_protege once a candidate has been chosen:
#
#   nobles    no penalty; the candidate is one of their own being advanced
#   burghers  the gold comes back; a merchant house funds its own
#   peasants  half the gold comes back
#   clergy    paid in religious influence instead
#   otherwise the nobility resents the crown taking an interest in someone it has not
#             vouched for, and nothing offsets the cost
###############################################################################

cc_xp_sponsor_protege = {
	type = owncountry

	price = price:cc_xp_sponsor_protege

	# The AI does not groom successors. It appoints from whoever is available, which is
	# what it did before this system existed, and the protege slots are a player tool.
	ai_tick = never

	potential = {
		scope:actor = { cc_xp_enabled = yes }
	}

	# Both clauses carry their own custom_tooltip. Without one the engine describes a bare
	# existence check with a placeholder key it has no string for, so a disabled button
	# gives the player nothing at all.
	allow = {
		scope:actor = {
			cc_xp_enabled = yes
			cc_xp_has_protege_room = yes
			custom_tooltip = {
				text = cc_xp_tt_needs_candidate_protege
				cc_xp_num_sponsorable >= 1
			}
		}
	}

	effect = {
		scope:actor = { trigger_event_silently = { id = cc_xp.85 } }
	}

	ai_will_do = {
		add = { value = 0 }
	}
}


###############################################################################
# SELL A WORK TO A DISCREET CLIENT
#
# The deliberate opposite of vanilla's request_work_of_art_purchase, which is a
# diplomatic action: it names a counterparty, lets them refuse (diplo_chance base = -30),
# and is actively harder to conclude the better the work is.
#
# This one names nobody, cannot be refused, and pays less for exactly that reason. The
# piece still goes somewhere real: cc_xp_sell_work picks a plausible buyer and calls
# vanilla's own move_art_and_owner, so the work continues to exist in the world rather
# than evaporating out of it.
#
# No price. This action raises money rather than spending it, which is also why cc_xp.84
# is the one choice event with a decline option.
###############################################################################

cc_xp_sell_art_discreetly = {
	type = owncountry

	ai_tick = never

	potential = {
		scope:actor = { cc_xp_enabled = yes }
	}

	allow = {
		scope:actor = {
			cc_xp_enabled = yes
			custom_tooltip = {
				text = cc_xp_tt_needs_sellable_art
				has_sellable_art = yes
			}
			custom_tooltip = {
				text = cc_xp_tt_needs_broker
				any_cabinet_character = { cc_xp_can_broker = yes }
			}
		}
	}

	effect = {
		scope:actor = { trigger_event_silently = { id = cc_xp.84 } }
	}

	ai_will_do = {
		add = { value = 0 }
	}
}
