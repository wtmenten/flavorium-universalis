
# Flavorium Universalis — Invasion of Mexico Generic Actions
# These actions appear in the situation panel for native and colonial powers.

################################################################################
# NATIVE ACTION 1 — Call for Confederation
# Allows a native Mesoamerican power to create or join the Mesoamerican Confederation.
################################################################################
ccim_call_confederation = {
	type = situation

	ai_tick = monthly
	ai_tick_frequency = 6

	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			is_capital_mesoamerica = yes
			is_subject = no
			NOT = { country_type = pop }
			NOT = { country_type = building }
			# Members use ccim_seek_tribal_alliance instead
			NOT = { is_member_of_international_organization = international_organization:cc_mesoamerican_confederation }
		}
	}

	allow = {
		scope:actor = {
			NOT = { has_variable = ccim_recently_called_confederation }
			at_war = no
		}
		situation:cc_invasion_mexico = { situation_is_active = yes }
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:cc_invasion_mexico = this
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			set_variable = {
				name = ccim_recently_called_confederation
				value = yes
				years = 10
			}
		}
		custom_tooltip = ccim_call_confederation_create_tt
		hidden_effect = {
			if = {
				limit = {
					NOT = { exists = international_organization:cc_mesoamerican_confederation }
				}
				scope:actor = {
					create_international_organization = {
						type = international_organization_type:cc_mesoamerican_confederation
						add_country_to_international_organization = root
						set_leader_country = root
					}
				}
			}
			else_if = {
				limit = {
					scope:actor = {
						NOT = { is_member_of_international_organization = international_organization:cc_mesoamerican_confederation }
						can_join_international_organization = international_organization:cc_mesoamerican_confederation
					}
				}
				international_organization:cc_mesoamerican_confederation = {
					add_country_to_international_organization = scope:actor
				}
			}
		}
		# Invite native neighbors — each neighbor only receives this once (2-year cooldown)
		scope:actor = {
			every_neighbor_country = {
				limit = {
					is_capital_mesoamerica = yes
					is_subject = no
					NOT = { is_member_of_international_organization = international_organization:cc_mesoamerican_confederation }
					can_join_international_organization = international_organization:cc_mesoamerican_confederation
					NOT = { has_variable = ccim_received_confederation_invite }
				}
				set_variable = { name = ccim_received_confederation_invite value = yes years = 2 }
				trigger_event_non_silently = { id = cc_invasion_mexico.2 days = 3 }
			}
		}
		scope:actor = {
			add_prestige = prestige_weak_penalty
		}
	}

	ai_will_do = {
		add = {
			value = -10
			desc = BASE_VALUE
		}
		# Worth forming if the IO doesn't exist yet
		if = {
			limit = { NOT = { exists = international_organization:cc_mesoamerican_confederation } }
			add = {
				desc = "CCIM_IO_NOT_YET_FORMED"
				value = 20
			}
		}
		if = {
			limit = {
				scope:actor = {
					any_neighbor_country = {
						NOT = { is_capital_mesoamerica = yes }
						OR = {
							has_presence_in = region:mesoamerica_region
							has_presence_in = region:central_america_region
						}
					}
				}
			}
			add = {
				desc = "ENEMY_ON_BORDER"
				value = 15
			}
		}
	}
}

################################################################################
# NATIVE ACTION 2 — Offer Tribute
# Pays gold to a neighboring colonizer to remove/prevent a CB and gain opinion.
################################################################################
ccim_offer_tribute = {
	type = situation

	ai_tick = monthly
	ai_tick_frequency = 12

	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			is_capital_mesoamerica = yes
			is_subject = no
			any_neighbor_country = {
				NOT = { is_capital_mesoamerica = yes }
				OR = {
					has_presence_in = region:mesoamerica_region
					has_presence_in = region:central_america_region
					has_presence_in = region:caribbean_region
				}
			}
		}
	}

	allow = {
		scope:actor = {
			NOT = { has_variable = ccim_recently_offered_tribute }
			at_war = no
			gold >= 50
		}
		situation:cc_invasion_mexico = { situation_is_active = yes }
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:cc_invasion_mexico = this
			situation_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = target
		name = "ccim_choose_tribute_target"
		column = { data = name }
		visible = {
			NOT = { is_capital_mesoamerica = yes }
			OR = {
				has_presence_in = region:mesoamerica_region
				has_presence_in = region:central_america_region
				has_presence_in = region:caribbean_region
			}
			is_neighbor_of = scope:actor
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:actor = {
				set_variable = {
					name = ccim_recently_offered_tribute
					value = yes
					years = 5
				}
				add_gold = -50
			}
			scope:target = {
				add_gold = 50
				add_opinion = {
					modifier = opinion_ccim_tribute_received
					target = scope:actor
				}
			}
		}
	}

	ai_will_do = {
		add = {
			value = -20
			desc = BASE_VALUE
		}
		if = {
			limit = {
				scope:actor = {
					any_neighbor_country = {
						NOT = { is_capital_mesoamerica = yes }
						has_casus_belli_of_type_on = {
							target = scope:actor
							type = casus_belli:cc_seize_mexico_native
						}
					}
				}
			}
			add = {
				desc = "ENEMY_HOLDS_CB"
				value = 50
			}
		}
	}
}

# NATIVE ACTION 3 — Fortify Sacred Sites
# Pays gold to apply a defensive modifier to self; also grants blessing to all
# confederation members if the actor belongs to the confederation.

	type = situation

	ai_tick = monthly
	ai_tick_frequency = 6

	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			is_capital_mesoamerica = yes
			is_subject = no
			NOT = { country_type = pop }
			NOT = { country_type = building }
		}
	}

	allow = {
		scope:actor = {
			NOT = { has_variable = ccim_recently_fortified_sacred_sites }
			gold >= 60
		}
		situation:cc_invasion_mexico = { situation_is_active = yes }

	effect = {
		scope:actor = {
			set_variable = {
				name = ccim_recently_fortified_sacred_sites
				value = yes
				years = 8
			}
			add_gold = -60
			add_country_modifier = {
				modifier = ccim_sacred_fortifications
				years = 5

					every_international_organization_member = {
						add_country_modifier = {
							modifier = ccim_divine_blessing
							years = 3
							mode = add_and_extend
						}
					}
				}
			}
		}
	}

	ai_will_do = {
		add = {
			value = 10
			desc = BASE_VALUE
		}
		if = {
			limit = {

					any_neighbor_country = {
						NOT = { is_capital_mesoamerica = yes }

							has_presence_in = region:mesoamerica_region

			}
			add = {
				desc = "ENEMY_ON_BORDER"
				value = 30
			}
		}
	}

	type = situation

	ai_tick = monthly
	ai_tick_frequency = 6

	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			NOT = { is_capital_mesoamerica = yes }
			NOT = { country_type = pop }
			NOT = { country_type = building }
			OR = {
				has_presence_in = region:mesoamerica_region
				has_presence_in = region:central_america_region
				has_presence_in = region:caribbean_region
			}
		}
	}

	allow = {
		scope:actor = {
			NOT = { has_variable = ccim_recently_commissioned_expedition }
			is_during_bankruptcy = no
			gold >= 75
		}
		situation:cc_invasion_mexico = { situation_is_active = yes }
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:cc_invasion_mexico = this
			situation_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = target
		name = "ccim_choose_expedition_target"
		column = { data = name }
		column = { data = population }
		visible = {
			is_capital_mesoamerica = yes
			NOT = { is_at_war_with = scope:actor }
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:actor = {
				set_variable = {
					name = ccim_recently_commissioned_expedition
					value = yes
					years = 5
				}
				add_gold = -75
				add_casus_belli = {
					type = casus_belli:cc_seize_mexico_native
					target = scope:target
					years = 8
				}
				trigger_event_non_silently = { id = cc_invasion_mexico.30 days = 5 }
			}
		}
	}

	ai_will_do = {
		add = {
			value = 10
			desc = BASE_VALUE
		}
		if = {
			limit = {
				exists = scope:target
				scope:target = { is_capital_mesoamerica = yes }
			}
			add = {
				desc = "RELATIVE_STRENGTH"
				value = 100
				multiply = "scope:actor.relative_strength(scope:target)"
			}
		}
		if = {
			limit = { scope:actor = { at_war = yes } }
			add = {
				desc = "AT_WAR_PENALTY"
				value = -40
			}
		}
	}

	effect = {
		scope:actor = {
			set_variable = {
				name = ccim_recently_extracted_tribute

################################################################################
# IO LEADER ACTION 1 — Call War Council
# Confederation leader calls all members to a war council, granting a military
# boost at the cost of prestige and 15 confederation unity.
################################################################################
ccim_io_war_council = {
	type = internationalorganization

	ai_tick = monthly
	ai_tick_frequency = 6

	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			is_leader_of_international_organization = international_organization:cc_mesoamerican_confederation
			NOT = { country_type = pop }
			NOT = { country_type = building }
		}
	}

	allow = {
		scope:actor = {
			NOT = { has_variable = ccim_recently_called_war_council }
			at_war = yes
		}
		international_organization:cc_mesoamerican_confederation.var:ccim_confederation_unity >= 15
		situation:cc_invasion_mexico = { situation_is_active = yes }
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		column = { data = name }
		visible = {
			international_organization_type = international_organization_type:cc_mesoamerican_confederation
		}
	}

	effect = {
		scope:actor = {
			set_variable = {
				name = ccim_recently_called_war_council
				value = yes
				years = 5
			}
			add_prestige = prestige_mild_penalty
		}
		international_organization:cc_mesoamerican_confederation = {
			change_variable = { name = ccim_confederation_unity add = -15 }
			every_international_organization_member = {
				add_country_modifier = {
					modifier = ccim_war_council_boost
					years = 2
					mode = add_and_extend
				}
			}
		}
	}

	ai_will_do = {
		add = {
			value = 20
			desc = BASE_VALUE
		}
	}
}

################################################################################
# IO LEADER ACTION 4 — Strike Against the Invaders
# Confederation leader selects a neighboring colonial power. If the leader holds
# cc_repel_invaders against them, declares war immediately and boosts all members.
# If not, grants the CB to use at will. Either way gains 20 confederation unity.
################################################################################
ccim_io_strike_invaders = {
	type = internationalorganization

	ai_tick = monthly
	ai_tick_frequency = 6

	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		scope:actor = {
			is_leader_of_international_organization = international_organization:cc_mesoamerican_confederation
			NOT = { country_type = pop }
			NOT = { country_type = building }
		}
	}

	allow = {
		scope:actor = {
			NOT = { has_variable = ccim_recently_struck_invaders }
			at_war = no
		}
		international_organization:cc_mesoamerican_confederation.var:ccim_confederation_unity >= 15
		situation:cc_invasion_mexico = { situation_is_active = yes }
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		column = { data = name }
		visible = {
			international_organization_type = international_organization_type:cc_mesoamerican_confederation
		}
	}

	select_trigger = {
		looking_for_a = country
		source = actor
		target_flag = target
		name = "ccim_choose_strike_target"
		column = { data = name }
		column = { data = population }
		visible = {
			NOT = { is_capital_mesoamerica = yes }
			OR = {
				has_presence_in = region:mesoamerica_region
				has_presence_in = region:central_america_region
				has_presence_in = region:caribbean_region
			}
			is_neighbor_of = scope:actor
			NOT = { is_at_war_with = scope:actor }
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:actor = {
				set_variable = {
					name = ccim_recently_struck_invaders
					value = yes
					years = 10
				}
			}
			international_organization:cc_mesoamerican_confederation = {
				change_variable = { name = ccim_confederation_unity add = 20 }
				every_international_organization_member = {
					add_country_modifier = {
						modifier = ccim_war_council_boost
						years = 3
						mode = add_and_extend
					}
				}
			}
			# Declare war if CB already held; otherwise grant it for later use
			if = {
				limit = {
					scope:actor = {
						has_casus_belli_of_type_on = {
							type = casus_belli:cc_repel_invaders
							target = scope:target
						}
					}
				}
				scope:actor = {
					declare_war_with_cb = {
						target = scope:target
						type = casus_belli:cc_repel_invaders
					}
				}
			}
			else = {
				scope:actor = {
					add_casus_belli = {
						type = casus_belli:cc_repel_invaders
						target = scope:target
						years = 5
					}
				}
			}
		}
	}

	ai_will_do = {
		add = {
			value = 15
			desc = BASE_VALUE
		}
		if = {
			limit = {
				exists = scope:target
				scope:actor = {
					has_casus_belli_of_type_on = {
						type = casus_belli:cc_repel_invaders
						target = scope:target
					}
				}
			}
			add = {
				desc = "HAS_CB_READY"
				value = 25
			}
		}
	}
}
