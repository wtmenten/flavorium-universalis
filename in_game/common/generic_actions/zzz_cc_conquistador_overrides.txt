
# Flavorium Universalis: create_conquistador override (Invasion of Mexico)
#
# Copy of vanilla in_game/common/generic_actions/conquistadors.txt with these edits only:
#   1. Character picker: religion = scope:actor.religion instead of religion:catholic, so the
#      non-Iberian Christian colonizers that cc_invasion_mexico grants allow_conquistadors to
#      (ccim_conquistador_capitulations) can recruit their own explorers. Unchanged for Iberia.
#   2. Target army cap: conquistador_max_target_army_size -> ccim_conquistador_max_target_army_size
#      (both in the area picker and the effect's target ordering). 60 for Mesoamerican owners
#      while the situation runs, vanilla 18 otherwise.
#   3. Same-religion foothold rule: inside Mesoamerica while the situation runs, an area is only
#      excluded if the actor itself (or a subject) already holds land there, so several
#      Christian powers can compete for it. Vanilla rule everywhere else.
#   4. ai_will_do: +40 for Mesoamerican areas while the situation runs, applied only when the
#      engine's own conquistador_utility already rates the target positively.
# Re-check against vanilla on every EU5 update.

REPLACE:create_conquistador = {
	type = owncountry


	ai_tick = monthly
	ai_tick_frequency = 6

	automation_tick = monthly
	automation_tick_frequency = 6
	player_automated_category = colonies

	allow = {
		scope:actor = {
			modifier:allow_conquistadors = yes
		}
	}

	#we don't execute the price for this action because it creates a construction, which needs its own price information in order to be able to restore the price if it is cancelled. We leave the price in here though for display
	should_execute_price = no


	price = price:recruit_conquistador


	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = target
		name = "recruit_conquistador_select_char_title"
		none_available_msg_key = "create_conquistador_no_characters"
		column = {
			data = character_explorer
		}
		visible = {
			is_valid_for_exploration = yes
			is_married = no
			NOT = { has_estate = estate_type:crown_estate }
			religion = scope:actor.religion		# CC edit 1 (vanilla: religion:catholic)
		}
	}

	select_trigger = {
		top_widget = conquistador_character_top
		looking_for_a = area
		target_flag = target_1
		cache_targets = yes #this list of targets doesn't rely on the previous target so we only need to do it once
		name = "recruit_conquistador_select_area_title"
		none_available_msg_key = "create_conquistador_no_areas"
		column = {
			data = name
		}
		visible = {
			within_colonial_range_of = scope:actor

			continent = continent:america
			scope:actor = {
				has_discovered_area = prev
			}
			is_area_coastal_sea = no

			any_location_in_area = {
				has_owner = yes
				NOT = { top_owner = scope:actor }
				scope:actor = {
					not = {
						is_subject_or_below_of = prev.top_owner
					}
				}
				NOT = { dominant_religion.group = scope:actor.religion.group }
				NOT = { owner ?= { is_subject_type = colonial_nation } }
				NOT = {
					owner ?= {
						receives_defensive_support_from = scope:actor
					}
				}
				owner ?= { potential_army_size < ccim_conquistador_max_target_army_size }		# CC edit 2
			}

			# CC edit 3. Vanilla: don't let a conquistador onto an area where another power of the
			# actor's religion group already has a foothold (direct holding or via colonial
			# nation / trading company subject). Relaxed inside Mesoamerica during the invasion.
			OR = {
				AND = {
					situation:cc_invasion_mexico = { situation_is_active = yes }
					any_location_in_area = { region = region:mesoamerica_region }
					NOT = {
						any_location_in_area = {
							top_owner ?= scope:actor
						}
					}
				}
				NOT = {
					any_location_in_area = {
						top_owner ?= {
							religion.group = scope:actor.religion.group
						}
					}
				}
			}
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:target_1
			}

			scope:target_1 = {
				ordered_location_in_area = {
					order_by = {
						value = population
						if = {
							limit = { owner ?= { potential_army_size < ccim_conquistador_max_target_army_size } }		# CC edit 2
							add = 1000000
						}
					}
					max = 1
					limit = {
						is_coastal = yes
						has_owner = yes
						NOT = { top_owner = scope:actor }
						scope:actor = {
							not = {
								is_subject_or_below_of = prev.top_owner
							}
						}
						NOT = { dominant_religion.group = scope:actor.religion.group }

					}
					save_scope_as = target_location
				}

				if = {
					limit = {
						not = { exists = scope:target_location }
					}

					#try again, but this time ignore coastal
					ordered_location_in_area = {
						order_by = {
							value = population
							if = {
								limit = { owner ?= { potential_army_size < ccim_conquistador_max_target_army_size } }		# CC edit 2
								add = 1000000
							}
						}
						max = 1
						limit = {
							has_owner = yes
							NOT = { top_owner = scope:actor }
							scope:actor = {
								not = {
									is_subject_or_below_of = prev.top_owner
								}
							}
							NOT = { dominant_religion.group = scope:actor.religion.group }
						}
						save_scope_as = target_location
					}
				}
			}

			scope:actor = {
				start_conquistador = {
					area = scope:target_1
					location = scope:target_location
					character = scope:target
					price = scope:price
					price_modifier = scope:price_modifier
				}
			}
		}
	}

	ai_will_do = {
		scope:actor = {
			add = {
				value = "conquistador_utility(scope:target_1)"
				save_temporary_value_as = utility_val
				if = {
					limit = {
						scope:utility_val > 0
					}
					add = {
						value = scope:target.total_abilities
						subtract = 150
					}
					# CC edit 4
					if = {
						limit = {
							situation:cc_invasion_mexico = { situation_is_active = yes }
							scope:target_1 = {
								any_location_in_area = { region = region:mesoamerica_region }
							}
						}
						add = 40
					}
				}
			}
		}
	}
}
