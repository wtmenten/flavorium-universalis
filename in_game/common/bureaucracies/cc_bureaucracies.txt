
cc_shadow_court_bureau = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_advance = shadow_diplomacy_advance
	}

	allow = {
		has_advance = shadow_diplomacy_advance
		nobles_estate_power >= 45
		OR = {
			has_estate_privilege = ep_noble_shadow_network
			has_estate_privilege = ep_noble_elite_enclave
		}
	}

	estates_that_like = {
		nobles_estate
		crown_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	neutral_modifier = {
		power_projection = 5
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		power_projection = 15
		diplomatic_reputation = 2
	}

	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_rebel_growth = 0.01
		nobles_estate_satisfaction_decay = 0.02
	}
}
