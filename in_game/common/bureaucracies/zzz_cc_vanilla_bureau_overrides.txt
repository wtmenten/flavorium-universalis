# VANILLA OVERRIDE of the ten generic bureaucracies and China's four, from
# game/in_game/common/bureaucracies/generic.txt and china.txt.
#
# THIS IS THE MOD'S SECOND AUTHORISED VANILLA OVERRIDE, and the larger one. The first is the
# Rhomania submod's zzz_cc_byz_vanilla_overrides.txt, covering the ten Byzantine bureaucracies.
# Between them, every bureaucracy vanilla ships is now under our control, which is deliberate:
# the bureau system is only coherent if every bureau plays by the same rules, and half a system
# would be worse than none.
#
# PATCH RISK, STATED PLAINLY. Any EU5 update touching generic.txt or china.txt diverges from
# this file silently and nothing will tell you. Re-check on every game update by diffing
# against those two files and re-applying the four edits listed below. That is the price of
# this feature and it was accepted knowingly.
#
#
# THE REPLACE: PREFIX IS LOAD-BEARING. A bare `key = { ... }` for a key vanilla already defines
# does NOT win on load order. The engine refuses it and keeps vanilla, logging
#   [gamedatabase.h:408]: Duplicated key <name> will not be created from file: ...
# while the file looks entirely correct. INJECT: would merge at the entry's top level but
# cannot reach inside an existing positive_modifier to alter its scale, which is why this file
# uses REPLACE:.
#
#
# ENTRENCHMENT IS THE ENGINE'S JOB. scope:entrenchment DOES NOT EXIST HERE.
#
# The engine ALREADY multiplies both the positive and the negative modifier of every
# bureaucracy by entrenchment. See game_concept_bureaucracy_desc, "increasing the power of both
# positive and negative modifiers", and the "Entrenchment +N%" line under a bureaucracy
# tooltip. Doing it again on the boon side would square it, so the boon side is left alone.
#
# The malus side used to carry one deliberate extra factor of scope:entrenchment, to soften a
# brand new unfunded office. That was wrong twice over. Squaring aside, the scope is not pushed
# into these blocks at all: the readme in game/in_game/common/bureaucracies/ lists it next to
# scope:maintenance, but no vanilla bureaucracy uses it anywhere and the engine reports
#
#   Undefined event target 'entrenchment'
#   Value of wrong type in '...' Got value of type 'none'
#
# The undefined read poisoned the enclosing `scale`, so every negative_modifier carrying the
# factor evaluated to none and the maluses were not applying at all. The scale is vanilla's
# again. scope:maintenance IS real; vanilla relies on it 20 times in generic.txt.
#
#
# WHAT CHANGED, AND NOTHING ELSE. Vanilla text is otherwise untouched, so this re-syncs by
# diffing and re-applying these four per entry:
#
#   1. on_maintenance_changed firing cc_bureau.60 on a cut to <= 0.25. Vanilla leaves this hook
#      undeclared on all fourteen. THE HOOK IS DELIBERATELY UNGUARDED HERE: fourteen entries
#      carry it, so an emperor economising across the board would fire the same dialogue
#      fourteen times in an afternoon. The cooldown lives on cc_bureau.60's own trigger, where
#      one guard covers every call site and cannot be missed on a fifteenth copy. This is the
#      same arrangement the Byzantine file uses and for the same reason.
#   2. maintenance_price_modifier: APPENDED a steward discount, conditional on a per-type
#      variable. See the comment at each site for why it cannot be done any other way.
#   3. positive_modifier body: a small crown power cost, since a funded office runs
#      increasingly to its own standards. The positive SCALE is untouched.
#   4. negative_modifier body: a little noble power, universal to all fourteen. The negative
#      SCALE is untouched; it once carried an entrenchment factor, see the note above.
#
# ONE ARITHMETIC MERGE was needed, because vanilla already sets the same key in that block and
# a repeated key inside one modifier block does not stack:
#   audit_bureau  positive  global_crown_estate_power  0.1 - 0.02 = 0.08
#
#
# THE AXIS DRIFT IS NOT HERE, and that is the significant difference from the first draft of
# this system. It used to sit inside each definition, one line per side. A definition carries a
# single positive_modifier block and can therefore express drift for exactly one axis, which
# broke as soon as two existed: a Byzantine emperor holding a generic bureaucracy would have
# been pushed on an axis he does not have, and monthly_towards_cc_order on a country without
# cc_order_vs_cc_magnates is the error vanilla's own byzantium.txt wraps every entry in
# has_societal_value to avoid. Drift lives in auto_modifiers/cc_bureau_axis.txt, which takes
# potential_trigger, with a mirror set in the submod for the Byzantine axis.

REPLACE:board_of_revenue_bureaucracy = {
	on_maintenance_changed = {
		if = {
			limit = {
				scope:new_maintenance < scope:old_maintenance
				scope:new_maintenance <= 0.25
			}
			trigger_event_non_silently = cc_bureau.60
		}
	}
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = cc_bureau_maintenance_base
		multiply = 0.004

		# STEWARD DISCOUNT. cc_assign_bureau_steward puts a cabinet member in charge of one named
		# bureau, and this is the only place a per-SLOT cost reduction can live:
		# maintenance_price_modifier is evaluated in COUNTRY scope (bureaucracies/readme.txt) and
		# therefore cannot ask which bureau it is pricing. Each definition naming its own variable
		# is what makes it per-slot rather than global.
		if = {
			limit = { has_variable = cc_steward_board_of_revenue_bureaucracy }
			multiply = cc_bureau_steward_discount
		}
	}

	potential = {
		has_embraced_institution = institution:confessionalism
	}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	#neutral
	neutral_modifier = {
		government_size = 1
	}

	#positive
	positive_modifier = {
		scale = { value = scope:maintenance }
		tax_income_efficiency = 0.1

		# A funded office runs increasingly to its own standards rather than the crown's.
		global_crown_estate_power = -0.02
	}

	#negative
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		estate_enrichment = 0.1

		# Any starved office becomes somebody's private asset, whatever it was founded for.
		# Small per bureau: this applies to all of them and stacks.
		global_nobles_estate_power = 0.03
	}
}
