## Cabinet Duty: Free Hands
# The default duty — minister attends to whatever the court requires.
# All idle cabinet events fire at their baseline rate while on this duty.
# Multiple ministers may be on Free Hands simultaneously.
# NOTE: idle events must check cc_minister_is_available = yes, not has_cabinet_action = no,
# so that ministers on this duty are still reached by those events.

cc_duty_free_hands = {
	ability = adm
	allow_multiple = yes

	# NOTE: is_finished cannot reliably detect per-seat empty state for allow_multiple = yes
	# without a select_trigger target providing per-instance scope. The on_activate guard
	# covers activation-time empty seats (via stop_cabinet_action); mid-action departures
	# are handled in events via cc_minister_is_available.

	on_activate = {
		# root = cabinet seat; root.cabinet_member = the character (nil if slot is empty)
		if = {
			limit = { NOT = { exists = root.cabinet_member } }
			root = { stop_cabinet_action = {} }
		}
		else = {
			root = {
				set_variable = { name = cc_cabinet_character value = cabinet_member }
			}
		}
	}

	on_deactivate = {
		root = { remove_variable = cc_cabinet_character }
	}

	ai_will_do = {
		# AI defaults here when no specialised duty applies
		add = { value = 1 }
	}
}
