# ADM PATH: indirect_rule -> colonial_assembly_advance -> federal_constitution_advance

colonial_assembly_advance = {
	age = age_6_revolutions
	icon = diplomatic_range_age_1
	for = adm
	requires = indirect_rule

	potential = {
		OR = {
			any_subject = { is_subject_type = colonial_nation }
			any_subject = { is_subject_type = client_state }
			any_subject = { is_subject_type = shadow_state }
			any_subject = { is_subject_type = chartered_company }
		}
	}
	allow = {
		OR = {
			any_subject = { is_subject_type = colonial_nation }
			any_subject = { is_subject_type = client_state }
			any_subject = { is_subject_type = shadow_state }
			any_subject = { is_subject_type = chartered_company }
		}
	}

	subject_loyalty = 5
	global_integration_speed_modifier = 0.10
}

federal_constitution_advance = {
	age = age_6_revolutions
	icon = diplomatic_influence
	for = adm
	requires = colonial_assembly_advance

	potential = {
		has_advance = colonial_assembly_advance
	}
	allow = {
		has_advance = colonial_assembly_advance
	}

	unlock_subject_type = imperial_council_member
	unlock_subject_type = equal_federal_member
	unlock_subject_type = lesser_federal_member

	subject_loyalty = 5
	global_integration_speed_modifier = 0.10
}
