puppet_state_advance = {
	age = age_5_absolutism
	icon = expansionism
	requires = diplomatic_range_age_5
	for = dip
	
	# potential = "could research the prerequisite", allow = "has researched it". Using
	# has_advance_available in allow meant the prerequisite was never actually required: both
	# prerequisites are early advances gated only on cc_subject_types_available, so they are
	# available to nearly everyone and allow passed without either being taken. Cf.
	# naval_administration_advance and military_march_advance, which already split it this way.
	potential = {
		cc_subject_types_available = yes
		OR = {
			has_advance_available = shadow_diplomacy_advance
			has_advance_available = client_state_advance
		}
	}
	allow = {
		OR = {
			has_advance = shadow_diplomacy_advance
			has_advance = client_state_advance
		}
	}

	unlock_subject_type = puppet_state

	
	# Bias research toward large realms and those with prior techs 
	ai_weight = {
		if = {
			limit = {
				OR = {
					is_great_power = yes
					is_regional_power = yes
					has_advance = shadow_diplomacy_advance
					has_advance = client_state_advance
				}
			}
			add = 100
		}
	}
}
