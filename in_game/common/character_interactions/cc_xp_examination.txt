###############################################################################
# Flavorium Universalis — Cabinet Experience: the court examination
#
# A set piece rather than a routine tool: an expensive, decade-cooldown event that
# credits the whole court at once and singles one person out.
#
# Modelled as a character interaction (you choose who presides) because the engine has
# no country-level equivalent that would let you pick the examiner, and who presides
# over an examination is exactly the sort of decision worth making.
#
# The prize does NOT go to the presider or to the most senior minister. It goes to one
# participant at random, which is the point of examining people: occasionally the
# results are not the ones the court expected.
###############################################################################

cc_xp_court_examination = {
	message = yes
	on_own_nation = yes

	price = price:cc_xp_examination

	cooldown = {
		type = cc_xp_court_examination
		years = 10
	}

	# The AI does not hold examinations. It would pay a large sum for an outcome it has
	# no way to exploit, and the interaction exists to give the player a set piece.
	ai_tick = never

	potential = {
		scope:actor = { cc_xp_enabled = yes }
	}

	# A presider, and at least two other people to examine. Three total, since the presider
	# is themselves examinable: a sitting of one is not an examination, it is a chat.
	#
	# This block used to read `any_character = { count = 2  cc_xp_is_examinable = yes }`,
	# meaning EXACTLY two examinable characters, so a court of five ministers could never
	# hold an examination at all. `count` on an any_ iterator is an exact match in EU5 and
	# has no at-least form, which is why cc_xp_num_examinable counts and this compares.
	allow = {
		scope:actor = {
			cc_xp_enabled = yes
			custom_tooltip = {
				text = cc_xp_tt_needs_presider
				any_cabinet_character = { cc_xp_can_preside = yes }
			}
			custom_tooltip = {
				text = cc_xp_tt_needs_candidates
				cc_xp_num_examinable >= 3
			}
		}
	}

	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
		name = "cc_xp_choose_examiner"
		none_available_msg_key = "cc_xp_none_presider"
		column = { data = name }

		visible = {
			in_cabinet = yes
			is_alive = yes
		}

		enabled = {
			cc_xp_can_preside = yes
		}
	}

	# This only COMMISSIONS the examination. Questions have to be set and sealed and
	# everyone has to be in the same place on the same day, so the sitting itself is
	# months away and happens in cc_xp.11, which is where every consequence lives.
	#
	# The presider carries cc_xp_examining until then, which both identifies them to the
	# delayed event and marks them as occupied: someone organising an examination is not
	# also available to be sent abroad.
	effect = {
		# Guarded on `exists`. A character interaction's effect block is evaluated to
		# build its tooltip BEFORE a target is chosen, and scope:recipient is not set
		# during that pass any more than a second target is. Without this the log fills
		# with "Undefined event target 'recipient'" every time the panel is opened.
		if = {
			limit = { exists = scope:recipient }
			scope:recipient = {
				set_variable = {
					name = cc_xp_examining
					value = yes
					months = 36
				}
			}

			scope:actor = {
				trigger_event_silently = {
					id = cc_xp.11
					months = { 5 7 }
				}
			}
		}
	}

	ai_will_do = {
		add = { value = 0 }
	}
}
