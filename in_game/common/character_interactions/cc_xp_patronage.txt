###############################################################################
# Flavorium Universalis — Cabinet Experience: patronage of the arts
#
# Connects the cabinet to vanilla's existing art system rather than building a parallel
# one. That system already provides everything structural: artist_types, artist_work,
# the commission_art and dismiss_artist interactions, an every_artist iterator, five
# work_of_art iterators, and art_quality / art_price / art_is_priceless.
#
# NOTHING HERE ANNOTATES INDIVIDUAL WORKS. A country can hold hundreds, and per-work
# variables would be both enormous and pointless. Works are reached through the engine's
# own `looking_for_a = work_of_art` selector, which enumerates them with no bookkeeping
# at all, and counted in the panel through Country.GetNumWorksOfArt.
#
# The only state this adds is one flag per sponsorship:
#   artist   cc_xp_patron        character ref, read by the GUI to show the patron's name
#   minister cc_xp_is_patron     timed flag, read by script for the yearly payoff
#
# Two variables for one relationship, deliberately. The GUI can dereference a character
# stored in a variable (GetVariable(...).GetCharacter is proven); doing the equivalent in
# script is not, so the script side never has to.
###############################################################################


###############################################################################
# SPONSOR AN ARTIST
#
# A minister puts their name and money behind an artist. While it lasts the crown gains
# a slow prestige trickle and the minister gains diplomatic experience: patronage is
# reputation work, conducted at dinner tables.
###############################################################################

cc_xp_sponsor_artist = {
	message = yes
	on_own_nation = yes

	price = price:cc_xp_patronage

	ai_tick = never

	potential = {
		scope:actor = { cc_xp_enabled = yes }
	}

	allow = {
		scope:actor = {
			cc_xp_enabled = yes
			custom_tooltip = {
				text = cc_xp_tt_needs_patron
				any_cabinet_character = { cc_xp_can_patronise = yes }
			}
			custom_tooltip = {
				text = cc_xp_tt_needs_artist
				any_artist = {
					is_alive = yes
					NOT = { has_variable = cc_xp_patron }
				}
			}
		}
	}

	# THE ARTIST is the recipient, not the patron. That is the reverse of how this read
	# before, and it is deliberate: the patronage tab has an artist ROW and no minister
	# row, and a panel action_button can only drive an interaction whose target it is able
	# to supply. Handing the artist from their own row makes the button work; the minister
	# who puts their name to it is chosen among options in cc_xp.83.
	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
		name = "cc_xp_choose_artist"
		none_available_msg_key = "cc_xp_none_artist"
		column = { data = name }

		visible = {
			is_artist = yes
			is_alive = yes
		}

		enabled = {
			is_artist = yes
			is_alive = yes
			NOT = { has_variable = cc_xp_patron }
		}
	}

	effect = {
		if = {
			limit = { exists = scope:recipient }

			scope:recipient = {
				set_variable = { name = cc_xp_awaiting_patron  value = yes  months = 2 }
				save_scope_as = cc_xp_artist
			}

			scope:actor = { trigger_event_silently = { id = cc_xp.83 } }
		}
	}

	ai_will_do = {
		add = { value = 0 }
	}
}
