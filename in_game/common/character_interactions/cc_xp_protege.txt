###############################################################################
# Flavorium Universalis — Cabinet Experience: proteges and mentorship
#
# A protege is a character outside the cabinet whom the crown has taken an interest in.
# They accumulate experience slowly (cc_xp_protege_tick), can be sent to be trained like
# any minister, and can be mentored by a senior one. The point is to have a successor
# ready before a seat opens, rather than appointing whoever is standing nearby.
#
# Capacity is three, counted live by cc_xp_has_protege_room rather than kept in a running
# total, so a protege who dies or gets appointed does not silently burn a slot forever.
# The cap exists for the same reason the cabinet has one: attention is finite.
###############################################################################


###############################################################################
# RELEASE A PROTEGE
#
# Frees a slot. Their experience is not touched: if they are picked up again later, or
# appointed outright, everything they learned is still on their record.
###############################################################################

cc_xp_release_protege = {
	message = yes
	on_own_nation = yes

	ai_tick = never

	potential = {
		scope:actor = { cc_xp_enabled = yes }
	}

	allow = {
		scope:actor = {
			cc_xp_enabled = yes
			any_character = { has_variable = cc_protege }
		}
	}

	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
		name = "cc_xp_choose_released_protege"
		none_available_msg_key = "cc_xp_none_released_protege"
		column = { data = name }

		visible = {
			has_variable = cc_protege
		}

		enabled = {
			has_variable = cc_protege
		}
	}

	effect = {
		# Guarded on `exists`. A character interaction's effect block is evaluated to
		# build its tooltip BEFORE a target is chosen, and scope:recipient is not set
		# during that pass any more than a second target is. Without this the log fills
		# with "Undefined event target 'recipient'" every time the panel is opened.
		if = {
			limit = { exists = scope:recipient }
			scope:recipient = {
				remove_variable = cc_protege
			}

			scope:actor = {
				cc_xp_sync_slots = yes
			}
		}
	}

	ai_will_do = {
		add = { value = 0 }
	}
}


###############################################################################
# ASSIGN A MENTOR
#
# Puts a senior minister's experience at a junior's disposal for a decade. The student
# gains experience yearly in the MENTOR's dominant track, which is the whole point: this
# is how a career is steered, not merely accelerated.
#
# The mentor's track is copied onto the student at assignment rather than looked up each
# year. That keeps the yearly pulse free of any need to dereference a character stored
# in a variable, and it means a mentorship survives its mentor: what they taught does
# not stop being true when they die.
###############################################################################

cc_xp_assign_mentor = {
	message = yes
	on_own_nation = yes

	price = price:cc_xp_mentorship

	ai_tick = never

	potential = {
		scope:actor = { cc_xp_enabled = yes }
	}

	# Two independently tooltipped halves: a teacher and a pupil. They fail for entirely
	# different reasons (nobody senior enough versus nobody junior enough who is free), and
	# an undescribed pair of any_ checks told the player neither.
	#
	# The pupil pool is deliberately wide: seated ministers below level 5, sponsored
	# proteges, and crown-estate courtiers who are not in the system at all. That last group
	# is the cheapest way into the ladder for the large pool of people standing around the
	# court that nothing else looks at, and the effect below initialises them.
	allow = {
		scope:actor = {
			cc_xp_enabled = yes
			custom_tooltip = {
				text = cc_xp_tt_needs_mentor
				any_cabinet_character = { cc_xp_can_mentor = yes }
			}
			# Not cc_xp_tt_needs_student: that key already belongs to school enrollment.
			custom_tooltip = {
				text = cc_xp_tt_needs_pupil
				any_character = { cc_xp_can_be_mentored = yes }
			}
		}
	}

	# ONE target flag only, and the panel supplies it from the mentor's own row.
	#
	# There used to be a second selector here for the pupil. A panel action_button cannot
	# open an interaction's target picker at all: the engine evaluates the action against
	# the parameters the button hands it, so a flag that is never supplied leaves the
	# action unperformable and the button dead with no stated reason. Vanilla never invokes
	# a two-flag interaction from a button for exactly this reason (ennoble and
	# assume_fort_command have no GUI call site anywhere). The pupil is chosen in cc_xp.80
	# instead, which keeps the choice and keeps the button alive.
	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
		name = "cc_xp_choose_mentor"
		none_available_msg_key = "cc_xp_none_mentor"
		column = { data = name }

		visible = {
			in_cabinet = yes
			is_alive = yes
		}

		enabled = {
			cc_xp_can_mentor = yes
		}
	}

	effect = {
		if = {
			limit = { exists = scope:recipient }

			# The mentor is identified to cc_xp.80 by a timed flag rather than a saved
			# scope, for the reason the rest of this system does the same: a scope is not
			# something to rely on across a queued event, and the flag also marks them as
			# occupied while the pupil is being settled on.
			scope:recipient = {
				set_variable = { name = cc_xp_offering_mentorship  value = yes  months = 2 }
				save_scope_as = cc_xp_mentor
			}

			scope:actor = { trigger_event_silently = { id = cc_xp.80 } }
		}
	}

	ai_will_do = {
		add = { value = 0 }
	}
}
