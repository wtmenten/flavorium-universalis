
# Balance of Power — the Balancing Coalition's congress.
# The bloc begins with its congress DORMANT (cc_bop_coalition_dormant): an alliance of
# convenience with no standing assembly. The chief balancer must formally convene the
# congress (the cc_bop_io_convene_the_congress leader action swaps the parliament form to
# cc_bop_coalition_assembly) before any law or parliament-issue vote can take place.

# DORMANT — the congress has not yet been called into session.
cc_bop_coalition_dormant = {
	type = international_organization
	potential = {
		international_organization_type = international_organization_type:cc_bop_coalition
	}
	locked = { parliament_type = parliament_type:cc_bop_coalition_dormant }
	modifier = {
		pure_tooltip_entry = pte_no_international_parliament	# has_international_parliament = no
	}
}

# ASSEMBLY — the standing congress, once convened.
# A lightweight international parliament (modelled on union_royal_assembly, not the HRE
# elector chambers): with no participation restrictions every member votes once and a simple
# majority carries a policy (see the default branch in resolutions/policy_vote.txt). The chief
# balancer initiates the votes (see can_initiate_policy_votes on the IO) and tables the
# parliament issues; the bloc must vote itself real teeth — mutual defence, a pre-emptive
# doctrine, congress powers, and finally the coup-de-grâce — as the aspirant grows dangerous.
cc_bop_coalition_assembly = {
	type = international_organization
	potential = {
		international_organization_type = international_organization_type:cc_bop_coalition
	}
	locked = { parliament_type = parliament_type:cc_bop_coalition_assembly }
	modifier = {
		has_international_parliament = yes
		policy_vote_required_vote_ratio = 0.5
	}
}
