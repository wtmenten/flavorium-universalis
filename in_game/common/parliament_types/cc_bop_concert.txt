
# Concert of Europe — the standing congress of the great powers.
# uses_parliament_for_law_votes routes vote weight through special_status_power, so the
# president casts two votes and each member one (see cc_bop_concert special statuses). A simple
# majority of weighted votes carries a policy.
cc_concert_assembly = {
	type = international_organization
	potential = {
		international_organization_type = international_organization_type:cc_bop_concert
	}
	locked = { parliament_type = parliament_type:cc_concert_assembly }
	modifier = {
		has_international_parliament = yes
		uses_parliament_for_law_votes = yes
		policy_vote_required_vote_ratio = 0.5
		cc_concert_president_can_participate_in_parliament = yes
		cc_concert_seat_can_participate_in_parliament = yes
	}
}
