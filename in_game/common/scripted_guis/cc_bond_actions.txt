# Subject Relations Ledger — engage actions
# Invoked from in_game/gui/panels/situation/cc_subject_bonds.gui buttons.
# root  = the overlord (the viewing player), set via GuiScope.SetRoot(Player.MakeScope)
# scope:subject = the subject country whose row was clicked
#
# Each scripted_gui is a thin OPENER: is_shown/is_valid gate the button (game rule,
# affordability, dimension cap, and a shared per-subject 2-year cooldown), and the
# effect saves the subject as scope:cc_engage_subject and fires a visible country
# event (cc_bonds.300-306). The event presents flavor and a confirm/decline choice,
# and its confirm option applies the cost, the bond change, and the cooldown stamp.

############################ CARROTS — one per dimension ############################

# Economic — Underwrite the Local Treasury (gold) -> cc_bonds.300
cc_bond_underwrite = {
	saved_scopes = { subject }
	is_shown = {
		has_game_rule = cc_subject_bonds_on
		scope:subject = { has_variable = cc_bond_initialized }
	}
	is_valid = {
		gold >= scope:subject.cc_bond_gift_cost
		scope:subject = {
			var:cc_bond_eco < 5
			NOT = { has_variable = cc_bond_action_cd }
		}
	}
	effect = {
		root = {
			scope:subject = { save_scope_as = cc_engage_subject }
			trigger_event_silently = { id = cc_bonds.300 }
		}
	}
}

############################ LEDGER VIEW TOGGLE ############################

# Flip the ledger between the short-term (current) and long-term (lifetime
# average) view. Backed by a flag on the viewing player that the panel reads via
# .IsSet to switch which score each row renders. root = the viewing player.
cc_bond_toggle_ltm = {
	is_shown = { has_game_rule = cc_subject_bonds_on }
	is_valid = { always = yes }
	effect = {
		if = {
			limit = { has_variable = cc_bond_gui_show_ltm }
			remove_variable = cc_bond_gui_show_ltm
		}
		else = {
			set_variable = cc_bond_gui_show_ltm
		}
	}
}
