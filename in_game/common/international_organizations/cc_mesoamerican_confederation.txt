
# Mesoamerican Confederation
# A defensive alliance of native Mesoamerican powers formed in response to colonial encroachment.
# Created via events or the ccim_call_confederation generic action; never from the IO screen directly.

cc_mesoamerican_confederation = {
	has_target = no
	unique = yes
	show_on_diplomatic_map = yes

	expel_members_who_are_targets_of_other_members = yes
	expel_members_who_are_attackers_at_war_with_other_members = yes

	has_leader_country = yes
	leader_type = country
	disband_if_no_leader = yes

	has_military_access = {
		scope:recipient ?= {
			is_member_of_international_organization = root
		}
		scope:actor ?= {
			is_member_of_international_organization = root
		}
	}

	can_declare_war = {
		trigger_if = {
			limit = { scope:attacker ?= { is_member_of_international_organization = scope:recipient } }
			scope:defender ?= { NOT = { is_member_of_international_organization = scope:recipient } }
		}
		trigger_else = {
			always = yes
		}
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		always = yes
	}

	can_lead_trigger = {
		is_leader_of_international_organization = international_organization:cc_mesoamerican_confederation
	}

	can_join_trigger = {
		is_capital_mesoamerica = yes
		is_subject = no
		is_rebel_country = no
		NOT = { is_member_of_international_organization = international_organization:cc_mesoamerican_confederation }
	}

	can_leave_trigger = {
		NOT = { is_leader_of_international_organization = international_organization:cc_mesoamerican_confederation }
	}

	auto_leave_trigger = {
		OR = {
			NOT = { is_capital_mesoamerica = yes }
			is_subject = yes
		}
	}

	auto_disband_trigger = {
		total_members < 2
	}

	disband_minimum_member_count = 1

	on_joined = {
		# root = joining country, scope:recipient = IO
		scope:recipient = {
			change_variable = { name = ccim_confederation_unity add = 5 }
		}
		add_country_modifier = {
			modifier = ccim_newly_joined_bonus
			years = 2
			mode = add_and_extend
		}
	}

	on_left = {
		# root = leaving country, scope:recipient = IO
		scope:recipient = {
			change_variable = { name = ccim_confederation_unity add = -10 }
		}
		add_country_modifier = {
			modifier = ccim_confederation_betrayer
			years = 3
			mode = add_and_extend
		}
	}

	leader_modifier = {
		monthly_prestige = 0.03
	}

	non_leader_modifier = {
		land_morale = 0.05
	}

	monthly_effect = {
		# Apply tiered unity bonuses/penalties to all members (2-month refresh; expires naturally when tier exits)
		if = {
			limit = { var:ccim_confederation_unity >= 70 }
			every_international_organization_member = {
				add_country_modifier = {
					modifier = ccim_high_unity_bonus
					months = 2
					mode = add_and_extend
				}
			}
		}
		else_if = {
			limit = { var:ccim_confederation_unity < 30 }
			every_international_organization_member = {
				add_country_modifier = {
					modifier = ccim_low_unity_penalty
					months = 2
					mode = add_and_extend
				}
			}
		}
	}

	variables = {
		ccim_confederation_unity = {
			format = "CCIM_UNITY_FORMAT"
			change_format = "VARIABLE_CHANGE_FORMAT"
			min = 0
			max = 100
			start = 50
			monthly_change = {
				add = {
					desc = "CCIM_UNITY_NATURAL"
					value = 0.2
				}
				add = {
					desc = "CCIM_UNITY_MEMBER_SIZE"
					value = total_members
					multiply = 0.1
				}
				add = {
					desc = "CCIM_UNITY_COMMON_THREAT"
					if = {
						limit = {
							leader_country = {
								at_war = yes
							}
						}
						value = 0.5
					}
					else = { value = 0 }
				}
				every_international_organization_member = {
					limit = { at_war = yes }
					add = -0.3
				}
			}
		}
	}

	#(root = potential joiner, actor = potential leader, recipient = existing organization if it already exists)
	ai_desire_to_join = {
		add = {
			value = -40
			desc = BASE_VALUE
		}
		# Urgency: colonizer directly borders our territory
		if = {
			limit = {
				any_neighbor_country = {
					NOT = { is_capital_mesoamerica = yes }
					OR = {
						has_presence_in = region:mesoamerica_region
						has_presence_in = region:central_america_region
						has_presence_in = region:caribbean_region
					}
				}
			}
			add = {
				desc = "CCIM_COLONIAL_BORDER"
				value = 30
			}
		}
		# War urgency — already under attack, seek allies desperately
		if = {
			limit = { at_war = yes }
			add = {
				desc = "CCIM_WAR_URGENCY"
				value = 50
			}
		}
		# Bribed nobles counsel neutrality and appeasement
		if = {
			limit = { has_country_modifier = ccim_native_weakness }
			add = {
				desc = "CCIM_BRIBED_NOBLES"
				value = -40
			}
		}
		# Relations with existing members — old grudges still matter
		if = {
			limit = {
				exists = scope:recipient
				scope:recipient = { total_members > 0 }
			}
			add = {
				desc = "CCIM_MEMBER_RELATIONS"
				scope:recipient = {
					every_international_organization_member = {
						root = {
							add = {
								value = this_opinion_of_prev
								multiply = 0.15
							}
						}
					}
				}
			}
		}
	}

	#(root = international organization, actor = potential new member, target = target if available)
	ai_desire_to_allow_new_member = {
		add = {
			desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 20
		}
	}
}
