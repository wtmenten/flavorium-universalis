cc_promote_integration = {
	estate = nobles_estate

	modifier_when_in_debate = {
		global_integration_speed_modifier = 2
	}

	allow = {
		OR = {
			parliament_type = parliament_type:council
			parliament_type = parliament_type:estate_parliament
			parliament_type = parliament_type:dutch_parliament
			parliament_type = parliament_type:plc_parliament
			parliament_type = parliament_type:autocratic_parliament
			parliament_type = parliament_type:constitutional_parliament
		}
		any_owned_location = {
			integration_level = conquered
			integration_progress > 0
			integration_progress <= 60
		}
	}

	chance = {
		add = 1
		add = {
			if = {
				limit = {
					at_war = no
				}
				value = 2
			}
		}
		every_province = {
			limit = { province_average_integration < 1 }
			add = 1
		}
		
		multiply = {
			if = {
				limit = {
					at_war = yes
				}
				value = 0.5
			}
			else = {
				value = 1	#We need that else or otherwise the whole chance is multiplied with 0
			}
		}
	}
	
	on_debate_passed = {
		ordered_province = {
			limit = { province_average_integration < 1 }
			order_by = population
			max = 1
			check_range_bounds = no
			every_location_in_province = {
				limit = { integration_level = conquered }
				change_integration_progress = 33
			}
		}
	}
	
	on_debate_failed = {
		parliament_debate_failed_effect = yes
	}
}

cc_create_naval_administration = {
	estate = crown_estate
	# estate = nobles_estate

	potential = {
		always = yes
		has_advance = naval_charter_advance
	}
	allow = {
		always = yes
		has_advance = naval_charter_advance
		NOT = {
			any_subject = {
				is_subject_type = naval_administration
			}
		}
		any_owned_location = {
			continent = ROOT.capital.continent
			# is_port = yes
			NOT = {OR = { location_rank = location_rank:rural_settlement is_capital = no }}
			dominant_culture = {
				is_primary_or_accepted_in = ROOT
			} 
		}
	}

	wants_this_parliament_issue_bias = {
		add = 100000000
	}

	modifier_when_in_debate = {
		global_sailors_modifier = 0.20
		navy_light_ship_build_cost_modifier = -0.33
		navy_heavy_ship_build_cost_modifier = -0.33
		exploration_maintenance_cost = -0.2
		exploration_preparation_time_modifier = -0.5
	}

	chance = {
		add = 8
		every_province = {
			limit = {
				OR = {
					NOT = { continent = ROOT.capital.continent }
					# is_overseas_for_owner = yes	
				}
			}
			add = 1
		}
	}
	
	on_debate_passed = {
		if = {
			limit = {
				NOT = {
					any_subject = {
						is_subject_type = naval_administration
					}
				}
				any_owned_location = {
					continent = ROOT.capital.continent
					is_port = yes
					NOT = {OR = { location_rank = location_rank:rural_settlement is_capital = yes }}
				}
			}
			random_owned_location = {
				limit = { 
					continent = ROOT.capital.continent
					is_port = yes
					NOT = {
						OR = { 
							location_rank = location_rank:rural_settlement 
							is_capital = yes 
						}
					}
				}
				weight = {
					base = 0 
					modifier = {
						add = modifier:natural_harbor_suitability
					}

					modifier = {
						factor = 100
						add = 10
					}
				}
				create_country_from_location = {
					set_capital = prev
					make_subject_of = {  target = ROOT type = subject_type:naval_administration }
					change_government_type = government_type:monarchy
				}

			}
		}
	}

	on_debate_failed = {
		parliament_debate_failed_effect = yes
	}

}
