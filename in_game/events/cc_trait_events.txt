###############################################################################
# INTEGRATION TRAIT GRANT
# Fires yearly when a cabinet member is actively integrating but has no
# integration trait.
###############################################################################

cc_traits.1 = {
	hide_portraits = yes
	type = country_event
	title = cc_traits.1.title
	desc = cc_traits.1.desc
	outcome = neutral

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			has_cabinet_action = yes
			OR = {
				cabinet_action = cabinet_action:integrate_area
				cabinet_action = cabinet_action:integrate_province
			}
			NOR = {
				has_trait = fumbling_integrator
				has_trait = able_integrator
				has_trait = master_integrator
				has_trait = iron_integrator
			}
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				has_cabinet_action = yes
				OR = {
					cabinet_action = cabinet_action:integrate_area
					cabinet_action = cabinet_action:integrate_province
				}
				NOR = {
					has_trait = fumbling_integrator
					has_trait = able_integrator
					has_trait = master_integrator
					has_trait = iron_integrator
				}
			}
			save_scope_as = integrating_minister
		}
	}

	option = {
		name = cc_traits.1.a
		scope:integrating_minister = {
			if = {
				limit = { adm > 66  mil > 66  dip > 66 }
				add_trait = trait:iron_integrator
			}
			else_if = {
				limit = { adm > 66  dip > 66 }
				add_trait = trait:master_integrator
			}
			else_if = {
				limit = { adm > 33  mil > 33  dip > 33 }
				add_trait = trait:able_integrator
			}
			else = {
				add_trait = trait:fumbling_integrator
			}
		}
	}
	option = {
		name = cc_traits.1.b
	}
}

###############################################################################
# AGE-SPECIFIC TRAIT SPAWNING
# One event per age. Each fires via monthly random_events.
# Options are gated by capital sub_continent, religion group, government type,
# active cabinet actions, and minister estate type.
###############################################################################

# Age of Renaissance — culturally branched intellectual trait
# Western/Eastern Europe → renaissance_humanist
# East/SE Asia          → eastern_pragmatist
# Muslim world          → sufi_scholar
cc_traits.10 = {
	type = country_event
	title = cc_traits.10.title
	desc = cc_traits.10.desc
	outcome = neutral

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			NOR = {
				has_trait = renaissance_humanist
				has_trait = eastern_pragmatist
				has_trait = sufi_scholar
			}
		}
		OR = {
			current_age = age_2_renaissance
			current_age = age_3_discovery
			current_age = age_4_reformation
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
		# Fire only when at least one cultural option would be visible
		OR = {
			capital.sub_continent = sub_continent:western_europe
			capital.sub_continent = sub_continent:eastern_europe
			capital.sub_continent = sub_continent:east_asia
			capital.sub_continent = sub_continent:south_east_asia
			religion.group = religion_group:muslim
		}
	}

	weight_multiplier = {
		base = 10
		modifier = {
			add = -3
			NOT = { current_age = age_2_renaissance }
		}
		modifier = {
			add = -3
			NOR = {
				current_age = age_2_renaissance
				current_age = age_3_discovery
			}
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				NOR = {
					has_trait = renaissance_humanist
					has_trait = eastern_pragmatist
					has_trait = sufi_scholar
				}
			}
			save_scope_as = minister
		}
	}

	# Western/Eastern Europe
	option = {
		name = cc_traits.10.a
		trigger = {
			OR = {
				capital.sub_continent = sub_continent:western_europe
				capital.sub_continent = sub_continent:eastern_europe
			}
		}
		scope:minister = { add_trait = trait:renaissance_humanist }
	}

	# East/SE Asia
	option = {
		name = cc_traits.10.b
		trigger = {
			OR = {
				capital.sub_continent = sub_continent:east_asia
				capital.sub_continent = sub_continent:south_east_asia
			}
		}
		scope:minister = { add_trait = trait:eastern_pragmatist }
	}

	# Muslim world
	option = {
		name = cc_traits.10.c
		trigger = {
			religion.group = religion_group:muslim
		}
		scope:minister = { add_trait = trait:sufi_scholar }
	}

	# Dismiss
	option = {
		name = cc_traits.10.d
	}
}

# Age of Discovery — pioneer (bold), skeptic (cautious), or cautious_merchant
# cautious_merchant only available to burgher estate ministers
# Event weighted higher when a minister is actively running colonial cabinet actions
cc_traits.11 = {
	type = country_event
	title = cc_traits.11.title
	desc = cc_traits.11.desc
	outcome = neutral

	trigger = {
		OR = {
			current_age = age_3_discovery
			current_age = age_4_reformation
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
		any_cabinet_character = {
			in_cabinet = yes
			NOR = {
				has_trait = age_of_discovery_pioneer
				has_trait = age_of_discovery_skeptic
				has_trait = cautious_merchant
			}
		}
	}

	weight_multiplier = {
		base = 10
		modifier = {
			factor = 0.3
			NOT = { current_age = age_3_discovery }
		}
		modifier = {
			add = 3
			any_cabinet_character = {
				in_cabinet = yes
				has_cabinet_action = yes
				OR = {
					cabinet_action = cabinet_action:send_people_to_the_colonies
					cabinet_action = cabinet_action:send_people_to_colonial_charter
					cabinet_action = cabinet_action:settle_the_frontier
				}
			}
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				NOR = {
					has_trait = age_of_discovery_pioneer
					has_trait = age_of_discovery_skeptic
					has_trait = cautious_merchant
				}
			}
			save_scope_as = minister
		}
	}

	option = {
		name = cc_traits.11.a
		scope:minister = { add_trait = trait:age_of_discovery_pioneer }
	}

	option = {
		name = cc_traits.11.b
		scope:minister = { add_trait = trait:age_of_discovery_skeptic }
	}

	# Burgher minister only: trade-focused alternative to colonial adventure
	option = {
		name = cc_traits.11.c
		trigger = {
			scope:minister = { has_estate = estate_type:burghers_estate }
		}
		scope:minister = { add_trait = trait:cautious_merchant }
	}
}

# Age of Reformation — split by religion group
# Christian → preacher (tolerant) or inquisitor (zealous)
# Non-Christian → reformist_theologian (renewal) or orthodox_defender (tradition)
# Clergy estate ministers weighted higher for selection
cc_traits.12 = {
	type = country_event
	title = cc_traits.12.title
	desc = cc_traits.12.desc
	outcome = neutral

	trigger = {
		OR = {
			current_age = age_4_reformation
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
		any_cabinet_character = {
			in_cabinet = yes
			NOR = {
				has_trait = reformation_preacher
				has_trait = reformation_inquisitor
				has_trait = reformist_theologian
				has_trait = orthodox_defender
			}
		}
	}

	weight_multiplier = {
		base = 10
		modifier = {
			factor = 0.3
			NOT = { current_age = age_4_reformation }
		}
		# More likely when a clergy minister in cabinet lacks a reformation trait
		modifier = {
			add = 3
			any_cabinet_character = {
				in_cabinet = yes
				has_estate = estate_type:clergy_estate
				NOR = {
					has_trait = reformation_preacher
					has_trait = reformation_inquisitor
					has_trait = reformist_theologian
					has_trait = orthodox_defender
				}
			}
		}
	}

	immediate = {
		# Prefer clergy estate ministers; fall back to any qualifying minister
		if = {
			limit = {
				any_cabinet_character = {
					in_cabinet = yes
					has_estate = estate_type:clergy_estate
					NOR = {
						has_trait = reformation_preacher
						has_trait = reformation_inquisitor
						has_trait = reformist_theologian
						has_trait = orthodox_defender
					}
				}
			}
			random_cabinet_character = {
				limit = {
					in_cabinet = yes
					has_estate = estate_type:clergy_estate
					NOR = {
						has_trait = reformation_preacher
						has_trait = reformation_inquisitor
						has_trait = reformist_theologian
						has_trait = orthodox_defender
					}
				}
				save_scope_as = minister
			}
		}
		else = {
			random_cabinet_character = {
				limit = {
					in_cabinet = yes
					NOR = {
						has_trait = reformation_preacher
						has_trait = reformation_inquisitor
						has_trait = reformist_theologian
						has_trait = orthodox_defender
					}
				}
				save_scope_as = minister
			}
		}
	}

	# Christian countries
	option = {
		name = cc_traits.12.a
		trigger = { religion.group = religion_group:christian }
		scope:minister = { add_trait = trait:reformation_preacher }
	}

	option = {
		name = cc_traits.12.b
		trigger = { religion.group = religion_group:christian }
		scope:minister = { add_trait = trait:reformation_inquisitor }
	}

	# Non-Christian countries
	option = {
		name = cc_traits.12.c
		trigger = { NOT = { religion.group = religion_group:christian } }
		scope:minister = { add_trait = trait:reformist_theologian }
	}

	option = {
		name = cc_traits.12.d
		trigger = { NOT = { religion.group = religion_group:christian } }
		scope:minister = { add_trait = trait:orthodox_defender }
	}
}

# Age of Absolutism — one of five governance postures
# noble_patron gated to monarchies
# parliamentarian gated to countries with a parliamentary system
# crown_theologian gated to clergy estate ministers
cc_traits.13 = {
	type = country_event
	title = cc_traits.13.title
	desc = cc_traits.13.desc
	outcome = neutral

	trigger = {
		OR = {
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
		any_cabinet_character = {
			in_cabinet = yes
			NOR = {
				has_trait = absolute_administrator
				has_trait = noble_patron
				has_trait = parliamentarian
				has_trait = privilege_reformer
				has_trait = crown_theologian
			}
		}
	}

	weight_multiplier = {
		base = 10
		modifier = {
			factor = 0.3
			NOT = { current_age = age_5_absolutism }
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				NOR = {
					has_trait = absolute_administrator
					has_trait = noble_patron
					has_trait = parliamentarian
					has_trait = privilege_reformer
					has_trait = crown_theologian
				}
			}
			save_scope_as = minister
		}
	}

	option = {
		name = cc_traits.13.a
		scope:minister = { add_trait = trait:absolute_administrator }
	}

	# Monarchies only — noble patronage makes little sense in a republic
	option = {
		name = cc_traits.13.b
		trigger = { government_type = government_type:monarchy }
		scope:minister = { add_trait = trait:noble_patron }
	}

	# Countries with a parliamentary system only
	option = {
		name = cc_traits.13.c
		trigger = { modifier:has_a_parliamentary_system = yes }
		scope:minister = { add_trait = trait:parliamentarian }
	}

	option = {
		name = cc_traits.13.d
		scope:minister = { add_trait = trait:privilege_reformer }
	}

	# Clergy estate ministers only
	option = {
		name = cc_traits.13.e
		trigger = {
			scope:minister = { has_estate = estate_type:clergy_estate }
		}
		scope:minister = { add_trait = trait:crown_theologian }
	}
}

# Age of Revolutions — agitator, counterrevolutionary, or privilege champion

					has_estate = estate_type:burghers_estate

# Ruler Teaches a Cabinet Member
# The ruler's expertise shapes the development of a young minister.
cc_traits.20 = {
	type = country_event
	title = cc_traits.20.title
	desc = cc_traits.20.desc
	outcome = neutral

	trigger = {
		has_ruler = yes
		ruler ?= {
			adm > 55
			dip > 55
			mil > 55
			OR = {
				has_trait = war_hawk
				has_trait = iron_disciplinarian
				has_trait = ecumenist
				has_trait = progressive_reformist
				has_trait = shadow_counselor
				has_trait = loyal_steward
			}
		}
		any_cabinet_character = {
			in_cabinet = yes
			character_age < 40
		}
	}

	immediate = {
		ruler ?= { save_scope_as = the_ruler }
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				character_age < 40
			}
			save_scope_as = student
		}
	}

	option = {
		name = cc_traits.20.a
		trigger = {
			scope:the_ruler ?= {
				OR = {
					has_trait = war_hawk
					has_trait = iron_disciplinarian
				}
			}
		}
		scope:student = { add_trait = trait:tactical_advisor }
	}

	option = {
		name = cc_traits.20.b
		trigger = {
			scope:the_ruler ?= {
				OR = {
					has_trait = ecumenist
					has_trait = progressive_reformist
				}
			}
		}
		scope:student = { add_trait = trait:learned_counselor }
	}

	option = {
		name = cc_traits.20.c
		trigger = {
			scope:the_ruler ?= {
				OR = {
					has_trait = shadow_counselor
					has_trait = loyal_steward
				}
			}
		}
		scope:student = { add_trait = trait:loyal_steward }
	}

	option = {
		name = cc_traits.20.d
	}
}

# Peer Learning — an idle young minister develops from observing peers
cc_traits.21 = {
	type = country_event
	title = cc_traits.21.title
	desc = cc_traits.21.desc
	outcome = neutral

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			has_cabinet_action = no
			character_age < 35
			adm > 50
			dip > 50
			mil > 50
			num_of_traits_of_category = { type = cabinet  value < 1 }
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				has_cabinet_action = no
				character_age < 35
				adm > 50
				dip > 50
				mil > 50
				num_of_traits_of_category = { type = cabinet  value < 1 }
			}
			save_scope_as = learner
		}
	}

	option = {
		name = cc_traits.21.a
		scope:learner = { add_trait = trait:learned_counselor }
	}

	option = {
		name = cc_traits.21.b
		trigger = {
			scope:learner ?= { mil > 55 }
		}
		scope:learner = { add_trait = trait:tactical_advisor }
	}

	option = {
		name = cc_traits.21.c
		scope:learner = { add_trait = trait:loyal_steward }
	}
}
