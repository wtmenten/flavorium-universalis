###############################################################################
# INTEGRATION TRAIT GRANT
# Fires yearly when a cabinet member is actively integrating but has no
# integration trait. Variable cc_integration_done prevents re-offering the
# same minister after they dismiss.
###############################################################################

cc_traits.1 = {
	hide_portraits = yes
	type = country_event
	title = cc_traits.1.title
	desc = cc_traits.1.desc
	outcome = neutral

	trigger = {
		OR = {
			has_advance = state_efficiency
			has_advance = tolerance_idea
			has_advance = experienced_diplomats
		}
		any_cabinet_character = {
			in_cabinet = yes
			has_cabinet_action = yes
			OR = {
				cabinet_action = cabinet_action:integrate_area
				cabinet_action = cabinet_action:integrate_province
			}
			OR = {
				AND = {
					NOT = { has_trait = fumbling_integrator }
					adm < 33
					dip < 33
					mil < 33
				}
				AND = {
					NOT = { has_trait = able_integrator }
					adm > 33
					dip > 33
					mil > 33
				}
				AND = {
					NOT = { has_trait = master_integrator }
					adm > 66
					dip > 66
				}
				AND = {
					NOT = { has_trait = iron_integrator }
					adm > 66
					dip > 66
					mil > 66
				}
			}
			NOT = { has_variable = cc_integration_done }
			NOT = { has_variable = cc_recent_trait_assignment }
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				has_cabinet_action = yes
				OR = {
					cabinet_action = cabinet_action:integrate_area
					cabinet_action = cabinet_action:integrate_province
				}
				OR = {
					AND = {
						NOT = { has_trait = fumbling_integrator }
						adm < 33
						dip < 33
						mil < 33
					}
					AND = {
						NOT = { has_trait = able_integrator }
						adm > 33
						dip > 33
						mil > 33
					}
					AND = {
						NOT = { has_trait = master_integrator }
						adm > 66
						dip > 66
					}
					AND = {
						NOT = { has_trait = iron_integrator }
						adm > 66
						dip > 66
						mil > 66
					}
				}
				NOT = { has_variable = cc_integration_done }
				NOT = { has_variable = cc_recent_trait_assignment }
			}
			save_scope_as = minister
		}
	}

	option = {
		name = cc_traits.1.a
		trigger = {
			scope:minister = {
				adm > 66  mil > 66  dip > 66
			}
		}
		scope:minister = {
			set_variable = cc_integration_done
			add_trait = trait:iron_integrator
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
		}
	}
	option = {
		name = cc_traits.1.b
		trigger = {
			scope:minister = {
				adm > 66
				dip > 66
				mil < 66
				mil > 33
			}
		}
		scope:minister = {
			set_variable = cc_integration_done
			add_trait = trait:master_integrator
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
		}
	}
	option = {
		name = cc_traits.1.c
		trigger = {
			scope:minister = {
				OR = {
					OR = {
						has_variable = cc_cond_30_done          # Completed an integration cabinet action
						years_of_service_in_cabinet >= 5        # Long-service bypass
					}
					AND = {
						OR = {
							adm < 66
							dip < 66
						}
						mil < 66
					}
				}
				adm > 33
				dip > 33
				mil > 33
			}
		}
		scope:minister = {
			set_variable = cc_integration_done
			add_trait = trait:able_integrator
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
		}
	}
	option = {
		name = cc_traits.1.d
		trigger = {
			scope:minister = {
				AND = {
					adm > 33
					dip > 33
					mil > 33
					OR = {
						adm < 66
						dip < 66
					}
				}
			}
		}
		scope:minister = {
			set_variable = cc_integration_done
			add_trait = trait:fumbling_integrator
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
		}
	}
	option = {
		name = cc_traits.1.e
		scope:minister = { set_variable = cc_integration_done }
	}
}

###############################################################################
# AGE-SPECIFIC TRAIT SPAWNING
# One event per age. Each fires via monthly random_events.
# Options are gated by capital sub_continent, religion group, government type,
# active cabinet actions, and minister estate type.
# Variable cc_age_trait_NN_done prevents re-offering a dismissed minister.
###############################################################################

# Age of Renaissance — culturally branched intellectual trait
# Western/Eastern Europe → renaissance_humanist   (adm > 35)
# East/SE Asia          → eastern_pragmatist      (adm > 35)
# Muslim world          → sufi_scholar            (dip > 35)
cc_traits.10 = {
	type = country_event
	title = cc_traits.10.title
	desc = cc_traits.10.desc
	outcome = neutral

	trigger = {
		any_cabinet_character = {
			NOT = { has_variable = cc_age_trait_10_done }
			NOT = { has_variable = cc_recent_trait_assignment }
			in_cabinet = yes

			# Fire only when at least one cultural option would be visible
			OR = {
				AND = {
					OR = {
						scope:country.capital.sub_continent = sub_continent:western_europe
						scope:country.capital.sub_continent = sub_continent:eastern_europe
					}
					scope:country = { has_advance = renaissance_thought }
					NOT = { has_trait = renaissance_humanist }
					adm > 50
				}
				AND = {
					OR = {
						scope:country.capital.sub_continent = sub_continent:east_asia
						scope:country.capital.sub_continent = sub_continent:south_east_asia
					}
					NOT = { has_trait = eastern_pragmatist }
					adm > 50
				}
				AND = {
					religion.group = religion_group:muslim
					NOT = { has_trait = sufi_scholar }
					dip > 50
				}
			}
		}
		OR = {
			current_age = age_2_renaissance
			current_age = age_3_discovery
			current_age = age_4_reformation
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
	}

	weight_multiplier = {
		base = 10
		modifier = {
			add = -3
			NOT = { current_age = age_2_renaissance }
		}
		modifier = {
			add = -3
			NOR = {
				current_age = age_2_renaissance
				current_age = age_3_discovery
			}
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				NOT = { has_variable = cc_age_trait_10_done }
				NOT = { has_variable = cc_recent_trait_assignment }
				in_cabinet = yes

				# Fire only when at least one cultural option would be visible
				OR = {
					AND = {
						OR = {
							scope:country.capital.sub_continent = sub_continent:western_europe
							scope:country.capital.sub_continent = sub_continent:eastern_europe
						}
						scope:country = { has_advance = renaissance_thought }
						NOT = { has_trait = renaissance_humanist }
						adm > 50
					}
					AND = {
						OR = {
							scope:country.capital.sub_continent = sub_continent:east_asia
							scope:country.capital.sub_continent = sub_continent:south_east_asia
						}
						NOT = { has_trait = eastern_pragmatist }
						adm > 50
					}
					AND = {
						religion.group = religion_group:muslim
						NOT = { has_trait = sufi_scholar }
						dip > 50
					}
				}
			}
			save_scope_as = minister
		}
	}

	# Western/Eastern Europe — adm > 50 required
	option = {
		name = cc_traits.10.a
		trigger = {
			OR = {
				capital.sub_continent = sub_continent:western_europe
				capital.sub_continent = sub_continent:eastern_europe
			}
			scope:minister = { adm > 50 }
			has_advance = renaissance_thought
		}
		scope:minister = {
			set_variable = cc_age_trait_10_done
			add_trait = trait:renaissance_humanist
			set_variable = { name = cc_traits10_scholar_project value = yes }
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
		}
		trigger_event_silently = { id = cc_traits.110 days = 240 }
	}

	# East/SE Asia — adm > 50 required
	option = {
		name = cc_traits.10.b
		trigger = {
			OR = {
				capital.sub_continent = sub_continent:east_asia
				capital.sub_continent = sub_continent:south_east_asia
			}
			scope:minister = { adm > 50 }
		}
		scope:minister = {
			set_variable = cc_age_trait_10_done
			add_trait = trait:eastern_pragmatist
			set_variable = { name = cc_traits10_scholar_project value = yes }
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
		}
		trigger_event_silently = { id = cc_traits.110 days = 240 }
	}

	# Muslim world — dip > 50 required
	option = {
		name = cc_traits.10.c
		trigger = {
			religion.group = religion_group:muslim
			scope:minister = { dip > 50 }
		}
		scope:minister = {
			set_variable = cc_age_trait_10_done
			add_trait = trait:sufi_scholar
			set_variable = { name = cc_traits10_scholar_project value = yes }
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
		}
		trigger_event_silently = { id = cc_traits.110 days = 240 }
	}

	# Dismiss — the ruler takes notes anyway; small random benefit
	option = {
		name = cc_traits.10.d
		scope:minister = { set_variable = cc_age_trait_10_done }
		random_list = {
			50 = { add_country_modifier = { modifier = cc_temp_renaissance_observation years = 5 } }
			50 = { }
		}
	}
}

# Age of Discovery — pioneer (bold), skeptic (cautious), or cautious_merchant
# pioneer only available to adm > 30 + mil > 30 ministers
# cautious_merchant only available to burgher estate ministers with dip > 40
# skeptic is the always-available fallback
cc_traits.11 = {
	type = country_event
	title = cc_traits.11.title
	desc = cc_traits.11.desc
	outcome = neutral

	trigger = {
		OR = {
			current_age = age_3_discovery
			current_age = age_4_reformation
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
		any_cabinet_character = {
			NOT = { has_variable = cc_age_trait_11_done }
			NOT = { has_variable = cc_recent_trait_assignment }
			in_cabinet = yes
			OR = {
				AND = {
					NOT = { has_trait = age_of_discovery_pioneer }
					scope:country = { has_advance = explorer_commisions_advance }
					adm > 50
					dip > 33
					mil > 33
				}
				AND = {
					NOT = { has_trait = age_of_discovery_skeptic }
				}
				AND = {
					NOT = { has_trait = cautious_merchant }
					has_estate = estate_type:burghers_estate
					dip > 40
				}
			}
		}
	}

	weight_multiplier = {
		base = 10
		modifier = {
			factor = 0.3
			NOT = { current_age = age_3_discovery }
		}
		modifier = {
			add = 3
			any_cabinet_character = {
				in_cabinet = yes
				has_cabinet_action = yes
				OR = {
					cabinet_action = cabinet_action:send_people_to_the_colonies
					cabinet_action = cabinet_action:send_people_to_colonial_charter
					cabinet_action = cabinet_action:settle_the_frontier
				}
			}
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				NOT = { has_variable = cc_age_trait_11_done }
				NOT = { has_variable = cc_recent_trait_assignment }
				in_cabinet = yes
				OR = {
					AND = {
						NOT = { has_trait = age_of_discovery_pioneer }
						scope:country = { has_advance = explorer_commisions_advance }
						adm > 50
						dip > 33
						mil > 33
					}
					AND = {
						NOT = { has_trait = age_of_discovery_skeptic }
					}
					AND = {
						NOT = { has_trait = cautious_merchant }
						has_estate = estate_type:burghers_estate
						dip > 40
					}
				}
			}
			save_scope_as = minister
		}
	}

	# Bold adventurer — explorer_commisions_advance adm > 50 dip > 33 mil > 33
	option = {
		name = cc_traits.11.a
		trigger = {
			has_advance = explorer_commisions_advance
			scope:minister = { adm > 50 dip > 33 mil > 33 }
		}
		scope:minister = {
			set_variable = cc_age_trait_11_done
			add_trait = trait:age_of_discovery_pioneer
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
		}
	}

	# Cautious analyst — always available fallback
	option = {
		name = cc_traits.11.b
		scope:minister = {
			set_variable = cc_age_trait_11_done
			add_trait = trait:age_of_discovery_skeptic
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
		}
	}

	# Burgher minister only, dip > 40 — trade-focused alternative to colonial adventure
	option = {
		name = cc_traits.11.c
		trigger = {
			scope:minister = {
				has_estate = estate_type:burghers_estate
				dip > 40
			}
		}
		scope:minister = {
			set_variable = cc_age_trait_11_done
			add_trait = trait:cautious_merchant
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
		}
	}

	# to the docks - ship building and explo cost modifier 
	option = {
		name = cc_traits.11.d
		scope:minister = {
			set_variable = cc_age_trait_11_done
		}
		trigger = {
			num_of_ports > 0
		}

		add_country_modifier = {
			modifier = cc_exploration_enthusiasm
			years = 15
			size = 1
			mode = set_to_largest_and_extend
		}

	}
}

# Age of Reformation — split by religion group
# Christian: preacher (dip > 60) or inquisitor (Catholic fallback)
# Non-Christian: reformist_theologian (always available) or orthodox_defender (mil > 66)
# Clergy estate ministers weighted higher for selection
cc_traits.12 = {
	type = country_event
	title = cc_traits.12.title
	desc = cc_traits.12.desc
	outcome = neutral

	trigger = {
		OR = {
			current_age = age_4_reformation
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
		any_cabinet_character = {
			NOT = { has_variable = cc_age_trait_12_done }
			NOT = { has_variable = cc_recent_trait_assignment }
			in_cabinet = yes
			OR = {
				AND = {
					NOT = { has_trait = reformation_preacher }
					religion.group = religion_group:christian
					has_estate = estate_type:clergy_estate
					OR = {
						dip > 60
						AND = {
							NOT = { religion = religion:catholic }
							dip > 33
						}
					}
				}
				AND = {
					NOT = { has_trait = reformation_inquisitor }
					religion = religion:catholic
				}
				AND = {
					NOT = { has_trait = reformist_theologian }
					NOT = { religion.group = religion_group:christian }
					has_estate = estate_type:clergy_estate
				}
				AND = {
					NOT = { has_trait = orthodox_defender }
					NOT = { religion.group = religion_group:christian }
					mil > 66
				}
				# Universal fallback: any minister without a reformation or dogmatic trait
				AND = {
					NOR = {
						has_trait = reformation_preacher
						has_trait = reformation_inquisitor
						has_trait = reformist_theologian
						has_trait = orthodox_defender
						has_trait = dogmatic_zealot
					}
				}
			}
		}
	}

	weight_multiplier = {
		base = 10
		modifier = {
			factor = 0.3
			NOT = { current_age = age_4_reformation }
		}
		# More likely when a clergy minister in cabinet lacks a reformation trait
		modifier = {
			add = 3
			any_cabinet_character = {
				in_cabinet = yes
				has_estate = estate_type:clergy_estate
				NOR = {
					has_trait = reformation_preacher
					has_trait = reformation_inquisitor
					has_trait = reformist_theologian
					has_trait = orthodox_defender
				}
				NOT = { has_variable = cc_age_trait_12_done }
			}
		}
	}

	immediate = {
		# Prefer clergy estate ministers; fall back to any qualifying minister
		if = {
			limit = {
				any_cabinet_character = {
					NOT = { has_variable = cc_age_trait_12_done }
					NOT = { has_variable = cc_recent_trait_assignment }
					in_cabinet = yes
					has_estate = estate_type:clergy_estate
					OR = {
						AND = {
							NOT = { has_trait = reformation_preacher }
							religion.group = religion_group:christian
							has_estate = estate_type:clergy_estate
							OR = {
								dip > 60
								AND = {
									NOT = { religion = religion:catholic }
									dip > 33
								}
							}
						}
						AND = {
							NOT = { has_trait = reformation_inquisitor }
							religion = religion:catholic
						}
						AND = {
							NOT = { has_trait = reformist_theologian }
							NOT = { religion.group = religion_group:christian }
							has_estate = estate_type:clergy_estate
						}
						AND = {
							NOT = { has_trait = orthodox_defender }
							NOT = { religion.group = religion_group:christian }
							mil > 66
						}
					}
				}
			}
			random_cabinet_character = {
				limit = {
					NOT = { has_variable = cc_age_trait_12_done }
					NOT = { has_variable = cc_recent_trait_assignment }
					in_cabinet = yes
					has_estate = estate_type:clergy_estate
					OR = {
						AND = {
							NOT = { has_trait = reformation_preacher }
							religion.group = religion_group:christian
							has_estate = estate_type:clergy_estate
							OR = {
								dip > 60
								AND = {
									NOT = { religion = religion:catholic }
									dip > 33
								}
							}
						}
						AND = {
							NOT = { has_trait = reformation_inquisitor }
							religion = religion:catholic
						}
						AND = {
							NOT = { has_trait = reformist_theologian }
							NOT = { religion.group = religion_group:christian }
							has_estate = estate_type:clergy_estate
						}
						AND = {
							NOT = { has_trait = orthodox_defender }
							NOT = { religion.group = religion_group:christian }
							mil > 66
						}
					}
				}
				save_scope_as = minister
			}
		}
		else = {
			random_cabinet_character = {
				limit = {
					NOT = { has_variable = cc_age_trait_12_done }
					NOT = { has_variable = cc_recent_trait_assignment }
					in_cabinet = yes
					OR = {
						AND = {
							NOT = { has_trait = reformation_preacher }
							religion.group = religion_group:christian
							has_estate = estate_type:clergy_estate
							OR = {
								dip > 60
								AND = {
									NOT = { religion = religion:catholic }
									dip > 33
								}
							}
						}
						AND = {
							NOT = { has_trait = reformation_inquisitor }
							religion = religion:catholic
						}
						AND = {
							NOT = { has_trait = reformist_theologian }
							NOT = { religion.group = religion_group:christian }
							has_estate = estate_type:clergy_estate
						}
						AND = {
							NOT = { has_trait = orthodox_defender }
							NOT = { religion.group = religion_group:christian }
							mil > 66
						}
						# Universal: any minister without a reformation or dogmatic trait
						AND = {
							NOR = {
								has_trait = reformation_preacher
								has_trait = reformation_inquisitor
								has_trait = reformist_theologian
								has_trait = orthodox_defender
								has_trait = dogmatic_zealot
							}
						}
					}
				}
				save_scope_as = minister
			}
		}
	}

	# Christian — tolerant preacher (premium: dip > 40)
	option = {
		name = cc_traits.12.a
		trigger = {
			religion.group = religion_group:christian
			scope:minister = {
				has_estate = estate_type:clergy_estate
				OR = {
					dip > 60
					AND = {
						NOT = { religion = religion:catholic }
						dip > 33
					}
				}
			}
		}
		scope:minister = {
			set_variable = cc_age_trait_12_done
			add_trait = trait:reformation_preacher
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
		}
	}

	# Catholic — zealous inquisitor (always available fallback for Catholics)
	option = {
		name = cc_traits.12.b
		trigger = { religion = religion:catholic }
		scope:minister = {
			set_variable = cc_age_trait_12_done
			add_trait = trait:reformation_inquisitor
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
		}
	}

	# Non-Christian — reformist theologian (clergy estate)
	option = {
		name = cc_traits.12.c
		trigger = {
			NOT = { religion.group = religion_group:christian }
			scope:minister = {
				has_estate = estate_type:clergy_estate
			}
		}
		scope:minister = {
			set_variable = cc_age_trait_12_done
			add_trait = trait:reformist_theologian
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
		}
	}

	# Non-Christian — orthodox defender (premium: mil > 66)
	option = {
		name = cc_traits.12.d
		trigger = {
			NOT = { religion.group = religion_group:christian }
			scope:minister = { mil > 66 }
		}
		scope:minister = {
			set_variable = cc_age_trait_12_done
