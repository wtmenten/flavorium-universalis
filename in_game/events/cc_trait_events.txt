###############################################################################
# INTEGRATION TRAIT GRANT
# Fires yearly when a cabinet member is actively integrating but has no
# integration trait. Variable cc_integration_done prevents re-offering the
# same minister after they dismiss.
###############################################################################

cc_traits.1 = {
	hide_portraits = yes
	type = country_event
	title = cc_traits.1.title
	desc = cc_traits.1.desc
	outcome = neutral

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			has_cabinet_action = yes
			OR = {
				cabinet_action = cabinet_action:integrate_area
				cabinet_action = cabinet_action:integrate_province
			}
			NOR = {
				has_trait = fumbling_integrator
				has_trait = able_integrator
				has_trait = master_integrator
				has_trait = iron_integrator
			}
			NOT = { has_variable = cc_integration_done }
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				has_cabinet_action = yes
				OR = {
					cabinet_action = cabinet_action:integrate_area
					cabinet_action = cabinet_action:integrate_province
				}
				NOR = {
					has_trait = fumbling_integrator
					has_trait = able_integrator
					has_trait = master_integrator
					has_trait = iron_integrator
				}
				NOT = { has_variable = cc_integration_done }
			}
			save_scope_as = integrating_minister
		}
	}

	option = {
		name = cc_traits.1.a
		scope:integrating_minister = {
			set_variable = cc_integration_done
			if = {
				limit = { adm > 66  mil > 66  dip > 66 }
				add_trait = trait:iron_integrator
			}
			else_if = {
				limit = { adm > 66  dip > 66 }
				add_trait = trait:master_integrator
			}
			else_if = {
				limit = { adm > 33  mil > 33  dip > 33 }
				add_trait = trait:able_integrator
			}
			else = {
				add_trait = trait:fumbling_integrator
			}
		}
	}
	option = {
		name = cc_traits.1.b
		scope:integrating_minister = { set_variable = cc_integration_done }
	}
}

###############################################################################
# AGE-SPECIFIC TRAIT SPAWNING
# One event per age. Each fires via monthly random_events.
# Options are gated by capital sub_continent, religion group, government type,
# active cabinet actions, and minister estate type.
# Variable cc_age_trait_NN_done prevents re-offering a dismissed minister.
###############################################################################

# Age of Renaissance — culturally branched intellectual trait
# Western/Eastern Europe → renaissance_humanist   (adm > 35)
# East/SE Asia          → eastern_pragmatist      (adm > 35)
# Muslim world          → sufi_scholar            (dip > 35)
cc_traits.10 = {
	type = country_event
	title = cc_traits.10.title
	desc = cc_traits.10.desc
	outcome = neutral

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			NOR = {
				has_trait = renaissance_humanist
				has_trait = eastern_pragmatist
				has_trait = sufi_scholar
			}
			NOT = { has_variable = cc_age_trait_10_done }
		}
		OR = {
			current_age = age_2_renaissance
			current_age = age_3_discovery
			current_age = age_4_reformation
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
		# Fire only when at least one cultural option would be visible
		OR = {
			capital.sub_continent = sub_continent:western_europe
			capital.sub_continent = sub_continent:eastern_europe
			capital.sub_continent = sub_continent:east_asia
			capital.sub_continent = sub_continent:south_east_asia
			religion.group = religion_group:muslim
		}
	}

	weight_multiplier = {
		base = 10
		modifier = {
			add = -3
			NOT = { current_age = age_2_renaissance }
		}
		modifier = {
			add = -3
			NOR = {
				current_age = age_2_renaissance
				current_age = age_3_discovery
			}
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				NOR = {
					has_trait = renaissance_humanist
					has_trait = eastern_pragmatist
					has_trait = sufi_scholar
				}
				NOT = { has_variable = cc_age_trait_10_done }
			}
			save_scope_as = minister
		}
	}

	# Western/Eastern Europe — adm > 35 required
	option = {
		name = cc_traits.10.a
		trigger = {
			OR = {
				capital.sub_continent = sub_continent:western_europe
				capital.sub_continent = sub_continent:eastern_europe
			}
			scope:minister = { adm > 35 }
		}
		scope:minister = {
			set_variable = cc_age_trait_10_done
			add_trait = trait:renaissance_humanist
		}
	}

	# East/SE Asia — adm > 35 required
	option = {
		name = cc_traits.10.b
		trigger = {
			OR = {
				capital.sub_continent = sub_continent:east_asia
				capital.sub_continent = sub_continent:south_east_asia
			}
			scope:minister = { adm > 35 }
		}
		scope:minister = {
			set_variable = cc_age_trait_10_done
			add_trait = trait:eastern_pragmatist
		}
	}

	# Muslim world — dip > 35 required
	option = {
		name = cc_traits.10.c
		trigger = {
			religion.group = religion_group:muslim
			scope:minister = { dip > 35 }
		}
		scope:minister = {
			set_variable = cc_age_trait_10_done
			add_trait = trait:sufi_scholar
		}
	}

	# Dismiss — always available; marks minister as offered so they aren't re-selected
	option = {
		name = cc_traits.10.d
		scope:minister = { set_variable = cc_age_trait_10_done }
	}
}

# Age of Discovery — pioneer (bold), skeptic (cautious), or cautious_merchant
# pioneer only available to adm > 30 + mil > 30 ministers
# cautious_merchant only available to burgher estate ministers with dip > 40
# skeptic is the always-available fallback
cc_traits.11 = {
	type = country_event
	title = cc_traits.11.title
	desc = cc_traits.11.desc
	outcome = neutral

	trigger = {
		OR = {
			current_age = age_3_discovery
			current_age = age_4_reformation
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
		any_cabinet_character = {
			in_cabinet = yes
			NOR = {
				has_trait = age_of_discovery_pioneer
				has_trait = age_of_discovery_skeptic
				has_trait = cautious_merchant
			}
			NOT = { has_variable = cc_age_trait_11_done }
		}
	}

	weight_multiplier = {
		base = 10
		modifier = {
			factor = 0.3
			NOT = { current_age = age_3_discovery }
		}
		modifier = {
			add = 3
			any_cabinet_character = {
				in_cabinet = yes
				has_cabinet_action = yes
				OR = {
					cabinet_action = cabinet_action:send_people_to_the_colonies
					cabinet_action = cabinet_action:send_people_to_colonial_charter
					cabinet_action = cabinet_action:settle_the_frontier
				}
			}
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				NOR = {
					has_trait = age_of_discovery_pioneer
					has_trait = age_of_discovery_skeptic
					has_trait = cautious_merchant
				}
				NOT = { has_variable = cc_age_trait_11_done }
			}
			save_scope_as = minister
		}
	}

	# Bold adventurer — adm > 30 and mil > 30
	option = {
		name = cc_traits.11.a
		trigger = {
			scope:minister = { adm > 30  mil > 30 }
		}
		scope:minister = {
			set_variable = cc_age_trait_11_done
			add_trait = trait:age_of_discovery_pioneer
		}
	}

	# Cautious analyst — always available fallback
	option = {
		name = cc_traits.11.b
		scope:minister = {
			set_variable = cc_age_trait_11_done
			add_trait = trait:age_of_discovery_skeptic
		}
	}

	# Burgher minister only, dip > 40 — trade-focused alternative to colonial adventure
	option = {
		name = cc_traits.11.c
		trigger = {
			scope:minister = {
				has_estate = estate_type:burghers_estate
				dip > 40
			}
		}
		scope:minister = {
			set_variable = cc_age_trait_11_done
			add_trait = trait:cautious_merchant
		}
	}
}

# Age of Reformation — split by religion group
# Christian: preacher (dip > 40) or inquisitor (always available fallback)
# Non-Christian: reformist_theologian (always available) or orthodox_defender (mil > 35)
# Clergy estate ministers weighted higher for selection
cc_traits.12 = {
	type = country_event
	title = cc_traits.12.title
	desc = cc_traits.12.desc
	outcome = neutral

	trigger = {
		OR = {
			current_age = age_4_reformation
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
		any_cabinet_character = {
			in_cabinet = yes
			NOR = {
				has_trait = reformation_preacher
				has_trait = reformation_inquisitor
				has_trait = reformist_theologian
				has_trait = orthodox_defender
			}
			NOT = { has_variable = cc_age_trait_12_done }
		}
	}

	weight_multiplier = {
		base = 10
		modifier = {
			factor = 0.3
			NOT = { current_age = age_4_reformation }
		}
		# More likely when a clergy minister in cabinet lacks a reformation trait
		modifier = {
			add = 3
			any_cabinet_character = {
				in_cabinet = yes
				has_estate = estate_type:clergy_estate
				NOR = {
					has_trait = reformation_preacher
					has_trait = reformation_inquisitor
					has_trait = reformist_theologian
					has_trait = orthodox_defender
				}
				NOT = { has_variable = cc_age_trait_12_done }
			}
		}
	}

	immediate = {
		# Prefer clergy estate ministers; fall back to any qualifying minister
		if = {
			limit = {
				any_cabinet_character = {
					in_cabinet = yes
					has_estate = estate_type:clergy_estate
					NOR = {
						has_trait = reformation_preacher
						has_trait = reformation_inquisitor
						has_trait = reformist_theologian
						has_trait = orthodox_defender
					}
					NOT = { has_variable = cc_age_trait_12_done }
				}
			}
			random_cabinet_character = {
				limit = {
					in_cabinet = yes
					has_estate = estate_type:clergy_estate
					NOR = {
						has_trait = reformation_preacher
						has_trait = reformation_inquisitor
						has_trait = reformist_theologian
						has_trait = orthodox_defender
					}
					NOT = { has_variable = cc_age_trait_12_done }
				}
				save_scope_as = minister
			}
		}
		else = {
			random_cabinet_character = {
				limit = {
					in_cabinet = yes
					NOR = {
						has_trait = reformation_preacher
						has_trait = reformation_inquisitor
						has_trait = reformist_theologian
						has_trait = orthodox_defender
					}
					NOT = { has_variable = cc_age_trait_12_done }
				}
				save_scope_as = minister
			}
		}
	}

	# Christian — tolerant preacher (premium: dip > 40)
	option = {
		name = cc_traits.12.a
		trigger = {
			religion.group = religion_group:christian
			scope:minister = { dip > 40 }
		}
		scope:minister = {
			set_variable = cc_age_trait_12_done
			add_trait = trait:reformation_preacher
		}
	}

	# Christian — zealous inquisitor (always available fallback for Christians)
	option = {
		name = cc_traits.12.b
		trigger = { religion.group = religion_group:christian }
		scope:minister = {
			set_variable = cc_age_trait_12_done
			add_trait = trait:reformation_inquisitor
		}
	}

	# Non-Christian — reformist theologian (always available fallback)
	option = {
		name = cc_traits.12.c
		trigger = { NOT = { religion.group = religion_group:christian } }
		scope:minister = {
			set_variable = cc_age_trait_12_done
			add_trait = trait:reformist_theologian
		}
	}

	# Non-Christian — orthodox defender (premium: mil > 35)
	option = {
		name = cc_traits.12.d
		trigger = {
			NOT = { religion.group = religion_group:christian }
			scope:minister = { mil > 35 }
		}
		scope:minister = {
			set_variable = cc_age_trait_12_done
			add_trait = trait:orthodox_defender
		}
	}
}

# Age of Absolutism — one of five governance postures
# absolute_administrator and privilege_reformer always available (fallbacks)
# noble_patron: monarchy + dip > 40 + noble_patronage_advance available
# parliamentarian: parliamentary system + dip > 45
# crown_theologian: clergy estate + adm > 40
cc_traits.13 = {
	type = country_event
	title = cc_traits.13.title
	desc = cc_traits.13.desc
	outcome = neutral

	trigger = {
		OR = {
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
		any_cabinet_character = {
			in_cabinet = yes
			NOR = {
				has_trait = absolute_administrator
				has_trait = noble_patron
				has_trait = parliamentarian
				has_trait = privilege_reformer
				has_trait = crown_theologian
			}
			NOT = { has_variable = cc_age_trait_13_done }
		}
	}

	weight_multiplier = {
		base = 10
		modifier = {
			factor = 0.3
			NOT = { current_age = age_5_absolutism }
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				NOR = {
					has_trait = absolute_administrator
					has_trait = noble_patron
					has_trait = parliamentarian
					has_trait = privilege_reformer
					has_trait = crown_theologian
				}
				NOT = { has_variable = cc_age_trait_13_done }
			}
			save_scope_as = minister
		}
	}

	# Centralizing administrator — always available
	option = {
		name = cc_traits.13.a
		scope:minister = {
			set_variable = cc_age_trait_13_done
			add_trait = trait:absolute_administrator
		}
	}

	# Monarchies + dip > 40 + noble patronage advance available
	option = {
		name = cc_traits.13.b
		trigger = {
			government_type = government_type:monarchy
			scope:minister = { dip > 40 }
			has_advance_available = noble_patronage_advance
		}
		scope:minister = {
			set_variable = cc_age_trait_13_done
			add_trait = trait:noble_patron
		}
	}

	# Parliamentary system + dip > 45
	option = {
		name = cc_traits.13.c
		trigger = {
			modifier:has_a_parliamentary_system = yes
			scope:minister = { dip > 45 }
		}
		scope:minister = {
			set_variable = cc_age_trait_13_done
			add_trait = trait:parliamentarian
		}
	}

	# Privilege broker — always available
	option = {
		name = cc_traits.13.d
		scope:minister = {
			set_variable = cc_age_trait_13_done
			add_trait = trait:privilege_reformer
		}
	}

	# Clergy estate + adm > 40
	option = {
		name = cc_traits.13.e
		trigger = {
			scope:minister = {
				has_estate = estate_type:clergy_estate
				adm > 40
			}
		}
		scope:minister = {
			set_variable = cc_age_trait_13_done
			add_trait = trait:crown_theologian
		}
	}
}

# Ruler Teaches a Cabinet Member
# The ruler's expertise shapes the development of a young minister.
# Variable cc_teach_20_done prevents repeat offers to the same student.
cc_traits.20 = {
	type = country_event
	title = cc_traits.20.title
	desc = cc_traits.20.desc
	outcome = neutral

	trigger = {
		has_ruler = yes
		ruler ?= {
			adm > 55
			dip > 55
			mil > 55
			OR = {
				has_trait = war_hawk
				has_trait = iron_disciplinarian
				has_trait = ecumenist
				has_trait = progressive_reformist
				has_trait = shadow_counselor
				has_trait = loyal_steward
			}
		}
		any_cabinet_character = {
			in_cabinet = yes
			character_age < 40
			NOT = { has_variable = cc_teach_20_done }
		}
	}

	immediate = {
		ruler ?= { save_scope_as = the_ruler }
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				character_age < 40
				NOT = { has_variable = cc_teach_20_done }
			}
			save_scope_as = student
		}
	}

	# Military tradition — ruler has war_hawk/iron_disciplinarian + student mil > 25
	option = {
		name = cc_traits.20.a
		trigger = {
			scope:the_ruler ?= {
				OR = {
					has_trait = war_hawk
					has_trait = iron_disciplinarian
				}
			}
			scope:student = { mil > 25 }
		}
		scope:student = {
			set_variable = cc_teach_20_done
			add_trait = trait:tactical_advisor
		}
	}

	# Scholarly tradition — ruler has ecumenist/progressive_reformist + student adm > 25
	option = {
		name = cc_traits.20.b
		trigger = {
			scope:the_ruler ?= {
				OR = {
					has_trait = ecumenist
					has_trait = progressive_reformist
				}
			}
			scope:student = { adm > 25 }
		}
		scope:student = {
			set_variable = cc_teach_20_done
			add_trait = trait:learned_counselor
		}
	}

	# Statecraft tradition — ruler has shadow_counselor/loyal_steward + student dip > 25
	option = {
		name = cc_traits.20.c
		trigger = {
			scope:the_ruler ?= {
				OR = {
					has_trait = shadow_counselor
					has_trait = loyal_steward
				}
			}
			scope:student = { dip > 25 }
		}
		scope:student = {
			set_variable = cc_teach_20_done
			add_trait = trait:loyal_steward
		}
	}

	# Dismiss — marks student so they won't be re-selected
	option = {
		name = cc_traits.20.d
		scope:student = { set_variable = cc_teach_20_done }
	}
}

# Peer Learning — an idle young minister develops from observing peers
# loyal_steward option now gated to dip > 50.
# Variable cc_teach_21_done prevents re-offering.
cc_traits.21 = {
	type = country_event
	title = cc_traits.21.title
	desc = cc_traits.21.desc
	outcome = neutral

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			has_cabinet_action = no
			character_age < 35
			adm > 50
			dip > 50
			mil > 50
			num_of_traits_of_category = { type = cabinet  value < 1 }
			NOT = { has_variable = cc_teach_21_done }
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				has_cabinet_action = no
				character_age < 35
				adm > 50
				dip > 50
				mil > 50
				num_of_traits_of_category = { type = cabinet  value < 1 }
				NOT = { has_variable = cc_teach_21_done }
			}
			save_scope_as = learner
		}
	}

	# Scholarly focus — always available
	option = {
		name = cc_traits.21.a
		scope:learner = {
			set_variable = cc_teach_21_done
			add_trait = trait:learned_counselor
		}
	}

	# Military focus — mil > 55
	option = {
		name = cc_traits.21.b
		trigger = {
			scope:learner ?= { mil > 55 }
		}
		scope:learner = {
			set_variable = cc_teach_21_done
			add_trait = trait:tactical_advisor
		}
	}

	# Diplomatic focus — dip > 50 (always met by trigger but confirms intent)
	option = {
		name = cc_traits.21.c
		trigger = {
			scope:learner = { dip > 50 }
		}
		scope:learner = {
			set_variable = cc_teach_21_done
			add_trait = trait:loyal_steward
		}
	}
}
