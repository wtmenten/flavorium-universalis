
namespace = cc_bonds

###############################################################################
# cc_bonds.1 — Hidden Monitor (10-year passive cycle)
# Root = overlord. Iterates every tracked subject via the registry map.
# Applies passive dimension adjustments, sets priming flags, syncs score maps.
###############################################################################

cc_bonds.1 = {
	type = country_event
	title = cc_bonds.1.title
	desc = cc_bonds.1.desc
	hidden = yes

	trigger = {
		has_game_rule = cc_subject_bonds_on
		NOT = { has_variable = cc_bond_monitor_cd }
		any_subject = { has_variable = cc_bond_initialized }
	}

	immediate = {
		set_variable = { name = cc_bond_monitor_cd  value = yes  years = 10 }

		every_key_in_variable_map = {
			variable = cc_subject_registry_map

			if = {
				limit = { has_variable = cc_bond_initialized }
				save_scope_as = cc_sync_subj

					# --- 0. Upgrade subjects from saves predating the LTM system ---
					if = {
						limit = { NOT = { has_variable = cc_bond_ltm_sweeps } }
						cc_bonds_ensure_ltm_stores = yes
					}

				# --- 0b. Per-cycle regression toward zero (BEFORE this cycle's drivers) ---
				# Decay first, then apply drivers: a quiet subject rests at 0, a single
				# sustained +1 driver rests at ~+1, and event spikes bleed off over time.
				cc_bonds_decay_dimension = { DIM = eco }
				cc_bonds_decay_dimension = { DIM = mil }
				cc_bonds_decay_dimension = { DIM = pol }
				cc_bonds_decay_dimension = { DIM = cul }
				cc_bonds_decay_dimension = { DIM = per }

				# --- 0c. Subsidies → eco (sustained while the overlord subsidizes the subject) ---
				# opinion_subsidies is held by the recipient toward the subsidizer, so the
				# subject holds it toward root (the overlord) while subsidies are active.
				if = {
					limit = { has_opinion = { target = root  modifier = opinion_subsidies } }
					change_variable = { name = cc_bond_eco  add = 1 }
					clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
				}

				# --- 1. Foreign buildings → eco ±1 ---
				set_local_variable = { name = lv_invest  value = 0 }
				prev = {
					every_owned_foreign_building_location = {
						limit = { owner = scope:cc_sync_subj }
						change_local_variable = { name = lv_invest  add = 1 }
					}
				}
				if = {
					limit = { local_var:lv_invest >= 2 }
					change_variable = { name = cc_bond_eco  add = 1 }
					clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
				}
				else_if = {
					limit = { local_var:lv_invest = 0 }
					change_variable = { name = cc_bond_eco  add = -1 }
					clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
				}

				# --- 2. Culture conversion detection (one-shot) ---
				if = {
					limit = {
						has_variable = cc_bond_mem_cultures_differ
						culture = prev.culture
					}
					# Cultures now match — likely forced conversion
					change_variable = { name = cc_bond_cul  add = -2 }
					clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
					remove_variable = cc_bond_mem_cultures_differ
				}
				else_if = {
					limit = {
						has_variable = cc_bond_mem_cultures_differ
						NOT = { culture = prev.culture }
					}
					# Subject retained distinct culture — slight positive
					change_variable = { name = cc_bond_cul  add = 1 }
					clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
				}

				# --- 3. Province growth → eco ±1 ---
				if = {
					limit = { num_provinces > var:cc_bond_init_provinces }
					change_variable = { name = cc_bond_eco  add = 1 }
					clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
				}
				else_if = {
					limit = { num_provinces < var:cc_bond_init_provinces }
					change_variable = { name = cc_bond_eco  add = -1 }
					clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
				}

				# --- 4. War-call frequency → mil ±1 then reset counter ---
				if = {
					limit = { var:cc_bond_war_calls >= 4 }
					change_variable = { name = cc_bond_mil  add = -1 }
					clamp_variable = { name = cc_bond_mil  max = 5  min = -5 }
					change_variable = { name = cc_bond_eco  add = -1 }
					clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
				}
				else_if = {
					# Recovery only: a peaceful decade nudges mil back toward neutral, but
					# never builds a standing positive (otherwise near-every subject pins +mil).
					limit = { var:cc_bond_war_calls = 0  var:cc_bond_mil < 0 }
					change_variable = { name = cc_bond_mil  add = 1 }
					clamp_variable = { name = cc_bond_mil  max = 5  min = -5 }
				}
				set_variable = { name = cc_bond_war_calls  value = 0 }

				# --- 5. Unjust wars accumulation → mil/cul, counter decays ---
				if = {
					limit = { var:cc_bond_unjust_wars >= 3 }
					change_variable = { name = cc_bond_mil  add = -1 }
					clamp_variable = { name = cc_bond_mil  max = 5  min = -5 }
				}
				if = {
					limit = { var:cc_bond_unjust_wars >= 5 }
					change_variable = { name = cc_bond_cul  add = -1 }
					clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
				}
				if = {
					limit = { var:cc_bond_unjust_wars > 0 }
					change_variable = { name = cc_bond_unjust_wars  add = -1 }
				}

				# --- 6. Institution direction tracking (updated each 10-year cycle) ---
				remove_variable = cc_bond_insti_to_subject
				remove_variable = cc_bond_insti_to_overlord
				remove_variable = cc_bond_insti_both

				# Overlord → subject: overlord has an institution this subject lacks
				if = {
					limit = {
						root = {
							any_institutions_embraced = {
								NOT = { scope:cc_sync_subj = { has_embraced_institution = prev } }
							}
						}
					}
					set_variable = { name = cc_bond_insti_to_subject  value = yes }
				}

				# Subject → overlord: this subject has an institution the overlord lacks
				if = {
					limit = {
						any_institutions_embraced = {
							NOT = { root = { has_embraced_institution = prev } }
						}
					}
					set_variable = { name = cc_bond_insti_to_overlord  value = yes }
				}

				# Both directions active simultaneously
				if = {
					limit = {
						has_variable = cc_bond_insti_to_subject
						has_variable = cc_bond_insti_to_overlord
					}
					set_variable = { name = cc_bond_insti_both  value = yes }
				}

				# Apply passive bond adjustments
				if = {
					limit = { has_variable = cc_bond_insti_both }
					change_variable = { name = cc_bond_eco  add = 1 }
					clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
					change_variable = { name = cc_bond_cul  add = 1 }
					clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
				}
				else_if = {
					limit = { has_variable = cc_bond_insti_to_subject }
					change_variable = { name = cc_bond_eco  add = 1 }
					clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
				}
				else_if = {
					limit = { has_variable = cc_bond_insti_to_overlord }
					# Subject contributing institutions upward — economic link but cultural tension
					change_variable = { name = cc_bond_eco  add = 1 }
					clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
					change_variable = { name = cc_bond_cul  add = -1 }
					clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
				}

				# --- 7. Priming flags ---
				if = {
					limit = {
						var:cc_bond_eco >= 2
						var:cc_bond_pol >= 2
						var:cc_bond_per >= 2
					}
					set_variable = { name = cc_bond_loyal_primed  value = yes }
					remove_variable = cc_bond_disloyal_primed
					remove_variable = cc_bond_revolutionary_primed
				}
				else_if = {
					limit = {
						OR = {
							AND = {
								var:cc_bond_eco <= -2
								var:cc_bond_pol <= -2
							}
							AND = {
								var:cc_bond_eco <= -2
								var:cc_bond_cul <= -2
							}
							AND = {
								var:cc_bond_mil <= -2
								var:cc_bond_pol <= -2
							}
						}
					}
					set_variable = { name = cc_bond_disloyal_primed  value = yes }
					remove_variable = cc_bond_loyal_primed
					if = {
						limit = {
							var:cc_bond_eco <= -2
							var:cc_bond_cul <= -2
						}
						set_variable = { name = cc_bond_revolutionary_primed  value = yes }
					}
					else = {
						remove_variable = cc_bond_revolutionary_primed
					}
				}
				else = {
					remove_variable = cc_bond_loyal_primed
					remove_variable = cc_bond_disloyal_primed
					remove_variable = cc_bond_revolutionary_primed
				}

				# --- 7b. Long-term memory: peak tracking (every cycle) + sweep (long cadence) ---
				cc_bonds_track_peak = { DIM = eco }
				cc_bonds_track_peak = { DIM = mil }
				cc_bonds_track_peak = { DIM = pol }
				cc_bonds_track_peak = { DIM = cul }
				cc_bonds_track_peak = { DIM = per }
				if = {
					limit = { NOT = { has_variable = cc_bond_ltm_sweep_cd } }
					change_variable = { name = cc_bond_ltm_sweeps  add = 1 }
					cc_bonds_sweep_dimension = { DIM = eco }
					cc_bonds_sweep_dimension = { DIM = mil }
					cc_bonds_sweep_dimension = { DIM = pol }
					cc_bonds_sweep_dimension = { DIM = cul }
					cc_bonds_sweep_dimension = { DIM = per }
					# Next sweep in 60-80 years (jittered so subjects do not all sweep at once).
					random_list = {
						33 = { set_variable = { name = cc_bond_ltm_sweep_cd  value = yes  years = 60 } }
						34 = { set_variable = { name = cc_bond_ltm_sweep_cd  value = yes  years = 70 } }
						33 = { set_variable = { name = cc_bond_ltm_sweep_cd  value = yes  years = 80 } }
					}
				}

				# --- 8. Sync scores (STM + effective + peak-low) to overlord variable maps ---
				cc_bonds_sync_dimension = { DIM = eco }
				cc_bonds_sync_dimension = { DIM = mil }
				cc_bonds_sync_dimension = { DIM = pol }
				cc_bonds_sync_dimension = { DIM = cul }
				cc_bonds_sync_dimension = { DIM = per }
			}
		}
	}
}
