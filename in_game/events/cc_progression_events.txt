# CABINETS AND CHOICES — STEPPING STONE PROGRESSION EVENTS (Pillar 12)
#
# Chain A (Reformer's Path):     cc_prog.1–4   fumbling_reformist → master_reformer
# Chain C (Diplomat's Ascent):   cc_prog.10–12 tentative_envoy → master_statesman
# Chain D (Military Hardening):  cc_prog.20–22 green_adjutant → standing_army/defensive
# Chain F (Fiscal Path):         cc_prog.30–32 clumsy_accountant → prosperity/treasury
# Chain E (Colonial Pioneer):    cc_prog.40    frontier_administrator → returned_colonial_governor
#
# Progress events (cc_prog.1/.10/.20/.30) fire via cc_cabinet_yearly_prog_progress.
# Advance events (cc_prog.2–4/.11–12/.21–22/.31–32/.40) sweep yearly.
###############################################################################


###############################################################################
# CHAIN A — THE REFORMER'S PATH
###############################################################################

# Fires randomly when a minister holds fumbling_reformist.
# Increments cc_prog_a_count; awards a skill bump.
cc_prog.1 = {
	type = country_event
	title = cc_prog.1.title
	desc = cc_prog.1.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
	}

	trigger = {
		any_cabinet_character = {
			has_trait = fumbling_reformist
			NOT = { has_variable = cc_prog_a_cooldown }
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				has_trait = fumbling_reformist
				NOT = { has_variable = cc_prog_a_cooldown }
			}
			if = {
				limit = { NOT = { has_variable = cc_prog_a_count } }
				set_variable = { name = cc_prog_a_count value = 0 }
			}
			set_variable = { name = cc_prog_a_cooldown value = yes months = 24 }
			save_scope_as = minister
		}
	}

	option = {
		name = cc_prog.1.a
		scope:minister = {
			add_adm = 3
			change_variable = { name = cc_prog_a_count add = 1 }
		}
	}

	option = {
		name = cc_prog.1.b
		add_stability = stability_weak_penalty
		scope:minister = {
			add_adm = 1
			change_variable = { name = cc_prog_a_count add = 1 }
		}
	}
}

# Sweep: fires when tactical_advisor + graduated + 5 years service.
# Branching advance: standing_army_advocate (Age 5+, mil > 60), offensive_strategist, or defensive_commander.
cc_prog.22 = {
	type = country_event
	title = cc_prog.22.title
	desc = cc_prog.22.desc
	outcome = positive

	illustration_tags = {
		10 = military
		10 = army
	}

	trigger = {
		any_cabinet_character = {
			has_trait = tactical_advisor
			has_variable = cc_chain_d_graduated
			years_of_service_in_cabinet >= 5
			mil > 50
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				has_trait = tactical_advisor
				has_variable = cc_chain_d_graduated
				years_of_service_in_cabinet >= 5
				mil > 50
			}
			save_scope_as = minister
		}
	}

	option = {
		name = cc_prog.22.a
		trigger = {
			scope:minister = {
				mil > 60
				OR = {
					current_age = age_5_absolutism
					current_age = age_6_revolutions
				}
			}
		}
		scope:minister = {
			add_mil = 3
			remove_trait = trait:tactical_advisor
			add_trait = trait:standing_army_advocate
		}
	}

	option = {
		name = cc_prog.22.b
		scope:minister = {
			add_mil = 2
			remove_trait = trait:tactical_advisor
			add_trait = trait:offensive_strategist
		}
	}

	option = {
		name = cc_prog.22.c
		scope:minister = {
			add_mil = 2
			remove_trait = trait:tactical_advisor
			add_trait = trait:defensive_commander
		}
	}
}
