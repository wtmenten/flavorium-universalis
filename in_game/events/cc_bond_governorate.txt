# cc_bonds.20 — "The Governor's First Annual Report" (one-shot)
# Option B (flexible admin) fires a false negative. Option C sets a memory flag.
cc_bonds.20 = {
	type = country_event
	title = cc_bonds.20.title
	desc = cc_bonds.20.desc
	illustration_tags = { 10 = interior }

	trigger = {
		has_game_rule = cc_subject_bonds_on
		any_subject = {
			is_subject_type = provincial_governorate
			num_locations >= 20
			has_variable = cc_bond_initialized
			NOT = { has_variable = cc_bond_20_cd }
		}
	}

	immediate = {
		random_subject = {
			limit = {
				is_subject_type = provincial_governorate
				num_locations >= 20
				has_variable = cc_bond_initialized
				NOT = { has_variable = cc_bond_20_cd }
			}
			save_scope_as = bond_subj
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_20_cd  value = yes }
		}
	}

	# Demand strict revenue targets — gold income, eco/pol damage
	option = {
		name = cc_bonds.20.a
		add_gold = { value = scope:bond_subj.monthly_income_trade_and_tax  multiply = 0.30  min = 65 }
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = -2 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = -1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
		}
	}

	# Allow flexible administration — eco/pol gain, false negative fires
	option = {
		name = cc_bonds.20.b
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = 2 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = 1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.75  months = 4 }
	}

	# Establish oversight council — pol gain, sets memory flag
	option = {
		name = cc_bonds.20.c
		scope:bond_subj = {
			set_variable = { name = cc_bond_mem_council  value = yes }
			change_variable = { name = cc_bond_pol  add = 1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
		}
	}
}

# cc_bonds.26 — "Corruption Reports" (cooldown 15 yrs)
# Requires strong relationship (pol or eco ≥ 3) — the corruption emerges from
# a comfortable relationship, not a strained one.
# Option A (ignore) looks cost-free — but the ongoing invisible drain is
# abstracted as a hidden eco hit. Option C (send gift) is the same as A except
# the gold cost is visible, making it look worse.
cc_bonds.26 = {
	type = country_event
	title = cc_bonds.26.title
	desc = cc_bonds.26.desc
	illustration_tags = { 10 = interior }

	trigger = {
		has_game_rule = cc_subject_bonds_on
		any_subject = {
			is_subject_type = provincial_governorate
			num_locations >= 20
			has_variable = cc_bond_initialized
			NOT = { has_variable = cc_bond_26_cd }
			OR = {
				var:cc_bond_pol >= 3
				var:cc_bond_eco >= 3
			}
		}
	}

	immediate = {
		random_subject = {
			limit = {
				is_subject_type = provincial_governorate
				num_locations >= 20
				has_variable = cc_bond_initialized
				NOT = { has_variable = cc_bond_26_cd }
				OR = {
					var:cc_bond_pol >= 3
					var:cc_bond_eco >= 3
				}
			}
			save_scope_as = bond_subj
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_26_cd  value = yes  years = 15 }
		}
	}

	# Ignore it — looks free, conceals ongoing eco drain
	option = {
		name = cc_bonds.26.a
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = -1 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
		}
	}

	# Audit and punish — stops drain, pol/per cost
	option = {
		name = cc_bonds.26.b
		scope:bond_subj = {
			change_variable = { name = cc_bond_pol  add = -2 }

			change_variable = { name = cc_bond_per  add = -2 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
	}

	# Send a "gift" — gold cost visible, outcome same as ignoring it
	option = {
		name = cc_bonds.26.c
		add_gold = { value = monthly_income_trade_and_tax  multiply = -0.30  max = -65  min = -1000 }
