
namespace = cc_bonds

###############################################################################
# Bond Engage Events — cc_bonds.300-306
# Fired by the Subject Relations Ledger panel's engage buttons (via the
# scripted_guis in common/scripted_guis/cc_bond_actions.txt). The clicked
# subject is passed as scope:cc_engage_subject. The confirm option pays the
# cost, nudges the relevant bond dimension (clamped -5..+5) on the subject and
# stamps the shared 2-year per-subject cooldown (cc_bond_action_cd); the decline
# option does nothing. The scripted_gui's is_valid has already verified the
# game rule, affordability, the dimension cap and the cooldown.
###############################################################################

# cc_bonds.300 — Underwrite the Local Treasury (Economic +1, -75 gold)
cc_bonds.300 = {
	type = country_event
	title = cc_bonds.300.t
	desc = cc_bonds.300.d
	outcome = positive
	illustration_tags = { 10 = economy }

	option = {
		name = cc_bonds.300.a
		add_gold = { value = scope:cc_engage_subject.cc_bond_gift_cost  multiply = -2 }
		scope:cc_engage_subject = {
			add_gold = { value = scope:cc_engage_subject.cc_bond_gift_cost multiply = 2 }
			change_variable = { name = cc_bond_eco  add = 1 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
			set_variable = { name = cc_bond_action_cd  value = yes  years = 2 }
		}
	}
	option = {
		name = cc_bonds.300.b
	}
}
