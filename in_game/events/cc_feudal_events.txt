###############################################################################
# FEUDAL COURT EVENTS (Age 1 only)
# cc_feudal.1  — Feudal Trait Grant (biyearly, Age 1)
# cc_feudal.10 — The Old Court Gives Way (fires once at Age 2 start)
# cc_feudal.2  — Feast or Fast (nobles demand celebration)
# cc_feudal.3  — The King's Confessor (chaplain advises on war)
# cc_feudal.4  — The Tax Survey (master of coin survey)
# cc_feudal.5  — Tournament of the Court
# cc_feudal.6  — A Learned Foreigner (mini-synergy seed event)
# cc_feudal.7  — The Call to Holy War (faith-gated, one-time)
###############################################################################

# cc_feudal.1 ("A Servant of the Court Finds Their Calling") and cc_feudal.10 ("The Old
# Court Gives Way") were removed when household offices replaced the six office traits.
#
# cc_feudal.1 granted an office trait at random. Appointment is now a paid, tier-gated
# player action (cc_office_appoint_to_household), so a random grant would be the same
# bypass the wider migration exists to close.
#
# cc_feudal.10 converted an office trait into a Renaissance trait when age 2 began. That
# job is now done by each office's own obsoleting advance: the Seneschal is retired by
# chancery_records, the Master of Coin by standardized_coins, and so on. See
# tools/generate_offices.py.

# "Feast or Fast"
# Treasury > 15x monthly income. Nobles demand celebration.
cc_feudal.2 = {
	type = country_event
	title = cc_feudal.2.title
	desc = cc_feudal.2.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
		10 = happy
	}

	trigger = {
		current_age = age_1_traditions
		gold >= {
			value = monthly_income_trade_and_tax
			multiply = 15
		}
		NOT = { has_variable = cc_feudal_feast_fired }
	}

	immediate = {
		set_variable = { name = cc_feudal_feast_fired value = yes }
	}

	option = {
		name = cc_feudal.2.a
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = -0.75
			min = -200
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = 0.08 }
		add_prestige = prestige_mild_bonus
	}

	option = {
		name = cc_feudal.2.b
		add_prestige = prestige_weak_penalty
		add_estate_satisfaction = { type = estate_type:nobles_estate value = -0.05 }
		add_country_modifier = { modifier = cc_cabinet_ministerial_investment years = 5 }
	}
}

# "A Learned Foreigner Seeks Service"
# Scholar from distant lands. Mini-synergy seed: if court_chaplain exists, unlock
# a court_of_learning modifier early (precursor to Renaissance version).
cc_feudal.6 = {
	type = country_event
	title = cc_feudal.6.title
	desc = cc_feudal.6.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
	}

	trigger = {
		current_age = age_1_traditions
		NOT = { has_variable = cc_feudal_foreigner_fired }
	}

	immediate = {
		# Fallback: use root country's own culture and religion
		culture = { save_scope_as = neighbor_culture }
		religion = { save_scope_as = neighbor_religion }
		# Override with a neighbor's if any exist
		if = {
			limit = { any_neighbor_country = { always = yes } }
			random_neighbor_country = {
				culture = { save_scope_as = neighbor_culture }
				religion = { save_scope_as = neighbor_religion }
			}
		}
		create_character = {
			trait_category = cabinet
			religion = scope:neighbor_religion
			culture = scope:neighbor_culture
			age = { 25 45 }
			adm = { 50 90 }
			dip = { 40 80 }
			mil = { 0 40 }
			create_in_limbo = yes
			add_trait = trait:learned_courtier
			save_scope_as = foreigner
		}
		set_variable = { name = cc_feudal_foreigner_fired value = yes }
	}

	# Appoint the foreigner — create a new character with the learned_courtier trait
	option = {
		name = cc_feudal.6.a
		scope:foreigner = {
			move_country = root
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_penalty }
	}

	# If court_chaplain exists — scholarly debate unlocks early learning modifier
	option = {
		name = cc_feudal.6.b
		trigger = {
			any_cabinet_character = {
				in_cabinet = yes
				OR = {
					has_variable = cc_office_held_court_chaplain
					has_variable = cc_office_held_court_astrologer
				}
			}
		}
		scope:foreigner = {
			move_country = root
		}
		add_country_modifier = { modifier = cc_synergy_court_of_learning years = 5 }
		add_prestige = prestige_mild_bonus
	}

	# Decline — prestige for keeping the court pure
	option = {
		name = cc_feudal.6.c
		add_prestige = prestige_weak_bonus
		hidden_effect = { kill_character_silently = scope:foreigner }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = 0.03 }
	}
}


# "The Call to Holy War"
# One-time event. Fires when NOT at war and faith conditions met.
cc_feudal.7 = {
	type = country_event
	title = cc_feudal.7.title
	desc = cc_feudal.7.desc
	outcome = neutral

	illustration_tags = {
		10 = military
		10 = exterior
	}

	trigger = {
		NOT = { at_war = yes }
		NOT = { has_variable = cc_holy_war_called }
		any_cabinet_character = {
			in_cabinet = yes
			OR = {
				has_variable = cc_office_held_court_chaplain
				has_variable = cc_office_held_court_astrologer
				has_trait = sufi_scholar
				has_trait = brahmin_counselor
				has_trait = dharma_minister
				has_trait = ecumenical_patriarch_advisor
			}
		}
		OR = {
			# Catholic crusade path
			AND = {
				religion = religion:catholic
				papacy_active = yes
				current_papal_authority >= { value = 50 }
				any_neighbor_country = {
					NOT = { religion.group = religion_group:christian }
				}
			}
			# Muslim path
			AND = {
				religion.group = religion_group:muslim
				any_neighbor_country = {
					NOT = { religion.group = religion_group:muslim }
				}
			}
			# Generic faithful path
			AND = {
				NOT = { religion = religion:catholic }
				NOT = { religion.group = religion_group:muslim }
				any_neighbor_country = {
					NOT = { religion.group = PREV.religion.group }
				}
			}
		}
	}

	immediate = {
		set_variable = { name = cc_holy_war_called value = yes }
		# Save a target: the nearest non-coreligionist neighbor
		if = {
			limit = { religion = religion:catholic }
			random_neighbor_country = {
				limit = { NOT = { religion.group = religion_group:christian } }
				save_scope_as = holy_war_target
			}
		}
		else_if = {
			limit = { religion.group = religion_group:muslim }
			random_neighbor_country = {
				limit = { NOT = { religion.group = religion_group:muslim } }
				save_scope_as = holy_war_target
			}
		}
		else = {
			random_neighbor_country = {
				limit = { NOT = { religion.group = PREV.religion.group } }
				save_scope_as = holy_war_target
			}
		}
	}

	# Catholics: call the actual crusade (single-nation crusade IO)
	option = {
		name = cc_feudal.7.a
		trigger = { religion = religion:catholic }
		crusade_effect = { target = scope:holy_war_target }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = 0.10 }
		add_prestige = prestige_mild_bonus
	}

	# Muslims: preparation modifier (cb_jihad is hardcoded, not player-triggerable)
	option = {
		name = cc_feudal.7.a
		trigger = { religion.group = religion_group:muslim }
		add_country_modifier = { modifier = cc_jihad_preparation years = 5 }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = 0.10 }
		add_prestige = prestige_mild_bonus
	}

	# All faiths: generic spiritual mandate preparation
	option = {
		name = cc_feudal.7.a
		trigger = {
			NOT = { religion = religion:catholic }
			NOT = { religion.group = religion_group:muslim }
		}
		add_country_modifier = { modifier = cc_spiritual_mandate years = 8 }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = 0.08 }
		add_prestige = prestige_mild_bonus
	}

	# Secular realpolitik — decline the call
	option = {
		name = cc_feudal.7.b
		add_prestige = prestige_mild_bonus
		add_country_modifier = { modifier = cc_synergy_court_of_learning years = 3 }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = -0.08 }
		remove_variable = cc_holy_war_called
	}
}
