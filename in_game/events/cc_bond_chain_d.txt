###############################################################################
# Chain D: "The Scholar's Name" — cc_bonds.130–132
# A single mind shapes how the subject sees itself and its overlord.
# Spans 20–45 years. Applies to all subject types with cc_bond_initialized.
# Root = overlord. scope:bond_subj = the subject.
###############################################################################

# cc_bonds.130 — "A Brilliant Student" (fires 10–15 yrs in, one-shot)
cc_bonds.130 = {
	type = country_event
	title = cc_bonds.130.title
	desc = cc_bonds.130.desc
	illustration_tags = { 10 = interior }

	trigger = {
		has_game_rule = cc_subject_bonds_on
		NOT = { has_variable = cc_bond_130_cd_overlord }
		any_subject = {
			OR = {
				AND = {
					capital.development > 10
					num_locations >= 20
				}
				AND = {
					capital.development > 15
					num_locations >= 10
				}
				capital.development > 25
			}
			has_variable = cc_bond_initialized
			NOT = { has_variable = cc_bond_130_cd }
		}
	}

	immediate = {
		save_scope_as = overlord
		set_variable = { name = cc_bond_130_cd_overlord  value = yes years = { 30 80 } }
		random_subject = {
			limit = {
				OR = {
					AND = {
						capital.development > 10
						num_locations >= 20
					}
					AND = {
						capital.development > 15
						num_locations >= 10
					}
					capital.development > 25
				}
				has_variable = cc_bond_initialized
				NOT = { has_variable = cc_bond_130_cd }
			}
			save_scope_as = bond_subj
			culture = {
				save_scope_as = bond_subj_culture
			}
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_130_cd  value = yes }
		}
		create_character = {
			create_in_limbo = yes
			culture = scope:bond_subj.culture
			religion = scope:bond_subj.religion
			add_trait = trait:learned_counselor
			artist = writer
			adm = { integer_range = { min = 45  max = 75 } }
			dip = { integer_range = { min = 40  max = 65 } }
			mil = { integer_range = { min = 10  max = 35 } }
			save_scope_as = target_character
		}
		# Store reference so cc_bonds.131 and 132 can retrieve the scholar
		scope:bond_subj = {
			set_variable = { name = cc_bond_scholar_char  value = scope:target_character }
		}

	}

	# Bring to overlord's court — cul-1, per+2
	option = {
		name = cc_bonds.130.a
		add_gold = { value = monthly_income_trade_and_tax  multiply = -4 min = -100 max = -50 }
		scope:bond_subj = {
			add_gold = { value = ROOT.monthly_income_trade_and_tax  multiply = 4 min = 50 max = 100 }
			set_variable = { name = cc_bond_mem_scholar_at_court  value = yes }
			change_variable = { name = cc_bond_cul  add = -1 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = 2 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		scope:target_character = {
			move_country = ROOT
		}
		ROOT = { capital ?= { create_art = { type = work_of_art_type:chronicle  artist = scope:target_character  quality = { 55 75 }  key = cc_scholar_early_notes } } }
		trigger_event_silently = { id = cc_bonds.131  years = 10 }
	}

	# Fund education at home, let publish freely — cul+2, pol+1
	option = {
		name = cc_bonds.130.b
		add_gold = { value = monthly_income_trade_and_tax  multiply = -2 min = -75 max = -25 }
		scope:bond_subj = {
			add_gold = { value = ROOT.monthly_income_trade_and_tax  multiply = 2 min = 25 max = 75 }
			set_variable = { name = cc_bond_mem_scholar_at_home  value = yes }
			change_variable = { name = cc_bond_cul  add = 2 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = 1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			# Overlord's institutional culture already permeating — scholar joins shared knowledge economy
			if = {
				limit = { has_variable = cc_bond_insti_to_subject }
				change_variable = { name = cc_bond_eco  add = 1 }
				clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
				add_opinion = { target = ROOT modifier = cc_intl_scholarly_bond scale = 1.5 }
			}
			else = {
				add_opinion = { target = ROOT modifier = cc_intl_scholarly_bond }
			}
			capital ?= { create_art = { type = work_of_art_type:chronicle  artist = scope:target_character  quality = { 60 80 }  key = cc_scholar_early_notes } }
		}
		scope:target_character = {
			move_country = scope:bond_subj
		}
		add_prestige = prestige_weak_bonus
		trigger_event_silently = { id = cc_bonds.131  years = 10 }
	}

	# Offer employment as crown propagandist — cul-2, per+1
	option = {
		name = cc_bonds.130.c
		add_gold = { value = monthly_income_trade_and_tax  multiply = -2 min = -150 max = -25 }
		scope:bond_subj = {
			add_gold = { value = ROOT.monthly_income_trade_and_tax  multiply = 2 min = 25 max = 150 }
			set_variable = { name = cc_bond_mem_scholar_propagandist  value = yes }
			if = { # Overlord's institutional culture already permeating — scholars departure is prestigious
				limit = { has_variable = cc_bond_insti_to_subject }
				change_variable = { name = cc_bond_cul  add = 1 }
			}
			change_variable = { name = cc_bond_cul  add = -2 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = 1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		ROOT = { capital ?= { create_art = { type = work_of_art_type:chronicle  artist = scope:target_character  quality = { 45 65 }  key = cc_scholar_early_notes } } }
		trigger_event_silently = { id = cc_bonds.131  years = 8 }
		capital.continent = {
			random_location_in_continent = {
				limit = {
					has_owner = yes
					owner = ROOT
				}
				province = {
					every_location_in_province = {
						add_location_modifier = {
							modifier = cc_cabinet_provincial_scholar
							years = 5
							mode = set_to_largest_and_extend
						}
					}
				}
			}
		}
	}

	# Ignore — no bond change; scholar found own way
	option = {
		name = cc_bonds.130.d
		trigger_event_silently = { id = cc_bonds.131  years = 15 }
	}
}

# cc_bonds.131 — "The Scholar's First Work" (fires 8–15 years after cc_bonds.130)
cc_bonds.131 = {
	type = country_event
	title = cc_bonds.131.title

	trigger = {
		exists = scope:bond_subj
		scope:bond_subj = {
			has_variable = cc_bond_initialized
			exists = var:cc_bond_scholar_char
		}
	}

	immediate = {
		# Retrieve scholar from stored reference; fall back to subject ruler as artistic proxy
		scope:bond_subj = { var:cc_bond_scholar_char ?= { save_scope_as = scholar_char } }
		if = {
			limit = { NOT = { exists = scope:scholar_char } }
			scope:bond_subj = { ruler ?= { save_scope_as = scholar_char } }
		}
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:bond_subj = { has_variable = cc_bond_mem_scholar_at_court } }
				desc = cc_bonds.131.desc_court
			}
			triggered_desc = {
				trigger = { scope:bond_subj = { has_variable = cc_bond_mem_scholar_at_home } }
				desc = cc_bonds.131.desc_home
			}
			triggered_desc = {
				trigger = { scope:bond_subj = { has_variable = cc_bond_mem_scholar_propagandist } }
				desc = cc_bonds.131.desc_prop
			}
			desc = cc_bonds.131.desc_free
		}
	}
	illustration_tags = { 10 = interior }

	# At court: work glorifies the empire
	option = {
		name = cc_bonds.131.a_use
		ai_chance = { base = 40 }
		trigger = {
			scope:bond_subj = { has_variable = cc_bond_mem_scholar_at_court }
		}
		scope:bond_subj = {
			change_variable = { name = cc_bond_cul  add = 1 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = 1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = -1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		ROOT = { capital ?= { create_art = { type = work_of_art_type:chronicle  artist = scope:scholar_char  quality = { 65 80 }  key = cc_scholar_imperial_chronicle } } }
		trigger_event_silently = { id = cc_bonds.132  years = 25 }
	}

	option = {
		name = cc_bonds.131.a_standalone
		ai_chance = { base = 60 }
		trigger = {
			scope:bond_subj = { has_variable = cc_bond_mem_scholar_at_court }
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_mem_scholar_legacy_aligned  value = yes }
			change_variable = { name = cc_bond_cul  add = 1 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = 2 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		ROOT = { capital ?= { create_art = { type = work_of_art_type:chronicle  artist = scope:scholar_char  quality = { 75 90 }  key = cc_scholar_imperial_chronicle } } }
		trigger_event_silently = { id = cc_bonds.132  years = 25 }
	}

	# At home: work celebrates subject's cultural heritage — cul+3, pol-1
	option = {
		name = cc_bonds.131.b_sponsor
		ai_chance = { base = 30 }
		trigger = {
			scope:bond_subj = { has_variable = cc_bond_mem_scholar_at_home }
		}
		add_gold = { value = monthly_income_trade_and_tax  multiply = -0.30  min = -65 }
		scope:bond_subj = {
			set_variable = { name = cc_bond_mem_scholar_legacy_contested  value = yes }
			change_variable = { name = cc_bond_cul  add = 3 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = 1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_eco  add = -1 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
			capital ?= { create_art = { type = work_of_art_type:chronicle  artist = scope:scholar_char  quality = { 70 85 }  key = cc_scholar_homeland_chronicle } }
		}
		trigger_event_silently = { id = cc_bonds.132  years = 22 }
	}

	option = {
		name = cc_bonds.131.b_allow
		ai_chance = { base = 50 }
		trigger = {
			scope:bond_subj = { has_variable = cc_bond_mem_scholar_at_home }
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_mem_scholar_legacy_contested  value = yes }
			change_variable = { name = cc_bond_cul  add = 3 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = -1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			capital ?= { create_art = { type = work_of_art_type:chronicle  artist = scope:scholar_char  quality = { 65 80 }  key = cc_scholar_homeland_chronicle } }
		}
		trigger_event_silently = { id = cc_bonds.132  years = 22 }
	}

	option = {
		name = cc_bonds.131.b_suppress
		ai_chance = { base = 15 }
		trigger = {
			scope:bond_subj = { has_variable = cc_bond_mem_scholar_at_home }
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_mem_scholar_suppressed  value = yes }
			change_variable = { name = cc_bond_cul  add = -2 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = -2 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = -2 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.132  years = 28 }
	}

	# Propagandist: work is transparently political — cul-2, pol-1
	option = {
		name = cc_bonds.131.c_more
		ai_chance = { base = 30 }
		trigger = {
			scope:bond_subj = { has_variable = cc_bond_mem_scholar_propagandist }
		}
		scope:bond_subj = {
			change_variable = { name = cc_bond_cul  add = -2 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = -1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
		}
		ROOT = { capital ?= { create_art = { type = work_of_art_type:chronicle  artist = scope:scholar_char  quality = { 40 60 }  key = cc_scholar_propaganda_work } } }
		trigger_event_silently = { id = cc_bonds.132  years = 20 }
	}

	option = {
		name = cc_bonds.131.c_distance
		ai_chance = { base = 70 }
		trigger = {
			scope:bond_subj = { has_variable = cc_bond_mem_scholar_propagandist }
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_mem_scholar_legacy_counter  value = yes }
			change_variable = { name = cc_bond_cul  add = -2 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = -1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = 1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.132  years = 20 }
	}

	# Free scholar: ambivalent tone — cul+1 only
	option = {
		name = cc_bonds.131.d_note
		ai_chance = { base = 100 }
		trigger = {
			scope:bond_subj = {
				NOT = { has_variable = cc_bond_mem_scholar_at_court }
				NOT = { has_variable = cc_bond_mem_scholar_at_home }
				NOT = { has_variable = cc_bond_mem_scholar_propagandist }
			}
		}
		scope:bond_subj = {
			change_variable = { name = cc_bond_cul  add = 1 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.132  years = 25 }
	}
}

# cc_bonds.132 — "The Scholar's Legacy" (terminal — fires 20–28 years after cc_bonds.131)
cc_bonds.132 = {
	type = country_event
	title = cc_bonds.132.title

	trigger = {
		exists = scope:bond_subj
		scope:bond_subj = { has_variable = cc_bond_initialized }
	}

	immediate = {
		# Retrieve scholar; fall back to overlord ruler as patron proxy
		scope:bond_subj = { var:cc_bond_scholar_char ?= { save_scope_as = scholar_char } }
		if = {
			limit = { NOT = { exists = scope:scholar_char } }
			ROOT = { ruler ?= { save_scope_as = scholar_char } }
		}
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:bond_subj = { has_variable = cc_bond_mem_scholar_legacy_aligned } }
				desc = cc_bonds.132.desc_aligned
			}
			triggered_desc = {
				trigger = { scope:bond_subj = { has_variable = cc_bond_mem_scholar_legacy_contested } }
				desc = cc_bonds.132.desc_contested
			}
			triggered_desc = {
				trigger = { scope:bond_subj = { has_variable = cc_bond_mem_scholar_suppressed } }
				desc = cc_bonds.132.desc_suppressed
			}
			triggered_desc = {
				trigger = { scope:bond_subj = { has_variable = cc_bond_mem_scholar_legacy_counter } }
				desc = cc_bonds.132.desc_counter
			}
			desc = cc_bonds.132.desc_default
		}
	}
	illustration_tags = { 10 = interior }

	# Aligned — scholar's ideas quoted by loyalists
	option = {
		name = cc_bonds.132.a
		ai_chance = { base = 100 }
		trigger = {
			scope:bond_subj = { has_variable = cc_bond_mem_scholar_legacy_aligned }
		}
		scope:bond_subj = {
			change_variable = { name = cc_bond_per  add = 2 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
			change_variable = { name = cc_bond_cul  add = 2 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
		}
		ROOT = { capital ?= { create_art = { type = work_of_art_type:chronicle  artist = scope:scholar_char  quality = { 80 95 }  key = cc_scholar_legacy_work } } }
	}

	# Contested — school of thought emphasizing subject's distinct identity
	option = {
		name = cc_bonds.132.b_embrace
		ai_chance = { base = 65 }
		trigger = {
			scope:bond_subj = { has_variable = cc_bond_mem_scholar_legacy_contested }
		}
		scope:bond_subj = {
			change_variable = { name = cc_bond_cul  add = 3 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			capital ?= { create_art = { type = work_of_art_type:chronicle  artist = scope:scholar_char  quality = { 65 80 }  key = cc_scholar_homeland_chronicle } }
		}
	}

	option = {
		name = cc_bonds.132.b_neutral
		ai_chance = { base = 35 }
		trigger = {
			scope:bond_subj = { has_variable = cc_bond_mem_scholar_legacy_contested }
		}
		scope:bond_subj = {
			change_variable = { name = cc_bond_cul  add = 1 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
		}
	}

	# Suppressed — underground copies circulate; ideas became exciting because forbidden
	option = {
		name = cc_bonds.132.c
		ai_chance = { base = 100 }
		trigger = {
			scope:bond_subj = { has_variable = cc_bond_mem_scholar_suppressed }
		}
		scope:bond_subj = {
			change_variable = { name = cc_bond_cul  add = -3 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = -2 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			capital ?= { create_art = { type = work_of_art_type:chronicle  artist = scope:scholar_char  quality = { 55 75 }  key = cc_scholar_forbidden_work } }
		}
	}

	# Counter-narrative: scholar distanced themselves from propaganda — work gained authenticity
	# The overlord took political distance in cc_bonds.131.c_distance; now the legacy is assessed.
	option = {
		name = cc_bonds.132.e
		ai_chance = { base = 100 }
		trigger = {
			scope:bond_subj = { has_variable = cc_bond_mem_scholar_legacy_counter }
		}
		scope:bond_subj = {
			remove_variable = cc_bond_mem_scholar_legacy_counter
			change_variable = { name = cc_bond_cul  add = 1 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = 1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
			capital ?= { create_art = { type = work_of_art_type:chronicle  artist = scope:scholar_char  quality = { 65 80 }  key = cc_scholar_legacy_work } }
		}
	}

	# Default / propagandist path
	option = {
		name = cc_bonds.132.d
		ai_chance = { base = 100 }
		trigger = {
			scope:bond_subj = {
				NOT = { has_variable = cc_bond_mem_scholar_legacy_aligned }
				NOT = { has_variable = cc_bond_mem_scholar_legacy_contested }
				NOT = { has_variable = cc_bond_mem_scholar_suppressed }
				NOT = { has_variable = cc_bond_mem_scholar_legacy_counter }
			}
		}
	}
}
