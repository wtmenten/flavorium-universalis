# cc_bonds.40 — "The Puppet Requests an Ambassador" (one-shot)
# Option C (refuse) fires a false positive.
cc_bonds.40 = {
	type = country_event
	title = cc_bonds.40.title
	desc = cc_bonds.40.desc
	illustration_tags = {  }

	trigger = {
		has_game_rule = cc_subject_bonds_on
		any_subject = {
			is_subject_type = puppet_state
			num_provinces >= 4
			has_variable = cc_bond_initialized
			NOT = { has_variable = cc_bond_40_cd }
		}
	}

	immediate = {
		random_subject = {
			limit = {
				is_subject_type = puppet_state
				num_provinces >= 4
				has_variable = cc_bond_initialized
				NOT = { has_variable = cc_bond_40_cd }
			}
			save_scope_as = bond_subj
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_40_cd  value = yes }
		}
	}

	# Full diplomatic mission — strong pol/per gain
	option = {
		name = cc_bonds.40.a
		scope:bond_subj = {
			change_variable = { name = cc_bond_pol  add = 2 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = 2 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
	}

	# Limited representative — moderate pol/per gain
	option = {
		name = cc_bonds.40.b
		scope:bond_subj = {
			change_variable = { name = cc_bond_pol  add = 1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = 1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
	}

	# Refuse — pol/per damage, false positive fires
	option = {
		name = cc_bonds.40.c
		scope:bond_subj = {
			change_variable = { name = cc_bond_pol  add = -2 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = -3 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.83  months = 4 }
	}
}
