# cc_bonds.50 — "The Royal Visit" (one-shot)
# Option C (send letter only) fires a false positive — dependency "sends warm reply."
cc_bonds.50 = {
	type = country_event
	title = cc_bonds.50.title
	desc = cc_bonds.50.desc
	illustration_tags = { 10 = exterior }

	trigger = {
		has_game_rule = cc_subject_bonds_on
		any_subject = {
			is_subject_type = crown_dependency
			num_locations >= 10
			has_variable = cc_bond_initialized
			NOT = { has_variable = cc_bond_50_cd }
		}
	}

	immediate = {
		random_subject = {
			limit = {
				is_subject_type = crown_dependency
				num_locations >= 10
				has_variable = cc_bond_initialized
				NOT = { has_variable = cc_bond_50_cd }
			}
			save_scope_as = bond_subj
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_50_cd  value = yes }
		}
	}

	# Visit in person — prestige, pol/cul/per gain
	option = {
		name = cc_bonds.50.a
		add_prestige = prestige_weak_bonus
		scope:bond_subj = {
			change_variable = { name = cc_bond_pol  add = 1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_cul  add = 1 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = 4 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
	}

	# Send royal envoy — per gain
	option = {
		name = cc_bonds.50.b
		scope:bond_subj = {
			change_variable = { name = cc_bond_per  add = 2 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
	}

	# Send a formal letter — per damage, false positive fires
	option = {
		name = cc_bonds.50.c
		scope:bond_subj = {
			change_variable = { name = cc_bond_per  add = -1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.86  months = 4 }
	}
}
