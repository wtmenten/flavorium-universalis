# AoR Payoff Events — Imperial Federalization (cc_bonds.140–145)
#
# Fired by cc_bonds_aor_pulse with scope:bond_subj selected and
# var:cc_bond_aor_tier already graded. Root = overlord. See cc_bond_aor.txt for
# the shared contract and cc_bond_aor_effects.txt for the grading, memory and
# revolution effects.
#
# Events 140–143: fire on EXISTING federal subject types.
# Events 144–145: fire on NON-federal subjects, opening the path to federation.
###############################################################################

# cc_bonds.140 — The Federal Member Stands with the Union
cc_bonds.140 = {
	type = country_event
	title = cc_bonds.140.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:cc_bond_aor_tier >= 3 }
				desc = cc_bonds.140.desc.deep
			}
			desc = cc_bonds.140.desc
		}
	}
	illustration_tags = { 10 = happy }

	trigger = { exists = scope:bond_subj }

	# Celebrate the union publicly.
	option = {
		name = cc_bonds.140.a
		cc_bonds_aor_ld_relief = yes
		cc_bonds_aor_tiered_modifier = { MOD = cc_federal_union_solidarity }
		cc_bonds_aor_suppress_revolution = yes
		cc_bonds_aor_prestige_gain = yes
	}

	# Formalize it. The federal parliament, not the crown, becomes the thing
	# holding the union together.
	option = {
		name = cc_bonds.140.b
		trigger = { var:cc_bond_aor_tier >= 2 }
		cc_bonds_aor_ld_relief = yes
		cc_bonds_aor_tiered_modifier = { MOD = cc_federal_union_solidarity }
		cc_bonds_aor_suppress_revolution = yes
		add_country_modifier = {
			modifier = cc_synergy_parliamentary_harmony
			years = 15
		}
		cc_bonds_aor_estate_nudge = { ESTATE = nobles_estate  VALUE = estate_satisfaction_mild_penalty }
	}
}

# cc_bonds.141 — The Federal Member Demands Equal Rights
cc_bonds.141 = {
	type = country_event
	title = cc_bonds.141.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:bond_subj = { has_variable = cc_bond_mem_scholar_suppressed } }
				desc = cc_bonds.141.desc.scholar
			}
			triggered_desc = {
				trigger = { var:cc_bond_aor_tier >= 3 }
				desc = cc_bonds.141.desc.deep
			}
			desc = cc_bonds.141.desc
		}
	}
	illustration_tags = { 10 = angry }

	trigger = { exists = scope:bond_subj }

	# Grant full representation. Costly and genuine.
	option = {
		name = cc_bonds.141.a
		cc_bonds_aor_pay_heavy = yes
		cc_bonds_aor_ld_relief = yes
		cc_bonds_aor_tiered_modifier_subject = { MOD = cc_aor_settled_grievance }
		add_country_modifier = {
			modifier = cc_aor_settled_grievance
			years = 15
		}
	}

	# Deny the demand. The union holds by force, and the argument spreads.
	option = {
		name = cc_bonds.141.b
		add_prestige = prestige_weak_bonus
		cc_bonds_aor_ld_spike = yes
		cc_bonds_aor_tiered_modifier_subject = { MOD = cc_revolutionary_sympathies }
		cc_bonds_aor_bolster_revolution = yes
	}

	# Dissolve the union rather than fight over it.
	option = {
		name = cc_bonds.141.c
		trigger = {
			var:cc_bond_aor_tier >= 3
			scope:bond_subj = { is_subject_of = root }
		}
		cancel_subject = scope:bond_subj
		add_prestige = prestige_mild_penalty
		add_country_modifier = {
			modifier = cc_aor_orderly_separation
			years = 20
		}
	}
}

# cc_bonds.142 — The Lesser Member Submits Its Petition
cc_bonds.142 = {
	type = country_event
	title = cc_bonds.142.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:cc_bond_aor_tier >= 3 }
				desc = cc_bonds.142.desc.deep
			}
			desc = cc_bonds.142.desc
		}
	}
	illustration_tags = { 10 = interior }

	trigger = { exists = scope:bond_subj }

	# Elevate them to a full member.
	option = {
		name = cc_bonds.142.a
		trigger = {
			has_advance = federal_constitution_advance
			var:cc_bond_aor_tier >= 2
			scope:bond_subj = { is_subject_of = root }
		}
		scope:bond_subj = {
			change_subject_type = subject_type:equal_federal_member
		}
		cc_bonds_aor_ld_relief = yes
		cc_bonds_aor_suppress_revolution = yes
		cc_bonds_aor_prestige_gain = yes
	}

	# Acknowledge the petition without acting on it.
	option = {
		name = cc_bonds.142.b
		cc_bonds_aor_pay_light = yes
		cc_bonds_aor_ld_relief = yes
	}

	# Dismiss it.
	option = {
		name = cc_bonds.142.c
		add_prestige = prestige_weak_bonus
		cc_bonds_aor_ld_spike = yes
		if = {
			limit = { scope:bond_subj = { has_variable = cc_bond_initialized } }
			scope:bond_subj = {
				change_variable = { name = cc_bond_pol  add = -2 }
				clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
				change_variable = { name = cc_bond_per  add = -1 }
				clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
			}
		}
	}
}

# cc_bonds.144 — The Colony Asserts Its Place
# Dispatched at a higher bar than cc_bonds.90 (both dimensions at 3 or better),
# and after it, so a colony that merely stayed quiet gets the loyalty event and
# only one that earned a seat gets this.
cc_bonds.144 = {
	type = country_event
	title = cc_bonds.144.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:bond_subj = { has_variable = cc_bond_mem_voice_granted } }
				desc = cc_bonds.144.desc.voice
			}
			triggered_desc = {
				trigger = { var:cc_bond_aor_tier >= 3 }
				desc = cc_bonds.144.desc.deep
			}
			desc = cc_bonds.144.desc
		}
	}
	illustration_tags = { 10 = interior }

	trigger = { exists = scope:bond_subj }

	# Grant full federal membership.
	option = {
		name = cc_bonds.144.a
		trigger = {
			has_advance = federal_constitution_advance
			var:cc_bond_aor_tier >= 2
			scope:bond_subj = { is_subject_of = root }
		}
		scope:bond_subj = {
			change_subject_type = subject_type:equal_federal_member
		}
		cc_bonds_aor_ld_relief = yes
		cc_bonds_aor_tiered_modifier = { MOD = cc_federal_union_solidarity }
		cc_bonds_aor_suppress_revolution = yes
		cc_bonds_aor_prestige_gain = yes
	}

	# Offer lesser membership instead.
	option = {
		name = cc_bonds.144.b
		trigger = {
			has_advance = federal_constitution_advance
			scope:bond_subj = { is_subject_of = root }
		}
		scope:bond_subj = {
			change_subject_type = subject_type:lesser_federal_member
		}
		add_prestige = prestige_weak_bonus
		cc_bonds_aor_suppress_revolution = yes
	}

	# A council seat as a stepping stone.
	option = {
		name = cc_bonds.144.c
		trigger = {
			has_advance = federal_constitution_advance
			scope:bond_subj = { is_subject_of = root }
		}
		scope:bond_subj = {
			change_subject_type = subject_type:imperial_council_member
		}
		cc_bonds_aor_suppress_revolution = yes
	}

	# Decline. They asked at the one moment refusing them was expensive.
	option = {
		name = cc_bonds.144.d
		cc_bonds_aor_ld_spike = yes
		if = {
			limit = { scope:bond_subj = { has_variable = cc_bond_initialized } }
			scope:bond_subj = {
				change_variable = { name = cc_bond_per  add = -2 }
				clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
			}
		}
	}
}
