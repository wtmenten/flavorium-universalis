###############################################################################
# AoR Payoff Events — Imperial Federalization (cc_bonds.140–145)
# Fire during age_6_revolutions via yearly_country_pulse.
# Root = overlord. Reads bond scores from variable maps on overlord.
# One-shot per subject via cc_bond_aor_14X_fired flag on subject.
#
# Events 140–143: fire on EXISTING federal subject types.
# Events 144–145: fire on NON-federal subjects, opening the path to federation.
###############################################################################

# cc_bonds.140 — The Federal Member Stands with the Union
# Triggers: equal_federal_member with pol >= 3 AND per >= 2
cc_bonds.140 = {
	type = country_event
	title = cc_bonds.140.title
	desc = cc_bonds.140.desc
	illustration_tags = { 10 = happy }

	trigger = {
		has_game_rule = cc_subject_bonds_on
		current_age = age_6_revolutions
		any_key_in_variable_map = {
			variable = cc_subject_registry_map
			OR = {
				AND = {
					is_subject_of = root
					is_subject_type = equal_federal_member
				}
				AND = {
					NOT = { is_subject_of = root }
					has_variable = cc_bond_pol_frozen
				}
			}
			"variable_map(cc_bond_pol_map|this)" >= 3
			"variable_map(cc_bond_per_map|this)" >= 2
			NOT = { has_variable = cc_bond_aor_140_fired }
		}
	}

	immediate = {
		random_key_in_variable_map = {
			variable = cc_subject_registry_map
			limit = {
				OR = {
					AND = {
						is_subject_of = root
						is_subject_type = equal_federal_member
					}
					AND = {
						NOT = { is_subject_of = root }
						has_variable = cc_bond_pol_frozen
					}
				}
				"variable_map(cc_bond_pol_map|this)" >= 3
				"variable_map(cc_bond_per_map|this)" >= 2
				NOT = { has_variable = cc_bond_aor_140_fired }
			}
			save_scope_as = bond_subj
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_aor_140_fired  value = yes }
		}
		if = {
			limit = { scope:bond_subj = { is_subject_of = root } }
			scope:bond_subj = {
				add_liberty_desire = liberty_desire_severe_minus
			}
		}
		add_country_modifier = {
			modifier = cc_federal_union_solidarity
			years = 20
		}
	}

	# Celebrate the union publicly
	option = {
		name = cc_bonds.140.a
		add_prestige = prestige_mild_bonus
	}

	# Formalize the bond — the federal parliament strengthens
	option = {
		name = cc_bonds.140.b
		if = {
			limit = { scope:bond_subj = { has_variable = cc_bond_initialized } }
			scope:bond_subj = {
				change_variable = { name = cc_bond_pol  add = 1 }
				clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
				change_variable = { name = cc_bond_per  add = 1 }
				clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
			}
		}
		add_country_modifier = {
			modifier = cc_synergy_parliamentary_harmony
			years = 10
		}
	}
}
