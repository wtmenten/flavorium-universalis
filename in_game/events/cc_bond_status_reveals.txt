###############################################################################
# First-Subject Major Events — cc_bonds.2–4
# Each fires exactly once per game (gated by global variables) the first time a
# broad subject category appears anywhere in the world.
# major = yes / major_trigger = { is_human = yes } sends a visible notification
# to all human players.
###############################################################################

# cc_bonds.2 — "A Colonial Horizon" (first colonial nation in the game)
cc_bonds.2 = {
	type = country_event
	title = cc_bonds.2.title
	desc = cc_bonds.2.desc
	outcome = positive
	major = yes
	major_trigger = { is_human = yes }

	illustration_tags = { 10 = exterior }

	immediate = {
		var:cc_first_subject_ref ?= { save_scope_as = bond_subj }
		remove_variable = cc_first_subject_ref
	}

	option = {
		name = cc_bonds.2.a
		add_prestige = prestige_weak_bonus
	}

	option = {
		name = cc_bonds.2.b
	}
}

###############################################################################
# Status Reveal Events — cc_bonds.5–9
# Rare player-visible windows into relationship health.
# Each has its own 15-year cooldown. Desc varies by current bond scores.
# Small option effects let the player nudge the relationship even here.
###############################################################################

# cc_bonds.5 — "Word from the Provinces"
# Generic reveal. Fires for any tracked subject. Flavor varies by eco/per state.
cc_bonds.5 = {
	type = country_event
	title = cc_bonds.5.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					any_subject = {
						has_variable = cc_bond_initialized
						var:cc_bond_eco >= 3
						var:cc_bond_per >= 3
					}
				}
				desc = cc_bonds.5.desc.thriving
			}
			triggered_desc = {
				trigger = {
					any_subject = {
						has_variable = cc_bond_initialized
						OR = {
							var:cc_bond_eco <= -3
							var:cc_bond_per <= -3
						}
					}
				}
				desc = cc_bonds.5.desc.strained
			}
			desc = cc_bonds.5.desc.neutral
		}
	}

	trigger = {
		has_game_rule = cc_subject_bonds_on
		NOT = { has_variable = cc_bond_5_cd }
		any_subject = { has_variable = cc_bond_initialized }
	}

	immediate = {
		set_variable = { name = cc_bond_5_cd  value = yes  years = 15 }
		random_subject = {
			limit = { has_variable = cc_bond_initialized }
			save_scope_as = bond_subj
		}
	}

	# Respond with interest — small personal bond gain
	option = {
		name = cc_bonds.5.a
		scope:bond_subj = {
			change_variable = { name = cc_bond_per  add = 1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		add_gold = { value = monthly_income_trade_and_tax multiply = -2 min = -100 max = -25 }
		add_prestige = prestige_weak_bonus
	}
	# File it away — no effect
	option = {
		name = cc_bonds.5.b
	}
}

# cc_bonds.6 — "The Diplomat's Report"
# Colonial nation bond reveal. Flavor based on economic dimension.
# desc triggers use scope:bond_subj (set in immediate) to avoid mismatches
# when multiple colonial nations exist.
cc_bonds.6 = {
	type = country_event
	title = cc_bonds.6.title

	trigger = {
		has_game_rule = cc_subject_bonds_on
		NOT = { has_variable = cc_bond_6_cd }
		any_subject = {
			is_subject_type = colonial_nation
			has_variable = cc_bond_initialized
		}
	}

	immediate = {
		set_variable = { name = cc_bond_6_cd  value = yes  years = 15 }
		random_subject = {
			limit = {
				is_subject_type = colonial_nation
				has_variable = cc_bond_initialized
			}
			save_scope_as = bond_subj
		}
	}
	
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:bond_subj = { var:cc_bond_eco >= 2 } }
				desc = cc_bonds.6.desc.content
			}
			triggered_desc = {
				trigger = { scope:bond_subj = { var:cc_bond_eco <= -2 } }
				desc = cc_bonds.6.desc.discontent
			}
			desc = cc_bonds.6.desc.neutral
		}
	}

	option = {
		name = cc_bonds.6.a
		add_gold = { value = monthly_income_trade_and_tax  multiply = -2  min = -100  max = -20 }
		add_prestige = prestige_weak_bonus
		scope:bond_subj = {
			change_variable = { name = cc_bond_per  add = 1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
	}
	option = {
		name = cc_bonds.6.b
	}
}
