
namespace = cc_bonds

###############################################################################
# Status Reveal Events — cc_bonds.5–9
# Rare player-visible windows into relationship health.
# Each has its own 15-year cooldown. Desc varies by current bond scores.
# Small option effects let the player nudge the relationship even here.
###############################################################################

# cc_bonds.5 — "Word from the Provinces"
# Generic reveal. Fires for any tracked subject. Flavor varies by eco/per state.
cc_bonds.5 = {
	type = country_event
	title = cc_bonds.5.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					any_subject = {
						has_variable = cc_bond_initialized
						var:cc_bond_eco >= 3
						var:cc_bond_per >= 3
					}
				}
				desc = cc_bonds.5.desc.thriving
			}
			triggered_desc = {
				trigger = {
					any_subject = {
						has_variable = cc_bond_initialized
						OR = {
							var:cc_bond_eco <= -3
							var:cc_bond_per <= -3
						}
					}
				}
				desc = cc_bonds.5.desc.strained
			}
			desc = cc_bonds.5.desc.neutral
		}
	}

	trigger = {
		has_game_rule = cc_subject_bonds_on
		NOT = { has_variable = cc_bond_5_cd }
		any_subject = { has_variable = cc_bond_initialized }
	}

	immediate = {
		set_variable = { name = cc_bond_5_cd  value = yes  years = 15 }
		random_subject = {
			limit = { has_variable = cc_bond_initialized }
			save_scope_as = bond_subj
		}
	}

	# Respond with interest — small personal bond gain
	option = {
		name = cc_bonds.5.a
		scope:bond_subj = {
			change_variable = { name = cc_bond_per  add = 1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
	}
	# File it away — no effect
	option = {
		name = cc_bonds.5.b
	}
}
