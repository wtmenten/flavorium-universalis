# ============================================================
# FLAVORIUM UNIVERSALIS — CONDITIONAL TRAIT SPAWNING EVENTS
# These events grant traits based on country-level conditions.

# ============================================================
# cc_cond.1 — Capital Development Traits
# Tier 1: dev > 30 (any age)    → urban_planner / prosperity_herald / merchant_syndic
# Tier 2: dev > 40 (age 3+)     → grand_architect / arts_patron / learned_courtier / reform_minded_counselor
# Tier 3: dev > 50 (age 5+)     → master_urbanist / enlightenment_herald / progressive_rationalist / revolutionary_herald
# ============================================================

cc_cond.1 = {
	type = country_event
	title = cc_cond.1.title
	desc = cc_cond.1.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			NOT = { has_variable = cc_recent_trait_assignment }
			NOT = { has_variable = cc_recent_dev_trait_event }

				has_trait = urban_planner
				has_trait = prosperity_herald
				has_trait = merchant_syndic
				has_trait = grand_architect
				has_trait = arts_patron
				has_trait = learned_courtier
				has_trait = reform_minded_counselor
				has_trait = master_urbanist
				has_trait = enlightenment_herald
				has_trait = progressive_rationalist
				has_trait = revolutionary_herald
			}
		}
		capital.development > 30
	}

	immediate = {
		save_scope_as = country
		if = {
			limit = { any_cabinet_character = { has_variable = cc_trait_dispatch_selected } }
			random_cabinet_character = {
				limit = { has_variable = cc_trait_dispatch_selected }
				remove_variable = cc_trait_dispatch_selected
				save_scope_as = minister
			}
		}
		else = {
			random_cabinet_character = {
				limit = {
					in_cabinet = yes
					NOT = { has_variable = cc_recent_trait_assignment }
					NOT = { has_variable = cc_recent_dev_trait_event }
					NOR = {
						has_trait = urban_planner
						has_trait = prosperity_herald
						has_trait = merchant_syndic
						has_trait = grand_architect
						has_trait = arts_patron
						has_trait = learned_courtier
						has_trait = reform_minded_counselor
						has_trait = master_urbanist
						has_trait = enlightenment_herald
						has_trait = progressive_rationalist
						has_trait = revolutionary_herald
					}
				}
				save_scope_as = minister
			}
		}
	}

	# --- Tier 3 options (dev > 50, age 5+) ---

	option = {
		name = cc_cond.1.a
		trigger = {
			scope:minister = { cc_xp_dispatch_ready = { TRACK = adm  TIER = 1 } }
			capital.development > 50
			OR = {
				current_age = age_5_absolutism
				current_age = age_6_revolutions
			}
		}
		scope:minister = {
			add_trait = trait:master_urbanist

			set_variable = { name = cc_recent_dev_trait_event value = yes years = 12 }
		}
	}

	option = {
		name = cc_cond.1.b
		trigger = {
			scope:minister = { cc_xp_dispatch_ready = { TRACK = adm  TIER = 3 } }
			capital.development > 50
			OR = {
				current_age = age_5_absolutism
				current_age = age_6_revolutions
			}
		}
		scope:minister = {
			add_trait = trait:enlightenment_herald
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
			set_variable = { name = cc_recent_dev_trait_event value = yes years = 12 }
		}
	}

	option = {
		name = cc_cond.1.c
		trigger = {
			scope:minister = { cc_xp_dispatch_ready = { TRACK = adm  TIER = 2 } }
			capital.development > 50
			OR = {

			}

		}
	}

	option = {
		name = cc_cond.1.d
		trigger = {
			capital.development > 50
			current_age = age_6_revolutions
		}
		scope:minister = {
			add_trait = trait:revolutionary_herald
			set_variable = { name = cc_recent_trait_assignment value = yes years = 2 }
			set_variable = { name = cc_recent_dev_trait_event value = yes years = 12 }
		}
	}

	# --- Tier 2 options (dev > 40, age 3+) ---

	option = {
		name = cc_cond.1.e
		trigger = {
			scope:minister = { cc_xp_dispatch_ready = { TRACK = adm  TIER = 2 } }

# cc_cond.101 — Urban Ambitions Take Shape
# Fires ~6 months after cc_cond.1 Tier 1 options (urban_planner / prosperity_herald / merchant_syndic).
cc_cond.101 = {
	type = country_event
	title = cc_cond.101.title
	desc = cc_cond.101.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
		10 = economy
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				has_variable = cc_cond1_urban_initiative
			}
			save_scope_as = minister
		}
	}

	after = {
		if = {
			limit = { exists = scope:minister }
			scope:minister = { remove_variable = cc_cond1_urban_initiative }
		}
	}

	option = {
		name = cc_cond.101.a

		random_list = {

			# ---- Development push — urban_planner biased ----
			30 = {
				trigger = {
					exists = scope:minister
					scope:minister = { has_trait = urban_planner }
				}
				capital = {
					change_development = development_weak_bonus
				}
				add_prestige = prestige_weak_bonus
			}

			# ---- Trade expansion — merchant_syndic biased ----
			25 = {
				trigger = {
					exists = scope:minister
					scope:minister = { has_trait = merchant_syndic }
				}
				add_country_modifier = { modifier = cc_cabinet_ministerial_investment years = 2 }
				add_prestige = prestige_weak_bonus
			}

			# ---- Provincial flourishing ----
			25 = {
				capital = {
					add_location_modifier = {
						modifier = cc_cabinet_provincial_flourishing
						years = 5
						mode = set_to_largest_and_extend
					}
				}
				add_prestige = prestige_weak_bonus
			}

			# ---- Costly project overrun ----
			20 = {
				add_gold = {
					value = monthly_income_trade_and_tax
					multiply = -3
					min = -100
					max = -30
				}
				add_prestige = prestige_weak_penalty
			}
		}
	}
}

# --- cc_cond.20: INTELLECTUAL / RELIGIOUS CLUSTER ---
# Axes: traditionalist_vs_innovative, spiritualist_vs_humanist, mysticism_vs_jurisprudence

cc_cond.20 = {
	type = country_event
	title = cc_cond.20.title
	desc = cc_cond.20.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			NOT = { has_variable = cc_recent_trait_assignment }
			NOT = { has_variable = cc_recent_social_trait_event }

				has_trait = innovative_counselor
				has_trait = traditionalist_guardian

				has_trait = mystical_counselor
				has_trait = legal_scholar

			add_trait = trait:traditionalist_guardian

			add_trait = trait:mystical_patriarch
