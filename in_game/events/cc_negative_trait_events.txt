namespace = cc_neg

###############################################################################
# NEGATIVE TRAIT EVENTS
# Grant events: cc_neg.1–9 (one per negative trait)
# Rehabilitation events: cc_neg.11–19 (one per negative trait)
# Fired from yearly pulse in cc_cabinet_pulse.txt (low weight, ~10%).
# Design rules:
#   - Always warn player before granting ("seems troubled")
#   - Never apply more than 1 negative trait per minister (check NOR in trigger)
#   - All are removable via paired rehabilitation events
###############################################################################

# ============================================================
# GRANT EVENTS
# ============================================================

# cc_neg.1 — Stagnant Counselor
# Minister in cabinet 20+ years; no capital dev growth
cc_neg.1 = {
	type = country_event
	title = cc_neg.1.title
	desc = cc_neg.1.desc
	outcome = negative

	illustration_tags = {
		10 = interior
	}

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			character_age > 50
			# No current negative traits
			NOR = {
				has_trait = stagnant_counselor
				has_trait = war_profiteer
				has_trait = court_schemer
				has_trait = inefficient_bureaucrat
				has_trait = corrupted_official
				has_trait = outdated_general
				has_trait = cynical_courtier
				has_trait = reckless_expander
			}
			# Long-serving: no cc_neg1 variable set (fired before)
			NOT = { has_variable = cc_neg1_warned }
		}
		capital.development < 25
		NOT = { has_variable = cc_neg1_fired }
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				character_age > 50
				NOR = {
					has_trait = stagnant_counselor
					has_trait = war_profiteer
					has_trait = court_schemer
					has_trait = inefficient_bureaucrat
					has_trait = corrupted_official
					has_trait = outdated_general
					has_trait = cynical_courtier
					has_trait = reckless_expander
				}
				NOT = { has_variable = cc_neg1_warned }
			}
			save_scope_as = troubled_minister
			set_variable = { name = cc_neg1_warned value = yes }
		}
	}

	# Act now — invest in the capital to re-invigorate the minister
	option = {
		name = cc_neg.1.a
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = -8
		}
		add_country_modifier = { modifier = cc_cabinet_provincial_flourishing years = 5 }
		set_variable = { name = cc_neg1_fired value = yes }
	}

	# Dismiss the minister before they stagnate
	option = {
		name = cc_neg.1.b
		remove_from_cabinet = scope:troubled_minister
		add_prestige = prestige_weak_bonus
		set_variable = { name = cc_neg1_fired value = yes }
	}

	# Ignore the warning — minister gains stagnant_counselor
	option = {
		name = cc_neg.1.c
		scope:troubled_minister = { add_trait = trait:stagnant_counselor }
		set_variable = { name = cc_neg1_fired value = yes }
	}
}


# cc_neg.2 — War Profiteer
# Military minister active during 2+ wars with high treasury
cc_neg.2 = {
	type = country_event
	title = cc_neg.2.title
	desc = cc_neg.2.desc
	outcome = negative

	illustration_tags = {
		10 = military
	}

	trigger = {
		gold >= {
			value = monthly_income_trade_and_tax
			multiply = 50
		}
		NOT = { at_war = yes }
		any_cabinet_character = {
			in_cabinet = yes
			OR = {
				has_trait = cavalry_marshal
				has_trait = commissariat_officer
				has_trait = siege_engineer
				has_trait = supreme_commander
				has_trait = marshal_of_the_court
			}
			NOR = {
				has_trait = war_profiteer
				has_trait = stagnant_counselor
				has_trait = court_schemer
				has_trait = inefficient_bureaucrat
				has_trait = corrupted_official
				has_trait = outdated_general
				has_trait = cynical_courtier
				has_trait = reckless_expander
			}
		}
		NOT = { has_variable = cc_neg2_fired }
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				OR = {
					has_trait = cavalry_marshal
					has_trait = commissariat_officer
					has_trait = siege_engineer
					has_trait = supreme_commander
					has_trait = marshal_of_the_court
				}
				NOR = {
					has_trait = war_profiteer
					has_trait = stagnant_counselor
					has_trait = court_schemer
					has_trait = inefficient_bureaucrat
					has_trait = corrupted_official
					has_trait = outdated_general
					has_trait = cynical_courtier
					has_trait = reckless_expander
				}
			}
			save_scope_as = troubled_minister
		}
		set_variable = { name = cc_neg2_fired value = yes }
	}

	option = {
		name = cc_neg.2.a
		# Pay them off — they stop profiteering
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = -15
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = 0.05 }
	}

	option = {
		name = cc_neg.2.b
		scope:troubled_minister = { add_trait = trait:war_profiteer }
	}
}
