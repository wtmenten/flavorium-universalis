namespace = cc_neg

###############################################################################
# NEGATIVE TRAIT EVENTS
# Grant events: cc_neg.1–9 (one per negative trait)
# Rehabilitation events: cc_neg.11–19 (one per negative trait)
# Fired from yearly pulse in cc_cabinet_pulse.txt (low weight, ~10%).
# Design rules:
#   - Always warn player before granting ("seems troubled")
#   - Never apply more than 1 negative trait per minister (check NOR in trigger)
#   - All are removable via paired rehabilitation events
###############################################################################

# ============================================================
# GRANT EVENTS
# ============================================================

# cc_neg.1 — Stagnant Counselor
# Minister in cabinet 20+ years; no capital dev growth
cc_neg.1 = {
	type = country_event
	title = cc_neg.1.title
	desc = cc_neg.1.desc
	outcome = negative

	illustration_tags = {
		10 = interior
	}

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			character_age > 50
			# No current negative traits
			NOR = {
				has_trait = stagnant_counselor
				has_trait = war_profiteer
				has_trait = court_schemer
				has_trait = inefficient_bureaucrat
				has_trait = corrupted_official
				has_trait = outdated_general
				has_trait = cynical_courtier
				has_trait = reckless_expander
			}
			# Long-serving: no cc_neg1 variable set (fired before)
			NOT = { has_variable = cc_neg1_warned }
		}
		capital.development < 25
		NOT = { has_variable = cc_neg1_fired }
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				character_age > 50
				NOR = {
					has_trait = stagnant_counselor
					has_trait = war_profiteer
					has_trait = court_schemer
					has_trait = inefficient_bureaucrat
					has_trait = corrupted_official
					has_trait = outdated_general
					has_trait = cynical_courtier
					has_trait = reckless_expander
				}
				NOT = { has_variable = cc_neg1_warned }
			}
			save_scope_as = minister
			set_variable = { name = cc_neg1_warned value = yes }
			set_variable = { name = cc_granting_trait value = yes }
		}
	}

	after = {
		scope:minister = {
			remove_variable = cc_granting_trait
		}
	}

	# Act now — invest in the capital to re-invigorate the minister
	option = {
		name = cc_neg.1.a
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = -0.40
			min = -100
		}
		scope:minister = { set_variable = { name = cc_neg1_new_project value = yes } }
		trigger_event_silently =  { id = cc_neg.100 days = 150 }
		set_variable = { name = cc_neg1_fired value = yes }
	}

	# Dismiss the minister before they stagnate
	option = {
		name = cc_neg.1.b
		remove_from_cabinet = scope:minister
		add_prestige = prestige_weak_bonus
		set_variable = { name = cc_neg1_fired value = yes }
	}

	# Ignore the warning — minister gains stagnant_counselor
	option = {
		name = cc_neg.1.c
		scope:minister = { add_trait = trait:stagnant_counselor }
		set_variable = { name = cc_neg1_fired value = yes }
	}
}


# ============================================================
# FOLLOW-UP EVENTS — fire after player intervenes in negative trait events
# cc_neg.100–103: one per negative trait that has an active intervention option.
# Minister is re-found via a dedicated character variable set in the triggering option.
# ============================================================

# cc_neg.100 — A Minister Seizes Initiative
# Fires ~5 months after cc_neg.1 option A (the heavy investment).
# Random outcomes weighted by minister estate, skills, and location availability.
# Costly overrun re-triggers this event up to 2 times before forcing resolution.
cc_neg.100 = {
	type = country_event
	title = cc_neg.100.title
	desc = cc_neg.100.desc
	outcome = neutral

	illustration_tags = {
		10 = exterior
		10 = economy
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				has_variable = cc_neg1_new_project
			}
			set_variable = { name = cc_granting_trait value = yes }
			save_scope_as = minister
		}
	}

	after = {
		if = {
			limit = { NOT = { has_variable = cc_neg100_retrigger } }
			if = {
				limit = { exists = scope:minister }
				scope:minister = { remove_variable = cc_neg1_new_project remove_variable = cc_granting_trait  }
			}
			remove_variable = cc_neg100_overrun_count
		}
		else = {
			remove_variable = cc_neg100_retrigger
		}
	}

	option = {
		name = cc_neg.100.a

		random_list = {

			# ---- POSITIVE: Entrepot — burghers estate favored ----
			30 = {
				trigger = {
					exists = scope:minister
					scope:minister = { has_estate = estate_type:burghers_estate }
				}
				random_owned_location = {
					limit = {
						owner = root
						can_build_building = building_type:entrepot
					}
					construct_building = { building_type = building_type:entrepot cost_multiplier = 1 }
				}
				add_prestige = prestige_mild_bonus
				add_gold = {
					value = monthly_income_trade_and_tax
					multiply = 0.15
					min = 30
				}
			}

			# ---- POSITIVE: Library — burghers or clergy favored ----
			20 = {
				trigger = {
					exists = scope:minister
					OR = {
						scope:minister = { has_estate = estate_type:burghers_estate }
						scope:minister = { has_estate = estate_type:clergy_estate }
					}
				}
				random_owned_location = {
					limit = {
						owner = root
						can_build_building = building_type:library
					}
					construct_building = { building_type = building_type:library cost_multiplier = 1 }
				}
				add_prestige = prestige_weak_bonus
			}

			# ---- POSITIVE: Temple — clergy estate favored ----
			30 = {
				trigger = {
					exists = scope:minister
					scope:minister = { has_estate = estate_type:clergy_estate }
				}
				random_owned_location = {
					limit = {
						owner = root
						can_build_building = building_type:temple
					}
					save_scope_as = cc_neg100_temple_loc
					construct_building = { building_type = building_type:temple cost_multiplier = 1 }
				}
				if = {
					limit = { exists = scope:cc_neg100_temple_loc }
					scope:cc_neg100_temple_loc = {
						every_pop = {
							add_pop_satisfaction = pop_satisfaction_weak_bonus
						}
					}
				}
				add_prestige = prestige_weak_bonus
			}

			# ---- POSITIVE: Irrigation systems — peasants estate favored ----
			30 = {
				trigger = {
					exists = scope:minister
					scope:minister = { has_estate = estate_type:peasants_estate }
				}
				random_owned_location = {
					limit = {
						owner = root
						can_build_building = building_type:irrigation_systems
					}
					construct_building = { building_type = building_type:irrigation_systems cost_multiplier = 1 }
				}
				add_prestige = prestige_weak_bonus
			}

			# ---- POSITIVE: Bailiff — nobles estate favored ----
			30 = {
				trigger = {
					exists = scope:minister
					scope:minister = { has_estate = estate_type:nobles_estate }
				}
				random_owned_location = {
					limit = {
						owner = root
						can_build_building = building_type:bailiff
					}
					construct_building = { building_type = building_type:bailiff cost_multiplier = 1 }
				}
				add_estate_satisfaction = { type = estate_type:nobles_estate value = 0.05 }
				add_prestige = prestige_weak_bonus
			}

			# ---- POSITIVE: Provincial flourishing (generic) ----
			20 = {
				random_owned_location = {
					limit = {
						owner = root
						development > 5
					}
					add_location_modifier = {
						modifier = cc_cabinet_provincial_flourishing
						years = 5
						mode = set_to_largest_and_extend
					}
				}
				add_prestige = prestige_weak_bonus
			}

			# ---- NEGATIVE: Pop satisfaction hit ----
			12 = {
				random_owned_location = {
					limit = { owner = root }
					every_pop = {
						add_pop_satisfaction = pop_satisfaction_mild_penalty
					}
				}
			}

			# ---- NEGATIVE: Neighbor opinion hit ----
			10 = {
				random_neighbor_country = {
					limit = { NOT = { is_at_war_with = root } }
					add_opinion = { target = root modifier = cc_neg_project_overreach }
				}
				add_prestige = prestige_weak_penalty
			}

			# ---- NEGATIVE: Prosperity decrease ----
			10 = {
				random_owned_location = {
					limit = {
						owner = root
						development > 3
					}
					change_prosperity = prosperity_mild_penalty
				}
			}

			# ---- NEGATIVE: Stagnant counselor — forced ----
			8 = {
				trigger = {
					exists = scope:minister
					scope:minister = { in_cabinet = yes }
				}
				scope:minister = { add_trait = trait:stagnant_counselor }
				add_prestige = prestige_weak_penalty
			}

			# ---- NEGATIVE: Costly overrun — retriggers up to 2x ----
			8 = {
				trigger = {
					OR = {
						NOT = { has_variable = cc_neg100_overrun_count }
						var:cc_neg100_overrun_count < 2 
					}
				}
				add_gold = {
					value = monthly_income_trade_and_tax
					multiply = -0.20
					min = -200
					max = -50
				}
				add_prestige = prestige_weak_penalty
				if = {
					limit = { NOT = { has_variable = cc_neg100_overrun_count } }
					set_variable = { name = cc_neg100_overrun_count value = 0 }
				}
				change_variable = { name = cc_neg100_overrun_count add = 1 }
				if = {
					limit = { exists = scope:minister }
					scope:minister = { set_variable = { name = cc_neg1_new_project value = yes } }
				}
				set_variable = { name = cc_neg100_retrigger value = yes }
				trigger_event_silently =  { id = cc_neg.100 days = 90 }
			}
		}
	}
}


# cc_neg.2 — War Profiteer
# Military minister active during 2+ wars with high treasury
cc_neg.2 = {
	type = country_event
	title = cc_neg.2.title
	desc = cc_neg.2.desc
	outcome = negative

	illustration_tags = {
		10 = military
	}

	trigger = {
		gold >= {
			value = monthly_income_trade_and_tax
			multiply = 50
		}
		NOT = { at_war = yes }
		any_cabinet_character = {
			in_cabinet = yes
			OR = {
				has_trait = cavalry_marshal
				has_trait = commissariat_officer
				has_trait = siege_engineer
				has_trait = supreme_commander
				has_variable = cc_office_held_marshal_of_the_court
			}
			NOR = {
				has_trait = war_profiteer
				has_trait = stagnant_counselor
				has_trait = court_schemer
				has_trait = inefficient_bureaucrat
				has_trait = corrupted_official
				has_trait = outdated_general
				has_trait = cynical_courtier
				has_trait = reckless_expander
			}
		}
		NOT = { has_variable = cc_neg2_fired }
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				OR = {
					has_trait = cavalry_marshal
					has_trait = commissariat_officer
					has_trait = siege_engineer
					has_trait = supreme_commander
					has_variable = cc_office_held_marshal_of_the_court
				}
				NOR = {
					has_trait = war_profiteer
					has_trait = stagnant_counselor
					has_trait = court_schemer
					has_trait = inefficient_bureaucrat
					has_trait = corrupted_official
					has_trait = outdated_general
					has_trait = cynical_courtier
					has_trait = reckless_expander
				}
			}
			save_scope_as = minister
			set_variable = { name = cc_granting_trait value = yes }
		}
		set_variable = { name = cc_neg2_fired value = yes }
	}

	after = {
		scope:minister = {
			remove_variable = cc_granting_trait
		}
	}

	option = {
		name = cc_neg.2.a
		# Pay them off — they stop profiteering
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = -0.75
			min = -200
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = 0.05 }
	}

	option = {
		name = cc_neg.2.b
		scope:minister = {
			add_trait = trait:war_profiteer
		}
	}
}

# cc_neg.101 — The Reassigned Minister Reports

	type = country_event
	title = cc_neg.101.title
	desc = cc_neg.101.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				has_variable = cc_neg3_reassigned
			}
			save_scope_as = minister
		}
	}

	after = {
		if = {
			limit = { exists = scope:minister }
			scope:minister = { remove_variable = cc_neg3_reassigned }
		}
	}

	option = {
		name = cc_neg.101.a

		random_list = {

			# ---- Productive: minister channeled ambitions productively ----
			30 = {
				trigger = {
					exists = scope:minister
					scope:minister = { dip > 50 }
				}
				add_prestige = prestige_mild_bonus
				add_stability = stability_weak_bonus
			}

			# ---- Quietly scheming: busy but not reformed ----
			25 = {
				add_country_modifier = { modifier = cc_cabinet_ministerial_investment years = 2 }
			}

			# ---- Backfired: minister made the situation worse ----
			20 = {
				trigger = {
					exists = scope:minister
					scope:minister = { dip < 40 }
				}
				set_variable = { name = cc_tension_schemer_risk value = yes }
				if = {
					limit = { exists = scope:minister }
					scope:minister = { set_variable = { name = cc_schemer_risk_target value = yes } }
				}
				remove_variable = cc_neg3_fired
				add_prestige = prestige_weak_penalty
			}

			# ---- Defection: minister leaked court secrets ----
			15 = {
				trigger = {
					exists = scope:minister
					scope:minister = { dip < 40 }
				}
				random_neighbor_country = {
					limit = { NOT = { is_at_war_with = root } }
					add_opinion = { target = root modifier = cc_diplomatic_insult }
				}
				add_prestige = prestige_weak_penalty
			}

			# ---- Resignation: minister quits under pressure ----
			10 = {
				trigger = {
					exists = scope:minister
					scope:minister = { in_cabinet = yes }
				}
				remove_from_cabinet = scope:minister
				add_prestige = prestige_weak_bonus
			}
		}
	}
}
