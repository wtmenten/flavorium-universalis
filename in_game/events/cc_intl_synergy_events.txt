namespace = cc_intl

###############################################################################
# CROSS-COUNTRY SYNERGY EVENTS (cc_intl namespace)
# Fire from yearly_country_pulse when your minister/ruler trait combination
# matches a condition on a neighboring country — ruler, cabinet, religion, or
# government type. Effects cross the border: modifiers, CBs, opinion shifts,
# migration flavor, estate satisfaction, gold.
# Cooldowns prevent rapid re-fires; set on root or on the partner country.
###############################################################################

###############################################################################
# GROUP A — INTELLECTUAL & SCHOLARLY
###############################################################################

# cc_intl.1 — "The Philosopher's Exchange"
# Your enlightened minister + an innovative neighboring court.
# Flavor: Renaissance correspondence between scholars of rival courts.
cc_intl.1 = {
	type = country_event
	title = cc_intl.1.title
	desc = cc_intl.1.desc
	outcome = positive

	illustration_tags = {
		10 = interior
		10 = professional
	}

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			OR = {
				has_trait = enlightenment_herald
				has_trait = progressive_rationalist
			}
		}
		any_neighbor_country = {
			NOT = { has_variable = cc_intl_1_fired }
			societal_value:traditionalist_vs_innovative > 20
		}
	}

	immediate = {
		random_neighbor_country = {
			limit = {
				NOT = { has_variable = cc_intl_1_fired }
				societal_value:traditionalist_vs_innovative > 20
			}
			save_scope_as = target_nation
			set_variable = { name = cc_intl_1_fired value = yes years = 12 }
		}
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				OR = {
					has_trait = enlightenment_herald
					has_trait = progressive_rationalist
				}
			}
			save_scope_as = minister
		}
		scope:target_nation = { capital = { save_scope_as = cc_intl_origin } }
		capital = { save_scope_as = cc_intl_dest }
	}

	# Formal scholarly exchange — both courts benefit
	option = {
		name = cc_intl.1.a
		add_country_modifier = { modifier = cc_intl_scholar_influx years = 12 }
		scope:target_nation = {
			add_country_modifier = { modifier = cc_intl_scholar_influx years = 12 }
			add_opinion = { modifier = cc_intl_scholarly_bond target = root }
		}
		add_opinion = { target = scope:target_nation modifier = cc_intl_scholarly_bond }
		add_prestige = prestige_mild_bonus
	}

	# Invite their scholars exclusively — a one-way migration of talent
	option = {
		name = cc_intl.1.b
		add_country_modifier = { modifier = cc_intl_scholar_influx years = 15 }
		add_migration = {
			from_location = scope:cc_intl_origin
			to_location = scope:cc_intl_dest
			culture = scope:target_nation.culture
			religion = scope:target_nation.religion
			type = pop_type:burghers
			months = 24
			amount = 0.025
		}
		scope:target_nation = {
			add_prestige = prestige_mild_bonus
			add_opinion = { modifier = cc_intl_scholarly_bond target = root }
		}
	}

	# Decline — the court maintains its own intellectual independence
	option = {
		name = cc_intl.1.c
		add_prestige = prestige_weak_bonus
	}
}


# cc_intl.2 — "Scholars in Exile"
# Your learned minister attracts scholars fleeing a neighbor's war.
# Flavor: Flight of talent from warring states — historical in every era.
cc_intl.2 = {
	type = country_event
	title = cc_intl.2.title
	desc = cc_intl.2.desc
	outcome = positive

	illustration_tags = {
		10 = interior

	}

	trigger = {
		NOT = { has_variable = cc_intl_2_cooldown }
		any_cabinet_character = {
			in_cabinet = yes
			OR = {
				has_trait = learned_courtier
				has_trait = master_urbanist
				has_trait = grand_architect
			}
		}
		any_neighbor_country = { 
			at_war = yes 
			NOT = { exists = "war_with_country(ROOT)" }
			capital.continent = ROOT.capital.continent 
			save_temporary_scope_as = war_nation
		}
		scope:war_nation.capital.continent = {
			any_location_in_continent = {
				owner = scope:war_nation
				OR = {
					province.province_prosperity <= 0.33
					NOT = { controller = scope:war_nation }
				}
			}
		}
	}

	immediate = {
		set_variable = { name = cc_intl_2_cooldown value = yes years = 12 }
		random_neighbor_country = {
			limit = { 
				at_war = yes
				capital.continent = ROOT.capital.continent 
			}
			save_scope_as = war_nation
		}
		capital.continent = {
			random_location_in_continent = {
				limit = {
					owner = scope:war_nation
					OR = {
						province.province_prosperity <= 0.33
						NOT = { controller = scope:war_nation }
					}
				}
				save_scope_as = war_nation_province_1
			}
			random_location_in_continent = {
				limit = {
					owner = scope:war_nation
					OR = {
						province.province_prosperity <= 0.33
						NOT = { controller = scope:war_nation }
					}
				}
				save_scope_as = war_nation_province_2
			}
			random_location_in_continent = {
				limit = {
					owner = scope:war_nation
					OR = {
						province.province_prosperity <= 0.33
						NOT = { controller = scope:war_nation }
					}
				}
				save_scope_as = war_nation_province_3
			}
			
			random_location_in_continent = {
				limit = {
					owner = ROOT
					NOT = { location_rank = location_rank:rural_settlement } 
				}
				save_scope_as = cc_intl_dest
			}
		}
		
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				OR = {
					has_trait = learned_courtier
					has_trait = master_urbanist
					has_trait = grand_architect
				}
			}
			save_scope_as = minister
		}

	}

	# Open the court — scholars arrive and enrich the capital
	option = {
		name = cc_intl.2.a
		add_country_modifier = { modifier = cc_intl_scholar_influx years = 15  mode = set_to_largest_and_extend }
		add_migration = {
			from_location = scope:war_nation_province_1
			to_location = scope:cc_intl_dest
			culture = scope:war_nation.culture
			religion = scope:war_nation.religion
			type = pop_type:clergy
			months = 12
			amount = 0.004
		}
		add_migration = {
			from_location = scope:war_nation_province_2
			to_location = scope:cc_intl_dest
			culture = scope:war_nation.culture
			religion = scope:war_nation.religion
			type = pop_type:burghers
			months = 12
			amount = 0.010
		}
		add_migration = {
			from_location = scope:war_nation_province_3
			to_location = scope:cc_intl_dest
			culture = scope:war_nation.culture
			religion = scope:war_nation.religion
			type = pop_type:peasants
			months = 12
			amount = 0.020
		}
		scope:war_nation = {
			add_opinion = { modifier = cc_intl_scholarly_bond target = root }
		}
		add_prestige = prestige_mild_bonus
	}

	# Recruit selectively — fund targeted outreach for the best minds
	option = {
		name = cc_intl.2.b
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = -0.50
			min = -250
			max = -75
		}
		add_country_modifier = { 
			modifier = cc_intl_scholar_influx 
			years = 8 
			size  = 0.5 
			mode = set_to_largest_and_extend 
		}
		add_migration = {
			from_location = scope:war_nation_province_1
			to_location = scope:cc_intl_dest
			culture = scope:war_nation.culture
			religion = this.religion
			type = pop_type:clergy
			months = 8
			amount = 0.008
		}
		add_migration = {
			from_location = scope:war_nation_province_2
			to_location = scope:cc_intl_dest
			culture = this.culture
			religion = this.religion
			type = pop_type:burghers
			months = 8
			amount = 0.008
		}
		add_migration = {
			from_location = scope:war_nation_province_3
			to_location = scope:cc_intl_dest
			culture = scope:war_nation.culture
			religion = scope:war_nation.religion
			type = pop_type:burghers
			months = 8
			amount = 0.008
		}
		scope:war_nation = {
			add_opinion = { modifier = denied_aid target = root }
		}
	}

	# Remain neutral — do not invite refugees, maintain diplomatic stability
	option = {
		name = cc_intl.2.c
		add_prestige = prestige_weak_bonus
		scope:war_nation = {
			add_opinion = { modifier = denied_aid target = root scale = 1.5 }
		}
	}
}


# cc_intl.3 — "The Rival Philosopher"
# Your ruler has philosopher_king trait (or enlightened cabinet) + innovative neighbor.
# Flavor: Two enlightened courts competing for cultural supremacy.
cc_intl.3 = {
	type = country_event
	title = cc_intl.3.title
	desc = cc_intl.3.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
	}

	trigger = {
		NOT = { has_variable = cc_intl_3_cooldown }
		OR = {
			ruler = { has_trait = philosopher_king }
			any_cabinet_character = {
				in_cabinet = yes
				OR = {
					has_trait = progressive_rationalist
					has_trait = enlightenment_herald
				}
			}
		}
		any_neighbor_country = {
			societal_value:traditionalist_vs_innovative > 30
			NOT = { this = root }
		}
	}

	immediate = {
		set_variable = { name = cc_intl_3_cooldown value = yes years = 15 }
		random_neighbor_country = {
			limit = {
				societal_value:traditionalist_vs_innovative > 30
				NOT = { this = root }
			}
			save_scope_as = rival_court
		}
	}

	# Let the rivalry sharpen both courts — competitive innovation
	option = {
		name = cc_intl.3.a
		add_country_modifier = { modifier = cc_intl_intellectual_rivalry years = 10 }
		scope:rival_court = {
			add_country_modifier = { modifier = cc_intl_intellectual_rivalry years = 10 }
			add_opinion = { modifier = cc_intl_rival_court target = root }
		}
		add_opinion = { target = scope:rival_court modifier = cc_intl_rival_court }
		add_prestige = prestige_mild_bonus
	}

	# Assert cultural preeminence — challenge their standing openly
	option = {
		name = cc_intl.3.b
		add_country_modifier = { modifier = cc_intl_intellectual_rivalry years = 12 }
		culture = { add_cultural_influence = cultural_influence_mild_bonus }
		scope:rival_court = {
			add_opinion = { modifier = cc_intl_rival_court target = root }
		}
		add_prestige = prestige_mild_bonus
	}

	# Propose a joint philosophical academy — costs gold, benefits both
	option = {
		name = cc_intl.3.c
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = -1.50
			min = -350
		}
		add_country_modifier = { modifier = cc_intl_scholar_influx years = 15 }
		scope:rival_court = {
			add_country_modifier = { modifier = cc_intl_scholar_influx years = 15 }
			add_opinion = { modifier = cc_intl_scholarly_bond target = root }
		}
		add_opinion = { target = scope:rival_court modifier = cc_intl_scholarly_bond }
		add_prestige = prestige_severe_bonus
	}
}
