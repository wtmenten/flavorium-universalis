###############################################################################
# FLAVORIUM UNIVERSALIS — CABINET EXPERIENCE EVENTS
#
# cc_xp.1    Service record readout, opened from the cc_xp_inspect_service_record
#            character interaction. Displays only; changes nothing.
# cc_xp.10   Ladder advancement notification. The advancement itself has already
#            happened in cc_xp_ladder_advance; this only reports it.
# cc_xp.11   Court examination sitting. Queued months earlier by the interaction;
#            this is where the experience is actually awarded.
# cc_xp.40   Weariness at the top: a very senior minister is worn down.
# cc_xp.41   Institutional memory: a great minister dies and someone inherits.
# cc_xp.42   Rivalry of equals: two accomplished specialists of the same track.
# cc_xp.30   Return from the chancery      (administrative)
# cc_xp.31   Return from the embassy circuit (diplomatic)
# cc_xp.32   Return from the staff college   (military)
#
# cc_xp.20-28 are the generated ladder branch events (cc_xp_branch_events.txt).
#
# THE RETURN EVENTS all follow one shape: find the character carrying this posting's
# away flag, clear it, award the experience, report. They identify the minister by flag
# rather than by a scope saved at dispatch, because a scope is not something to rely on
# surviving a year of queued game time. Clearing the flag as part of selection means two
# ministers sent on the same posting resolve to different people.
#
# Each posting can go well, ordinarily, or badly; the branch is in the immediate block
# and the desc picks its text from the outcome flag afterwards.
#
# A POSTING IS ALSO HOW A CAREER STARTS. If the minister who came back holds no ladder
# trait at all, the option chains into that track's career picker (cc_xp.90/.91/.92) and
# the player chooses which of the track's careers they are now on. Before this, rung 0 was
# reachable only by the monthly trait dispatcher's random roll, and four of the fifteen
# ladders were reachable by nothing at all.
#
# A POOR POSTING GRANTS NO CAREER. Wash out of the chancery and you keep the reduced
# experience and get no commission; train them again. That is what stops career entry
# from being a straight purchase, and it is the only thing the outcome roll decides
# besides the size of the award.
###############################################################################

# Requires scope:cc_xp_subject, set by the interaction.
cc_xp.1 = {
	type = country_event
	title = cc_xp.1.title
	desc = cc_xp.1.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
	}

	option = {
		name = cc_xp.1.a
	}
}


###############################################################################
# LADDER ADVANCEMENT NOTIFICATION
#
# Fired once per year by cc_xp_yearly_advance when at least one of the player's own
# ministers moved up a rung. The swap has already been applied; nothing here changes
# state, so the player closing this event cannot lose the advancement.
#
# Names one minister rather than listing all of them: more than one advancing in the
# same year is rare (the four-year cooldown sees to that), and a single named person
# reads better than a list. The rest are visible on their cards.
###############################################################################

cc_xp.10 = {
	type = country_event
	title = cc_xp.10.title
	desc = cc_xp.10.desc
	outcome = positive

	illustration_tags = {
		10 = interior
	}

	trigger = {
		any_character = {
			has_variable = cc_xp_advanced_recently
		}
	}

	immediate = {
		random_character = {
			limit = {
				has_variable = cc_xp_advanced_recently
			}
			save_scope_as = cc_xp_subject
		}
	}

	option = {
		name = cc_xp.10.a
	}
}


###############################################################################
# THE COURT EXAMINATION SITS
#
# Queued by cc_xp_court_examination months earlier. The interaction only commissioned
# it and charged for it; the examination itself, and every consequence of it, happens
# here. The presider is found by the cc_xp_examining flag rather than a saved scope.
###############################################################################

cc_xp.11 = {
	type = country_event
	title = cc_xp.11.title
	desc = cc_xp.11.desc
	outcome = positive

	illustration_tags = {
		10 = interior
	}

	trigger = {
		any_character = { has_variable = cc_xp_examining }
	}

	immediate = {
		random_character = {
			limit = { has_variable = cc_xp_examining }
			remove_variable = cc_xp_examining
			save_scope_as = cc_xp_examiner
		}

		# Everyone sat it, so everyone learned something from it.
		every_character = {
			limit = {
				cc_xp_is_examinable = yes
				NOT = { this = scope:cc_xp_examiner }
			}
			cc_xp_gain_dominant = { AMOUNT = cc_xp_exam_participant }
			cc_xp_recompute = yes
		}

		# One of them did conspicuously well. Not the presider, and not necessarily the
		# most senior: that is the entire point of examining people.
		random_character = {
			limit = {
				cc_xp_is_examinable = yes
				NOT = { this = scope:cc_xp_examiner }
			}
			cc_xp_gain_dominant = { AMOUNT = cc_xp_exam_winner }
			cc_xp_recompute = yes
			save_scope_as = cc_xp_subject
		}

		# Presiding over it is a visible mark of the crown's confidence.
		add_prestige = prestige_weak_bonus

		# Advancement decided by examination rather than by birth. The burghers approve
		# for exactly the reason the nobility does not.
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_weak_bonus
		}
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_weak_penalty
		}
	}

	option = {
		name = cc_xp.11.a
	}
}

# The chancery.
cc_xp.30 = {
	type = country_event
	title = cc_xp.30.title
	outcome = positive

	illustration_tags = {
		10 = interior
	}

	desc = {
		# first_valid, not a bare list. A bare `desc = { triggered_desc ... }` renders
		# EVERY matching entry; vanilla wraps mutually-exclusive ones in first_valid in
		# 300 places for exactly this reason. Without it the unconditional entry below
		# is appended to the good/poor text instead of standing in for it.
		first_valid = {
			triggered_desc = {
				trigger = { scope:cc_xp_subject ?= { has_variable = cc_xp_outcome_good } }
				desc = cc_xp.30.desc.good
			}
			triggered_desc = {
				trigger = { scope:cc_xp_subject ?= { has_variable = cc_xp_outcome_poor } }
				desc = cc_xp.30.desc.poor
			}
			triggered_desc = {
				desc = cc_xp.30.desc
			}
	
		}
	}

	trigger = {
		any_character = { has_variable = cc_xp_away_adm }
	}

	immediate = {
		random_character = {
			limit = { has_variable = cc_xp_away_adm }
			remove_variable = cc_xp_away_adm
			save_scope_as = cc_xp_subject

			# How the posting went. An already accomplished minister gets more out of it
			# and flounders less often, so training compounds rather than leveling the
			# court out. The outcome flags are timed and exist only for the desc above,
			# which renders after this block has run.
			random_list = {
				20 = {
					modifier = { factor = 1.5  cc_xp_tier_at_least = { TRACK = adm  TIER = 2 } }
					set_variable = { name = cc_xp_outcome_good  value = yes  months = 2 }
					cc_xp_gain_adm = { AMOUNT = cc_xp_training_gain_good }
				}
				60 = {
					cc_xp_gain_adm = { AMOUNT = cc_xp_training_gain }
				}
				20 = {
					modifier = { factor = 0.5  cc_xp_tier_at_least = { TRACK = adm  TIER = 2 } }
					set_variable = { name = cc_xp_outcome_poor  value = yes  months = 2 }
					cc_xp_gain_adm = { AMOUNT = cc_xp_training_gain_poor }
				}
			}

			cc_xp_recompute = yes

			# Offer a career if they have none. Rolled here rather than in the picker so
			# the offer is stamped on the MINISTER: two ministers back from the chancery in
			# the same month get their own menus instead of sharing whichever was rolled last.
			if = {
				limit = {
					cc_xp_is_unladdered = yes
					NOT = { has_variable = cc_xp_outcome_poor }
					# Every entry trait has a real allow block, and add_trait against a
					# failing one is refused silently. Without this a minister carrying an
					# age trait, or a protege (seven entry traits want in_cabinet = yes),
					# would get a picker whose every option grants nothing.
					cc_xp_can_enter_career_adm = yes
				}
				cc_xp_offer_career_adm = yes
			}
		}
	}

	option = {
		name = cc_xp.30.a
		# Chained from the option rather than the immediate so the player reads the report
		# before choosing, and so the two events cannot open on top of each other.
		if = {
			limit = { any_character = { has_variable = cc_xp_awaiting_career_adm } }
			trigger_event_silently = { id = cc_xp.90 }
		}
	}
}

###############################################################################
# WEARINESS AT THE TOP
#
# The counterweight to a system that otherwise only ever rewards keeping a good
# minister in post forever. Someone at the very top of their career, still working,
# eventually runs down. The crown chooses whether to spend the productivity now or
# preserve the person.
#
# Fired from cc_xp_yearly_weariness. Gated through the scripted triggers rather than
# raw var: reads, because an event's top-level trigger is evaluated before immediate.
###############################################################################

cc_xp.40 = {
	type = country_event
	title = cc_xp.40.title
	desc = cc_xp.40.desc
	outcome = negative

	illustration_tags = {
		10 = interior
	}

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			cc_xp_level_at_least = { LEVEL = 7 }
			years_of_service_in_cabinet >= 20
			NOT = { cc_xp_is_away = yes }
			NOT = { has_character_modifier = cc_xp_weary }
			NOT = { has_variable = cc_xp_weary_cd }
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				cc_xp_level_at_least = { LEVEL = 7 }
				years_of_service_in_cabinet >= 20
				NOT = { cc_xp_is_away = yes }
				NOT = { has_character_modifier = cc_xp_weary }
				NOT = { has_variable = cc_xp_weary_cd }
			}
			set_variable = { name = cc_xp_weary_cd  value = yes  years = 8 }
			save_scope_as = cc_xp_subject
		}
	}

	# Send them away to rest. Costs a year of their service and nothing else.
	#
	# Uses its own cc_xp_on_leave flag rather than borrowing a posting flag: the posting
	# flags are claimed by the return events, which would award training experience for
	# what is supposed to be a year of doing nothing. This one simply expires.
	option = {
		name = cc_xp.40.a
		scope:cc_xp_subject = {
			set_variable = { name = cc_xp_on_leave  value = yes  months = 12 }
		}
	}

	# Keep them at the desk. They stay, and they are worse at it from here on.
	option = {
		name = cc_xp.40.b
		scope:cc_xp_subject = {
			add_character_modifier = {
				modifier = cc_xp_weary
				years = -1
				mode = replace
			}
		}
	}
}


###############################################################################
# INSTITUTIONAL MEMORY
#
# What a long career leaves behind. When a genuinely accomplished minister dies, the
# people they worked alongside keep some of it: the filing system they built, the
# contacts they introduced, the way they did things.
#
# Chained from on_cabinet_death via cc_xp_on_death_legacy. root = country,
# scope:target = the dying minister.
###############################################################################

cc_xp.41 = {
	type = country_event
	title = cc_xp.41.title
	desc = cc_xp.41.desc
	outcome = positive

	illustration_tags = {
		10 = interior
	}

	trigger = {
		# Someone must be left to inherit it.
		any_character = {
			is_alive = yes
			OR = {
				in_cabinet = yes
				has_variable = cc_protege
			}
			cc_xp_is_tracked = yes
			NOT = { cc_xp_level_at_least = { LEVEL = 7 } }
		}
	}

	immediate = {
		random_character = {
			limit = {
				is_alive = yes
				OR = {
					in_cabinet = yes
					has_variable = cc_protege
				}
				cc_xp_is_tracked = yes
				NOT = { cc_xp_level_at_least = { LEVEL = 7 } }
			}
			cc_xp_gain_dominant = { AMOUNT = cc_xp_legacy_transfer }
			cc_xp_recompute = yes
			save_scope_as = cc_xp_subject
		}
	}

	option = {
		name = cc_xp.41.a
	}
}


###############################################################################
# RIVALRY OF EQUALS
#
# Two accomplished ministers of the same speciality, who between them can only hold one
# reputation. Distinct from the mod's existing court_tension rivalry, which is about
# opposed ideologies: this one is about two people being good at the same thing.
#
# Fired from cc_xp_yearly_rivalry.
###############################################################################

cc_xp.42 = {
	type = country_event
	title = cc_xp.42.title
	desc = cc_xp.42.desc
	outcome = negative

	illustration_tags = {
		10 = interior
	}

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			cc_xp_level_at_least = { LEVEL = 5 }
			NOT = { has_character_modifier = cc_xp_professional_rival }
			NOT = { has_variable = cc_xp_rivalry_cd }
		}
		# Two or more, counted. This used to be `any_cabinet_character = { count = 2 ... }`,
		# which is an EXACT match in EU5: a court with three senior ministers, which is
		# precisely the court a rivalry belongs in, could never fire this event.
		cc_xp_num_senior_ministers >= 2
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				cc_xp_level_at_least = { LEVEL = 5 }
				NOT = { has_variable = cc_xp_rivalry_cd }
			}
			set_variable = { name = cc_xp_rivalry_cd  value = yes  years = 10 }
			save_scope_as = cc_xp_subject
		}

		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				cc_xp_level_at_least = { LEVEL = 5 }
				NOT = { this = scope:cc_xp_subject }
			}
			set_variable = { name = cc_xp_rivalry_cd  value = yes  years = 10 }
			save_scope_as = cc_xp_rival
		}
	}

	# Back one of them. The favored one flourishes; the other takes it badly.
	option = {
		name = cc_xp.42.a
		scope:cc_xp_subject = {
			add_character_modifier = { modifier = cc_minister_favoured  years = 10  mode = replace }
		}
		scope:cc_xp_rival ?= {
			add_character_modifier = { modifier = cc_minister_slighted  years = 10  mode = replace }
			set_variable = { name = cc_minister_resentful  value = yes  years = 10 }
		}
	}

	# Let them fight it out. Both are diminished while it lasts.
	option = {
		name = cc_xp.42.b
		scope:cc_xp_subject = {
			add_character_modifier = { modifier = cc_xp_professional_rival  years = 10  mode = replace }
		}
		scope:cc_xp_rival ?= {
			add_character_modifier = { modifier = cc_xp_professional_rival  years = 10  mode = replace }
		}
	}

	# Set them to the same problem from opposite ends. Costs prestige to arbitrate.
	option = {
		name = cc_xp.42.c
		add_prestige = prestige_weak_penalty
		scope:cc_xp_subject = {
			add_character_modifier = { modifier = cc_minister_reconciled  years = 10  mode = replace }
			cc_xp_gain_dominant = { AMOUNT = cc_xp_exam_participant }
			cc_xp_recompute = yes
		}
		scope:cc_xp_rival ?= {
			add_character_modifier = { modifier = cc_minister_reconciled  years = 10  mode = replace }
			cc_xp_gain_dominant = { AMOUNT = cc_xp_exam_participant }
			cc_xp_recompute = yes
		}
	}
}
