namespace = cc_dual

###############################################################################
# DUAL-LAYER SYNERGY EVENTS (Cabinet × Religious_Figure on SAME minister)
# Fire from yearly pulse when a specific (cabinet, religious_figure) pair is
# present on the SAME minister simultaneously.
# Each is one-time per minister (gated by character variable).
# Uses cc_dual.1–10 range.
###############################################################################

# cc_dual.1 — "The Warrior Mystic"
# cavalry_marshal + sufi_scholar: martial skill and Sufi devotion united.
cc_dual.1 = {
	type = country_event
	title = cc_dual.1.title
	desc = cc_dual.1.desc
	outcome = positive

	illustration_tags = {
		10 = military
	}

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			has_trait = cavalry_marshal
			has_trait = sufi_scholar
			NOT = { has_variable = cc_dual_1_done }
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				has_trait = cavalry_marshal
				has_trait = sufi_scholar
				NOT = { has_variable = cc_dual_1_done }
			}
			save_scope_as = minister
			set_variable = { name = cc_dual_1_done value = yes }
		}
	}

	# Warrior mystic modifier — war enthusiasm + conversion in conquered land
	option = {
		name = cc_dual.1.a
		add_country_modifier = { modifier = cc_dual_warrior_mystic years = 15 }
		add_prestige = prestige_mild_bonus
	}

	# Prestige + conversion speed boost
	option = {
		name = cc_dual.1.b
		add_prestige = prestige_mild_bonus
		add_country_modifier = { modifier = cc_cabinet_pop_conversion years = 8 }
	}
}

# cc_dual.10 — "A Crisis of Faith"
# reformation_inquisitor + reformist_theologian: irreconcilable convictions.
# TENSION event — these two traits on the same minister create internal conflict.
# Outcome: suppress doubt (zealous_fanatic risk) OR emerge transformed (both removed, crown_theologian added)
cc_dual.10 = {
	type = country_event
	title = cc_dual.10.title
	desc = cc_dual.10.desc
	outcome = negative

	illustration_tags = {
		10 = interior
	}

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			has_trait = reformation_inquisitor
			has_trait = reformist_theologian
			NOT = { has_variable = cc_dual_10_done }
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				has_trait = reformation_inquisitor
				has_trait = reformist_theologian
				NOT = { has_variable = cc_dual_10_done }
			}
			save_scope_as = minister
			set_variable = { name = cc_dual_10_done value = yes }
			set_variable = { name = cc_granting_trait value = yes }
		}
	}

	after = {
		scope:minister = {
			remove_variable = cc_granting_trait
		}
	}

	# Suppress the doubt — minister doubles down; zealous_fanatic risk
	option = {
		name = cc_dual.10.a
		scope:minister = {
			add_trait = trait:zealous_fanatic
		}
		add_estate_satisfaction = { type = estate_type:clergy_estate value = 0.05 }
	}

	# Emerge transformed — both removed, replaced by crown_theologian (rf synthesis)
	option = {
		name = cc_dual.10.b
		scope:minister = {
			remove_trait = trait:reformation_inquisitor
			remove_trait = trait:reformist_theologian
			add_trait = trait:crown_theologian
		}
		add_prestige = prestige_severe_bonus
		custom_tooltip = cc_dual.10.b.tt
	}

	# Dismiss the minister before the crisis deepens
	option = {
		name = cc_dual.10.c
		remove_from_cabinet = scope:minister
		add_prestige = prestige_weak_penalty
	}
}
