# Chain B: "The Merchant and the Sea" — cc_bonds.110–113
# A merchant from a colonial subject expands into a dynasty spanning multiple
# trading partners, potentially including foreign colonial nations.
# Spans ~32–50 years. Root = overlord throughout.
#
# State storage (on ROOT/overlord):
#   cc_bond_chain_b_active           — chain is in progress
#   cc_bond_chain_b_subj             — country-ref to primary subject
#   cc_bond_chain_b_secondary_map    — map of all secondary trading targets
#   cc_bond_merchant_rep             — reputation score 0–6
#   cc_bond_mem_merchant_patronized  — 110 option A chosen
#   cc_bond_mem_merchant_allowed     — 110 option B chosen
#   cc_bond_mem_merchant_denied      — 110 option C chosen
#   cc_bond_mem_merchant_backed      — 111 option A chosen (funded)
#   cc_bond_mem_merchant_rival       — 111 option C chosen (obstructed)
#   cc_bond_mem_crisis_managed       — 112 good option taken
#   cc_bond_mem_second_is_foreign    — secondary map contains a foreign nation
#   cc_bond_trade_link_active        — timed; keeps cc_merchant_trade_boost alive
#
# Per-secondary-target variables (stored on each target country):
#   cc_bond_secondary_is_foreign     — this country is not a subject of ROOT
#   cc_bond_merchant_interaction     — +1 backed, 0 neutral, -1 rival
###############################################################################

# cc_bonds.110 — "A Merchant Seeks Passage"
# Fires from cc_bonds_narrative_pulse (one-shot via cc_bond_110_cd).
cc_bonds.110 = {
	type = country_event
	title = cc_bonds.110.title
	desc = cc_bonds.110.desc
	illustration_tags = { 10 = bank }

	trigger = {
		has_game_rule = cc_subject_bonds_on
		NOT = { has_variable = cc_bond_110_cd_overlord }
		any_subject = {
			OR = {
				is_subject_type = colonial_nation
				is_subject_type = puppet_state
				is_subject_type = crown_dependency
			}
			num_locations >= 20
			has_variable = cc_bond_initialized
			NOT = { has_variable = cc_bond_110_cd }
		}
	}

	immediate = {
		set_variable = { name = cc_bond_110_cd_overlord  value = yes  years = { 30 80 } }
		random_subject = {
			limit = {
				OR = {
					is_subject_type = colonial_nation
					is_subject_type = puppet_state
					is_subject_type = crown_dependency
				}
				num_locations >= 20
				has_variable = cc_bond_initialized
				NOT = { has_variable = cc_bond_110_cd }
			}
			save_scope_as = bond_subj
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_110_cd  value = yes }
		}
		# Initialize chain state on overlord
		set_variable = { name = cc_bond_chain_b_subj    value = scope:bond_subj }
		set_variable = { name = cc_bond_merchant_rep    value = 0 }
		set_variable = { name = cc_bond_chain_b_active  value = yes }
	}

	# Option A: Grant full passage rights and a trade monopoly — eco+1, pol-1, rep+2
	option = {
		name = cc_bonds.110.a
		ai_chance = { base = 45 }
		set_variable = { name = cc_bond_mem_merchant_patronized  value = yes }
		change_variable = { name = cc_bond_merchant_rep  add = 2 }
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = 1 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = -1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.111  years = 10 }
	}

	# Option B: Grant passage only, no monopoly — eco+1, rep+1
	option = {
		name = cc_bonds.110.b
		ai_chance = { base = 45 }
		set_variable = { name = cc_bond_mem_merchant_allowed  value = yes }
		change_variable = { name = cc_bond_merchant_rep  add = 1 }
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = 1 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.111  years = 12 }
	}

	# Option C: Deny passage — eco-1, per-1; he finds another route (slower chain)
	option = {
		name = cc_bonds.110.c
		ai_chance = { base = 10 }
		set_variable = { name = cc_bond_mem_merchant_denied  value = yes }
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = -1 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = -1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.111  years = 14 }
	}
}

# cc_bonds.111 — "The Merchant's Expansion"
# The merchant has grown and is reaching beyond the primary subject's borders.
# One secondary trading partner is identified — an own qualifying subject or a
# foreign colonial not at war with ROOT. Options act directly on this single
# target. Options A and B persist the target in cc_bond_chain_b_secondary_map
# for use in 113; Option C does not (obstruction severs the tie).
cc_bonds.111 = {
	type = country_event
	title = cc_bonds.111.title
	illustration_tags = { 10 = bank }

	immediate = {
		# Recover primary subject
		var:cc_bond_chain_b_subj = { save_scope_as = bond_subj }

		# Pick one secondary target: own qualifying subject OR foreign colonial
		random_country = {
			limit = {
				OR = {
					AND = {
						is_subject_of = ROOT
						OR = {
							is_subject_type = colonial_nation
							is_subject_type = puppet_state
							is_subject_type = crown_dependency
						}
						NOT = { this = scope:bond_subj }
						num_locations >= 15
						has_variable = cc_bond_initialized
					}
					AND = {
						is_subject_type = colonial_nation
						NOT = { is_subject_of = ROOT }
						NOT = { this = scope:bond_subj }
						num_locations >= 15
						NOT = { is_at_war_with = ROOT }
					}
				}
			}
			save_scope_as = bond_secondary
			set_variable = { name = cc_bond_merchant_interaction  value = 0 }
			if = {
				limit = { NOT = { is_subject_of = ROOT } }
				set_variable = { name = cc_bond_secondary_is_foreign  value = yes }
				ROOT = { set_variable = { name = cc_bond_mem_second_is_foreign  value = yes } }
			}
		}
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = cc_bond_mem_merchant_patronized }
				desc = cc_bonds.111.desc_patronized
			}
			triggered_desc = {
				trigger = { has_variable = cc_bond_mem_merchant_allowed }
				desc = cc_bonds.111.desc_allowed
			}
			desc = cc_bonds.111.desc_denied
		}
	}
	# Option A: Fund the expansion — gold cost; 75% backed, 25% neutral; persisted to map
	option = {
		name = cc_bonds.111.a_back
		ai_chance = {
			base = 45
			modifier = {
				factor = 1.5
				has_variable = cc_bond_mem_merchant_patronized
			}
		}
		add_gold = { value = monthly_income_trade_and_tax  multiply = -0.25  min = -500 max = -250 }
		set_variable = { name = cc_bond_mem_merchant_backed  value = yes }
		change_variable = { name = cc_bond_merchant_rep  add = 1 }
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = 1 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
		}
		scope:bond_secondary = {
			random_list = {
				75 = {
					set_variable = { name = cc_bond_merchant_interaction  value = 1 }
					if = {
						limit = { NOT = { has_variable = cc_bond_secondary_is_foreign } }
						change_variable = { name = cc_bond_eco  add = 1 }
						clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
					}
				}
				25 = { }	# Funding spent but this target didn't convert
			}
		}
		# Persist: we backed it, so this contact remains in the network
		if = {
			limit = { scope:bond_secondary = { has_variable = cc_bond_secondary_is_foreign } }
			add_to_variable_map = { name = cc_bond_chain_b_secondary_map  key = scope:bond_secondary  value = 2 }
		}
		else = {
			add_to_variable_map = { name = cc_bond_chain_b_secondary_map  key = scope:bond_secondary  value = 1 }
		}
		trigger_event_silently = { id = cc_bonds.112  years = 12 }
	}

	# Option B: Observe — no cost; 40% backed, 20% rival, 40% neutral; persisted to map
	option = {
		name = cc_bonds.111.b_observe
		ai_chance = { base = 35 }
		scope:bond_secondary = {
			random_list = {
				40 = {
					# Merchant succeeds independently — acts as backed without our credit
					set_variable = { name = cc_bond_merchant_interaction  value = 1 }
					if = {
						limit = { NOT = { has_variable = cc_bond_secondary_is_foreign } }
						change_variable = { name = cc_bond_eco  add = 1 }
						clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
					}
				}
				20 = {
					# Merchant cuts us out of this route
					set_variable = { name = cc_bond_merchant_interaction  value = -1 }
					if = {
						limit = { NOT = { has_variable = cc_bond_secondary_is_foreign } }
						change_variable = { name = cc_bond_eco  add = -1 }
						clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
					}
				}
				40 = { }	# Stalemate — no movement on this target
			}
		}
		# Persist: we watched, so the relationship exists in the network regardless
		if = {
			limit = { scope:bond_secondary = { has_variable = cc_bond_secondary_is_foreign } }
			add_to_variable_map = { name = cc_bond_chain_b_secondary_map  key = scope:bond_secondary  value = 2 }
		}
		else = {
			add_to_variable_map = { name = cc_bond_chain_b_secondary_map  key = scope:bond_secondary  value = 1 }
		}
		trigger_event_silently = { id = cc_bonds.112  years = 14 }
	}

	# Option C: Clip his ambitions — 50% rival, 50% neutral; NOT persisted (obstruction severs the tie)
	option = {
		name = cc_bonds.111.c_discourage
		ai_chance = {
			base = 20
			modifier = {
				factor = 2.0
				has_variable = cc_bond_mem_merchant_denied
			}
		}
		set_variable = { name = cc_bond_mem_merchant_rival  value = yes }
		change_variable = { name = cc_bond_merchant_rep  add = -1 }
		# Local political interests satisfied; primary subject feels the squeeze
		change_variable = { name = cc_bond_pol  add = 1 }
		clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
		scope:bond_subj = {
			add_liberty_desire = liberty_desire_mild_plus
			change_variable = { name = cc_bond_per  add = -1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		scope:bond_secondary = {
			random_list = {
				50 = {
					# Obstruction works: own subjects shrink, foreign targets fill the gap
					set_variable = { name = cc_bond_merchant_interaction  value = -1 }
					if = {
						limit = { has_variable = cc_bond_secondary_is_foreign }
						# Foreign nation fills the gap left by our obstruction
						change_variable = { name = cc_bond_merchant_interaction  add = 2 }	# flip to +1
					}
					if = {
						limit = { NOT = { has_variable = cc_bond_secondary_is_foreign } }
						change_variable = { name = cc_bond_eco  add = -1 }
						clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
					}
				}
				50 = { }	# Merchant rerouted — obstruction had no lasting effect
			}
			# NOT persisted: clean up per-country vars since this target won't enter the map
			remove_variable = cc_bond_secondary_is_foreign
			remove_variable = cc_bond_merchant_interaction
		}
		remove_variable = cc_bond_mem_second_is_foreign
		trigger_event_silently = { id = cc_bonds.112  years = 12 }
	}
}

				trigger = { var:cc_bond_112_crisis = 2 }

		var:cc_bond_chain_b_subj = { save_scope_as = bond_subj }
		if = {
			limit = { has_variable = cc_bond_mem_merchant_backed }
			# Backed merchants are more exposed — piracy and rivals more likely

				30 = { set_variable = { name = cc_bond_112_crisis  value = 1 } }	# piracy

				10 = { set_variable = { name = cc_bond_112_crisis  value = 5 } }	# port blockade
			}
		}
		else = {
			random_list = {
				20 = { set_variable = { name = cc_bond_112_crisis  value = 1 } }
				15 = { set_variable = { name = cc_bond_112_crisis  value = 2 } }
				25 = { set_variable = { name = cc_bond_112_crisis  value = 3 } }
				25 = { set_variable = { name = cc_bond_112_crisis  value = 4 } }
				15 = { set_variable = { name = cc_bond_112_crisis  value = 5 } }

		scope:bond_subj = {
			change_variable = { name = cc_bond_mil  add = 1 }
			clamp_variable = { name = cc_bond_mil  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.113  years = 10 }
	}

	# Bad: Demand the subject handle its own security — bond_subj eco-1
	option = {
		name = cc_bonds.112.b_1_demand
		ai_chance = { base = 45 }
		trigger = { var:cc_bond_112_crisis = 1 }
		change_variable = { name = cc_bond_merchant_rep  add = -1 }

	############################################################

	option = {
		name = cc_bonds.112.a_2_buyout
		ai_chance = { base = 50 }
		trigger = { var:cc_bond_112_crisis = 2 }
		add_gold = { value = monthly_income_trade_and_tax  multiply = -0.30  min = -60 }

# cc_bonds.113 — "The Merchant's Legacy"
# Terminal event. Reputation score determines outcome tier (1/2/3).
# Grants cc_merchant_trade_boost (scripted relation) to bond_subj,
# and per-target modifiers to secondary targets based on stored interaction data.
cc_bonds.113 = {
	type = country_event
	title = cc_bonds.113.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:cc_bond_113_outcome = 3 }
				desc = cc_bonds.113.desc_flourish
			}
			triggered_desc = {
				trigger = { var:cc_bond_113_outcome = 2 }
				desc = cc_bonds.113.desc_modest
			}
			desc = cc_bonds.113.desc_collapse
		}
	}
	illustration_tags = { 10 = bank }

	immediate = {
		var:cc_bond_chain_b_subj = { save_scope_as = bond_subj }
		if = {
			limit = { var:cc_bond_merchant_rep >= 4 }
			set_variable = { name = cc_bond_113_outcome  value = 3 }
		}
		else_if = {
			limit = { var:cc_bond_merchant_rep >= 2 }
			set_variable = { name = cc_bond_113_outcome  value = 2 }
		}
		else = {
			set_variable = { name = cc_bond_113_outcome  value = 1 }
		}
	}

	after = {
		# Clean up chain state
		remove_variable = cc_bond_chain_b_active
		remove_variable = cc_bond_chain_b_subj
		remove_variable = cc_bond_merchant_rep
		remove_variable = cc_bond_113_outcome
		remove_variable = cc_bond_mem_merchant_patronized
		remove_variable = cc_bond_mem_merchant_allowed
		remove_variable = cc_bond_mem_merchant_denied
		remove_variable = cc_bond_mem_merchant_backed
		remove_variable = cc_bond_mem_merchant_rival
		remove_variable = cc_bond_mem_crisis_managed
		remove_variable = cc_bond_mem_second_is_foreign
		# Clean up per-country variables on each secondary target, then clear the map
		every_key_in_variable_map = {
			variable = cc_bond_chain_b_secondary_map
			remove_variable = cc_bond_secondary_is_foreign
			remove_variable = cc_bond_merchant_interaction
		}
		remove_variable = cc_bond_chain_b_secondary_map
	}

	############################################################
	# Option A: Grant the house a formal trading charter (outcome >= 2)
	# Grants cc_merchant_trade_boost to primary subject.
	# Iterates secondary map: own subjects get relation, foreign nations get modifiers.
	############################################################
	option = {
		name = cc_bonds.113.a_charter
		ai_chance = {
			base = 60
			modifier = {
				factor = 0
				var:cc_bond_113_outcome = 1
			}
		}
		trigger = { var:cc_bond_113_outcome >= 2 }

		# Primary subject — grant trade boost relation
		create_relation = {
			type = relation_type:cc_merchant_trade_boost
			first = ROOT
			second = scope:bond_subj
		}
		# Timed variable that keeps the relation alive (will_expire_trigger watches this)
		set_variable = { name = cc_bond_trade_link_active  value = yes  years = { 20 40 } }
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = 2 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = 1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
		}

		# Secondary targets — iterate map for per-target effects
		every_key_in_variable_map = {
			variable = cc_bond_chain_b_secondary_map
			save_scope_as = secondary_target

			# Own subject with positive interaction → trade boost relation
			if = {
				limit = {
					var:cc_bond_merchant_interaction >= 1
					NOT = { has_variable = cc_bond_secondary_is_foreign }
				}
				ROOT = {
					create_relation = {
						type = relation_type:cc_merchant_trade_boost
						first = ROOT
						second = scope:secondary_target
					}
				}
			}

			# Foreign target with positive interaction — overlord DID fund it
			# Overlord places a merchant trade post in the foreign nation's market center.
			# The building's foreign_country_modifier provides ongoing trade benefits to ROOT.
			if = {
				limit = {
					var:cc_bond_merchant_interaction >= 1
					has_variable = cc_bond_secondary_is_foreign
					ROOT = { has_variable = cc_bond_mem_merchant_backed }
				}
				# Foreign nation gets preferential trading access
				add_country_modifier = { modifier = cc_merchant_trading_rights  years = { 20 35 } }
				# Place the trade post in the market center of this nation's capital market
				capital = {
					market = {
						random_location_in_market = {
							limit = { is_market_center = yes }
							change_building_level_in_location = {
								building = building_type:cc_merchant_trade_post
								value = 1
								owner = root
							}
						}
					}
				}
				# Track this target so the cleanup event can find and remove the building
				ROOT = {
					set_variable = { name = cc_bond_trade_post_target  value = scope:secondary_target }
					trigger_event_silently = { id = cc_bonds.115  years = { 20 40 } }
				}
			}

			# Foreign target with positive interaction — overlord did NOT fund it
			# Only the foreign nation benefits from the organic merchant connection
			if = {
				limit = {
					var:cc_bond_merchant_interaction >= 1
					has_variable = cc_bond_secondary_is_foreign
					ROOT = { NOT = { has_variable = cc_bond_mem_merchant_backed } }
				}
				add_country_modifier = { modifier = cc_merchant_trading_rights  years = { 15 25 } }
			}
		}
	}

	############################################################
	# Option B: Accept the legacy as it stands (outcome >= 2)
	# Simpler — trade boost to primary only; secondary targets get trading rights.
	############################################################
	option = {
		name = cc_bonds.113.b_accept
		ai_chance = {
			base = 40
			modifier = {
				factor = 0
				var:cc_bond_113_outcome = 1
			}
		}
		trigger = { var:cc_bond_113_outcome >= 2 }

		create_relation = {
			type = relation_type:cc_merchant_trade_boost
			first = ROOT
			second = scope:bond_subj
		}
		set_variable = { name = cc_bond_trade_link_active  value = yes  years = { 20 25 } }
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = 1 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
		}

		# Secondary targets with any positive interaction get trading rights
		every_key_in_variable_map = {
			variable = cc_bond_chain_b_secondary_map
			if = {
				limit = { var:cc_bond_merchant_interaction >= 1 }
				add_country_modifier = { modifier = cc_merchant_trading_rights  years = { 15 20 } }
			}
		}
	}

	############################################################
	# Option C: The house stands alone (any outcome; only option when outcome = 1)
	############################################################
	option = {
		name = cc_bonds.113.c_decline
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				var:cc_bond_113_outcome = 1
			}
		}
		if = {
			limit = { var:cc_bond_113_outcome = 1 }
			scope:bond_subj = {
				change_variable = { name = cc_bond_eco  add = -1 }
				clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
				change_variable = { name = cc_bond_per  add = -1 }
				clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
				add_liberty_desire = liberty_desire_mild_plus
			}
		}
	}
}

# cc_bonds.115 — Merchant Trade Post Cleanup (hidden)
# Fires on ROOT (overlord) after 20–40 years, scheduled by event 113 option A.
# Removes all cc_merchant_trade_post buildings ROOT placed in foreign nations.
# Uses every_owned_foreign_building_location to find them without needing to
# know exactly which country's market center holds the building today.
cc_bonds.115 = {
	type = country_event
	hidden = yes
	title = cc_bonds.115.title

	immediate = {
		# Remove any cc_merchant_trade_post buildings owned by ROOT in foreign locations
		every_owned_foreign_building_location = {
			limit = {
				any_foreign_buildings_in_location = {
					building_type = building_type:cc_merchant_trade_post
					owner = root
				}
			}
			every_foreign_buildings_in_location = {
				limit = {
					building_type = building_type:cc_merchant_trade_post
					owner = root
				}
				location = {
					destroy_building = prev
				}
			}
		}
		remove_variable = cc_bond_trade_post_target
	}

	option = { name = cc_bonds.115.a }
}
