# cc_bonds.60 — "The March Lord's Formal Investiture" (one-shot)
# Option C (minimal ceremony) fires a false positive — lord "proves himself."
cc_bonds.60 = {
	type = country_event
	title = cc_bonds.60.title
	desc = cc_bonds.60.desc
	illustration_tags = { 10 = military }

	trigger = {
		has_game_rule = cc_subject_bonds_on
		any_subject = {
			is_subject_type = military_march
			num_locations >= 10
			has_variable = cc_bond_initialized
			NOT = { has_variable = cc_bond_60_cd }
		}
	}

	immediate = {
		random_subject = {
			limit = {
				is_subject_type = military_march
				num_locations >= 10
				has_variable = cc_bond_initialized
				NOT = { has_variable = cc_bond_60_cd }
			}
			save_scope_as = bond_subj
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_60_cd  value = yes }
		}
	}

	# Grand ceremony, full noble privileges — prestige cost, mil/pol/per gain
	option = {
		name = cc_bonds.60.a
		add_prestige = prestige_weak_penalty
		scope:bond_subj = {
			change_variable = { name = cc_bond_mil  add = 1 }
			clamp_variable = { name = cc_bond_mil  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = 1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = 3 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
	}

	# Standard investiture — mil/per gain
	option = {
		name = cc_bonds.60.b
		scope:bond_subj = {
			change_variable = { name = cc_bond_mil  add = 1 }
			clamp_variable = { name = cc_bond_mil  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = 1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
	}

	# Minimal ceremony — mil/per damage, false positive fires
	option = {
		name = cc_bonds.60.c
		scope:bond_subj = {
			change_variable = { name = cc_bond_mil  add = -1 }
			clamp_variable = { name = cc_bond_mil  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = -2 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.70  months = 8 }
	}
}
