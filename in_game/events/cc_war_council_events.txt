###############################################################################
# FLAVORIUM UNIVERSALIS — WAR COUNCIL EVENTS (Pillar 04)
#
# Section A — Active-war council events:
#   cc_wc.1  The Council of War       (yearly while at war; military traits shape options)
#   cc_wc.2  Financing the War        (yearly while at war and treasury is strained)
#   cc_wc.3  The General's Request    (yearly while at war; supreme_commander or outdated_general)
#
# Section B — Post-war military reform proposals (require cc_military_reform_opportunity):
#   cc_wc.10 Standing Army Reform     (standing_army_advocate present)
#   cc_wc.11 Supply Reform            (commissariat_officer present)
#   cc_wc.12 Fortification Doctrine   (siege_engineer present)
#
# Section C — Outdated general resolution:
#   cc_wc.20 Formal Council Retirement (outdated_general present, at war)
#
# Pulse hooks: cc_cabinet_yearly_war_council (A events) and
#              cc_cabinet_yearly_war_reform (B events) in cc_cabinet_pulse.txt.
# On-action hook: cc_wc_on_winning_war in cc_war_on_actions.txt sets the reform flag.
###############################################################################


###############################################################################
# SECTION A — ACTIVE-WAR COUNCIL EVENTS
###############################################################################

# "The Council of War" — fires yearly while at war when a military minister is present.
# Ministers with standing_army_advocate, defensive_commander, or hawk_minister shape the options.
cc_wc.1 = {
	type = country_event
	title = cc_wc.1.title
	desc = cc_wc.1.desc
	outcome = neutral

	illustration_tags = {
		10 = military
		10 = armed
	}

	trigger = {
		at_war = yes
		NOT = { has_variable = cc_wc_council_cooldown }
		any_cabinet_character = {
			OR = {
				has_trait = standing_army_advocate
				has_trait = defensive_commander
				has_trait = hawk_minister
				has_trait = tactical_advisor
				has_trait = commissariat_officer
				has_trait = siege_engineer
				has_trait = supreme_commander
				has_trait = offensive_strategist
			}
		}
	}

	immediate = {
		set_variable = { name = cc_wc_council_cooldown value = yes months = 18 }
		random_cabinet_character = {
			limit = {
				OR = {
					has_trait = standing_army_advocate
					has_trait = defensive_commander
					has_trait = hawk_minister
					has_trait = tactical_advisor
					has_trait = commissariat_officer
					has_trait = siege_engineer
					has_trait = supreme_commander
					has_trait = offensive_strategist
				}
			}
			save_scope_as = minister
		}
	}

	# Option A — standing_army_advocate or offensive_strategist: push for aggressive campaign
	option = {
		name = cc_wc.1.a
		trigger = {
			scope:minister = {
				OR = {
					has_trait = standing_army_advocate
					has_trait = offensive_strategist
				}
			}
		}
		add_country_modifier = {
			modifier = cc_wc_offensive_surge
			years = 4
			mode = set_to_largest_and_extend
		}
		add_stability = stability_weak_penalty
	}

	# Option B — defensive_commander: counsel patience and dig in
	option = {
		name = cc_wc.1.b
		trigger = {
			scope:minister = { has_trait = defensive_commander }
		}
		add_country_modifier = {
			modifier = cc_wc_defensive_posture
			years = 4
			mode = set_to_largest_and_extend
		}
	}

	# Option C — hawk_minister: pledge no peace short of total victory
	option = {
		name = cc_wc.1.c
		trigger = {
			scope:minister = { has_trait = hawk_minister }
		}
		add_prestige = prestige_mild_bonus
		set_variable = { name = cc_hawk_war_pledge value = yes months = 24 }
	}

	# Option D — tactical_advisor, commissariat_officer, siege_engineer, or supreme_commander: competent counsel
	option = {
		name = cc_wc.1.d
		trigger = {
			scope:minister = {
				NOT = { has_trait = standing_army_advocate }
				NOT = { has_trait = offensive_strategist }
				NOT = { has_trait = defensive_commander }
				NOT = { has_trait = hawk_minister }
			}
		}
		add_prestige = prestige_weak_bonus
		scope:minister = { add_mil = 1 }
	}
}

# "Financing the War" — fires yearly while at war and treasury is under pressure.
# Ministers with treasury expertise offer spending strategies; without them only painful defaults remain.
cc_wc.2 = {
	type = country_event
	title = cc_wc.2.title
	desc = cc_wc.2.desc
	outcome = neutral

	illustration_tags = {
		10 = bank
		10 = economy
	}

	trigger = {
		at_war = yes
		NOT = { has_variable = cc_wc_finance_cooldown }
		OR = {
			gold < 200
			NOT = { gold >= 0 }
		}
		any_cabinet_character = {
			OR = {
				has_trait = efficiency_administrator
				has_trait = treasury_enforcer
				has_trait = prosperity_planner
			}
		}
	}

	immediate = {
		set_variable = { name = cc_wc_finance_cooldown value = yes months = 24 }
		random_cabinet_character = {
			limit = {
				OR = {
					has_trait = efficiency_administrator
					has_trait = treasury_enforcer
					has_trait = prosperity_planner
				}
			}
			save_scope_as = minister
		}
	}

	# Option A — efficiency_administrator: squeeze the population (manpower at cost of loyalty)
	option = {
		name = cc_wc.2.a
		trigger = {
			scope:minister = {
				OR = {
					has_trait = efficiency_administrator
					has_trait = treasury_enforcer
				}
			}
		}
		add_gold = { value = monthly_income_trade_and_tax  multiply = 3  min = 60 }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_penalty }
		add_stability = stability_weak_penalty
	}

	# Option B — prosperity_planner: take war loans (gold now, inflation later)
	option = {
		name = cc_wc.2.b
		trigger = {
			scope:minister = { has_trait = prosperity_planner }
		}
		add_gold = { value = monthly_income_trade_and_tax  multiply = 5  min = 100 }
		add_inflation = inflation_mild_penalty
	}

	# Option C — weather the shortage (always available)
	option = {
		name = cc_wc.2.c
		add_stability = stability_weak_penalty
	}
}

# "The General's Request" — fires when a supreme_commander or outdated_general is in cabinet during war.
# Granting resources boosts morale; refusing sours the relationship.
cc_wc.3 = {
	type = country_event
	title = cc_wc.3.title
	desc = cc_wc.3.desc
	outcome = neutral

	illustration_tags = {
		10 = military
		10 = armed
	}

	trigger = {
		at_war = yes
		NOT = { has_variable = cc_wc_general_request_cooldown }
		any_cabinet_character = {
			OR = {
				has_trait = supreme_commander
				has_trait = outdated_general
			}
		}
	}

	immediate = {
		set_variable = { name = cc_wc_general_request_cooldown value = yes months = 30 }
		random_cabinet_character = {
			limit = {
				OR = {
					has_trait = supreme_commander
					has_trait = outdated_general
				}
			}
			save_scope_as = minister
		}
	}

	# Option A — grant resources (costs gold, awards morale surge)
	option = {
		name = cc_wc.3.a
		add_gold = { value = monthly_income_trade_and_tax  multiply = -2  max = -50 }
		add_country_modifier = {
			modifier = cc_wc_offensive_surge
			years = 3
			mode = set_to_largest_and_extend
		}
		scope:minister = { add_mil = 1 }
	}

	# Option B — refuse (minister relationship sours)
	option = {
		name = cc_wc.3.b
		scope:minister = {
			set_variable = { name = cc_minister_defied value = yes }
			add_character_modifier = { modifier = cc_minister_defied_modifier  years = 12  mode = add_and_extend }
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_penalty }
	}
}

# "The Standing Army Proposal" — fires after a victorious war if a standing_army_advocate is present.
# Nobles resist the loss of levy influence; the reform grants lasting discipline.
cc_wc.10 = {
	type = country_event
	title = cc_wc.10.title
	desc = cc_wc.10.desc
	outcome = positive

	illustration_tags = {
		10 = military
		10 = army
	}

	trigger = {
		has_variable = cc_military_reform_opportunity
		NOT = { has_country_modifier = cc_standing_army_reform }
		any_cabinet_character = { has_trait = standing_army_advocate }
	}

	immediate = {
		random_cabinet_character = {
			limit = { has_trait = standing_army_advocate }
			save_scope_as = minister
		}
	}

	# Option A — enact standing army reform (nobles displeasure, lasting discipline)
	option = {
		name = cc_wc.10.a
		add_country_modifier = {
			modifier = cc_standing_army_reform
			years = 15
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }
		remove_variable = cc_military_reform_opportunity
	}

	# Option B — decline for now
	option = {
		name = cc_wc.10.b
		remove_variable = cc_military_reform_opportunity
	}
}

# "The Hawk's Reckoning" — fires from on_pre_ending_war via cc_hawk_reckoning_pre_end
# when the country ends a war while a hawk_minister's pledge (cc_hawk_war_pledge) is active.
# The hawk swore no peace would be made short of total victory. Peace has been made.
cc_wc.50 = {
	type = country_event
	title = cc_wc.50.title
	desc = cc_wc.50.desc
	outcome = negative

	illustration_tags = {
		10 = military
		10 = angry
	}

	trigger = {
		any_cabinet_character = { has_trait = hawk_minister }
	}

	immediate = {
		remove_variable = cc_hawk_war_pledge
		random_cabinet_character = {
			limit = { has_trait = hawk_minister }
			save_scope_as = minister
		}
	}

	# Assert authority — the peace was necessary; the minister must accept it
	option = {
		name = cc_wc.50.a
		add_prestige = prestige_weak_bonus
		scope:minister = {
			set_variable = { name = cc_minister_resentful value = yes }
			add_character_modifier = { modifier = cc_minister_slighted  years = 15  mode = add_and_extend }
			cc_minister_estate_slight = yes
		}
	}

	# Acknowledge the peace was unfortunate — offer the hawk symbolic concession
	option = {
		name = cc_wc.50.b
		add_prestige = prestige_weak_penalty
		scope:minister = { add_mil = 1 }
	}

	# Promise the next campaign — hawk stays loyal, but you accept an obligation
	option = {
		name = cc_wc.50.c
		add_prestige = prestige_mild_penalty
		add_country_modifier = {
			modifier = cc_wc_offensive_surge
			years = 5
			mode = set_to_largest_and_extend
		}
	}
}

# "The General's Resentment" — fires biyearly when a cabinet character has cc_minister_defied.
# Set by cc_wc.3 option B (ruler refused the general's resource request during war).
# The minister's grievance has festered long enough to require resolution.
cc_wc.51 = {
	type = country_event
	title = cc_wc.51.title
	desc = cc_wc.51.desc
	outcome = negative

	illustration_tags = {
		10 = military
		10 = angry
	}

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			has_variable = cc_minister_defied
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				has_variable = cc_minister_defied
			}
			save_scope_as = minister
		}
	}

	# Apologize and compensate — gold cost, minister forgiven, add mil skill
	option = {
		name = cc_wc.51.a
		add_gold = { value = monthly_income_trade_and_tax  multiply = -3  min = -100 }
		scope:minister = {
			remove_variable = cc_minister_defied
			remove_character_modifier = cc_minister_defied_modifier
			add_mil = 1
		}
	}

	# Grant a symbolic recognition — prestige cost; the active grievance is defused, but a
	# residue of friction lingers (a decaying marker, not the full resentment of option d).
	option = {
		name = cc_wc.51.b
		add_prestige = prestige_weak_penalty
		scope:minister = {
			remove_variable = cc_minister_defied
			remove_character_modifier = cc_minister_defied_modifier
			set_variable = { name = cc_minister_grievance_noted value = yes years = 3 }
		}
	}

	# Dismiss the insubordination — minister leaves resentful
	option = {
		name = cc_wc.51.c
		add_stability = stability_weak_penalty
		scope:minister = {
			remove_variable = cc_minister_defied
			remove_character_modifier = cc_minister_defied_modifier
			set_variable = { name = cc_dismissed_resentful value = yes }
		}
	}

	# Ignore it entirely — minister escalates to full resentment, mil penalty
	option = {
		name = cc_wc.51.d
		scope:minister = {
			remove_variable = cc_minister_defied
			remove_character_modifier = cc_minister_defied_modifier
			set_variable = { name = cc_minister_resentful value = yes }
			add_character_modifier = { modifier = cc_minister_slighted  years = 15  mode = add_and_extend }
			cc_minister_estate_slight = yes
		}
		add_prestige = prestige_weak_penalty
	}
}

# "A Council Decision on the Old General" — fires during war when an outdated_general is present.
# The council can formally retire him with dignity, clearing the negative trait.
cc_wc.20 = {
	type = country_event
	title = cc_wc.20.title
	desc = cc_wc.20.desc
	outcome = neutral

	illustration_tags = {
		10 = military
	}

	trigger = {
		at_war = yes
		NOT = { has_variable = cc_wc_outdated_general_cooldown }
		any_cabinet_character = { has_trait = outdated_general }
	}

	immediate = {
		set_variable = { name = cc_wc_outdated_general_cooldown value = yes months = 36 }
		random_cabinet_character = {
			limit = { has_trait = outdated_general }
			save_scope_as = minister
		}
	}

	# Option A — formal council retirement (clear the trait; costs prestige)
	option = {
		name = cc_wc.20.a
		add_prestige = prestige_mild_penalty
		scope:minister = {
			remove_trait = trait:outdated_general
		}
	}

	# Option B — let the old soldier serve out the campaign
	option = {
		name = cc_wc.20.b
		add_prestige = prestige_weak_bonus
	}
}
