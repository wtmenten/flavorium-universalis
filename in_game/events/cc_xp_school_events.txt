# Flavorium Universalis — Cabinet Experience: school notifications
#
#   cc_xp_school.1   A school has moved from a lost seat to another university
#   cc_xp_school.2   A school has lost its seat and has nowhere to go: exile
#   cc_xp_school.3   An exiled school has a university again
#   cc_xp_school.11-17  Disputations, one per curriculum (Hold a Disputation)
#   cc_xp_school.20  Change Curriculum
#   cc_xp_school.30-32  Renown draws a scholar, a foreign student, an artist
#   cc_xp_school.40-42  The rival university poaches a master, draws students; first place
#
# Fired by cc_xp_school_find_seat (scripted_effects/cc_xp_school_effects.txt), human
# courts only. Each finds its school by a short-lived per-slot flag rather than a saved
# scope, the same way the cc_xp choice events find their subject, so two schools moving
# in the same month raise two events and each names its own school.
###############################################################################


# Character and seats of the first slot carrying cc_xp_school_N_$FLAG$, saved for loc,
# and that slot's number in cc_xp_school_active for the options. Country scope (the
# event's root).
scripted_effect cc_xp_school_take_notice = {
	if = {
		limit = { has_variable = cc_xp_school_1_$FLAG$ }
		remove_variable = cc_xp_school_1_$FLAG$
		set_variable = { name = cc_xp_school_active  value = 1 }
		var:cc_xp_school_1_founder ?= { save_scope_as = cc_xp_school_founder }
		var:cc_xp_school_1_loc ?= { save_scope_as = cc_xp_school_new_seat }
		var:cc_xp_school_1_prev_loc ?= { save_scope_as = cc_xp_school_old_seat }
	}
	else_if = {
		limit = { has_variable = cc_xp_school_2_$FLAG$ }
		remove_variable = cc_xp_school_2_$FLAG$
		set_variable = { name = cc_xp_school_active  value = 2 }
		var:cc_xp_school_2_founder ?= { save_scope_as = cc_xp_school_founder }
		var:cc_xp_school_2_loc ?= { save_scope_as = cc_xp_school_new_seat }
		var:cc_xp_school_2_prev_loc ?= { save_scope_as = cc_xp_school_old_seat }
	}
	else_if = {
		limit = { has_variable = cc_xp_school_3_$FLAG$ }
		remove_variable = cc_xp_school_3_$FLAG$
		set_variable = { name = cc_xp_school_active  value = 3 }
		var:cc_xp_school_3_founder ?= { save_scope_as = cc_xp_school_founder }
		var:cc_xp_school_3_loc ?= { save_scope_as = cc_xp_school_new_seat }
		var:cc_xp_school_3_prev_loc ?= { save_scope_as = cc_xp_school_old_seat }
	}
}

###############################################################################
# SHARED HELPERS FOR THE ACTIVE SCHOOL
#
# cc_xp_school_take_notice records which slot an event is about in cc_xp_school_active.
# These resolve that number into the slot effects and triggers, which take N literally.
###############################################################################

# Country scope. Members of the active school gain AMOUNT in its discipline.
scripted_effect cc_xp_school_active_members_gain = {
	if = {
		limit = { var:cc_xp_school_active ?= 1 }
		cc_xp_school_members_gain = { N = 1  AMOUNT = $AMOUNT$ }
	}
	else_if = {
		limit = { var:cc_xp_school_active ?= 2 }
		cc_xp_school_members_gain = { N = 2  AMOUNT = $AMOUNT$ }
	}
	else_if = {
		limit = { var:cc_xp_school_active ?= 3 }
		cc_xp_school_members_gain = { N = 3  AMOUNT = $AMOUNT$ }
	}
}

# Country scope. Push the active school's renown and quality.
scripted_effect cc_xp_school_active_change = {
	if = {
		limit = { var:cc_xp_school_active ?= 1 }
		cc_xp_school_change = { N = 1  RENOWN = $RENOWN$  QUALITY = $QUALITY$ }
	}
	else_if = {
		limit = { var:cc_xp_school_active ?= 2 }
		cc_xp_school_change = { N = 2  RENOWN = $RENOWN$  QUALITY = $QUALITY$ }
	}
	else_if = {
		limit = { var:cc_xp_school_active ?= 3 }
		cc_xp_school_change = { N = 3  RENOWN = $RENOWN$  QUALITY = $QUALITY$ }
	}
}

# Country scope. The active school's quality is at least Q.
scripted_trigger cc_xp_school_active_quality = {
	custom_tooltip = {
		text = cc_xp_tt_school_quality_$Q$
		OR = {
			AND = { var:cc_xp_school_active ?= 1  has_variable = cc_xp_school_1_quality  var:cc_xp_school_1_quality >= $Q$ }
			AND = { var:cc_xp_school_active ?= 2  has_variable = cc_xp_school_2_quality  var:cc_xp_school_2_quality >= $Q$ }
			AND = { var:cc_xp_school_active ?= 3  has_variable = cc_xp_school_3_quality  var:cc_xp_school_3_quality >= $Q$ }
		}
	}
}

# Country scope. Enrol WHO in the active school.
scripted_effect cc_xp_school_active_enroll = {
	if = {
		limit = { var:cc_xp_school_active ?= 1 }
		$WHO$ = {
			set_variable = { name = cc_xp_school        value = 1 }
			set_variable = { name = cc_xp_school_track  value = root.var:cc_xp_school_1_track }
		}
	}
	else_if = {
		limit = { var:cc_xp_school_active ?= 2 }
		$WHO$ = {
			set_variable = { name = cc_xp_school        value = 2 }
			set_variable = { name = cc_xp_school_track  value = root.var:cc_xp_school_2_track }
		}
	}
	else_if = {
		limit = { var:cc_xp_school_active ?= 3 }
		$WHO$ = {
			set_variable = { name = cc_xp_school        value = 3 }
			set_variable = { name = cc_xp_school_track  value = root.var:cc_xp_school_3_track }
		}
	}
}

scripted_trigger cc_xp_school_disputing = {
	OR = {
		has_variable = cc_xp_school_1_disputing
		has_variable = cc_xp_school_2_disputing
		has_variable = cc_xp_school_3_disputing
	}
}

###############################################################################
# CHANGE CURRICULUM (cc_xp_school.20)
#
# A school chooses among the five curricula of its discipline. Some are gated by what the
# school can teach: Natural Philosophy needs quality 40, Medicine quality 50, Navigation a
# coastal seat, Gunnery the Age of Discovery. A school keeps its curriculum if its quality
# later falls; the gate is only on choosing it.
###############################################################################

scripted_effect cc_xp_school_active_set_curriculum = {
	if = {
		limit = { var:cc_xp_school_active ?= 1 }
		cc_xp_school_set_curriculum = { N = 1  C = $C$  KEY = $KEY$ }
	}
	else_if = {
		limit = { var:cc_xp_school_active ?= 2 }
		cc_xp_school_set_curriculum = { N = 2  C = $C$  KEY = $KEY$ }
	}
	else_if = {
		limit = { var:cc_xp_school_active ?= 3 }
		cc_xp_school_set_curriculum = { N = 3  C = $C$  KEY = $KEY$ }
	}
}

# Country scope. The active school currently has setting KEY at value V.
scripted_trigger cc_xp_school_active_is = {
	OR = {
		AND = { var:cc_xp_school_active ?= 1  var:cc_xp_school_1_$KEY$ ?= $V$ }
		AND = { var:cc_xp_school_active ?= 2  var:cc_xp_school_2_$KEY$ ?= $V$ }
		AND = { var:cc_xp_school_active ?= 3  var:cc_xp_school_3_$KEY$ ?= $V$ }
	}
}

				create_in_limbo = yes
				save_scope_as = cc_xp_school_scholar
			}
		}

				OR = {
					AND = { var:cc_xp_school_active ?= 1  var:cc_xp_school_1_renown >= 60 }
					AND = { var:cc_xp_school_active ?= 2  var:cc_xp_school_2_renown >= 60 }

				dip = { 45 75 }

				culture = scope:cc_xp_school_new_seat.dominant_culture

				save_scope_as = cc_xp_school_scholar
