namespace = cc_legend

###############################################################################
# LEGEND QUEST CHAINS
# Each quest fires once when conditions are fully met.
# Legendary traits:   allow = { has_variable = cc_granting_trait } — granted only here.
# Legendary reforms:  gated by has_variable set in these events.
# Variable keys set: cc_legend_X_unlocked (checked in reform potential/allow)
###############################################################################

# Quest 1: "The Philosopher's Court"
# Requires: dev > 50, innovative > 50, cabinet has enlightened/master_urbanist, all cabinet members admin
cc_legend.1 = {
	type = country_event
	title = cc_legend.1.title
	desc = cc_legend.1.desc
	outcome = positive

	illustration_tags = {
		10 = interior
	}

	trigger = {
		capital.development > 50
		societal_value:traditionalist_vs_innovative > 50
		any_cabinet_character = {
			in_cabinet = yes
			OR = {
				has_trait = enlightenment_herald
				has_trait = master_urbanist
				has_trait = progressive_rationalist
			}
		}
		NOT = { has_variable = cc_legend_1_fired }
	}

	immediate = {
		set_variable = { name = cc_legend_1_fired value = yes }
		ruler = {
			set_variable = { name = cc_granting_trait value = yes }
		}
	}
	after = {
		ruler = {
			remove_variable = cc_granting_trait
		}
	}

	# Philosophers Court Reform + ruler legendary trait
	option = {
		name = cc_legend.1.a
		set_variable = { name = cc_legend_philosophers_court_unlocked value = yes }
		ruler = {
			add_trait = trait:philosopher_king
		}
		add_prestige = prestige_severe_bonus
		custom_tooltip = cc_legend.1.a.tt
	}

	# Prestige only
	option = {
		name = cc_legend.1.b
		add_prestige = prestige_severe_bonus
		add_country_modifier = { modifier = cc_synergy_enlightened_court years = 20 }
	}
}

# Quest 2: "The Holy Empire"
# Requires: theocracy or christian, clergy >= 60%, crown_theologian + reformation_preacher/orthodox_defender in cabinet, age 5-6
cc_legend.2 = {
	type = country_event
	title = cc_legend.2.title
	desc = cc_legend.2.desc
	outcome = positive

	illustration_tags = {
		10 = interior
	}

	trigger = {
		OR = {
			government_type = government_type:theocracy
			religion.group = religion_group:christian
		}
		"estate_power(estate_type:clergy_estate)" >= 0.60
		any_cabinet_character = {
			in_cabinet = yes
			has_trait = mystical_patriarch
		}
		any_cabinet_character = {
			in_cabinet = yes
			OR = {
				has_trait = ecumenical_patriarch_advisor
				has_trait = traditionalist_guardian
			}
		}
		OR = {
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
		NOT = { has_variable = cc_legend_2_fired }
	}

	immediate = {
		set_variable = { name = cc_legend_2_fired value = yes }
	}

	# Clergy privilege + permanent holy empire modifier
	option = {
		name = cc_legend.2.a
		grant_estate_privilege = estate_privilege:clergy_strengthen_faith
		add_country_modifier = { modifier = cc_legend_holy_empire years = 999 }
		add_prestige = prestige_severe_bonus
		custom_tooltip = cc_legend.2.a.tt
	}

	# Prestige + clergy satisfaction
	option = {
		name = cc_legend.2.b
		add_prestige = prestige_severe_bonus
		add_estate_satisfaction = { type = estate_type:clergy_estate value = 0.15 }
	}
}

# Quest 3: "The Grand Bazaar"
# Requires: muslim, >= 2 chartered_company or associated_republic subjects, grand_chancellor in cabinet
cc_legend.3 = {
	type = country_event
	title = cc_legend.3.title
	desc = cc_legend.3.desc
	outcome = positive

	illustration_tags = {
		10 = interior
	}

	trigger = {
		religion.group = religion_group:muslim
		any_cabinet_character = {
			in_cabinet = yes
			has_trait = grand_chancellor
		}
		any_subject = {
			OR = {
				is_subject_type = chartered_company
				is_subject_type = associated_republic
			}
		}
		NOT = { has_variable = cc_legend_3_fired }
	}

	immediate = {
		set_variable = { name = cc_legend_3_fired value = yes }
		random_cabinet_character = {
			limit = { in_cabinet = yes has_trait = grand_chancellor }
			set_variable = { name = cc_granting_trait value = yes }
			save_scope_as = minister
		}
	}

	after = {
		ruler = {
			remove_variable = cc_granting_trait
		}
	}

	# Grand Bazaar Reform + legendary grand vizier trait
	option = {
		name = cc_legend.3.a
		set_variable = { name = cc_legend_grand_bazaar_unlocked value = yes }
		scope:minister = {
			add_trait = trait:grand_vizier_legendary
		}
		add_prestige = prestige_severe_bonus
		custom_tooltip = cc_legend.3.a.tt
	}

	# Prestige + diplomatic capacity
	option = {
		name = cc_legend.3.b
		add_prestige = prestige_severe_bonus
		add_country_modifier = { modifier = cc_synergy_diplomatic_network years = 20 }
	}
}

# Quest 4: "The Iron Hegemon"
# Requires: military_march subject, supreme_commander minister, iron_disciplinarian ruler, nobles >= 45%, >= 3 military cost traits in cabinet
cc_legend.4 = {
	type = country_event
	title = cc_legend.4.title
	desc = cc_legend.4.desc
	outcome = positive

	illustration_tags = {
		10 = military
	}

	trigger = {
		any_subject = { is_subject_type = military_march }
		ruler = { has_trait = iron_disciplinarian }
		any_cabinet_character = {
			in_cabinet = yes
			has_trait = supreme_commander
		}
		"estate_power(estate_type:nobles_estate)" >= 0.45
		# At least 2 different military cost specialists
		any_cabinet_character = {
			in_cabinet = yes
			OR = {
				has_trait = cavalry_marshal
				has_trait = commissariat_officer
				has_trait = siege_engineer
				has_trait = infantry_drillmaster
			}
		}
		any_cabinet_character = {
			in_cabinet = yes
			OR = {
				has_trait = scout_captain
				has_trait = military_engineer
				has_trait = elite_trainer
				has_trait = continental_marshal
			}
		}
		NOT = { has_variable = cc_legend_4_fired }
	}

	immediate = {
		set_variable = { name = cc_legend_4_fired value = yes }
	}

	# Iron Hegemony Reform
	option = {
		name = cc_legend.4.a
		set_variable = { name = cc_legend_iron_hegemony_unlocked value = yes }
		add_prestige = prestige_severe_bonus
		custom_tooltip = cc_legend.4.a.tt
	}

	# Iron Hegemony modifier instead
	option = {
		name = cc_legend.4.b
		add_country_modifier = { modifier = cc_synergy_iron_hegemony years = 999 }
		add_prestige = prestige_mild_bonus
	}
}

# Quest 8: "The Age of Exploration"
# Requires: age 3+, exploration-trait minister, chartered_company subject, maritime_champion minister
# Grants: legendary_navigator to the exploring minister + permanent explorer modifier + explorer guild reform unlock
cc_legend.8 = {
	type = country_event
	title = cc_legend.8.title
	desc = cc_legend.8.desc
	outcome = positive

	illustration_tags = {
		10 = exterior
		10 = army
	}

	trigger = {
		OR = {
			current_age = age_3_discovery
			current_age = age_4_reformation
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
		any_cabinet_character = {
			in_cabinet = yes
			OR = {
				has_trait = new_world_pioneer
				has_trait = expansionist_advisor
				has_trait = age_of_discovery_pioneer
			}
		}
		any_cabinet_character = {
			in_cabinet = yes
			has_trait = maritime_champion
		}
		any_subject = { is_subject_type = chartered_company }
		NOT = { has_variable = cc_legend_8_fired }
	}

	immediate = {
		set_variable = { name = cc_legend_8_fired value = yes }
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				OR = {
					has_trait = new_world_pioneer
					has_trait = expansionist_advisor
					has_trait = age_of_discovery_pioneer
				}
			}
			save_scope_as = minister
		}
	}

	# Age of Exploration legend — legendary_navigator + permanent modifier + reform unlock
	option = {
		name = cc_legend.8.a
		set_variable = { name = cc_legend_exploration_unlocked value = yes }
		scope:minister = { add_trait = trait:legendary_navigator }
		add_country_modifier = { modifier = cc_legend_age_of_exploration years = -1 }
		add_prestige = prestige_severe_bonus
		custom_tooltip = cc_legend.8.a.tt
	}

	# Fallback — prestige + colonial syndicate modifier
	option = {
		name = cc_legend.8.b
		add_prestige = prestige_severe_bonus
		add_country_modifier = { modifier = cc_synergy_colonial_syndicate years = 20 }
	}
}
