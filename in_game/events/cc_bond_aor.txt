###############################################################################
# Age of Revolutions Payoff Events — cc_bonds.90–99
#
# Fired by cc_bonds_aor_pulse, which selects the subject, saves it as
# scope:bond_subj, and grades the relationship into var:cc_bond_aor_tier (1-3)
# before triggering the event. Selection and grading live in the pulse rather
# than here because triggered_desc is evaluated before immediate runs, so both
# the description and the option triggers need the tier to already exist.
#
# root = overlord. Scores are read from variable maps on the overlord so that a
# subject which has already been released is still judged on the record it left.
#
# Tier 1 is a local outcome, tier 2 is a serious one, tier 3 reaches into
# vanilla's revolution machinery (see cc_bond_aor_effects.txt).
###############################################################################

# cc_bonds.90 — A Loyal Colony in Turbulent Times
cc_bonds.90 = {
	type = country_event
	title = cc_bonds.90.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:bond_subj = { has_variable = cc_bond_mem_voice_granted } }
				desc = cc_bonds.90.desc.voice
			}
			triggered_desc = {
				trigger = { var:cc_bond_aor_tier >= 3 }
				desc = cc_bonds.90.desc.deep
			}
			desc = cc_bonds.90.desc
		}
	}
	illustration_tags = { 10 = happy }

	trigger = { exists = scope:bond_subj }

	# Formally elevate the colony. The settlement holds, and the crown gets the
	# benefit of it at home.
	option = {
		name = cc_bonds.90.a
		cc_bonds_aor_ld_relief = yes
		cc_bonds_aor_tiered_modifier = { MOD = cc_subject_loyal_bulwark }
		cc_bonds_aor_suppress_revolution = yes
		cc_bonds_aor_prestige_gain = yes
	}

	# Press emergency levies. Real manpower now, but you spend the goodwill
	# instead of banking it: no revolution insurance, and the bond takes damage.
	option = {
		name = cc_bonds.90.b
		cc_bonds_aor_levy_subject = yes
		if = {
			limit = { scope:bond_subj = { is_subject_of = root } }
			scope:bond_subj = {
				add_liberty_desire = liberty_desire_mild_minus
				if = {
					limit = { has_variable = cc_bond_initialized }
					change_variable = { name = cc_bond_mil  add = -2 }
					clamp_variable = { name = cc_bond_mil  max = 5  min = -5 }
				}
			}
		}
	}

	# Invite them into the Imperial Council. Structural, and only available to a
	# bond deep enough to survive the promotion.
	option = {
		name = cc_bonds.90.c
		trigger = {
			has_advance = federal_constitution_advance
			var:cc_bond_aor_tier >= 2
			scope:bond_subj = { is_subject_of = root }
		}
		scope:bond_subj = {
			change_subject_type = subject_type:imperial_council_member
		}
		cc_bonds_aor_ld_relief = yes
		cc_bonds_aor_tiered_modifier = { MOD = cc_federal_union_solidarity }
		cc_bonds_aor_suppress_revolution = yes
	}
}

# cc_bonds.91 — The Colony Finds Its Voice
cc_bonds.91 = {
	type = country_event
	title = cc_bonds.91.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:bond_subj = { has_variable = cc_bond_mem_scholar_suppressed } }
				desc = cc_bonds.91.desc.scholar
			}
			triggered_desc = {
				trigger = { scope:bond_subj = { has_variable = cc_bond_mem_religion_forced } }
				desc = cc_bonds.91.desc.faith
			}
			triggered_desc = {
				trigger = { var:cc_bond_aor_tier >= 3 }
				desc = cc_bonds.91.desc.deep
			}
			desc = cc_bonds.91.desc
		}
	}
	illustration_tags = { 10 = angry }

	trigger = { exists = scope:bond_subj }

	# Crack down. The colony radicalizes and the argument travels home.
	option = {
		name = cc_bonds.91.a
		cc_bonds_aor_prestige_gain = yes
		cc_bonds_aor_ld_spike = yes
		cc_bonds_aor_tiered_modifier_subject = { MOD = cc_revolutionary_sympathies }
		cc_bonds_aor_bolster_revolution = yes
	}

	# Concede. Expensive, late, and it works: the contagion does not reach the
	# metropole, though the colony keeps what it won.
	option = {
		name = cc_bonds.91.b
		cc_bonds_aor_pay_light = yes
		cc_bonds_aor_ld_relief = yes
		cc_bonds_aor_tiered_modifier_subject = { MOD = cc_aor_settled_grievance }
		add_country_modifier = {
			modifier = cc_aor_settled_grievance
			years = 15
		}
	}

	# Let them go before the argument crosses the ocean. Only on the worst
	# relationships, where holding on is the thing that costs you.
	option = {
		name = cc_bonds.91.c
		trigger = {
			var:cc_bond_aor_tier >= 3
			scope:bond_subj = { is_subject_of = root }
		}
		cancel_subject = scope:bond_subj
		add_prestige = prestige_mild_penalty
		add_country_modifier = {
			modifier = cc_aor_orderly_separation
			years = 20
		}
	}
}

# cc_bonds.99 — The Dependency Looks Elsewhere

# conditions were written bare rather than inside an OR block the way every other
# hostile payoff wrote them. The shared cc_bonds_aor_hostile_gate now supplies the
# intended OR.

	type = country_event
	title = cc_bonds.99.title

			}

		}
