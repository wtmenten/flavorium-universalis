
namespace = cc_bonds

###############################################################################
# Age of Revolutions Payoff Events — cc_bonds.90–99
# Fire during age_6_revolutions via yearly_country_pulse.
# Root = overlord. Each reads bond scores from variable maps on overlord
# to handle subjects that may have already been released.
# One-shot per subject via cc_bond_aor_fired flag on overlord (keyed per subject).
###############################################################################

# cc_bonds.90 — A Loyal Colony in Turbulent Times
# Triggers: colonial_nation with eco >= 2 AND pol >= 2
cc_bonds.90 = {
	type = country_event
	title = cc_bonds.90.title
	desc = cc_bonds.90.desc
	illustration_tags = { 10 = happy }

	trigger = {
		has_game_rule = cc_subject_bonds_on
		current_age = age_6_revolutions
		any_key_in_variable_map = {
			variable = cc_subject_registry_map
			OR = {
				AND = {
					is_subject_of = root
					is_subject_type = colonial_nation
				}
				AND = {
					NOT = { is_subject_of = root }
					has_variable = cc_bond_eco_frozen
				}
			}
			"variable_map(cc_bond_eco_eff_map|this)" >= 2
			"variable_map(cc_bond_pol_eff_map|this)" >= 2
			NOT = { has_variable = cc_bond_aor_90_fired }
		}
	}

	immediate = {
		random_key_in_variable_map = {
			variable = cc_subject_registry_map
			limit = {
				OR = {
					AND = {
						is_subject_of = root
						is_subject_type = colonial_nation
					}
					AND = {
						NOT = { is_subject_of = root }
						has_variable = cc_bond_eco_frozen
					}
				}
				"variable_map(cc_bond_eco_eff_map|this)" >= 2
				"variable_map(cc_bond_pol_eff_map|this)" >= 2
				NOT = { has_variable = cc_bond_aor_90_fired }
			}
			save_scope_as = bond_subj
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_aor_90_fired  value = yes }
		}
		if = {
			limit = { scope:bond_subj = { is_subject_of = root } }
			scope:bond_subj = {
				add_liberty_desire = liberty_desire_severe_minus
			}
		}
		add_country_modifier = {
			modifier = cc_subject_loyal_bulwark
			years = 20
		}
	}

	# Formally elevate the colony
	option = {
		name = cc_bonds.90.a
		add_prestige = prestige_mild_bonus
	}

	# Press emergency levies anyway (relationship damage even in loyalty scenario)
	option = {
		name = cc_bonds.90.b
		add_manpower = { value = scope:bond_subj.monthly_manpower multiply = 6}
		scope:bond_subj = {
			add_manpower = { value = scope:bond_subj.monthly_manpower multiply = -6}
		}
		if = {
			limit = { scope:bond_subj = { has_variable = cc_bond_initialized } }
			scope:bond_subj = {
				change_variable = { name = cc_bond_mil  add = -2 }
				clamp_variable = { name = cc_bond_mil  max = 5  min = -5 }
			}
		}
	}

	# Invite them into the Imperial Council
	option = {
		name = cc_bonds.90.c
		trigger = {
			has_advance = federal_constitution_advance
			scope:bond_subj = { is_subject_of = root }
		}
		scope:bond_subj = {
			change_subject_type = subject_type:imperial_council_member
		}
		add_country_modifier = {
			modifier = cc_federal_union_solidarity
			years = 20
		}
	}
}
