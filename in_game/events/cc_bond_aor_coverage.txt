###############################################################################
# Age of Revolutions Payoff Events — cc_bonds.150-155
#
# Coverage for the three gaps in the original payoff set:
#  150/151 puppet_state, which had a full seven-event chain (cc_bonds.40-46) and
#          no terminal event at all.
#  152/153 imperial_council_member, which three separate payoff options promote
#          into and which previously terminated in nothing, making the promotion
#          read as a dead end.
#  154/155 the cul dimension, which is tracked, driven by religion changes,
#          forced-conversion detection, assimilation events and institution flow,
#          and was read by none of the payoffs.
#
# Same contract as cc_bond_aor.txt: fired by cc_bonds_aor_pulse with
# scope:bond_subj selected and var:cc_bond_aor_tier already graded.
###############################################################################

# cc_bonds.150 — The Client Regime Holds
cc_bonds.150 = {
	type = country_event
	title = cc_bonds.150.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:cc_bond_aor_tier >= 3 }
				desc = cc_bonds.150.desc.deep
			}
			desc = cc_bonds.150.desc
		}
	}
	illustration_tags = { 10 = interior }

	trigger = { exists = scope:bond_subj }

	# Keep the arrangement exactly as it is.
	option = {
		name = cc_bonds.150.a
		cc_bonds_aor_ld_relief = yes
		cc_bonds_aor_tiered_modifier = { MOD = cc_subject_loyal_bulwark }
		cc_bonds_aor_suppress_revolution = yes
	}

	# Put the regime on a formal footing so it survives without our garrison.
	option = {
		name = cc_bonds.150.b
		trigger = {
			var:cc_bond_aor_tier >= 2
			has_advance = federal_constitution_advance
			scope:bond_subj = { is_subject_of = root }
		}
		scope:bond_subj = {
			change_subject_type = subject_type:imperial_council_member
		}
		cc_bonds_aor_ld_relief = yes
		cc_bonds_aor_suppress_revolution = yes
		cc_bonds_aor_prestige_gain = yes
	}
}

# cc_bonds.151 — The Client Regime Falls
cc_bonds.151 = {
	type = country_event
	title = cc_bonds.151.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:bond_subj = { has_variable = cc_bond_mem_scholar_suppressed } }
				desc = cc_bonds.151.desc.scholar
			}
			triggered_desc = {
				trigger = { var:cc_bond_aor_tier >= 3 }
				desc = cc_bonds.151.desc.deep
			}
			desc = cc_bonds.151.desc
		}
	}
	illustration_tags = { 10 = angry }

	trigger = { exists = scope:bond_subj }

	# Prop the regime back up.
	option = {
		name = cc_bonds.151.a
		cc_bonds_aor_pay_heavy = yes
		cc_bonds_aor_ld_relief = yes
		cc_bonds_aor_tiered_modifier_subject = { MOD = cc_aor_settled_grievance }
	}

	# Install a new one by force.
	option = {
		name = cc_bonds.151.b
		add_prestige = prestige_weak_bonus
		add_country_modifier = {
			modifier = cc_aor_martial_law
			years = 20
		}
		cc_bonds_aor_ld_spike = yes
		cc_bonds_aor_tiered_modifier_subject = { MOD = cc_revolutionary_sympathies }
		cc_bonds_aor_bolster_revolution = yes
	}

	# Disown it. The client was never worth the credibility it cost.
	option = {
		name = cc_bonds.151.c
		if = {
			limit = { scope:bond_subj = { is_subject_of = root } }
			cancel_subject = scope:bond_subj
		}
		add_prestige = prestige_mild_penalty
		add_country_modifier = {
			modifier = cc_aor_orderly_separation
			years = 20
		}
	}
}

# cc_bonds.154 — One Nation, Two Languages

	type = country_event
	title = cc_bonds.154.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:bond_subj = { has_variable = cc_bond_mem_scholar_legacy_aligned } }
				desc = cc_bonds.154.desc.scholar
			}

				trigger = { var:cc_bond_aor_tier >= 3 }
				desc = cc_bonds.154.desc.deep
			}
			desc = cc_bonds.154.desc
		}

		cc_bonds_aor_ld_relief = yes
		cc_bonds_aor_suppress_revolution = yes
		cc_bonds_aor_prestige_gain = yes
