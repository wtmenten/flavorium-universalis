# cc_bonds.30 — "The Palatinate Lord's First Homage" (one-shot)
# Option C (waive ceremony) triggers a false positive — the lord sends a lavish
# gift, masking the personal damage of the snub.
cc_bonds.30 = {
	type = country_event
	title = cc_bonds.30.title
	desc = cc_bonds.30.desc
	illustration_tags = { 10 = interior }

	trigger = {
		has_game_rule = cc_subject_bonds_on
		any_subject = {
			is_subject_type = palatinate
			num_locations >= 15
			has_variable = cc_bond_initialized
			NOT = { has_variable = cc_bond_30_cd }
		}
	}

	immediate = {
		random_subject = {
			limit = {
				is_subject_type = palatinate
				num_locations >= 15
				has_variable = cc_bond_initialized
				NOT = { has_variable = cc_bond_30_cd }
			}
			save_scope_as = bond_subj
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_30_cd  value = yes }
		}
	}

	# Grand public ceremony — pol and strong per gain
	option = {
		name = cc_bonds.30.a
		scope:bond_subj = {
			change_variable = { name = cc_bond_pol  add = 1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = 2 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
	}

	# Accept private oath — moderate pol/per gain
	option = {
		name = cc_bonds.30.b
		scope:bond_subj = {
			change_variable = { name = cc_bond_pol  add = 1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = 1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
	}

	# Waive ceremony — pol/per damage, false positive fires
	option = {
		name = cc_bonds.30.c
		scope:bond_subj = {
			change_variable = { name = cc_bond_pol  add = -1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = -1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.79  months = 4 }
	}
}
