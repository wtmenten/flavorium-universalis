namespace = cc_bureau

# The bureau thread.
#
# WHY IT EXISTS. Bureaucracies shipped as an isolated system: implemented once, funded on a
# slider, and touched by nothing else in the game. This thread, the steward cabinet duty, the
# estate privileges and the two government reforms are the connections back.
#
# The spine is a running score, cc_bureau_venality, clamped 0..5. It rises whenever an office is
# treated as property rather than as a post and gates the later events. Nothing resets it
# cheaply, which is the point: venality is easy to permit and expensive to undo.
#
#   .1   a bureau is founded. Who staffs it.
#   .10  a bureau is dissolved. What becomes of its people.
#   .20  the offices are unfunded and finding their own income
#   .21  twelve years on, the fees are the custom            (delayed from .20)
#   .30  the clerks against the great houses
#   .40  entrenched and unfunded: the fiefdom
#   .50  entrenched and funded: a pillar of the state
#   .60  a funding cut lands                                 (from on_maintenance_changed)
#   .61  ten years on, what the cut actually cost            (delayed from .60)
#
# FUNDING x ENTRENCHMENT. Two independent numbers describe every bureau and the interesting
# content is in how they combine rather than in either alone:
#
#                  low entrenchment          high entrenchment
#   funded         a new office               a pillar of the state   (.50)
#   unfunded       a paper office             a FIEFDOM               (.40)
#
# Vanilla names the dangerous cell itself, in china.txt's negative_modifier comment:
# "entrenched ministries without funding become fiefdoms of powerful officials".
#
# Scales, verified against vanilla usage:
#   bureaucracy_maintenance   0..1    (vanilla compares >= 0.25)
#   entrenchment              0..100  (hire_advisor.txt:74 steps 10..60)
#
# SIGN WARNING on change_entrenchment. The named values invert: entrenchment_severe_penalty is
# +5 and entrenchment_extreme_bonus is -10, because HIGH entrenchment is treated as the penalty.
# Reducing entrenchment therefore uses a _bonus name. Read those blocks carefully before
# editing them.
#
# CHARACTER STAT RANGES. adm, dip and mil are each 0..100; total_abilities is their SUM and runs
# 0..300. A gate of adm >= 150 can never be satisfied and the option silently never appears.
# Roughly: >= 65 above average, >= 75 capable, >= 85 exceptional.


#############################################
# .1  A new bureau. Who staffs it?
#############################################
cc_bureau.1 = {
	type = country_event
	title = cc_bureau.1.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:cc_bureau_venality >= 3 }
				desc = cc_bureau.1.desc_venal
			}
			desc = cc_bureau.1.desc
		}
	}
	outcome = neutral

	trigger = {
		has_societal_value = societal_value_type:cc_order_vs_cc_magnates
		exists = scope:bureaucracy_type
	}

	illustration_tags = {
		10 = interior
		10 = professional
	}

	immediate = {
		cc_init_bureau_scores_effect = yes
	}

	# Salaried clerks who owe the post to the office rather than to a family.
	option = {
		name = cc_bureau.1.a
		ai_chance = { base = 40 }

		add_country_modifier = {
			modifier = cc_bureau_career_clerks
			years = 20
			mode = add_and_extend
		}
	}

	# The great houses' younger sons. Free, and every post is now owed to somebody.
	option = {
		name = cc_bureau.1.b
		ai_chance = { base = 35 }

		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_bonus
		}
		change_variable = { name = cc_bureau_venality  add = 1 }
		clamp_variable = { name = cc_bureau_venality  min = 0  max = 5 }
	}

	# Sell the posts. Only worth considering once the practice is already normal, which is what
	# the venality gate represents.
	option = {
		name = cc_bureau.1.c
		ai_chance = { base = 20 }
		trigger = {
			var:cc_bureau_venality >= 2
		}

		add_gold = 150
		set_variable = { name = cc_bureau_mem_sold_offices  value = yes }
		change_variable = { name = cc_bureau_venality  add = 2 }
		clamp_variable = { name = cc_bureau_venality  min = 0  max = 5 }
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_penalty
		}
	}

	after = {
		# Whoever staffed it, the men already in post read the appointment as a precedent and
		# act on it before anyone asks them. Not offered as a choice.
		hidden_effect = {
			if = {
				limit = { var:cc_bureau_venality >= 4 }
				add_country_modifier = {
					modifier = cc_bureau_offices_are_property
					years = 25
					mode = add_and_extend
				}
			}
		}
	}
}


#############################################
# .10  A bureau is dissolved
#############################################
cc_bureau.10 = {
	type = country_event
	title = cc_bureau.10.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = cc_bureau_mem_sold_offices }
				desc = cc_bureau.10.desc_sold
			}
			desc = cc_bureau.10.desc
		}
	}
	outcome = neutral

	trigger = {
		has_societal_value = societal_value_type:cc_order_vs_cc_magnates
		exists = scope:bureaucracy_type
	}

	illustration_tags = {
		10 = interior
		10 = angry
	}

	immediate = {
		cc_init_bureau_scores_effect = yes
	}

	# Reassign them. Expensive, and the service remembers it was not abandoned.
	option = {
		name = cc_bureau.10.a
		ai_chance = { base = 50 }

		change_gold_effect = { scale = -2 }
		add_country_modifier = {
			modifier = cc_bureau_career_clerks
			years = 15
			mode = add_and_extend
		}
	}

	# Dismiss them. Cheap now.
	option = {
		name = cc_bureau.10.b
		ai_chance = { base = 35 }

		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_penalty
		}
		change_variable = { name = cc_bureau_venality  add = 1 }
		clamp_variable = { name = cc_bureau_venality  min = 0  max = 5 }
	}

	# Purge them. Available to a ruler with the stomach for it, and it is why rulers keep
	# reaching for it: a purge is the one thing that resets venality outright.
	option = {
		name = cc_bureau.10.c
		ai_chance = { base = 15 }
		trigger = {
			ruler ?= { mil >= 70 }
		}

		set_variable = { name = cc_bureau_mem_purged  value = yes }
		set_variable = { name = cc_bureau_venality  value = 0 }
		add_stability = stability_mild_penalty
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_penalty
		}
	}
}


#############################################
# .20  The offices find their own income
#############################################
cc_bureau.20 = {
	type = country_event
	title = cc_bureau.20.title
	desc = cc_bureau.20.desc
	outcome = negative

	trigger = {
		has_societal_value = societal_value_type:cc_order_vs_cc_magnates
		any_current_bureaucracy = {
			bureaucracy_maintenance < 0.5
			count > 0
		}
	}

	illustration_tags = {
		10 = interior
		10 = economy
	}

	immediate = {
		cc_init_bureau_scores_effect = yes
	}

	# Pay the arrears. THE ONLY THING IN THE THREAD THAT REDUCES VENALITY, which is why it is
	# priced at several years of a small realm's income. Without it the score is a countdown
	# rather than a decision.
	option = {
		name = cc_bureau.20.a
		ai_chance = { base = 40 }

		change_gold_effect = { scale = -5 }
		add_country_modifier = {
			modifier = cc_bureau_paid_in_full
			years = 25
			mode = add_and_extend
		}
		change_variable = { name = cc_bureau_venality  add = -1 }
		clamp_variable = { name = cc_bureau_venality  min = 0  max = 5 }
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_bonus
		}
	}

	# Let them find their own income. THE FOLLOW-UP IS THE POINT: in twelve years the fees will
	# be the custom, and repealing a custom is a different problem from repealing a policy. The
	# player will have forgotten choosing this.
	option = {
		name = cc_bureau.20.b
		ai_chance = { base = 60 }

		change_variable = { name = cc_bureau_venality  add = 1 }
		clamp_variable = { name = cc_bureau_venality  min = 0  max = 5 }
		add_country_modifier = {
			modifier = cc_bureau_fees_became_custom
			years = 30
			mode = add_and_extend
		}
		add_estate_satisfaction = {
			type = estate_type:peasants_estate
			value = estate_satisfaction_severe_penalty
		}
		trigger_event_silently = { id = cc_bureau.21  years = 12 }
	}

	# Retrench without conceding the principle. Available to a capable administrator.
	option = {
		name = cc_bureau.20.c
		ai_chance = { base = 30 }
		trigger = {
			ruler ?= { adm >= 75 }
		}

		change_gold_effect = { scale = -2 }
		add_country_modifier = {
			modifier = cc_bureau_retrenched
			years = 30
			mode = add_and_extend
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_penalty
		}
	}
}


#############################################
# .21  The fees are now the custom
#
# Twelve years after .20b. THE DELAY IS THE CONTENT: an arrangement permitted once as an
# emergency has become the way the office works, and the men who benefit have had a decade to
# arrange their affairs around it.
#############################################
cc_bureau.21 = {
	type = country_event
	title = cc_bureau.21.title
	desc = cc_bureau.21.desc
	outcome = negative

	trigger = {
		has_societal_value = societal_value_type:cc_order_vs_cc_magnates
	}

	illustration_tags = {
		10 = interior
		10 = economy
	}

	immediate = {
		cc_init_bureau_scores_effect = yes
	}

	# Repeal it. Expensive, hated by everyone who has built a life on it, and it works.
	option = {
		name = cc_bureau.21.a
		ai_chance = { base = 30 }
		trigger = {
			ruler ?= { adm >= 70 }
		}

		change_gold_effect = { scale = -6 }
		remove_country_modifier = cc_bureau_fees_became_custom
		change_variable = { name = cc_bureau_venality  add = -1 }
		clamp_variable = { name = cc_bureau_venality  min = 0  max = 5 }
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_severe_penalty
		}
		add_stability = stability_mild_penalty
	}

	# Tax the fees instead of abolishing them. The treasury takes its share of an abuse it has
	# stopped pretending to disapprove of.
	option = {
		name = cc_bureau.21.b
		ai_chance = { base = 45 }

		add_gold = 350
		add_country_modifier = {
			modifier = cc_bureau_office_farmed_out
			years = 60
			mode = add_and_extend
		}
		change_variable = { name = cc_bureau_venality  add = 1 }
		clamp_variable = { name = cc_bureau_venality  min = 0  max = 5 }
	}

	# Say nothing. The custom hardens on its own.
	option = {
		name = cc_bureau.21.c
		ai_chance = { base = 25 }

		change_variable = { name = cc_bureau_venality  add = 1 }
		clamp_variable = { name = cc_bureau_venality  min = 0  max = 5 }
	}
}

# .30  The clerks against the great houses
#############################################
cc_bureau.30 = {
	type = country_event
	title = cc_bureau.30.title
	desc = cc_bureau.30.desc
	outcome = neutral

	trigger = {
		has_societal_value = societal_value_type:cc_order_vs_cc_magnates
		any_current_bureaucracy = {
			bureaucracy_maintenance > 0
			count > 0
		}
	}

	illustration_tags = {
		10 = interior
		10 = professional
	}

	immediate = {
		cc_init_bureau_scores_effect = yes
	}

	# Back the offices. The nobles are told the appointment stands.
	option = {
		name = cc_bureau.30.a
		ai_chance = { base = 40 }

		set_variable = { name = cc_bureau_mem_clerks_backed  value = yes }
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_severe_penalty
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_bonus
		}
		add_country_modifier = {
			modifier = cc_bureau_crown_backed_clerks
			years = 25
			mode = add_and_extend
		}
	}

	# Back the houses. Cheaper today in every currency the crown counts.
	option = {
		name = cc_bureau.30.b
		ai_chance = { base = 40 }

		set_variable = { name = cc_bureau_mem_houses_backed  value = yes }
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_bonus
		}
		change_variable = { name = cc_bureau_venality  add = 1 }
		clamp_variable = { name = cc_bureau_venality  min = 0  max = 5 }
	}

	# Refuse to hear it, and let both sides discover the crown is not an arbitrator they can
	# summon. Costs a capable diplomat nothing and everybody else a great deal.
	option = {
		name = cc_bureau.30.c
		ai_chance = { base = 20 }
		trigger = {
			ruler ?= { dip >= 70 }
		}

		add_prestige = prestige_mild_bonus
	}
}


#############################################
# .40  Entrenched and unfunded: the fiefdom
#############################################
cc_bureau.40 = {
	type = country_event
	title = cc_bureau.40.title
	desc = cc_bureau.40.desc
	outcome = negative

	trigger = {
		has_societal_value = societal_value_type:cc_order_vs_cc_magnates
		any_current_bureaucracy = {
			bureaucracy_maintenance < 0.35
			entrenchment >= 50
			count > 0
		}
	}

	illustration_tags = {
		10 = interior
		10 = angry
	}

	immediate = {
		cc_init_bureau_scores_effect = yes
		random_current_bureaucracy = {
			limit = {
				bureaucracy_maintenance < 0.35
				entrenchment >= 50
			}
			save_scope_as = cc_fief_bureau
		}
	}

	# Fund it properly and take it back. The expensive answer and the only one that works.
	option = {
		name = cc_bureau.40.a
		ai_chance = { base = 35 }

		change_gold_effect = { scale = -5 }
		add_country_modifier = {
			modifier = cc_bureau_paid_in_full
			years = 25
			mode = add_and_extend
		}
		change_variable = { name = cc_bureau_venality  add = -1 }
		clamp_variable = { name = cc_bureau_venality  min = 0  max = 5 }
	}

	# Break it up. Entrenchment falls, which is what the _bonus naming means here: see the sign
	# warning in the file header before editing this.
	option = {
		name = cc_bureau.40.b
		ai_chance = { base = 25 }
		trigger = {
			ruler ?= { mil >= 70 }
		}

		scope:cc_fief_bureau ?= {
			change_entrenchment = entrenchment_extreme_bonus
		}
		add_stability = stability_mild_penalty
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_penalty
		}
	}

	# Recognise what has already happened. The office becomes a family's property in law as
	# well as in fact, and the treasury is paid for the privilege.
	option = {
		name = cc_bureau.40.c
		ai_chance = { base = 40 }

		add_gold = 250
		set_variable = { name = cc_bureau_mem_fiefdom  value = yes }
		change_variable = { name = cc_bureau_venality  add = 2 }
		clamp_variable = { name = cc_bureau_venality  min = 0  max = 5 }
		add_country_modifier = {
			modifier = cc_bureau_office_farmed_out
			years = 60
			mode = add_and_extend
		}
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_bonus
		}
	}
}

# .50  Entrenched and funded: a pillar of the state
#############################################
cc_bureau.50 = {
	type = country_event
	title = cc_bureau.50.title
	desc = cc_bureau.50.desc
	outcome = positive

	trigger = {
		has_societal_value = societal_value_type:cc_order_vs_cc_magnates
		any_current_bureaucracy = {
			bureaucracy_maintenance >= 0.75
			entrenchment >= 60
			count > 0
		}
	}

	illustration_tags = {
		10 = interior
		10 = professional
	}

	immediate = {
		cc_init_bureau_scores_effect = yes
	}

	# Let it write its own procedure. Competent, and it will outlast the men who wrote it.
	option = {
		name = cc_bureau.50.a
		ai_chance = { base = 45 }

		add_country_modifier = {
			modifier = cc_bureau_standing_procedure
			years = -1
			mode = add
		}
	}

	# Bring its head into the cabinet. The office gains a voice and the crown gains a rival it
	# can at least see.
	option = {
		name = cc_bureau.50.b
		ai_chance = { base = 35 }

		add_country_modifier = {
			modifier = cc_bureau_head_in_council
			years = 30
			mode = add_and_extend
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_bonus
		}
	}

	# Rotate its officers before any of them takes root. Safe, and it costs the competence the
	# office spent thirty years accumulating.
	option = {
		name = cc_bureau.50.c
		ai_chance = { base = 20 }

		add_country_modifier = {
			modifier = cc_bureau_rotated_officers
			years = 20
			mode = add_and_extend
		}
	}
}


#############################################
# .60  A funding cut lands
#
# From on_maintenance_changed, which this mod declares on all fourteen overridden vanilla
# bureaucracies. scope:old_maintenance and scope:new_maintenance are provided.
#
# THE COOLDOWN LIVES HERE RATHER THAN IN THE FOURTEEN HOOKS. An emperor economising across the
# board in a bad year would otherwise fire this dialogue fourteen times in an afternoon, and one
# who oscillates funding would fire it forever. A directly-fired event still evaluates its own
# trigger (vanilla's d008_flavor_byz.2 and .3 depend on exactly that), so one guard here covers
# every call site and cannot be missed on a fifteenth copy.
#
# cc_bureau_cut_cd is TIMED and expires by itself. Nothing has to clear it and nothing can leak
# it, which is the same pattern the bond system uses for cc_bond_aor_cd.
#############################################
cc_bureau.60 = {
	type = country_event
	title = cc_bureau.60.title
	desc = cc_bureau.60.desc
	outcome = negative

	trigger = {
		has_societal_value = societal_value_type:cc_order_vs_cc_magnates
		NOT = { has_variable = cc_bureau_cut_cd }
	}

	illustration_tags = {
		10 = interior
		10 = economy
	}

	immediate = {
		cc_init_bureau_scores_effect = yes
		set_variable = { name = cc_bureau_cut_cd  value = yes  years = 5 }
	}

	# Say it plainly: the money is not there this year and the office will be told so.
	option = {
		name = cc_bureau.60.a
		ai_chance = { base = 40 }

		add_country_modifier = {
			modifier = cc_bureau_fees_became_custom
			years = 20
			mode = add_and_extend
		}
		change_variable = { name = cc_bureau_venality  add = 1 }
		clamp_variable = { name = cc_bureau_venality  min = 0  max = 5 }
		trigger_event_silently = { id = cc_bureau.61  years = 10 }
	}

	# Cut the establishment instead of the pay. Fewer clerks, still salaried, still the crown's.
	option = {
		name = cc_bureau.60.b
		ai_chance = { base = 35 }
		trigger = {
			ruler ?= { adm >= 65 }
		}

		add_country_modifier = {
			modifier = cc_bureau_retrenched
			years = 25
			mode = add_and_extend
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_penalty
		}
	}

	# Find the money somewhere else. It is always found somewhere else.
	option = {
		name = cc_bureau.60.c
		ai_chance = { base = 25 }

		change_gold_effect = { scale = -3 }
		add_country_modifier = {
			modifier = cc_bureau_paid_in_full
			years = 15
			mode = add_and_extend
		}
	}
}
