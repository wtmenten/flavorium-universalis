namespace = cc_pty

###############################################################################
# Flavorium Universalis - Factions: events
#
# The party system is meant to run quietly. Satisfaction, agenda results and the first
# two escalation steps show only in the Factions panel. Popups:
#
#   cc_pty.5   Generate party leaders            hidden
#   cc_pty.6   Leaders after setup               hidden
#   cc_pty.10  New parties (realignment)          budgeted (cc_pty_popup_cd)
#   cc_pty.20  A party sets an agenda             budgeted
#   cc_pty.30  A minister considers defecting     shares cc_rival_event_cooldown
#   cc_pty.40  The officers move (coup starts)    escalation outcome, always
#   cc_pty.41  The provinces rise                 escalation outcome, always
#   cc_pty.42  The liberal clubs rise             escalation outcome, always
#   cc_pty.43  The subjects pull away             escalation outcome, always
#   cc_pty.45  The coup ends
#   cc_pty.46  The uprising ends: defused (momentum 0) or the party is gone
#   cc_pty.47  The uprising boils over (momentum 100): constitution or civil war
#   cc_pty.48  The uprising ends after five years, unresolved
#   cc_pty.70-75  Liberal uprising crisis events  from the disaster's on_monthly
#   cc_pty.50  The Liberal Party forms            budgeted
#   cc_pty.60  Term limits adopted                rare (follows the dominant party)
#   cc_pty.61  Term limits lifted                 rare
#
# Budgeted popups set cc_pty_popup_cd (five years), so they come to about two a decade.
# Party names in loc come from customizable_localization/cc_pty_names.txt via
# [ROOT.GetCountry.Custom('...')].
#
# Party effects that compare scratch variables run through cc_pty_run, inside hidden_effect,
# with a custom_tooltip beside them. An option's effect is evaluated to build its tooltip,
# hidden_effect included, without committing any variable writes; cc_pty_run skips the body
# in that pass (see TOOLTIP GUARD in cc_pty_effects.txt).
###############################################################################


###############################################################################
# cc_pty.5 - GENERATE PARTY LEADERS (hidden)
#
# Fired by cc_pty_update_leaders for a player realm when a party has no minister and no
# living leader. Characters must be created in an event's immediate block.
###############################################################################

cc_pty.5 = {
	type = country_event
	hidden = yes

	immediate = {
		save_scope_as = cc_pty_country
		cc_pty_create_leader_dispatch = { SLOT = 1 }
		cc_pty_create_leader_dispatch = { SLOT = 2 }
		cc_pty_create_leader_dispatch = { SLOT = 3 }
		cc_pty_create_leader_dispatch = { SLOT = 4 }
	}
}


###############################################################################
# cc_pty.6 - LEADERS AFTER SETUP (hidden)
#
# Fired a day after cc_pty_init_country, once the player's country is known to be a
# player's, so its parties get leaders straight away instead of at the first cabinet change.
###############################################################################

cc_pty.6 = {
	type = country_event
	hidden = yes

	immediate = {
		if = {
			limit = { has_variable = cc_pty_active }
			cc_pty_affiliate_court = yes
			cc_pty_update_leaders = yes
		}
	}
}

		name = cc_pty.30.a

###############################################################################
# cc_pty.42 - THE LIBERAL CLUBS RISE
#
# Opens cc_pty_liberal_uprising.
###############################################################################

cc_pty.42 = {
	type = country_event
	title = cc_pty.42.title
	desc = cc_pty.42.desc
	outcome = negative

	illustration_tags = {
		10 = society
		10 = angry
	}

	# Opens the uprising. No option ends it: see cc_pty_liberal_uprising for the momentum
	# meter (starts at 60) and how it falls. The desc explains it to the player.

	# Concede some reforms.
	option = {
		name = cc_pty.42.a
		change_societal_value = { type = absolutism_vs_liberalism value = societal_value_minor_move_to_right }
		add_political_influence = political_influence_mild_penalty
		custom_tooltip = cc_pty_tt_liberal_concede
		custom_tooltip = cc_pty_lu_tt_down_10
		hidden_effect = {
			cc_pty_type_add_sat = { TYPE = 11 VALUE = 20 }
			cc_pty_lu_change = { VALUE = -10 }
		}
		ai_chance = { base = 45 }
	}

	# Close the clubs and arrest the leaders.
	option = {
		name = cc_pty.42.b
		add_stability = stability_severe_penalty
		custom_tooltip = cc_pty_tt_liberal_purge
		custom_tooltip = cc_pty_lu_tt_down_15
		custom_tooltip = cc_pty_lu_tt_repressed
		hidden_effect = {
			cc_pty_type_suppress = { TYPE = 11 }
			cc_pty_type_add_sat = { TYPE = 11 VALUE = -20 }
			cc_pty_lu_change = { VALUE = -15 }
			set_variable = cc_pty_lu_repressed
		}
		ai_chance = { base = 35 }
	}

	# Wait and see.
	option = {
		name = cc_pty.42.c
		custom_tooltip = cc_pty_lu_tt_up_5
		hidden_effect = {
			cc_pty_lu_change = { VALUE = 5 }
		}
		ai_chance = { base = 20 }
	}
}

###############################################################################
# cc_pty.70-75 - LIBERAL UPRISING CRISIS EVENTS
#
# Fired by cc_pty_lu_monthly while cc_pty_liberal_uprising runs. Each moves the momentum
# meter (cc_pty_lu_change, 0..100); bringing it down costs gold, stability, political
# influence or ground on Liberalism. Repression (cc_pty_lu_repressed) adds +0.5 a month
# to the drift and unlocks cc_pty.74.
###############################################################################

# The pamphlet war.
cc_pty.70 = {
	type = country_event
	title = cc_pty.70.title
	desc = cc_pty.70.desc
	outcome = neutral

	illustration_tags = {
		10 = society
		10 = interior
	}

	# Censor the presses.
	option = {
		name = cc_pty.70.a
		add_stability = stability_weak_penalty
		change_societal_value = { type = absolutism_vs_liberalism value = societal_value_tiny_move_to_left }
		custom_tooltip = cc_pty_lu_tt_down_8
		custom_tooltip = cc_pty_lu_tt_repressed
		hidden_effect = {
			cc_pty_lu_change = { VALUE = -8 }
			set_variable = cc_pty_lu_repressed
		}
		ai_chance = { base = 35 }

		name = cc_pty.70.b

		custom_tooltip = cc_pty_lu_tt_down_5
		hidden_effect = { cc_pty_lu_change = { VALUE = -5 } }

	}

		name = cc_pty.70.c
