# Colonial Nation Chain — cc_bonds.10–16
# Root = overlord. scope:bond_subj = the colonial nation subject.
# Primary dimensions affected: Economic, Cultural. Secondary: Political, Personal.
# cc_bond_col_gen_cd: 5-yr per-subject gate shared across all colonial events.
###############################################################################

# cc_bonds.10 — "The Colony's First Harvest" (one-shot)
# Intro event. Option A looks profitable but harms eco long-term (red herring 70
# fires making it seem fine). Option B costs gold but builds real eco capital
# (red herring 71 fires making it seem like a mistake).
cc_bonds.10 = {
	type = country_event
	title = cc_bonds.10.title
	desc = cc_bonds.10.desc
	illustration_tags = { 10 = economy }

	trigger = {
		has_game_rule = cc_subject_bonds_on
		any_subject = {
			is_subject_type = colonial_nation
			num_locations >= 15
			total_population > 60000
			has_variable = cc_bond_initialized
			NOT = { has_variable = cc_bond_10_cd }
			NOT = { has_variable = cc_bond_col_gen_cd }
		}
	}

	immediate = {
		random_subject = {
			limit = {
				is_subject_type = colonial_nation
				num_locations >= 15
				total_population > 60000
				has_variable = cc_bond_initialized
				NOT = { has_variable = cc_bond_10_cd }
				NOT = { has_variable = cc_bond_col_gen_cd }
			}
			save_scope_as = bond_subj
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_10_cd  value = yes }
			set_variable = { name = cc_bond_col_gen_cd  value = yes  years = 5 }
		}
	}

	# Heavy extraction — gold gain, eco bond damage (false positive fires)
	option = {
		name = cc_bonds.10.a
		add_gold = { value = scope:bond_subj.monthly_income_trade_and_tax  multiply = 3  min = 100 }
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = -2 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.70  months = 3 }
	}

	# Invest in infrastructure — gold cost, eco bond gain (false negative fires)
	option = {
		name = cc_bonds.10.b
		add_gold = { value = monthly_income_trade_and_tax  multiply = -0.30  max = -65  min = -300 }
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = 1 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
			random_province = {
				add_province_modifier = {
					modifier = cc_infrastructure_investment
					years = 20
					mode = set_to_largest

				}
			}
		}
		trigger_event_silently = { id = cc_bonds.71  months = 3 }
	}

	# Hands-off — prestige loss for visible neglect
	option = {
		name = cc_bonds.10.c
		add_prestige = prestige_weak_penalty
	}
}

# cc_bonds.12 — "The Governor Reports Unrest" (cooldown 15 yrs)
# Option A looks decisive but hits 4 dimensions. Option B costs gold but is
# the only genuinely good choice. Option C triggers a false positive.
cc_bonds.12 = {
	type = country_event
	title = cc_bonds.12.title
	desc = cc_bonds.12.desc
	illustration_tags = { 10 = angry }

	trigger = {
		has_game_rule = cc_subject_bonds_on
		any_subject = {
			is_subject_type = colonial_nation
			num_locations >= 15
			has_variable = cc_bond_initialized
			NOT = { has_variable = cc_bond_12_cd }
			NOT = { has_variable = cc_bond_col_gen_cd }
		}
	}

	immediate = {
		random_subject = {
			limit = {
				is_subject_type = colonial_nation
				num_locations >= 15
				has_variable = cc_bond_initialized
				NOT = { has_variable = cc_bond_12_cd }
				NOT = { has_variable = cc_bond_col_gen_cd }
			}
			save_scope_as = bond_subj
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_12_cd  value = yes  years = 15 }
			set_variable = { name = cc_bond_col_gen_cd  value = yes  years = 5 }
		}
	}

	# Send troops to suppress — prestige gain, 4-dimension damage, manpower drain scaled to subject size
	option = {
		name = cc_bonds.12.a
		add_prestige = prestige_mild_bonus
		add_manpower = { value = -5  multiply = { value = scope:bond_subj.num_locations  divide = 15 }  max = -5  min = -50 }
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = -1 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
			change_variable = { name = cc_bond_mil  add = -1 }
			clamp_variable = { name = cc_bond_mil  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = -2 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_cul  add = -1 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
		}
	}

	# Investigate and reform tax code — reduced gold cost, eco+pol improvement, short admin penalty
	option = {
		name = cc_bonds.12.b
		add_gold = { value = monthly_income_trade_and_tax  multiply = -0.20  max = -50  min = -750 }
		add_country_modifier = { modifier = cc_governor_unrest_reform  years = 10  mode = add_and_extend }
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = 2 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
			change_variable = { name = cc_bond_pol  add = 1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
		}
	}

	# Replace the governor — prestige loss, small pol/per damage, legitimacy, opinion hit, false positive fires
	option = {
		name = cc_bonds.12.c
		add_prestige = prestige_weak_penalty
		add_legitimacy = legitimacy_weak_bonus
		scope:bond_subj = {
			add_opinion = { target = ROOT modifier = cc_overlord_unrest_response }
			change_variable = { name = cc_bond_pol  add = -1 }
			clamp_variable = { name = cc_bond_pol  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = -1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.72  months = 6 }
	}

	# Appease with promises — no immediate bond change, but seeds future unrest in the colony
	option = {
		name = cc_bonds.12.d
		add_gold = { value = monthly_income_trade_and_tax  multiply = -0.10  max = -50  min = -300 }
		scope:bond_subj = {
			add_gold = { value = monthly_income_trade_and_tax  multiply = -0.10  min = 50  max = 300 }
			add_country_modifier = { modifier = cc_governor_unrest_appeasement  years = 15  mode = add_and_extend }
		}
	}
}
