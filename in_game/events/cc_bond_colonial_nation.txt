# cc_bonds.10 — "The Colony's First Harvest" (one-shot)
# Intro event. Option A looks profitable but harms eco long-term (red herring 70
# fires making it seem fine). Option B costs gold but builds real eco capital
# (red herring 71 fires making it seem like a mistake).
cc_bonds.10 = {
	type = country_event
	title = cc_bonds.10.title
	desc = cc_bonds.10.desc
	illustration_tags = { 10 = economy }

	trigger = {
		has_game_rule = cc_subject_bonds_on
		any_subject = {
			is_subject_type = colonial_nation
			num_provinces >= 3
			has_variable = cc_bond_initialized
			NOT = { has_variable = cc_bond_10_cd }
		}
	}

	immediate = {
		random_subject = {
			limit = {
				is_subject_type = colonial_nation
				num_provinces >= 3
				has_variable = cc_bond_initialized
				NOT = { has_variable = cc_bond_10_cd }
			}
			save_scope_as = bond_subj
		}
		scope:bond_subj = {
			set_variable = { name = cc_bond_10_cd  value = yes }
		}
	}

	# Heavy extraction — gold gain, eco bond damage (false positive fires)
	option = {
		name = cc_bonds.10.a
		add_gold = { value = monthly_income_trade_and_tax  multiply = 8 }
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = -2 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.70  months = 3 }
	}

	# Invest in infrastructure — gold cost, eco bond gain (false negative fires)
	option = {
		name = cc_bonds.10.b
		add_gold = { value = monthly_income_trade_and_tax  multiply = -5 }
		scope:bond_subj = {
			change_variable = { name = cc_bond_eco  add = 2 }
			clamp_variable = { name = cc_bond_eco  max = 5  min = -5 }
		}
		trigger_event_silently = { id = cc_bonds.71  months = 3 }
	}

	# Hands-off — no change
	option = {
		name = cc_bonds.10.c
	}
}
