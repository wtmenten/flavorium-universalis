# "The Minister Seeks Counsel"
# An idle cabinet member requests an audience to discuss statecraft.
cc_cabinet.1 = {
	type = country_event
	title = cc_cabinet.1.title
	desc = cc_cabinet.1.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		has_ruler = yes
		any_cabinet_character = {
			in_cabinet = yes
			cc_minister_is_available = yes
			NOT = { has_variable = cc_idle_cooldown }
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				cc_minister_is_available = yes
				NOT = { has_variable = cc_idle_cooldown }
			}
			save_scope_as = minister
		}
		scope:minister = { set_variable = { name = cc_idle_cooldown value = yes years = 2 } }
	}

	option = {
		name = cc_cabinet.1.a
		add_prestige = prestige_weak_bonus
		scope:minister = { add_trait = trait:learned_counselor }
	}

	option = {
		name = cc_cabinet.1.b
		add_prestige = prestige_weak_bonus
		scope:minister = { add_trait = trait:loyal_steward }
	}

	option = {
		name = cc_cabinet.1.c
		trigger = {
			scope:minister ?= { mil > 50 }
		}
		add_prestige = prestige_weak_bonus
		scope:minister = { add_trait = trait:tactical_advisor }
	}

	option = {
		name = cc_cabinet.1.d
	}
}

# "The Minister Responds to Refusal" (silent follow-up to cc_cabinet.2 option B)
# Pre-rolls one of three outcomes: grudging acceptance, quiet extraction, or political manoeuvring.
cc_cabinet.5 = {
	type = country_event
	title = cc_cabinet.5.title
	desc = cc_cabinet.5.desc
	outcome = negative

	illustration_tags = {
		10 = interior
		10 = angry
	}

	immediate = {
		var:cc_2b_minister ?= { save_scope_as = minister }
		if = {
			limit = {
				scope:minister ?= {
					OR = { adm > 50  has_trait = loyal_steward  has_trait = learned_counselor }
				}
			}
			random_list = {
				50 = { set_variable = { name = cc_5_outcome  value = 1 } }
				30 = { set_variable = { name = cc_5_outcome  value = 2 } }
				20 = { set_variable = { name = cc_5_outcome  value = 3 } }
			}
		}
		else = {
			random_list = {
				25 = { set_variable = { name = cc_5_outcome  value = 1 } }
				40 = { set_variable = { name = cc_5_outcome  value = 2 } }
				35 = { set_variable = { name = cc_5_outcome  value = 3 } }
			}
		}
	}

	after = {
		remove_variable = cc_2b_minister
		remove_variable = cc_5_outcome
	}

	# Outcome 1: Minister sulks — prestige dent, minor modifier
	option = {
		name = cc_cabinet.5.a
		trigger = { var:cc_5_outcome = 1 }
		add_prestige = prestige_weak_penalty
		add_country_modifier = {
			modifier = cc_cabinet_revoke_difficulty
			years = 3
			mode = add_and_extend
		}
	}

	# Outcome 2: Minister quietly taps the treasury
	option = {
		name = cc_cabinet.5.b
		trigger = { var:cc_5_outcome = 2 }
		add_gold = { value = monthly_income_trade_and_tax  multiply = -0.20  max = -40  min = -500 }
	}

	# Outcome 3: Minister lobbies nobles for private backing
	option = {
		name = cc_cabinet.5.c
		trigger = { var:cc_5_outcome = 3 }
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_weak_bonus
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_penalty
		}
	}
}

# "The Minister Puts the Funds to Work" (silent follow-up to cc_cabinet.2 option A)
# Resolves randomly.
cc_cabinet.3 = {
	title = cc_cabinet.3.title
	desc = cc_cabinet.3.desc
	type = country_event
	outcome = neutral

	option = {
		name = cc_cabinet.3.a
		random_list = {
			60 = {
				add_prestige = prestige_mild_bonus
				add_country_modifier = {
					modifier = cc_cabinet_ministerial_investment
					years = 5
					mode = add_and_extend
				}
			}
			40 = {
				add_gold = {
					value = monthly_income_trade_and_tax
					multiply = {
						value = { random_integer = { max = 200 } }
						modulo = 11		# → 0–10
						subtract = 5	# → -5–5
						multiply = 0.50	# scale variance relative to base
						divide = 10		# → -0.075–0.075
						add = 0.50		# → 0.075–0.225
					}
					min = 100
					max = 2000
				}
			}
		}
	}
}

# An idle minister uses their estate connections for political ends.
cc_cabinet.10 = {
	type = country_event
	title = cc_cabinet.10.title
	desc = cc_cabinet.10.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
	}

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			cc_minister_is_available = yes
			NOT = { has_variable = cc_idle_cooldown }
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				cc_minister_is_available = yes
				NOT = { has_variable = cc_idle_cooldown }
			}
			save_scope_as = minister
		}
		scope:minister = { set_variable = { name = cc_idle_cooldown value = yes years = 2 } }
	}

	option = {
		name = cc_cabinet.10.a
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_weak_bonus
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_penalty
		}
		set_variable = { name = cc_snubbed_estate_burghers value = yes }
		trigger_event_silently = { id = cc_cabinet.50 months = { 6 12 } }
	}

	option = {
		name = cc_cabinet.10.b
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_weak_bonus
		}
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_penalty
		}
		set_variable = { name = cc_snubbed_estate_nobles value = yes }
		trigger_event_silently = { id = cc_cabinet.50 months = { 6 12 } }
	}

	option = {
		name = cc_cabinet.10.c
		add_gold = { value = monthly_income_trade_and_tax  multiply = -0.25  max = -50  min = -750 }
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_weak_bonus
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_weak_bonus
		}
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_weak_bonus
		}
	}

	option = {
		name = cc_cabinet.10.d
		set_variable = { name = cc_52_minister value = scope:minister }
		if = {
			limit = {
				scope:minister = {
					OR = {
						dip > 55
						has_trait = diplomatic_attache
						has_trait = grand_chancellor
						has_trait = parliamentary_broker
						has_trait = prosperity_herald
						has_trait = loyal_steward
					}
				}
			}
			trigger_event_silently = { id = cc_cabinet.53 months = { 3 8 } }
		}
		else_if = {
			limit = {
				OR = {
					estate_satisfaction:nobles_estate < 0.40
					estate_satisfaction:burghers_estate < 0.40
					estate_satisfaction:clergy_estate < 0.40
				}
			}
			trigger_event_silently = { id = cc_cabinet.54 months = { 3 8 } }
		}
		else = {
			trigger_event_silently = { id = cc_cabinet.52 months = { 3 8 } }
		}
	}
}

# "Diplomatic Overtures"
# A minister pursues diplomacy with a neighboring state.
cc_cabinet.20 = {
	type = country_event
	title = cc_cabinet.20.title
	desc = cc_cabinet.20.desc
	outcome = neutral

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		any_cabinet_character = {
			in_cabinet = yes
			NOT = { has_variable = cc_idle_cooldown }
			OR = {
				cc_minister_is_available = yes
				cabinet_action = cabinet_action:diplomatic_endeavors
			}
		}
		any_neighbor_country = {
			NOT = { is_at_war_with = root }
		}
	}

	immediate = {
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				NOT = { has_variable = cc_idle_cooldown }
				OR = {
					cc_minister_is_available = yes
					cabinet_action = cabinet_action:diplomatic_endeavors
				}
			}
			save_scope_as = minister
		}
		random_neighbor_country = {
			limit = {
				NOT = { is_at_war_with = root }
			}
			save_scope_as = neighbor_country
		}
		scope:minister = { set_variable = { name = cc_idle_cooldown value = yes years = 2 } }
	}

	option = {
		name = cc_cabinet.20.a
		add_opinion = {
			target = scope:neighbor_country
			modifier = cc_diplomatic_favor
		}
		scope:neighbor_country = {
			add_opinion = {
				target = root
				modifier = cc_diplomatic_favor
			}
		}
	}

	option = {
		name = cc_cabinet.20.b
		add_country_modifier = {
			modifier = cc_cabinet_diplomatic_influence
			years = 5
			mode = add_and_extend
		}

	option = {
		name = cc_cabinet.20.c
	}
}
