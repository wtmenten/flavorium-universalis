# Iron Disciplinarian + Cavalry Marshal / Commissariat Officer
cc_synergy.1 = {
	type = country_event
	title = cc_synergy.1.title
	desc = cc_synergy.1.desc
	outcome = positive

	illustration_tags = {
		10 = military
	}

	trigger = {
		ruler = { has_trait = iron_disciplinarian }
		any_cabinet_character = {
			in_cabinet = yes
			OR = {
				has_trait = cavalry_marshal
				has_trait = commissariat_officer
			}
		}
		NOT = { has_variable = cc_synergy_1_fired }
	}

	immediate = {
		set_variable = { name = cc_synergy_1_fired value = yes }
		random_cabinet_character = {
			limit = {
				in_cabinet = yes
				OR = {
					has_trait = cavalry_marshal
					has_trait = commissariat_officer
				}
			}
			save_scope_as = minister
		}
	}

	# Grant iron army modifier — 15 years
	option = {
		name = cc_synergy.1.a
		add_country_modifier = { modifier = cc_synergy_iron_army years = 15 }
	}

	# Promote the minister — grant supreme_commander
	option = {
		name = cc_synergy.1.b
		scope:minister = { add_trait = trait:supreme_commander }
	}

	# Decline the synergy
	option = {
		name = cc_synergy.1.c
		add_prestige = prestige_weak_bonus
	}
}
