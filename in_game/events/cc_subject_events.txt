# CHAIN 2: PATRONAGE OF LEARNING
# Fired via on_activate in ep_clergy_scholarly_college.
###############################################################################

# "The Scholars Seek a Home"
cc.104 = {
	type = country_event
	title = cc.clergy.college.1.title
	desc = cc.clergy.college.1.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
	}

	# Re-offered from cc_scientific_college_pulse while the college can still be founded, and
	# fired immediately from ep_clergy_scholarly_college on_activate.
	trigger = {
		has_estate_privilege = estate_privilege:ep_clergy_scholarly_college
		"estate_power(estate_type:clergy_estate)" >= 0.30
		NOT = { any_subject = { is_subject_type = scientific_college } }
		NOT = { has_variable = cc_college_creation_cd }
		any_owned_location = {
			NOT = { is_capital = yes }
			dominant_culture = root.culture
			is_connected_to_through_realm = root.capital
		}
	}

	# EU5 does not permit releasing a subject in a cored (dominant-culture) location through
	# normal diplomacy — and a scientific college must share our culture. So we offer up to
	# three candidate seats here: locations of our dominant culture, connected to the capital,
	# excluding the capital. Selection is randomized but weighted heavily toward development,
	# so the great cities are favored without being guaranteed. A rural seat is granted town
	# rights on founding so the college has a proper urban home.
	immediate = {
		if = {
			limit = {
				NOT = { any_subject = { is_subject_type = scientific_college } }
			}
			# Draw three distinct seats, each weighted by development.
			random_owned_location = {
				limit = {
					NOT = { is_capital = yes }
					dominant_culture = root.culture
					is_connected_to_through_realm = root.capital
				}
				weight = {
					base = 1
					modifier = { add = { value = development multiply = 2 } }
				}
				save_scope_as = college_loc_1
			}
			if = {
				limit = { exists = scope:college_loc_1 }
				random_owned_location = {
					limit = {
						NOT = { is_capital = yes }
						dominant_culture = root.culture
						is_connected_to_through_realm = root.capital
						NOT = { this = scope:college_loc_1 }
					}
					weight = {
						base = 1
						modifier = { add = { value = development multiply = 2 } }
					}
					save_scope_as = college_loc_2
				}
			}
			if = {
				limit = { exists = scope:college_loc_2 }
				random_owned_location = {
					limit = {
						NOT = { is_capital = yes }
						dominant_culture = root.culture
						is_connected_to_through_realm = root.capital
						NOT = { this = scope:college_loc_1 }
						NOT = { this = scope:college_loc_2 }
					}
					weight = {
						base = 1
						modifier = { add = { value = development multiply = 2 } }
					}
					save_scope_as = college_loc_3
				}
			}
		}
	}

	# Found the college at the first candidate seat
	option = {
		name = cc.clergy.college.1.opt1
		trigger = { exists = scope:college_loc_1 }
		add_gold = { value = monthly_income_trade_and_tax multiply = -0.25 max = -250 min = -750 }
		scope:college_loc_1 = {
			if = {
				limit = { location_rank = location_rank:rural_settlement }
				change_location_rank = location_rank:town
			}
			create_country_from_location = {
				set_capital = prev
				change_government_type = government_type:theocracy
				change_culture = ROOT.culture
				change_religion = ROOT.religion
				make_subject_of = { target = ROOT type = subject_type:scientific_college }
				save_scope_as = new_college
			}
		}
		trigger_event_non_silently = { id = cc.105 years = 2 }
	}

	# Found the college at the second candidate seat
	option = {
		name = cc.clergy.college.1.opt2
		trigger = { exists = scope:college_loc_2 }
		add_gold = { value = monthly_income_trade_and_tax multiply = -0.25 max = -250 min = -750 }
		scope:college_loc_2 = {
			if = {
				limit = { location_rank = location_rank:rural_settlement }
				change_location_rank = location_rank:town
			}
			create_country_from_location = {
				set_capital = prev
				change_government_type = government_type:theocracy
				change_culture = ROOT.culture
				change_religion = ROOT.religion
				make_subject_of = { target = ROOT type = subject_type:scientific_college }
				save_scope_as = new_college
			}
		}
		trigger_event_non_silently = { id = cc.105 years = 2 }
	}

	# Found the college at the third candidate seat
	option = {
		name = cc.clergy.college.1.opt3
		trigger = { exists = scope:college_loc_3 }
		add_gold = { value = monthly_income_trade_and_tax multiply = -0.25 max = -250 min = -750 }
		scope:college_loc_3 = {
			if = {
				limit = { location_rank = location_rank:rural_settlement }
				change_location_rank = location_rank:town
			}
			create_country_from_location = {
				set_capital = prev
				change_government_type = government_type:theocracy
				change_culture = ROOT.culture
				change_religion = ROOT.religion
				make_subject_of = { target = ROOT type = subject_type:scientific_college }
				save_scope_as = new_college
			}
		}
		trigger_event_non_silently = { id = cc.105 years = 2 }
	}

	# Encourage the scholars to remain within our own institutions for now (re-offered later).
	option = {
		name = cc.clergy.college.1.b
		set_variable = { name = cc_college_creation_cd value = yes years = 5 }
		add_country_modifier = {
			modifier = cc_internalized_scholarship
			years = 10
			mode = add_and_extend
		}
		add_country_modifier = {
			modifier = cc_scholarly_exchange_modifier
			years = 5
			mode = add_and_extend
		}
	}
}

# "The Protectorate Claims the Holy War's Fruit"
# Fires from cc_holy_war_on_winning when overlord wins a holy war
# while holding a holy_protectorate subject.
cc.111 = {
	type = country_event
	title = cc.111.title
	desc = cc.111.desc
	outcome = neutral

	illustration_tags = {
		10 = armed
		10 = happy
	}

	immediate = {
		random_subject = {
			limit = { is_subject_type = holy_protectorate }
			save_scope_as = ward
		}
	}

	# Transfer nearby conquered locations to the protectorate
	option = {
		name = cc.111.a
		trigger = {
			any_owned_location = {
				integration_level = conquered
				"border_distance_to(scope:ward)" < 300
			}
		}
		every_owned_location = {
			limit = {
				integration_level = conquered
				"border_distance_to(scope:ward)" < 300
			}
			change_location_owner = scope:ward
		}
		scope:ward = {
			add_liberty_desire = liberty_desire_mild_minus
			add_prestige = prestige_mild_bonus
		}
	}

	# Keep the lands — prestige for the victory, costs for the broken expectation
	option = {
		name = cc.111.b
		add_prestige = prestige_mild_bonus
		add_religious_influence = -40
		add_opinion = {
			target = scope:ward
			modifier = cc_protectorate_lands_withheld
		}
		scope:ward = {
			add_liberty_desire = liberty_desire_mild_plus
		}
	}
}

			add_opinion = { modifier = opinion_mediated target = root }
		}
		scope:ward = {
			add_liberty_desire = liberty_desire_mild_plus
		}
	}

	# Buy him off — spend gold to redirect his energies; a costly middle path

		10 = interior
		10 = happy
	}

	trigger = {
		any_subject = {
			OR = {
				is_subject_type = protectorate
				is_subject_type = holy_protectorate
			}
			opinion = { target = root value >= 180 }
			subject_loyalty >= 65
			NOT = { has_variable = cc_ward_recent_subject }
			root = { has_primary_or_accepted_culture = prev.culture }
		}
		current_age_or_later = { age = age_4_reformation }
		NOT = { has_variable = cc_conversion_chain_cooldown }
	}

	immediate = {
		set_variable = { name = cc_conversion_chain_cooldown value = yes years = 10 }
		random_subject = {
			limit = {
				OR = {
					is_subject_type = protectorate
					is_subject_type = holy_protectorate
				}
				opinion = { target = root value >= 180 }
				subject_loyalty >= 65
				NOT = { has_variable = cc_ward_recent_subject }
				root = { has_primary_or_accepted_culture = prev.culture }
			}
			save_scope_as = ward
		}
	}

	# Accept the transformation — conversion to vassal (standard)
	option = {
		name = cc.ward_no_longer.a
		trigger = {
			NOT = { ruler_or_regent = { has_trait = cruel } }
			NOT = {
				AND = {
					ruler_or_regent = { has_trait = zealot }
					scope:ward = { is_subject_type = holy_protectorate }
				}
			}
		}
		scope:ward = {
			change_subject_type = subject_type:vassal
			add_liberty_desire = liberty_desire_mild_minus
		}
		add_prestige = prestige_severe_bonus

		if = {
			limit = {
				scope:ward = {
					is_subject_type = holy_protectorate
					religion = religion:catholic
				}
				exists = c:PAP
			}
			c:PAP = { add_opinion = { target = prev modifier = cc_papal_disapproval_neglect } }
		}
	}

	# Cruel ruler absorbs by decree — conversion but ward resents it
	option = {
		name = cc.ward_no_longer.a_cruel
		trigger = {
			ruler_or_regent = { has_trait = cruel }
		}

			change_subject_type = subject_type:vassal

		trigger = {
			NOT = { current_age_or_later = { age = age_5_absolutism } }
			scope:ward = { subject_loyalty >= 25 }
		}
		scope:ward = { add_liberty_desire = 5 }

			limit = { religion = religion:catholic }
			add_opinion = {
				target = c:PAP
				modifier = cc_papal_disapproval_neglect
			}

			}

		add_country_modifier = { modifier = cc_faithless_overlord_repute years = 5 }

		}

	option = {
		name = cc.ward_no_longer.d_honor

			current_age_or_later = { age = age_5_absolutism }

					has_trait = grand_chancellor
					has_trait = master_statesman

			}

		add_prestige = prestige_severe_bonus
		add_country_modifier = { modifier = cc_gracious_release years = 5 }
		# Pope approves of a gracious and ceremonial release — a dignified conclusion to a holy compact
		if = {
			limit = {
				scope:ward = { religion = religion:catholic }
				exists = c:PAP
			}

					OR = {

						has_trait = peacemaker_counselor
						has_trait = diplomatic_attache
					}

		}

# CROSS-REALM SUBJECT INTERACTIONS (cc.236 college, cc.240 commune)
# Scientific Colleges and Artists' Communes are one-per-realm, so "two of a kind"
# can only interact across borders: your subject meets a nearby foreign overlord's
# subject of the same type. Seen via the overlord (root). Each event offers a
# friendly route (collaboration — bonds rise) and a rivalrous route (competition /
# poaching — cultural gain at the cost of the personal bond and the rival's regard).
# A 15-year per-overlord cooldown; a border_distance cap keeps far-flung realms apart.
###############################################################################

# cc.236 — "A Correspondence of Scholars"
# Your Scientific College and a foreign overlord's College open contact.
cc.236 = {
	type = country_event
	title = cc.college.exchange.title
	desc = cc.college.exchange.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
	}

	trigger = {
		NOT = { has_variable = cc_college_exchange_cd }
		any_subject = { is_subject_type = scientific_college }
		any_country = {
			NOT = { this = root }
			is_subject = no
			"border_distance_to(root)" < 1000
			any_subject = { is_subject_type = scientific_college }
		}
	}

	immediate = {
		set_variable = { name = cc_college_exchange_cd value = yes years = 15 }
		random_subject = {
			limit = { is_subject_type = scientific_college }
			save_scope_as = cc_my_college
		}
		random_country = {
			limit = {
				NOT = { this = root }
				is_subject = no
				"border_distance_to(root)" < 1000
				any_subject = { is_subject_type = scientific_college }
			}
			save_scope_as = cc_rival_overlord
			random_subject = {
				limit = { is_subject_type = scientific_college }
				save_scope_as = cc_rival_college
			}
		}
	}

	# Friendly — sponsor a joint disputation and scholarly exchange.
	option = {
		name = cc.college.exchange.a
		add_gold = { value = monthly_income_trade_and_tax  multiply = -0.20  max = -40  min = -600 }
		cc_create_college_treatise = { type = work_of_art_type:treatise quality = 75 }
		add_country_modifier = { modifier = cc_scholastic_disputation  years = 5  mode = add_and_extend }
		add_prestige = prestige_mild_bonus
		scope:cc_my_college = {
			change_variable = { name = cc_bond_cul  add = 1 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  add = 1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		scope:cc_rival_overlord = {
			add_opinion = { target = root  modifier = cc_diplomatic_favor }
		}
	}

	# Rivalrous — drive your college to outshine the rival and poach its masters.
	option = {
		name = cc.college.exchange.b
		cc_create_college_treatise = { type = work_of_art_type:treatise quality = 85 }
		add_country_modifier = { modifier = cc_internalized_scholarship  years = 4  mode = add_and_extend }
		add_prestige = prestige_weak_bonus
		scope:cc_my_college = {
			change_variable = { name = cc_bond_cul  add = 1 }
			clamp_variable = { name = cc_bond_cul  max = 5  min = -5 }
			change_variable = { name = cc_bond_per  subtract = 1 }
			clamp_variable = { name = cc_bond_per  max = 5  min = -5 }
		}
		scope:cc_rival_overlord = {
			add_opinion = { target = root  modifier = opinion_harshly_treated }
		}
	}

	# Stay aloof — let the scholars correspond as they will.
	option = {
		name = cc.college.exchange.c
	}
}

# SPECIAL ONE-LOCATION SUBJECT — ANNEXATION STALL (cc.249 hidden handler + cc.250 popup)
# cc.249 is fired every month from on_annexing_subject_monthly_pulse (cc_subject_pulse.txt)
# while an annexation is in progress. root = overlord, scope:target = subject.
# It continuously suppresses the subject's opinion of the overlord below the subject type's
# annexation_stall_opinion so AI annexation never completes, and — for a human overlord —
# periodically surfaces the buy-out choice (cc.250). The AI never sees cc.250, so AI
# annexation of these subjects stays stalled indefinitely.
###############################################################################

cc.250 = {
	type = country_event
	title = cc.annex.resist.title
	desc = cc.annex.resist.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
	}

	# Respect their autonomy — the annexation stays stalled (the pulse keeps
	# suppressing opinion). The AI weight makes this the default if it ever fired.
	option = {
		name = cc.annex.resist.a
		ai_chance = { factor = 100 }
	}

	# Pay to force the integration through — gold scaled to the subject's tax base.
	option = {
		name = cc.annex.resist.b
		ai_chance = { factor = 0 }
		add_gold = {
			value = country_tax_base
			min = scope:target.country_tax_base
			multiply = -10
		}
		scope:target = {
			set_variable = { name = cc_annex_pushing value = yes years = 5 }
			add_opinion = { target = root modifier = cc_annex_pressure }
		}
	}

	# Spend prestige instead — lean on dynastic and diplomatic weight.
	option = {
		name = cc.annex.resist.c
		ai_chance = { factor = 0 }
		add_prestige = prestige_severe_penalty
		scope:target = {
			set_variable = { name = cc_annex_pushing value = yes years = 5 }
			add_opinion = { target = root modifier = cc_annex_pressure }
		}
	}
}
