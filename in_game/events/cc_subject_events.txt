# provincial_governorate: The Governor's Ambitions
# MTTH ~4 years, max every 10 years
cc.120 = {
	type = country_event
	title = cc.gov.1.title
	desc = cc.gov.1.desc
	outcome = negative

	illustration_tags = {
		10 = interior
	}

	trigger = {
		any_subject = {
			is_subject_type = provincial_governorate
			NOT = { has_variable = cc_gov_ambitions_cooldown }
		}
	}

	immediate = {
		random_subject = {
			limit = {
				is_subject_type = provincial_governorate
				NOT = { has_variable = cc_gov_ambitions_cooldown }
			}
			save_scope_as = governor
		}
	}

	# Pay off the governor's court
	option = {
		name = cc.gov.1.a
		add_gold = -100
		scope:governor = {
			set_variable = { name = cc_gov_ambitions_cooldown value = yes }
		}
	}

	# Suppress the ambitions — prestige penalty, subject opinion hit
	option = {
		name = cc.gov.1.b
		add_prestige = prestige_mild_penalty
		scope:governor = {
			add_opinion = { modifier = opinion_harshly_treated target = root }
			set_variable = { name = cc_gov_ambitions_cooldown value = yes }
		}
	}

	# Ignore the reports — autonomy pressure and liberty desire grow
	option = {
		name = cc.gov.1.c
		scope:governor = {
			add_country_modifier = { modifier = cc_subject_autonomy_pressure }
			add_liberty_desire = liberty_desire_mild_plus
		}
	}
}

# provincial_governorate: The Governor Eyes the Neighbors
# MTTH ~7 years, max every 15 years
cc.124 = {
	type = country_event
	title = cc.gov.2.title
	desc = cc.gov.2.desc
	outcome = negative

	illustration_tags = {
		10 = exterior
	}

	trigger = {
		any_subject = {
			is_subject_type = provincial_governorate
			NOT = { has_variable = cc_gov_neighbors_cooldown }
			any_neighbor_country = {
				NOT = { is_subject_of = root }
				NOT = { is_at_war_with = root }
				military_strength < prev.military_strength
			}
		}
	}

	immediate = {
		random_subject = {
			limit = {
				is_subject_type = provincial_governorate
				NOT = { has_variable = cc_gov_neighbors_cooldown }
				any_neighbor_country = {
					NOT = { is_subject_of = root }
					NOT = { is_at_war_with = root }
					military_strength < prev.military_strength
				}
			}
			save_scope_as = governor
		}
		scope:governor = {
			random_neighbor_country = {
				limit = {
					NOT = { is_subject_of = root }
					NOT = { is_at_war_with = root }
					military_strength < prev.military_strength
				}
				save_scope_as = target_neighbor
			}
		}
	}

	# Sanction expansion — neighbor opinion hit, subject liberty desire grows
	option = {
		name = cc.gov.2.a
		scope:target_neighbor = {
			add_opinion = { modifier = opinion_expansionist_threat target = root }
		}
		scope:governor = {
			add_liberty_desire = liberty_desire_mild_plus
			set_variable = { name = cc_gov_neighbors_cooldown value = yes years = 15 }
		}
	}

	# Forbid expansion — nobles displeasure, subject autonomy pressure
	option = {
		name = cc.gov.2.b
		add_estate_satisfaction = { type = estate_type:nobles_estate value = -0.05 }

			add_country_modifier = { modifier = cc_subject_autonomy_pressure }

		}
	}
}

					military_strength < prev.military_strength
				}
			}
			save_scope_as = march_lord
		}
		scope:march_lord = {
			random_neighbor_country = {
				limit = {
					NOT = { is_subject_of = root }
					NOT = { is_at_war_with = root }
					military_strength < prev.military_strength
				}
				save_scope_as = threatened_neighbor
			}
		}
	}

	# Back the march — prestige gain, antagonism hit from neighbor
	option = {
		name = cc.march.3.a
		add_prestige = prestige_mild_bonus
		scope:threatened_neighbor = {
			add_opinion = { modifier = opinion_backed_aggressor target = root }
		}
		scope:march_lord = {
			add_liberty_desire = liberty_desire_mild_minus
			set_variable = { name = cc_march_threat_cooldown value = yes }
			declare_war_with_cb = {
				target = scope:threatened_neighbor
				type = casus_belli:cb_humiliate_rival
			}
		}
	}

	# Denounce the march lord — march liberty desire grows, neighbor opinion restored
	option = {
		name = cc.march.3.b
		scope:march_lord = {
			add_liberty_desire = liberty_desire_mild_plus
			set_variable = { name = cc_march_threat_cooldown value = yes }
		}
		scope:threatened_neighbor = {
			add_opinion = { modifier = opinion_defended target = root }
		}
	}

	# Mediate the dispute — 100 gold, moderate outcome for all
	option = {
		name = cc.march.3.c
		add_gold = -100
		scope:march_lord = {
			set_variable = { name = cc_march_threat_cooldown value = yes }
		}
		scope:threatened_neighbor = {
			add_opinion = { modifier = opinion_mediated target = root }
		}
	}
}
