# "The Protectorate Claims the Holy War's Fruit"
# Fires from cc_holy_war_on_winning when overlord wins a holy war
# while holding a holy_protectorate subject.
cc.111 = {
	type = country_event
	title = cc.111.title
	desc = cc.111.desc
	outcome = neutral

	illustration_tags = {
		10 = armed
		10 = happy
	}

	immediate = {
		random_subject = {
			limit = { is_subject_type = holy_protectorate }
			save_scope_as = ward
		}
	}

	# Transfer nearby conquered locations to the protectorate
	option = {
		name = cc.111.a
		trigger = {
			any_owned_location = {
				integration_level = conquered
				"border_distance_to(scope:ward)" < 300
			}
		}
		every_owned_location = {
			limit = {
				integration_level = conquered
				"border_distance_to(scope:ward)" < 300
			}
			change_location_owner = scope:ward
		}
		scope:ward = {
			add_liberty_desire = liberty_desire_mild_minus
			add_prestige = prestige_mild_bonus
		}
	}

	# Keep the lands — prestige for the victory, costs for the broken expectation
	option = {
		name = cc.111.b
		add_prestige = prestige_mild_bonus
		add_religious_influence = -40
		add_opinion = {
			target = scope:ward
			modifier = cc_protectorate_lands_withheld
		}
		scope:ward = {
			add_liberty_desire = liberty_desire_mild_plus
		}
	}
}

# provincial_governorate: The Governor's Ambitions
# MTTH ~4 years, max every 10 years
cc.120 = {
	type = country_event
	title = cc.gov.1.title
	desc = cc.gov.1.desc
	outcome = negative

	illustration_tags = {
		10 = interior
	}

	trigger = {
		any_subject = {
			is_subject_type = provincial_governorate
			NOT = { has_variable = cc_gov_ambitions_cooldown }
		}
	}

	immediate = {
		random_subject = {
			limit = {
				is_subject_type = provincial_governorate
				NOT = { has_variable = cc_gov_ambitions_cooldown }
			}
			save_scope_as = governor
		}
	}

	# Pay off the governor's court
	option = {
		name = cc.gov.1.a
		add_gold = { value = monthly_income_trade_and_tax multiply = -0.40 max = -100 min = -1500 }
		scope:governor = {
			set_variable = { name = cc_gov_ambitions_cooldown value = yes }
		}
	}

	# Suppress the ambitions — prestige penalty, subject opinion hit
	option = {
		name = cc.gov.1.b
		add_prestige = prestige_mild_penalty
		scope:governor = {
			add_opinion = { modifier = opinion_harshly_treated target = root }
			set_variable = { name = cc_gov_ambitions_cooldown value = yes }
		}
	}

	# Ignore the reports — autonomy pressure and liberty desire grow
	option = {
		name = cc.gov.1.c
		scope:governor = {
			add_country_modifier = { modifier = cc_subject_autonomy_pressure }
			add_liberty_desire = liberty_desire_mild_plus
		}
	}
}

# provincial_governorate: The Governor Eyes the Neighbors
# MTTH ~7 years, max every 15 years
cc.124 = {
	type = country_event
	title = cc.gov.2.title
	desc = cc.gov.2.desc
	outcome = negative

	illustration_tags = {
		10 = exterior
	}

	trigger = {
		any_subject = {
			is_subject_type = provincial_governorate
			NOT = { has_variable = cc_gov_neighbors_cooldown }
			any_neighbor_country = {
				NOT = { is_subject_of = root }
				NOT = { is_at_war_with = root }
				military_strength < prev.military_strength
			}
		}
	}

	immediate = {
		random_subject = {
			limit = {
				is_subject_type = provincial_governorate
				NOT = { has_variable = cc_gov_neighbors_cooldown }

					NOT = { is_at_war_with = root }

				}
			}
			save_scope_as = governor
		}
		scope:governor = {
			random_neighbor_country = {
				limit = {
					NOT = { is_subject_of = root }
					NOT = { is_at_war_with = root }

	title = cc.emboldened_ward.title
	desc = cc.emboldened_ward.desc
	outcome = negative

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	trigger = {
		any_subject = {
			is_subject_type = protectorate
			NOT = { has_variable = cc_emboldened_cooldown }
			any_neighbor_country = {
				NOT = { is_subject_of = root }
				NOT = { this = root }
				opinion = { target = prev value < 0 }
				NOT = { is_at_war_with = prev }
			}
		}
		NOT = { has_variable = cc_emboldened_overlord_cooldown }
	}

	immediate = {
		set_variable = { name = cc_emboldened_overlord_cooldown value = yes years = 10 }
		random_subject = {
			limit = {
				is_subject_type = protectorate
				NOT = { has_variable = cc_emboldened_cooldown }
				any_neighbor_country = {
					NOT = { is_subject_of = root }
					NOT = { this = root }
					opinion = { target = prev value < 0 }
					NOT = { is_at_war_with = prev }
				}
			}
			save_scope_as = ward
		}
		scope:ward = {
			set_variable = { name = cc_emboldened_cooldown value = yes years = 8 }
			random_neighbor_country = {
				limit = {
					NOT = { is_subject_of = root }
					NOT = { this = root }
					opinion = { target = prev value < 0 }
					NOT = { is_at_war_with = prev }
				}
				save_scope_as = rival
			}
		}
	}

	# Defuse diplomatically (costs prestige and diplomatic reputation)
	option = {
		name = cc.emboldened_ward.a
		trigger = {
			NOT = {
				any_cabinet_character = {
					in_cabinet = yes
					OR = {
						has_trait = master_statesman
						has_trait = grand_chancellor
						has_trait = peacemaker_counselor
					}
				}
			}
		}
		add_prestige = prestige_mild_penalty
		add_country_modifier = { modifier = cc_diplomatic_scramble years = 3 }
	}

	# Masterful Mediation (grand_chancellor or master_statesman)
	option = {
		name = cc.emboldened_ward.a_master
		trigger = {
			any_cabinet_character = {
				in_cabinet = yes
				OR = {
					has_trait = master_statesman
					has_trait = grand_chancellor
				}
			}
		}
		scope:ward = { add_liberty_desire = -5 }
	}

	# Quiet Resolution (peacemaker_counselor)
	option = {
		name = cc.emboldened_ward.a_peace
		trigger = {
			any_cabinet_character = {
				in_cabinet = yes
				has_trait = peacemaker_counselor
			}
		}
		add_prestige = prestige_weak_penalty
		add_opinion = {
			target = scope:rival
			modifier = cc_de_escalation_overture
		}
	}

	# Stand firm → triggers cc.222
	option = {
		name = cc.emboldened_ward.b
		trigger_event = { id = cc.222 days = 1 }
	}

	# Step back → triggers cc.223
	option = {
		name = cc.emboldened_ward.c
		trigger_event = { id = cc.223 days = 1 }
	}
}

# "Ward Abandoned to Their Fate"

	type = country_event

	desc = cc.ward_abandoned.desc
	outcome = negative

	illustration_tags = {
		10 = angry
		10 = military
	}

	}

# "The Ward's Own Ambitions"
# A protectorate of high development and low loyalty pushes for a defined relationship on their terms.
cc.224 = {
	type = country_event
	title = cc.ward_ambitions.title
	desc = cc.ward_ambitions.desc
	outcome = negative

	illustration_tags = {
		10 = interior
		10 = armed
	}

	trigger = {
		any_subject = {
			is_subject_type = protectorate
			development >= 30
			loyalty < 45
			NOT = { has_variable = cc_ward_ambitions_cooldown }
		}
		NOT = { has_variable = cc_ward_ambitions_overlord_cooldown }
	}

	immediate = {
		set_variable = { name = cc_ward_ambitions_overlord_cooldown value = yes years = 10 }
		random_subject = {
			limit = {
				is_subject_type = protectorate
				development >= 30
				loyalty < 45
				NOT = { has_variable = cc_ward_ambitions_cooldown }
			}
			save_scope_as = ward
		}
		scope:ward = {
			set_variable = { name = cc_ward_ambitions_cooldown value = yes years = 8 }
		}
	}

	# Grant client state status
	option = {
		name = cc.ward_ambitions.a
		trigger = {
			NOT = { ruler_or_regent = { has_trait = cruel } }
		}
		scope:ward = { change_subject_type = subject_type:client_state }
		add_prestige = prestige_mild_bonus
		add_country_modifier = { modifier = cc_magnanimous_overlord years = 10 }
	}

	# Offer vassal status instead
	option = {
		name = cc.ward_ambitions.b
		scope:ward = {
			change_subject_type = subject_type:vassal
			add_liberty_desire = 10
		}
		add_prestige = prestige_weak_bonus
	}

	# "Your ambitions serve me well" — vassal with permanent loyalty modifier (diplo minister)
	option = {
		name = cc.ward_ambitions.b_minister
		trigger = {
			any_cabinet_character = {
				in_cabinet = yes
				OR = {
					has_trait = consolidation_minister
					has_trait = loyal_steward
				}
			}
		}
		scope:ward = {
			change_subject_type = subject_type:vassal
			add_liberty_desire = 5
			add_country_modifier = { modifier = cc_recognized_partner }
		}
		add_prestige = prestige_mild_bonus
	}

	# Deny the request (available until denied twice)
	option = {
		name = cc.ward_ambitions.c
		trigger = {
			scope:ward = {
				NOT = {
					AND = {
						has_variable = cc_ward_ambitions_denied_count
						var:cc_ward_ambitions_denied_count >= 2
					}
				}
			}
		}
		scope:ward = {
			add_liberty_desire = liberty_desire_severe_plus
			if = {
				limit = { NOT = { has_variable = cc_ward_ambitions_denied_count } }
				set_variable = { name = cc_ward_ambitions_denied_count value = 1 }
			}
			else = {
				change_variable = { name = cc_ward_ambitions_denied_count add = 1 }
			}
		}
	}

	# Independence war — denied too many times; friendly neighbors may join
	option = {
		name = cc.ward_ambitions.c_war
		trigger = {
			scope:ward = {
				has_variable = cc_ward_ambitions_denied_count
				var:cc_ward_ambitions_denied_count >= 2
			}
		}
		scope:ward = {
			declare_war_with_cb = { target = root type = casus_belli:cb_independence_war }
			random_current_war = {
				save_scope_as = cc_ambitions_war
			}
		}
		add_prestige = prestige_severe_penalty
		add_country_modifier = { modifier = cc_faithless_overlord_repute years = 5 }
		# Friendly neighbors of the ward may join the independence war
		if = {
			limit = { exists = scope:cc_ambitions_war }
			scope:ward = {
				every_neighbor_country = {
					limit = {
						NOT = { is_subject_of = root }
						NOT = { this = root }
						opinion = { target = prev value >= 50 }
						NOT = { at_war = yes }
					}
					join_war_as_attacker = { war = scope:cc_ambitions_war }
				}
			}
		}
	}
}

# "The Faith Grows Cold"

	desc = cc.faith_cold.desc

	}

		any_subject = {
			is_subject_type = holy_protectorate
			subject_loyalty < 40
			NOT = { has_variable = cc_faith_cold_cooldown }
		}
		NOT = { has_variable = cc_faith_cold_overlord_cooldown }

		}

		add_prestige = prestige_mild_bonus
		add_country_modifier = { modifier = cc_gracious_release years = 5 }
	}

	# Renew commitment — costs religious influence, restores loyalty
	option = {
		name = cc.faith_cold.b
		scope:ward = { add_liberty_desire = -15 }
		add_religious_influence = -20
		add_country_modifier = { modifier = cc_faith_renewed years = 3 }
	}

	# Refuse bluntly (available until refused twice)
	option = {
		name = cc.faith_cold.c
		trigger = {
			scope:ward = {
				NOT = {
					AND = {
						has_variable = cc_faith_cold_denied_count
						var:cc_faith_cold_denied_count >= 2
					}
				}
			}
		}
		scope:ward = {
			add_liberty_desire = liberty_desire_mild_plus
			if = {
				limit = { NOT = { has_variable = cc_faith_cold_denied_count } }
				set_variable = { name = cc_faith_cold_denied_count value = 1 }
			}
			else = {
				change_variable = { name = cc_faith_cold_denied_count add = 1 }
			}
		}
		add_prestige = prestige_weak_penalty
	}

	# "Sacred duty is non-negotiable" (zealot ruler)
	option = {
		name = cc.faith_cold.c_zealot
		trigger = {
			ruler_or_regent = { has_trait = zealot }
			scope:ward = {
				NOT = {
					AND = {
						has_variable = cc_faith_cold_denied_count
						var:cc_faith_cold_denied_count >= 2
					}
				}
			}
		}
		scope:ward = {
			add_liberty_desire = 10
			if = {
				limit = { NOT = { has_variable = cc_faith_cold_denied_count } }
				set_variable = { name = cc_faith_cold_denied_count value = 1 }
			}
			else = {
				change_variable = { name = cc_faith_cold_denied_count add = 1 }
			}
		}
		add_country_modifier = { modifier = cc_zealous_refusal years = 5 }
		add_stability = stability_mild_penalty
	}

	# Diplomatic solution (peacemaker_counselor or grand_chancellor)
	option = {
		name = cc.faith_cold.e
		trigger = {
			any_cabinet_character = {
				in_cabinet = yes
				OR = {
					has_trait = peacemaker_counselor
					has_trait = grand_chancellor
				}
			}
		}
		scope:ward = { add_liberty_desire = -15 }
		add_religious_influence = -10
		add_country_modifier = { modifier = cc_faith_renewed years = 3 }
	}

	# Seek Papal Mediation — Catholic only, requires Pope to have decent opinion of overlord
	option = {
		name = cc.faith_cold.d_papal
		trigger = {
			religion = religion:catholic
			scope:ward = { religion = religion:catholic }
			exists = c:PAP
			c:PAP = { opinion = { target = root value >= 80 } }
		}
		scope:ward = { add_liberty_desire = -20 }
		# Calling in a favor costs some goodwill with Rome
		c:PAP = { add_opinion = { target = prev modifier = cc_papal_mediation_called } }
		add_country_modifier = { modifier = cc_faith_renewed years = 3 }
	}

	# Independence war — too many refusals
	option = {
		name = cc.faith_cold.c_war
		trigger = {
			scope:ward = {
				has_variable = cc_faith_cold_denied_count
				var:cc_faith_cold_denied_count >= 2
			}
		}
		scope:ward = {
			declare_war_with_cb = { target = root type = casus_belli:cb_independence_war }
			random_current_war = {
				save_scope_as = cc_faith_war
			}
		}
		add_prestige = prestige_severe_penalty
		if = {
			limit = {
				scope:ward = { religion = religion:catholic }
				country_exists = c:PAP
				c:PAP = { NOT = { is_at_war_with = root } }
				exists = scope:cc_faith_war
			}
			c:PAP = { join_war_as_attacker = { war = scope:cc_faith_war } }
		}
	}
}

# "Patron of the Faith"
# Fires once per game when the overlord holds 2+ holy protectorates with sufficient religious influence.
cc.226 = {
	type = country_event
	title = cc.patron_faith.title
	desc = cc.patron_faith.desc
	outcome = positive

	illustration_tags = {
		10 = interior
		10 = happy
	}

	trigger = {
		any_subject = {
			count >= 2
			is_subject_type = holy_protectorate
		}
		NOT = { has_variable = cc_patron_of_faith_fired }
		religious_influence >= 50
		current_age_or_later = { age = age_3_discovery }
	}

	immediate = {
		set_variable = { name = cc_patron_of_faith_fired value = yes }
		random_subject = {
			limit = { is_subject_type = holy_protectorate }
			save_scope_as = ward_a
		}
		random_subject = {
			limit = {
				is_subject_type = holy_protectorate
				NOT = { this = scope:ward_a }
			}
			save_scope_as = ward_b
		}
	}

	# Embrace the title — prestige modifier + ward loyalty
	option = {
		name = cc.patron_faith.a
		add_country_modifier = { modifier = cc_patron_of_faith }
		scope:ward_a = { add_liberty_desire = -10 }
		scope:ward_b = { add_liberty_desire = -10 }
		set_variable = { name = cc_patron_of_faith_active value = yes }
	}

	# Champion of the Faith (zealot ruler — variant of A with war bonus)
	option = {
		name = cc.patron_faith.a_zealot
		trigger = {
			ruler_or_regent = { has_trait = zealot }
		}
		add_country_modifier = { modifier = cc_patron_of_faith }
		add_country_modifier = { modifier = cc_champion_of_faith years = 20 }
		scope:ward_a = { add_liberty_desire = -10 }
		scope:ward_b = { add_liberty_desire = -10 }
		set_variable = { name = cc_patron_of_faith_active value = yes }
	}

	# Receive Papal Recognition — Catholic overlord, Pope formally endorses the role
	option = {
		name = cc.patron_faith.a_papal
		trigger = {
			religion = religion:catholic
			exists = c:PAP
		}
		add_country_modifier = { modifier = cc_papal_patron }
		scope:ward_a = { add_liberty_desire = -10 }
		scope:ward_b = { add_liberty_desire = -10 }
		set_variable = { name = cc_patron_of_faith_active value = yes }
		# Pope formally recognizes the overlord's patronage — a prestigious endorsement
		c:PAP = { add_opinion = { target = prev modifier = cc_papal_patron_recognition } }
	}

	# Remain humble — prestige bonus, minor ward loyalty
	option = {
		name = cc.patron_faith.b
		add_prestige = prestige_severe_bonus
		scope:ward_a = { add_liberty_desire = -5 }
		scope:ward_b = { add_liberty_desire = -5 }
	}

	# Use the reputation as leverage — diplomatic reputation
	option = {
		name = cc.patron_faith.c
		add_country_modifier = { modifier = cc_gracious_release years = 10 }
	}
}

# "A Rift in the Sacred Bond"
# Fires when a holy protectorate no longer shares the overlord's religion.

	type = country_event
	title = cc.faith_rift.title
	desc = cc.faith_rift.desc
	outcome = negative

	illustration_tags = {
		10 = interior
		10 = angry
	}

			is_subject_type = holy_protectorate

			NOT = { has_variable = cc_faith_rift_cooldown }

		}

		add_country_modifier = { modifier = cc_righteous_intervention years = 3 }

	option = {
		name = cc.faith_rift.b
		cancel_subject = scope:ward
		add_prestige = prestige_mild_bonus

			limit = {

		name = cc.faith_rift.c

			add_liberty_desire = liberty_desire_mild_plus
			add_country_modifier = { modifier = cc_faith_rift_tension years = 10 }
		}
		add_prestige = prestige_weak_penalty
	}

	# Negotiate a formal exception — a skilled minister finds a face-saving arrangement
	option = {
		name = cc.faith_rift.c_master
		trigger = {
			any_cabinet_character = {
				in_cabinet = yes
				OR = {
					has_trait = grand_chancellor
					has_trait = master_statesman
					has_trait = peacemaker_counselor
				}
			}
		}
		scope:ward = {
			add_liberty_desire = 5
			add_country_modifier = { modifier = cc_faith_rift_tension years = 5 }
		}
	}

	# Seek Papal Guidance — Catholic overlord places the matter before Rome
	option = {
		name = cc.faith_rift.d_papal
		trigger = {
			religion = religion:catholic
			exists = c:PAP
			c:PAP = { opinion = { target = root value >= 60 } }
		}
		scope:ward = {
			add_liberty_desire = liberty_desire_mild_plus
			set_variable = { name = cc_reconversion_ultimatum value = yes years = 10 }
		}
		# Rome is pleased to be consulted rather than bypassed
		c:PAP = { add_opinion = { target = prev modifier = cc_de_escalation_overture } }
	}
}

# "The Ward Seeks Their Own Path"
# Follow-up to cc.227 when the ward received a reconversion ultimatum but loyalty remains too low
# to compel compliance. The ward demands independence, citing the broken spiritual covenant.
cc.228 = {
	type = country_event
	title = cc.faith_rift_escalate.title
	desc = cc.faith_rift_escalate.desc
	outcome = negative

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	trigger = {
		any_subject = {
			is_subject_type = holy_protectorate
			has_variable = cc_reconversion_ultimatum
			subject_loyalty < 25
		}
		NOT = { has_variable = cc_faith_rift_escalate_cooldown }
	}

	immediate = {
		set_variable = { name = cc_faith_rift_escalate_cooldown value = yes years = 5 }
		random_subject = {
			limit = {
				is_subject_type = holy_protectorate
				has_variable = cc_reconversion_ultimatum
				subject_loyalty < 25
			}
			save_scope_as = ward
		}
	}

	# Grant independence — accept the inevitable and release them with what dignity remains
	option = {
		name = cc.faith_rift_escalate.a
		cancel_subject = scope:ward
		add_prestige = prestige_mild_penalty
	}

	# Refuse — the ward declares independence and goes to war
	option = {
		name = cc.faith_rift_escalate.b
		scope:ward = {
			declare_war_with_cb = { target = root type = casus_belli:cb_independence_war }
			random_current_war = {
				save_scope_as = cc_faith_rift_war
			}
		}
		add_prestige = prestige_severe_penalty
		add_country_modifier = { modifier = cc_faithless_overlord_repute years = 5 }
		# If the ward is Catholic, the Pope backs their claim — a heretical overlord has no right to hold them
		if = {
			limit = {
				scope:ward = { religion = religion:catholic }
				country_exists = c:PAP
				c:PAP = { NOT = { is_at_war_with = root } }
				exists = scope:cc_faith_rift_war
			}
			c:PAP = { join_war_as_attacker = { war = scope:cc_faith_rift_war } }
		}
	}
}
