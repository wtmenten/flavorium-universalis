# "Whispers in the Court"
# Fires: yearly pulse, after shadow_diplomacy_advance; nobles power >= 45, power_projection >= 20.
cc.shadow.1 = {
	type = country_event
	title = cc.shadow.1.title
	desc = cc.shadow.1.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
	}

	trigger = {
		has_advance = shadow_diplomacy_advance
		nobles_estate_power >= 45
		power_projection >= 20
		NOT = { has_variable = cc_shadow_event_fired }
	}

	immediate = {
		set_variable = { name = cc_shadow_event_fired value = yes }
	}

	option = {
		name = cc.shadow.1.a
		add_estate_privilege = estate_privilege:ep_noble_shadow_network
		trigger_event = { id = cc.shadow.2 months = 12 }
	}

	option = {
		name = cc.shadow.1.b
		monthly_prestige = 10
	}
}

# "The Shadow Bureau Acts"
# Fires from yearly pulse when shadow_court_bureau is active + has shadow_state subjects.
# Offers discounted puppet_state upgrade for one shadow state.
cc.shadow.bureau.1 = {
	type = country_event
	title = cc.shadow.bureau.1.title
	desc = cc.shadow.bureau.1.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
	}

	trigger = {
		has_bureaucracy_of_type = bureaucracy_type:cc_shadow_court_bureau
		has_advance = puppet_state_advance
		any_subject_country = { is_subject_type = shadow_state }
		NOT = { has_variable = cc_shadow_bureau_event_fired }
	}

	immediate = {
		set_variable = { name = cc_shadow_bureau_event_fired value = yes years = 10 }
		random_subject_country = {
			limit = { is_subject_type = shadow_state }
			save_scope_as = shadow_target
		}
	}

	option = {
		name = cc.shadow.bureau.1.a
		# Upgrade shadow_state to puppet_state at reduced war score cost
		scope:shadow_target = {
			change_subject_type = puppet_state
		}
		power_projection = -5
	}

	option = {
		name = cc.shadow.bureau.1.b
		power_projection = 5
	}
}

# "The Scholars Seek a Home"
# Fires: from ep_clergy_scholarly_college on_activate.
cc.clergy.college.1 = {
	type = country_event
	title = cc.clergy.college.1.title
	desc = cc.clergy.college.1.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
	}

	trigger = {
		has_estate_privilege = estate_privilege:ep_clergy_scholarly_college
		clergy_estate_power >= 50
	}

	immediate = {
		random_neighbor_country = {
			limit = { government_type = government_type:theocracy }
			save_scope_as = college_candidate
		}
	}

	option = {
		name = cc.clergy.college.1.a
		trigger = {
			exists = scope:college_candidate
		}
		scope:college_candidate = {
			add_opinion = { target = root value = 25 }
		}
		custom_tooltip = cc.clergy.college.1.a.tt
		trigger_event = { id = cc.clergy.college.2 years = 2 }
	}

	option = {
		name = cc.clergy.college.1.b
		add_country_modifier = {
			name = cc_internalized_scholarship
			years = 10
		}
	}
}

###############################################################################
# CHAIN 5: THE HOLY LEAGUE
# Fires from yearly pulse when holy_patronage_advance + religious conditions.
###############################################################################

# "Coreligionists Seek Shelter"
# Fires: yearly pulse, after holy_patronage_advance; high religious influence; weak neighbor with same religion group.
cc.holy.1 = {
	type = country_event
	title = cc.holy.1.title
	desc = cc.holy.1.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
	}

	trigger = {
		has_advance = holy_patronage_advance
		monthly_religious_influence >= 0.3
		any_neighbor_country = {
			religion = {
				has_religion_group = root.religion.religion_group
			}
			NOT = { is_at_war_with = root }
			relative_power_to_country = {
				country = root
				value < -0.3
			}
		}
		NOT = { has_variable = cc_holy_event_fired }
	}

	immediate = {
		set_variable = { name = cc_holy_event_fired value = yes years = 15 }
		random_neighbor_country = {
			limit = {
				religion = {
					has_religion_group = root.religion.religion_group
				}
				NOT = { is_at_war_with = root }
				relative_power_to_country = {
					country = root
					value < -0.3
				}
			}
			save_scope_as = holy_candidate
		}
	}

	option = {
		name = cc.holy.1.a
		scope:holy_candidate = {
			change_subject_type = holy_protectorate
		}
		trigger_event = { id = cc.holy.2 years = 2 }
	}

	option = {
		name = cc.holy.1.b
		power_projection = -5
	}
}
