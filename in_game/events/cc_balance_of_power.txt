################################################################################
# EVENT 10 — Choosing Sides (to engaged, unaligned powers)
################################################################################
cc_balance_of_power.10 = {
	type = country_event
	category = situation_event
	title = cc_balance_of_power.10.title
	desc = cc_balance_of_power.10.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
		10 = ages
	}

	trigger = {
		situation:cc_balance_of_power = { situation_is_active = yes }
		is_subject = no
		NOT = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
		NOT = { is_member_of_international_organization = international_organization:cc_bop_coalition }
	}

	# Join the balancing coalition.
	option = {
		name = cc_balance_of_power.10.a
		ai_chance = {
			factor = 1
			modifier = {
				factor = 2.5
				exists = international_organization:cc_bop_preponderance
				international_organization:cc_bop_preponderance.leader_country ?= { is_rival_of = root }
			}
			modifier = { factor = 0.3  has_country_modifier = cc_bop_bought_ally }
		}
		hidden_effect = {
			if = {
				limit = {
					exists = international_organization:cc_bop_coalition
					can_join_international_organization = international_organization:cc_bop_coalition
				}
				international_organization:cc_bop_coalition = {
					add_country_to_international_organization = root
				}
			}
		}
		international_organization:cc_bop_coalition.leader_country ?= {
			add_opinion = { target = root  modifier = cc_bop_coalition_solidarity_opinion }
		}
	}

	# Join the preponderant bloc.
	option = {
		name = cc_balance_of_power.10.b
		ai_chance = {
			factor = 1
			modifier = {
				factor = 2
				exists = international_organization:cc_bop_preponderance
				international_organization:cc_bop_preponderance.leader_country ?= { has_royal_marriage_with = root }
			}
			modifier = { factor = 2  has_country_modifier = cc_bop_bought_ally }
		}
		hidden_effect = {
			if = {
				limit = {
					exists = international_organization:cc_bop_preponderance
					can_join_international_organization = international_organization:cc_bop_preponderance
				}
				international_organization:cc_bop_preponderance = {
					add_country_to_international_organization = root
				}
			}
		}
	}

	# Remain neutral — keep our hands free.
	option = {
		name = cc_balance_of_power.10.c
		ai_chance = { factor = 1 }
		add_prestige = prestige_weak_bonus
	}
}

################################################################################
# EVENT 20 — Pressure and Persuasion (preponderant leader coerces a court)
################################################################################
cc_balance_of_power.20 = {
	type = country_event
	category = situation_event
	title = cc_balance_of_power.20.title
	desc = cc_balance_of_power.20.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
		10 = armed
	}

	trigger = {
		is_leader_of_international_organization = international_organization:cc_bop_preponderance
		any_country = {
			cc_bop_eligible_country = yes
			is_subject = no
			has_variable = cc_bop_engaged
			NOT = { this = root }
			NOT = { is_leader_of_international_organization = international_organization:cc_bop_coalition }
			NOT = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
		}
	}

	immediate = {
		random_country = {
			limit = {
				cc_bop_eligible_country = yes
				is_subject = no
				has_variable = cc_bop_engaged
				NOT = { this = root }
				NOT = { is_leader_of_international_organization = international_organization:cc_bop_coalition }
				NOT = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
			}
			save_scope_as = bop_target
		}
	}

	# Threaten them into our bloc.
	option = {
		name = cc_balance_of_power.20.a
		ai_chance = { factor = 1 }
		hidden_effect = {
			scope:bop_target = {
				if = {
					limit = { is_member_of_international_organization = international_organization:cc_bop_coalition }
					international_organization:cc_bop_coalition = {
						remove_country_from_international_organization = scope:bop_target
					}
				}
			}
			if = {
				limit = { scope:bop_target = { can_join_international_organization = international_organization:cc_bop_preponderance } }
				international_organization:cc_bop_preponderance = {
					add_country_to_international_organization = scope:bop_target
				}
			}
		}
		scope:bop_target = {
			add_opinion = { target = root  modifier = cc_bop_coerced_resentment }
		}
	}

	# Buy their neutrality with gold.
	option = {
		name = cc_balance_of_power.20.b
		ai_chance = { factor = 1 }
		add_gold = { value = monthly_income_trade_and_tax  multiply = -0.50  min = -150  max = -10 }
		scope:bop_target = {
			add_country_modifier = {
				modifier = cc_bop_bought_ally
				years = 6
				mode = add_and_extend
			}
			add_opinion = { target = root  modifier = cc_bop_bought_neutrality_opinion }
		}
	}

	# Leave them be for now.
	option = {
		name = cc_balance_of_power.20.c
		ai_chance = { factor = 1 }
	}
}

			limit = {
				is_member_of_international_organization = international_organization:cc_bop_coalition

				years = 4

		international_organization:cc_bop_coalition = {
			change_variable = { name = cc_bop_resolve add = 5 }
		}
	}

	# Court a neutral power into the coalition.
	option = {
		name = cc_balance_of_power.40.b
		ai_chance = { factor = 1 }
		hidden_effect = {
			random_country = {
				limit = {
					cc_bop_eligible_country = yes
					is_subject = no
					has_variable = cc_bop_engaged
					can_join_international_organization = international_organization:cc_bop_coalition
				}
				save_scope_as = bop_recruit
				international_organization:cc_bop_coalition = {
					add_country_to_international_organization = scope:bop_recruit
				}
			}
		}
		if = {
			limit = { exists = scope:bop_recruit }
			scope:bop_recruit = {
				add_opinion = { target = root  modifier = cc_bop_coalition_solidarity_opinion }
			}
		}
	}

	# Stand pat and conserve our strength.
	option = {
		name = cc_balance_of_power.40.c
		ai_chance = { factor = 1 }
		add_prestige = prestige_weak_bonus
	}

# EVENT 83 — Playing Both Sides (to an unaligned court)
# Nudges the balance toward the weaker bloc and turns a modest profit.
cc_balance_of_power.83 = {
	type = country_event
	category = situation_event
	title = cc_balance_of_power.83.title
	desc = cc_balance_of_power.83.desc
	outcome = neutral
	illustration_tags = { 10 = interior  10 = trading }
	option = {
		name = cc_balance_of_power.83.a
		# Shake down BOTH bloc leaders: each pays 20% of its treasury, floored at 2.5x our
		# own monthly income; we pocket the combined sum.
		set_variable = { name = cc_bop_extort_total value = 0 }
		if = {
			limit = { exists = international_organization:cc_bop_preponderance }
			international_organization:cc_bop_preponderance.leader_country ?= {
				save_scope_as = bop_payer
				root = {
					set_variable = {
						name = cc_bop_extort_side
						value = {
							value = scope:bop_payer.gold
							multiply = 0.2
							min = { value = monthly_income_trade_and_tax  multiply = 2.5 }
						}
					}
					change_variable = { name = cc_bop_extort_total add = var:cc_bop_extort_side }
				}
				add_gold = { value = root.var:cc_bop_extort_side  multiply = -1 }
			}
		}
		if = {
			limit = { exists = international_organization:cc_bop_coalition }
			international_organization:cc_bop_coalition.leader_country ?= {
				save_scope_as = bop_payer
				root = {
					set_variable = {
						name = cc_bop_extort_side
						value = {
							value = scope:bop_payer.gold
							multiply = 0.2
							min = { value = monthly_income_trade_and_tax  multiply = 2.5 }
						}
					}
					change_variable = { name = cc_bop_extort_total add = var:cc_bop_extort_side }
				}
				add_gold = { value = root.var:cc_bop_extort_side  multiply = -1 }
			}
		}
		# Fallback if somehow neither bloc has a leader.
		if = {
			limit = { var:cc_bop_extort_total = 0 }
			change_variable = { name = cc_bop_extort_total add = 25 }
		}
		add_gold = { value = var:cc_bop_extort_total }
		remove_variable = cc_bop_extort_total
		remove_variable = cc_bop_extort_side
		if = {
			limit = { situation:cc_balance_of_power = { var:cc_bop_preponderance_share >= 50 } }
			situation:cc_balance_of_power = { change_variable = { name = cc_bop_share_offset subtract = 3 } }
		}
		else = {
			situation:cc_balance_of_power = { change_variable = { name = cc_bop_share_offset add = 3 } }
		}
	}
}

# EVENT 87 — Subsidise the Allies (coalition leader)
cc_balance_of_power.87 = {
	type = country_event
	category = situation_event
	title = cc_balance_of_power.87.title
	desc = cc_balance_of_power.87.desc
	outcome = positive
	illustration_tags = { 10 = interior  10 = economy }
	option = {
		name = cc_balance_of_power.87.a
		# Bankrolling the allies costs ~6 months of their combined income — the wealthier and
		# more numerous the coalition, the heavier the bill on the leader's treasury.
		set_variable = { name = cc_bop_subsidy_cost value = 0 }
		if = {
			limit = { exists = international_organization:cc_bop_coalition }
			international_organization:cc_bop_coalition = {
				every_international_organization_member = {
					limit = { NOT = { is_leader_of_international_organization = international_organization:cc_bop_coalition } }
					# Read the member's income as a native property in its own scope, then hand a clean
					# variable back to root. A cross-scope scriptvalue dot-chain in an `add` field can
					# resolve to none and poison the accumulator (then var:cc_bop_subsidy_cost reads none).
					set_variable = { name = cc_bop_member_income value = monthly_income_trade_and_tax }
					save_scope_as = sub_mem
					root = { change_variable = { name = cc_bop_subsidy_cost add = scope:sub_mem.var:cc_bop_member_income } }
					remove_variable = cc_bop_member_income
				}
			}
		}
		change_variable = { name = cc_bop_subsidy_cost multiply = 6 }
		clamp_variable = { name = cc_bop_subsidy_cost min = 50 max = 1500 }
		if = {
			limit = { has_variable = cc_bop_subsidy_cost }
			add_gold = { value = var:cc_bop_subsidy_cost  multiply = -1 }

# EVENT 100 — The Reversal of Alliances (to a wavering non-leader member)
cc_balance_of_power.100 = {
	type = country_event
	category = situation_event
	title = cc_balance_of_power.100.title
	desc = cc_balance_of_power.100.desc
	outcome = neutral
	illustration_tags = { 10 = interior  10 = professional }

	trigger = {
		OR = {
			AND = {
				is_member_of_international_organization = international_organization:cc_bop_coalition
				NOT = { is_leader_of_international_organization = international_organization:cc_bop_coalition }
			}
			AND = {
				is_member_of_international_organization = international_organization:cc_bop_preponderance
				NOT = { is_leader_of_international_organization = international_organization:cc_bop_preponderance }
			}
		}
	}

	immediate = {
		save_scope_as = bop_defector
		if = {
			limit = { is_member_of_international_organization = international_organization:cc_bop_coalition }
			international_organization:cc_bop_coalition.leader_country ?= { save_scope_as = bop_old_leader }
			international_organization:cc_bop_preponderance.leader_country ?= { save_scope_as = bop_new_leader }
		}
		else = {
			international_organization:cc_bop_preponderance.leader_country ?= { save_scope_as = bop_old_leader }
			international_organization:cc_bop_coalition.leader_country ?= { save_scope_as = bop_new_leader }
		}
	}

	# "Reverse our alliances — cross to the other camp." (the betrayal)
	option = {
		name = cc_balance_of_power.100.a
		# # A bloc closes ranks in wartime — one cannot change sides while under arms.
		# trigger = {
		# 	custom_tooltip = {
		# 		text = cc_bop_cannot_leave_at_war_tt
		# 		at_war = no
		# 	}
		# }
		ai_chance = {
			factor = 1
			modifier = {
				factor = 2.5
				any_cabinet_character = {
					OR = { has_trait = court_schemer  has_trait = cynical_courtier  has_trait = corrupted_official }
				}
			}
			modifier = { factor = 0.2  any_cabinet_character = { has_trait = cc_master_balancer } }
			modifier = { factor = 1.5  at_war = yes }
		}
		# A wartime defection: if we are still under arms against our new patron, make peace
		# FIRST so we may cross the lines without being expelled from the bloc we are joining.
		if = {
			limit = { exists = scope:bop_new_leader  is_at_war_with = scope:bop_new_leader }
			"war_with_country(scope:bop_new_leader)" ?= { save_scope_as = bop_switch_war }
			if = {
				limit = { exists = scope:bop_switch_war }
				white_peace = scope:bop_switch_war
			}
		}
		# Leave our bloc (on_left brands us a defector) and cross to the other.
		if = {
			limit = { is_member_of_international_organization = international_organization:cc_bop_coalition }
			international_organization:cc_bop_coalition = { remove_country_from_international_organization = root }
			if = {
				limit = { can_join_international_organization = international_organization:cc_bop_preponderance }
				international_organization:cc_bop_preponderance = { add_country_to_international_organization = root }
			}
			situation:cc_balance_of_power = { change_variable = { name = cc_bop_share_offset add = 8 } }
		}
		else = {
			international_organization:cc_bop_preponderance = { remove_country_from_international_organization = root }
			if = {
				limit = { can_join_international_organization = international_organization:cc_bop_coalition }
				international_organization:cc_bop_coalition = { add_country_to_international_organization = root }
			}
			situation:cc_balance_of_power = { change_variable = { name = cc_bop_share_offset subtract = 8 } }
		}
		# Complete the betrayal a month later (giving our armies time to disengage and re-form).
		trigger_event_silently = { id = cc_balance_of_power.103 days = 30 }
		# Our new patron pays handsomely (scaled to our economy); our betrayed former allies seethe.
		add_gold = { value = monthly_income_trade_and_tax  multiply = 8  min = 50  max = 400 }
		scope:bop_old_leader ?= {
			add_opinion = { target = root  modifier = cc_bop_concert_betrayal }
			trigger_event_non_silently = { id = cc_balance_of_power.101 days = 3 }
		}
	}

	# "Stand by our allies."
	option = {
		name = cc_balance_of_power.100.b
		ai_chance = {
			factor = 1
			modifier = { factor = 2.5  any_cabinet_character = { has_trait = cc_master_balancer } }
		}
		add_prestige = prestige_mild_bonus
		if = {
			limit = { is_member_of_international_organization = international_organization:cc_bop_coalition }
			international_organization:cc_bop_coalition = { change_variable = { name = cc_bop_resolve add = 8 } }
		}
		else = {
			international_organization:cc_bop_preponderance = { change_variable = { name = cc_bop_ambition add = 8 } }
		}
	}
}

# EVENT 103 — Complete the Betrayal (the defector, a month after crossing over)
# Fired 30 days after .100.a so the defector's armies have disengaged and re-formed. The
# scopes from .100 may have lapsed, so the old enemy is re-derived from current membership:
# the defector now sits in its new bloc, and the opposite bloc is its former camp.
cc_balance_of_power.103 = {
	type = country_event
	category = situation_event
	title = cc_balance_of_power.103.title
	desc = cc_balance_of_power.103.desc
	outcome = neutral
	illustration_tags = { 10 = exterior  10 = military }

	trigger = {
		OR = {
			is_member_of_international_organization = international_organization:cc_bop_preponderance
			is_member_of_international_organization = international_organization:cc_bop_coalition
		}
	}

	immediate = {
		# Identify our former camp's leader (the opposite bloc) as the target, and our
		# new patron (the bloc we just joined) whose banner we now march under.
		if = {
			limit = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
			international_organization:cc_bop_coalition.leader_country ?= { save_scope_as = bop_old_target }
			international_organization:cc_bop_preponderance.leader_country ?= { save_scope_as = bop_new_patron }
		}
		else = {
			international_organization:cc_bop_preponderance.leader_country ?= { save_scope_as = bop_old_target }
			international_organization:cc_bop_coalition.leader_country ?= { save_scope_as = bop_new_patron }
		}
		# Prefer the bloc's existing war against that leader, led by our new patron — we fall
		# in behind it rather than open a separate war of our own.
		if = {
			limit = {
				exists = scope:bop_old_target
				exists = scope:bop_new_patron
				scope:bop_new_patron = { is_at_war_with = scope:bop_old_target }
			}
			scope:bop_new_patron = {
				"war_with_country(scope:bop_old_target)" ?= { save_scope_as = bop_betrayal_war }
			}
		}
		# Failing that, a war we are already drawn into ourselves.
		else_if = {
			limit = { exists = scope:bop_old_target  is_at_war_with = scope:bop_old_target }
			"war_with_country(scope:bop_old_target)" ?= { save_scope_as = bop_betrayal_war }
		}
	}

	# "Take up arms against our former allies." — complete the wartime betrayal.
	option = {
		name = cc_balance_of_power.103.a
		trigger = { exists = scope:bop_old_target }
		ai_chance = {
			# Having taken the new patron's gold, the court is strongly inclined to earn it.
			base = 80
			modifier = {
				add = 15
				any_cabinet_character = { OR = { has_trait = war_hawk  has_trait = cc_master_balancer } }
			}
		}
		# Fall in behind the bloc's war if one is being fought; otherwise the turncoat does
		# NOT open a lone war on its former patron — that only splinters the front into a
		# dozen separate coalition wars. The betrayal still tells through the diplomacy below.
		if = {
			limit = { exists = scope:bop_betrayal_war }
			join_war_against = {
				war = scope:bop_betrayal_war
				target = scope:bop_old_target
				call_in_subjects = yes
			}
		}
		# The turncoat's weight tips the scale toward its new bloc.
		if = {
			limit = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
			situation:cc_balance_of_power = { change_variable = { name = cc_bop_share_offset add = 4 } }
		}
		else = {
			situation:cc_balance_of_power = { change_variable = { name = cc_bop_share_offset subtract = 4 } }
		}
	}

	# "A double game — keep our own counsel." — betray the new patron too and go it alone.
	option = {
		name = cc_balance_of_power.103.b
		ai_chance = {
			base = 20
			modifier = {
				add = 25
				any_cabinet_character = { OR = { has_trait = court_schemer  has_trait = cynical_courtier } }
			}
		}
		# Slip out of the bloc we only just joined — distrusted by all, beholden to none.
		if = {
			limit = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
			international_organization:cc_bop_preponderance = { remove_country_from_international_organization = root }
		}
		else_if = {
			limit = { is_member_of_international_organization = international_organization:cc_bop_coalition }
			international_organization:cc_bop_coalition = { remove_country_from_international_organization = root }
		}
		add_prestige = prestige_weak_bonus
	}
}

		international_organization:cc_bop_preponderance.leader_country ?= {
			NOT = { this = root }
			NOT = { is_at_war_with = root }
			# A truce binds the leader's hands — it cannot strike, so do not call the council.
			NOT = { has_truce_with = root }
		}
	}

	immediate = {
		international_organization:cc_bop_preponderance.leader_country ?= { save_scope_as = bop_hegemon }
	}

	# "Strike now, before the balance breaks for good."
	option = {
		name = cc_balance_of_power.110.a
		trigger = { exists = scope:bop_hegemon }
		ai_chance = {
			factor = 1
			modifier = { factor = 2  situation:cc_balance_of_power = { var:cc_bop_preponderance_share >= 64 } }
			modifier = { factor = 1.5  any_cabinet_character = { OR = { has_trait = war_hawk  has_trait = cc_master_balancer } } }
			modifier = { factor = 0.4  at_war = yes }
		}
		declare_war_with_cb = {
			target = scope:bop_hegemon
			type = casus_belli:cb_coalition
		}
		"war_with_country(scope:bop_hegemon)" ?= { save_scope_as = bop_coalition_war }
		if = {
			# The war actually opened — commit the coalition and pay out the cooldown in full.
			limit = { exists = scope:bop_coalition_war }
			set_variable = { name = cc_bop_recently_engaged_hegemon value = yes years = 3 }
			# Bring the rest of the coalition into the same war as co-attackers.
			international_organization:cc_bop_coalition = {
				every_international_organization_member = {
					limit = {
						NOT = { this = root }
						NOT = { is_at_war_with = scope:bop_hegemon }
					}
					join_war_as_attacker = { war = scope:bop_coalition_war  call_in_subjects = yes }
				}
				change_variable = { name = cc_bop_resolve add = 10 }
				every_international_organization_member = {
					add_country_modifier = { modifier = cc_bop_coalition_solidarity  years = 2  mode = add_and_extend }
				}
			}
			situation:cc_balance_of_power = { change_variable = { name = cc_bop_share_offset subtract = 5 } }
		}
		else = {
			# The blow could not land (truce or other bar) — short cooldown, try again soon.
			set_variable = { name = cc_bop_recently_engaged_hegemon value = yes years = 1 }
		}
	}

	# "Not yet — we are not ready to face the colossus."
	option = {
		name = cc_balance_of_power.110.b
		ai_chance = { factor = 1 }
		set_variable = { name = cc_bop_recently_engaged_hegemon value = yes years = 1 }
		international_organization:cc_bop_coalition = {
			change_variable = { name = cc_bop_resolve add = -12 }
		}
	}
}

# EVENT 111 — Honour the Union (coalition leader, defensive)
# A partner already fights the hegemon; the leader is called to its defence.
cc_balance_of_power.111 = {
	type = country_event
	category = situation_event
	title = cc_balance_of_power.111.title
	desc = cc_balance_of_power.111.desc
	outcome = neutral
	illustration_tags = { 10 = exterior  10 = army }

	trigger = {
		is_leader_of_international_organization = international_organization:cc_bop_coalition
		exists = international_organization:cc_bop_preponderance
		# The bloc must have voted itself at least a defensive pact before the union answers.
		international_organization:cc_bop_coalition = {
			OR = {
				international_organization_has_policy = policy:cc_bop_possible_defense
				international_organization_has_policy = policy:cc_bop_mutual_defense
			}
		}
		international_organization:cc_bop_preponderance.leader_country ?= {
			NOT = { this = root }
			NOT = { is_at_war_with = root }
			# A truce bars the leader from joining the war — do not raise a call it cannot answer.
			NOT = { has_truce_with = root }
		}
	}

	immediate = {
		international_organization:cc_bop_preponderance.leader_country ?= { save_scope_as = bop_hegemon }
		random_country = {
			limit = {
				is_member_of_international_organization = international_organization:cc_bop_coalition
				NOT = { is_leader_of_international_organization = international_organization:cc_bop_coalition }
				exists = scope:bop_hegemon
				is_at_war_with = scope:bop_hegemon
			}
			save_scope_as = bop_victim
		}
	}

	# "Honour the union — march to their defence."
	option = {
		name = cc_balance_of_power.111.a
		trigger = { exists = scope:bop_victim  exists = scope:bop_hegemon }
		ai_chance = {
			factor = 1
			modifier = { factor = 2  situation:cc_balance_of_power = { var:cc_bop_preponderance_share >= 62 } }
			modifier = { factor = 1.5  international_organization:cc_bop_coalition.var:cc_bop_resolve >= 50 }
			modifier = { factor = 1.5  any_cabinet_character = { has_trait = cc_master_balancer } }
			modifier = { factor = 0.5  at_war = yes }
		}
		scope:bop_victim = {
			"war_with_country(scope:bop_hegemon)" ?= { save_scope_as = bop_coalition_war }
		}
		if = {
			limit = { exists = scope:bop_coalition_war }
			join_war_against = {
				war = scope:bop_coalition_war
				target = scope:bop_hegemon
				call_in_subjects = yes
			}
		}
		if = {
			# The leader is now in the war — commit the bloc and pay the cooldown in full.
			limit = { is_at_war_with = scope:bop_hegemon }
			set_variable = { name = cc_bop_recently_engaged_hegemon value = yes years = 3 }
			# As coalition leader, assume command of every partner's war on the hegemon,
			# folding their separate fights under a single banner.
			international_organization:cc_bop_coalition = {
				every_international_organization_member = {
					limit = {
						NOT = { this = root }
						is_at_war_with = scope:bop_hegemon
					}
					root = { take_over_all_wars = prev }
				}
			}
			# Under MUTUAL DEFENCE the call is binding: rally every member not yet under arms
			# behind the now-unified war. Under POSSIBLE DEFENCE only the leader and the members
			# already fighting unite — the uncommitted "may be called, but need not answer".
			if = {
				limit = {
					international_organization:cc_bop_coalition = {
						international_organization_has_policy = policy:cc_bop_mutual_defense
					}
				}
				international_organization:cc_bop_coalition = {
					every_international_organization_member = {
						limit = {
							NOT = { this = root }
							NOT = { is_at_war_with = scope:bop_hegemon }
						}
						join_war_against = {
							war = scope:bop_coalition_war
							target = scope:bop_hegemon
							call_in_subjects = yes
						}
					}
				}
			}
			international_organization:cc_bop_coalition = {
				change_variable = { name = cc_bop_resolve add = 10 }
			}
			situation:cc_balance_of_power = { change_variable = { name = cc_bop_share_offset subtract = 4 } }
		}
		else = {
			# Could not join (truce or other bar) — short cooldown, the call may come again.
			set_variable = { name = cc_bop_recently_engaged_hegemon value = yes years = 1 }
		}
	}

	# "We dare not provoke the colossus — they must shift for themselves."
	option = {
		name = cc_balance_of_power.111.b
		ai_chance = { factor = 1 }
		set_variable = { name = cc_bop_recently_engaged_hegemon value = yes years = 1 }
		international_organization:cc_bop_coalition = {
			change_variable = { name = cc_bop_resolve add = -10 }
		}
		add_prestige = prestige_severe_penalty
		add_legitimacy = legitimacy_severe_penalty
		scope:bop_victim ?= {
			add_opinion = { target = root  modifier = cc_bop_coerced_resentment }
		}
	}
}

# EVENT 130 — Desperate Measures (coalition leader, hegemon nearing the threshold)
# Unlocked once the congress is empowered; offers extreme expedients to redress the balance.
cc_balance_of_power.130 = {
	type = country_event
	category = situation_event
	title = cc_balance_of_power.130.title
	desc = cc_balance_of_power.130.desc
	outcome = neutral
	illustration_tags = { 10 = exterior  10 = military }

	trigger = {
		is_leader_of_international_organization = international_organization:cc_bop_coalition
	}

	# "Buy a foreign sword into the war." — only when there is a live war to throw it into
	# and a strong outsider to be bought in.
	option = {
		name = cc_balance_of_power.130.a
		trigger = {
			exists = international_organization:cc_bop_preponderance.leader_country
			any_country = {
				is_member_of_international_organization = international_organization:cc_bop_coalition
				is_at_war_with = international_organization:cc_bop_preponderance.leader_country
			}
			any_country = {
				cc_bop_eligible_country = no
				is_subject = no
				is_great_power = yes
				NOT = { is_at_war_with = international_organization:cc_bop_preponderance.leader_country }
				NOT = { is_member_of_international_organization = international_organization:cc_bop_coalition }
				NOT = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
			}
		}
		ai_chance = {
			factor = 1
			modifier = { factor = 1.5  situation:cc_balance_of_power = { var:cc_bop_preponderance_share >= 63 } }
		}
		add_gold = { value = monthly_income_trade_and_tax  multiply = -8  min = -500  max = -60 }
		international_organization:cc_bop_preponderance.leader_country = { save_scope_as = bop_hegemon }
		# Find the coalition's war against the hegemon.
		random_country = {
			limit = {
				is_member_of_international_organization = international_organization:cc_bop_coalition
				is_at_war_with = scope:bop_hegemon
			}
			"war_with_country(scope:bop_hegemon)" ?= { save_scope_as = bop_foreign_war }
		}
		# Buy the strongest available outsider into that war against the hegemon.
		if = {
			limit = { exists = scope:bop_foreign_war }
			ordered_country = {
				limit = {
					cc_bop_eligible_country = no
					is_subject = no
					is_great_power = yes
					NOT = { is_at_war_with = scope:bop_hegemon }
					NOT = { is_member_of_international_organization = international_organization:cc_bop_coalition }
					NOT = { is_member_of_international_organization = international_organization:cc_bop_preponderance }
				}
				order_by = great_power_score
				max = 1
				check_range_bounds = no
				join_war_against = {
					war = scope:bop_foreign_war
					target = scope:bop_hegemon
					call_in_subjects = yes
				}
				add_opinion = { target = root  modifier = cc_bop_grateful_for_support }
			}
			situation:cc_balance_of_power = {
				change_variable = { name = cc_bop_share_offset subtract = 6 }
				clamp_variable = { name = cc_bop_share_offset min = -25 max = 25 }
			}
		}
		international_organization:cc_bop_coalition = { change_variable = { name = cc_bop_resolve add = 8 } }
		set_variable = { name = cc_bop_recently_desperate value = yes years = 4 }
	}

	# "Call a general mobilisation across the bloc."
	option = {
		name = cc_balance_of_power.130.b
		ai_chance = { factor = 1 }
		international_organization:cc_bop_coalition = {
			change_variable = { name = cc_bop_resolve add = -15 }
			every_international_organization_member = {
				add_country_modifier = { modifier = cc_bop_coalition_solidarity_empowered  years = 2  mode = add_and_extend }
			}
		}
		situation:cc_balance_of_power = {
			change_variable = { name = cc_bop_share_offset subtract = 3 }
			clamp_variable = { name = cc_bop_share_offset min = -25 max = 25 }
		}
		set_variable = { name = cc_bop_recently_desperate value = yes years = 4 }
	}

	# "Stay our hand — the cost is too dear."
	option = {
		name = cc_balance_of_power.130.c
		ai_chance = { factor = 0.4 }
		set_variable = { name = cc_bop_recently_desperate value = yes years = 1 }
	}
}
