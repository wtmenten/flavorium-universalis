
# Cabinets and Choices — Dynamic AI Personality Events
# Namespace: cc_personality
#
# IDs 1-5:   Periodic drift events (context-sensitive + broad catch-all)
# IDs 10-19: Ottoman (TUR) inflection events
# IDs 20-29: France (FRA) / England (ENG) inflection events
# IDs 30-39: Muscovy (MOS) inflection events
# IDs 40-49: Castile (CAS) inflection events
# IDs 50-59: Sweden (SWE) inflection events
#
# All events fire in country scope.
# AI option weights favor the historically-grounded choice (~70%) with a
# meaningful alternative (~30%) to preserve some game-to-game variation.
# Each option applies cc_ai_personality_guard to prevent drift overriding
# a freshly-set personality.

namespace = cc_personality

########################
# Drift Events (1-5)
########################

# cc_personality.1 — The Winds of War
# Fires when an AI country is at war and unguarded.
# Sharpens strategy toward aggression or survival instinct.
cc_personality.1 = {
	type = country_event
	title = cc_personality.1.title
	desc = cc_personality.1.desc
	outcome = neutral

	illustration_tags = {
		10 = military
		10 = armed
	}

	option = {
		name = cc_personality.1.a
		set_personality = ai_personality:ai_aggressive
		add_country_modifier = { modifier = cc_ai_personality_guard years = 15 }
		ai_chance = { base = 60 }
	}
	option = {
		name = cc_personality.1.b
		set_personality = ai_personality:ai_defensive
		add_country_modifier = { modifier = cc_ai_personality_guard years = 15 }
		ai_chance = { base = 40 }
	}
}

# cc_personality.5 — A Question of Strategy
# Broad catch-all drift event. Fires at 20% annual chance for any unguarded AI country.
# Uses weighted random_list covering all 7 personalities, adjusted by game state.
cc_personality.5 = {
	type = country_event
	title = cc_personality.5.title
	desc = cc_personality.5.desc
	outcome = neutral

	illustration_tags = {
		10 = interior
	}

	# For player-controlled countries: choose a strategic direction.
	# AI resolves via ai_chance weights on each option.

	option = {
		name = cc_personality.5.aggressive
		set_personality = ai_personality:ai_aggressive
		add_country_modifier = { modifier = cc_ai_personality_guard years = 15 }
		ai_chance = {
			base = 15
			modifier = { add = 10 at_war = yes }
			modifier = { add = 5 is_great_power = yes }
		}
	}
	option = {
		name = cc_personality.5.expansionist
		set_personality = ai_personality:ai_expansionist
		add_country_modifier = { modifier = cc_ai_personality_guard years = 15 }
		ai_chance = {
			base = 15
			modifier = { add = 10 is_great_power = yes }
			modifier = { add = 5 num_subjects < 3 }
		}
	}
	option = {
		name = cc_personality.5.defensive
		set_personality = ai_personality:ai_defensive
		add_country_modifier = { modifier = cc_ai_personality_guard years = 15 }
		ai_chance = {
			base = 15
			modifier = { add = 10 prestige < 30 }
			modifier = { add = 5 at_war = yes }
		}
	}
	option = {
		name = cc_personality.5.cautious
		set_personality = ai_personality:ai_cautious
		add_country_modifier = { modifier = cc_ai_personality_guard years = 15 }
		ai_chance = {
			base = 15
			modifier = { add = 5 prestige < 50 }
		}
	}
	option = {
		name = cc_personality.5.opportunistic
		set_personality = ai_personality:ai_opportunistic
		add_country_modifier = { modifier = cc_ai_personality_guard years = 15 }
		ai_chance = {
			base = 10
			modifier = { add = 10 gold > 100 }
		}
	}
	option = {
		name = cc_personality.5.isolationist
		set_personality = ai_personality:ai_isolationist
		add_country_modifier = { modifier = cc_ai_personality_guard years = 15 }
		ai_chance = {
			base = 10
			modifier = { add = 10 num_subjects < 2 }
			modifier = {
				add = 5
				capital = {
					OR = {
						sub_continent = sub_continent:east_asia
						sub_continent = sub_continent:south_east_asia
					}
				}
			}
		}
	}
	option = {
		name = cc_personality.5.friendly
		set_personality = ai_personality:ai_friendly
		add_country_modifier = { modifier = cc_ai_personality_guard years = 15 }
		ai_chance = {
			base = 10
			modifier = { add = 5 at_war = no }
		}
	}
}
