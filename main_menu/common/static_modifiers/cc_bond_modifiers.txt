# Subject Relations Ledger — engage-action modifiers
# Named country modifiers granted by the carrot actions in
# in_game/events/cc_bond_engage.txt (cc_bonds.300-304). Applied via
#   add_country_modifier = { modifier = X  months = 120  mode = add_and_extend }
# so re-engaging refreshes the duration rather than stacking the magnitude.
# All reuse vanilla modifier keys, so no modifier_type_definitions are required.

# Economic — Underwrite the Local Treasury (granted to the SUBJECT).
# The crown backs the subject's finances: cheaper credit and a richer mint.
cc_bond_underwritten = {
	game_data = {
		category = country
	}
	bank_interest = -0.01
	minting_income_factor = 0.10
}

# Military — Joint Military Review (granted to BOTH overlord and subject).
# Shared drill and logistics trim the upkeep of both armies and fleets.
cc_bond_drilled = {
	game_data = {
		category = country
	}
	army_maintenance_efficiency = 0.10
	navy_maintenance_efficiency = 0.10
}
