# CABINET COMPOSITION (country)
#
# Stacked: one party runs the cabinet, and it shows in efficiency. The cost is in
# satisfaction and rivalry (cc_pty_update_sat, cc_pty_feed_rivalry).
# Balanced: steadier and cheaper to manage, with weaker party bonuses
# (cc_pty_apply_mood) and a satisfaction floor.
###############################################################################

cc_pty_comp_stacked = {
	game_data = { category = country }
	country_cabinet_efficiency = 0.05
	# One party delivers the votes.
	monthly_political_influence_gain_modifier = 0.10
}

cc_pty_comp_balanced = {
	game_data = { category = country }
	stability_cost_efficiency = 0.05
	cc_pty_bribe_cost_modifier = -0.25
	cc_pty_attack_cost_modifier = -0.25
	# Brokering among the parties is itself a source of influence.
	monthly_political_influence = 0.5
}
