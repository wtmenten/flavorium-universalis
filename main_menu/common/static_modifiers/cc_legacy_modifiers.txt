# Cabinet Legacy Modifiers
# Permanent modifiers left behind when legendary ministers die.
# Applied by cc_legacy events; stack across the run.
# Each is unique per minister (set by character variable, applied to country).
########################

# Generic legacy shapes — flavored in loc by the character name
# Track A: Protege inheritance is trait-based; no modifier here.
# Track B: Legendary minister death → one of the following applied permanently.

# Left by a minister with grand_vizier_legendary
cc_legacy_grand_vizier = {
	game_data = {
		category = country
		decaying = no
	}

	diplomatic_reputation = 1
	antagonism_received_modifier = -0.05
	diplomatic_capacity_modifier = 0.05
}

# Left by a minister with tribune_of_the_people

		decaying = no

	monthly_rebel_growth = -0.0001
	legislative_efficiency = 0.03
}
