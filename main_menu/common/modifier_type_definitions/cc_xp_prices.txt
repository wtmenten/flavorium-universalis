###############################################################################
# Flavorium Universalis — Cabinet Experience: price cost modifier types
#
# Every entry in common/prices/ needs a matching <price_id>_cost_modifier registered
# here, or the game logs "Missing modifier type for price. <id>_cost_modifier" on load.
# Vanilla does the same for its own prices (head_of_cabinet_promotion_cost_modifier in
# 00_modifier_types.txt), and this mod already does it for the subject payment prices in
# cc_subject_prices.txt.
#
# Registering the type is what makes the cost scriptable: anything that wants to make
# training cheaper (an advance, a reform, a trait) can now apply
# cc_xp_training_cost_modifier as an ordinary country modifier.
#
# color=bad matches vanilla's convention for cost modifiers, where a positive value is
# a worse outcome for the player.
###############################################################################

cc_xp_training_cost_modifier = {
	color = bad
	percent = yes
	game_data = {
		category = country
	}
}
