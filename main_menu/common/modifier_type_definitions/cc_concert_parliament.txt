
# Concert of Europe — parliament participation modifier types.
# Each special status that may vote in an IO parliament needs a registered
# <status>_can_participate_in_parliament modifier type (see vanilla 00_modifier_types.txt,
# e.g. emperor_can_participate_in_parliament). The Concert weights votes through these two:
# the president (special_status_power 2) and ordinary members (special_status_power 1).
cc_concert_president_can_participate_in_parliament = {
	boolean = yes
	game_data = {
		category = internationalorganization
	}
}

cc_concert_seat_can_participate_in_parliament = {
	boolean = yes
	game_data = {
		category = internationalorganization
	}
}

# Each participating special status also needs an <status>_agenda_impact type
# (vanilla pairs every *_can_participate_in_parliament with an *_agenda_impact, percent=yes).
cc_concert_president_agenda_impact = {
	percent = yes
	game_data = {
		category = internationalorganization
	}
}

cc_concert_seat_agenda_impact = {
	percent = yes
	game_data = {
		category = internationalorganization
	}
}
