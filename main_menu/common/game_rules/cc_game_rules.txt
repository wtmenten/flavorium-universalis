# Should the mod's custom subject types be available?
# Gates every advance that unlocks a Cabinets & Choices subject type (and the
# estate privileges / government reforms bundled with them) via the
# cc_subject_types_available scripted trigger. When an advance is gated out it
# never appears in the tech tree, so the subject cannot be created and all its
# downstream events, decisions, and parliament issues stay dormant.
#   on          = available to every country
#   player_only = available to human-controlled countries only (AI keeps vanilla subjects)
#   off         = disabled for everyone
cc_subject_types = {
	default = cc_subject_types_on

	cc_subject_types_on = {
		flag = general_rule
	}

	cc_subject_types_player_only = {
		flag = general_rule
	}

	cc_subject_types_off = {
		flag = general_rule
	}
}
