# Flavorium Universalis - Factions: game rules
#
# Independent of the cabinet experience rule (cc_cabinet_xp) on purpose. Parties read cabinet
# membership directly and need nothing the XP system publishes, so switching XP off should
# not switch parties off with it.
# All of the mod's rules use flag = flavour_rule, so they sit together on the Flavour tab of
# the game rules screen rather than among vanilla's General rules. A tab of their own would
# need an override of vanilla in_game/gui/game_rules.gui, which defines the tabs inline.

cc_factions = {
	default = cc_factions_on

	cc_factions_on = {
		flag = flavour_rule
	}

	cc_factions_off = {
		flag = flavour_rule
	}
}
